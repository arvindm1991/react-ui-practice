// Fake API. Treat it like a real backend you can't change.
export type OrderStatus = 'Paid' | 'Baking' | 'Out for delivery' | 'Delivered';

export type Order = {
  id: string;
  number: number;
  customer: string;
  items: number;
  totalCents: number;
  status: OrderStatus;
};

export type OrdersPage = {
  orders: Order[];
  total: number; // total number of orders across ALL pages
};

const CUSTOMERS = ['Blue Door Café', 'Hotel Marlowe', 'Greenleaf Market', 'Oak & Ember', 'Little Fern Daycare', 'Sunday Social', 'Pier 9 Kitchen', 'The Reading Room'];
const STATUSES: OrderStatus[] = ['Paid', 'Baking', 'Out for delivery', 'Delivered'];

// 23 orders, newest first: #1123 down to #1101.
const ORDERS: Order[] = [];
for (let i = 0; i < 23; i++) {
  ORDERS.push({
    id: 'o' + (1123 - i),
    number: 1123 - i,
    customer: CUSTOMERS[i % CUSTOMERS.length],
    items: 4 + ((i * 7) % 19),
    totalCents: 4200 + ((i * 3517) % 26000),
    status: STATUSES[Math.min(3, Math.floor(i / 3))],
  });
}

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /orders?page=2&pageSize=5 -> { orders, total }
// Pages start at 1. A page past the end returns an empty list.
export async function getOrders(page: number, pageSize: number): Promise<OrdersPage> {
  await randomDelay(300, 900);
  const start = (page - 1) * pageSize;
  return {
    orders: ORDERS.slice(start, start + pageSize).map((o) => ({ ...o })),
    total: ORDERS.length,
  };
}

function OrdersTable() {
  const [orders, setOrders] = useState<Order[]>([]);
  const [total, setTotal] = useState(0);

import React, { useEffect, useState } from 'react';
import { getOrders } from './api';
import type { Order } from './api';
import { StatusBadge } from './components/StatusBadge';
import { formatPrice } from './utils/format';

const PAGE_SIZE = 5;

function TopNav() {
  return (
    <header className="topnav">
      <span className="logo">🥐 Crumb &amp; Co.</span>
      <span className="env">Wholesale admin</span>
    </header>
  );
}

function OrderRow({ order }: { order: Order }) {
  return (
    <tr>
      <td className="order-no">#{order.number}</td>
      <td>{order.customer}</td>
      <td>{order.items} items</td>
      <td><StatusBadge status={order.status} /></td>
      <td className="num">{formatPrice(order.totalCents)}</td>
    </tr>
  );
}

type PaginationProps = {
  page: number;
  pageCount: number;
  disabled: boolean;
  onChange: (page: number) => void;
};

function Pagination({ page, pageCount, disabled, onChange }: PaginationProps) {
  // [1, 2, ..., pageCount]
  const pages = Array.from({ length: pageCount }, (_, i) => {
    return i + 1;
  });

  const prevDisabled = disabled || page === 1;
  const nextDisabled = disabled || page === pageCount;

  return (
    <nav className="pager" aria-label="Pagination">
      <button type="button" className="pager-btn" disabled={prevDisabled} onClick={() => { onChange(page - 1); }}>← Prev</button>
      {pages.map((n) => {
        let cls = 'page-num';
        let current: 'page' | undefined = undefined;
        if (n === page) {
          cls = 'page-num is-current';
          current = 'page';
        }
        return (
          <button key={n} type="button" className={cls} aria-current={current} disabled={disabled} onClick={() => { onChange(n); }}>
            {n}
          </button>
        );
      })}
      <button type="button" className="pager-btn" disabled={nextDisabled} onClick={() => { onChange(page + 1); }}>Next →</button>
    </nav>
  );
}

function OrdersTable() {
  const [orders, setOrders] = useState<Order[]>([]);
  const [total, setTotal] = useState(0);
  const [loading, setLoading] = useState(true);
  const [page, setPage] = useState(1);

  // Re-runs whenever the page changes. The old rows stay on screen (dimmed) until the new ones arrive.
  useEffect(() => {
    let cancelled = false;
    setLoading(true);
    getOrders(page, PAGE_SIZE).then((data) => {
      if (cancelled) {
        return;
      }
      setOrders(data.orders);
      setTotal(data.total);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, [page]);

  function goTo(next: number) {
    setPage(next);
  }

  // derived
  const pageCount = Math.max(1, Math.ceil(total / PAGE_SIZE));
  const first = (page - 1) * PAGE_SIZE + 1;
  const last = Math.min(page * PAGE_SIZE, total);

  if (loading && orders.length === 0) {
    return <p className="msg">Loading orders…</p>;
  }

  let tableCls = 'table';
  if (loading) {
    tableCls = 'table is-loading';
  }

  return (
    <div className="card">
      <table className={tableCls} aria-busy={loading}>
        <thead>
          <tr>
            <th>Order</th>
            <th>Customer</th>
            <th>Items</th>
            <th>Status</th>
            <th className="num">Total</th>
          </tr>
        </thead>
        <tbody>
          {orders.map((o) => {
            return <OrderRow key={o.id} order={o} />;
          })}
        </tbody>
      </table>
      <footer className="table-foot">
        <span className="range">Showing {first}–{last} of {total}</span>
        <Pagination page={page} pageCount={pageCount} disabled={loading} onChange={(n) => { goTo(n); }} />
      </footer>
    </div>
  );
}

export default function App() {
  return (
    <div className="app">
      <TopNav />
      <main className="main">
        <div className="page-head">
          <h1>Orders</h1>
          <button type="button" className="btn">Export CSV</button>
        </div>
        <OrdersTable />
      </main>
    </div>
  );
}
