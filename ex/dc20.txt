// Fake API. Treat it like a real backend you can't change.
export type FileItem = {
  id: string;
  name: string;
  kind: 'pdf' | 'image' | 'doc' | 'sheet' | 'video';
  sizeBytes: number;
  modified: string; // "Oct 2"
  owner: string;
};

let FILES: FileItem[] = [
  { id: 'f1', name: 'Q3 roadmap.pdf', kind: 'pdf', sizeBytes: 2480000, modified: 'Oct 2', owner: 'You' },
  { id: 'f2', name: 'Brand photos (raw).zip', kind: 'image', sizeBytes: 812000000, modified: 'Sep 28', owner: 'Priya S.' },
  { id: 'f3', name: 'Offsite agenda.docx', kind: 'doc', sizeBytes: 64200, modified: 'Sep 27', owner: 'You' },
  { id: 'f4', name: 'Budget 2027.xlsx', kind: 'sheet', sizeBytes: 318000, modified: 'Sep 21', owner: 'Leo M.' },
  { id: 'f5', name: 'Product demo.mp4', kind: 'video', sizeBytes: 1430000000, modified: 'Sep 14', owner: 'You' },
  { id: 'f6', name: 'Hiring plan.docx', kind: 'doc', sizeBytes: 98000, modified: 'Sep 3', owner: 'You' },
];

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /files -> FileItem[]
export async function getFiles(): Promise<FileItem[]> {
  await randomDelay(300, 900);
  return FILES.map((f) => ({ ...f }));
}

// DELETE /files/:id -> { ok: true }
export async function deleteFile(id: string): Promise<{ ok: true }> {
  await randomDelay(400, 1000);
  FILES = FILES.filter((f) => f.id !== id);
  return { ok: true };
}

@@@ shared src/utils/formatBytes.ts
// formatBytes(2480000) -> "2.5 MB"   formatBytes(64200) -> "64 KB"   formatBytes(1430000000) -> "1.4 GB"
export function formatBytes(bytes: number): string {
  if (bytes >= 1e9) {
    return (bytes / 1e9).toFixed(1) + ' GB';
  }
  if (bytes >= 1e6) {
    return (bytes / 1e6).toFixed(1) + ' MB';
  }
  return Math.round(bytes / 1e3) + ' KB';
}

function FileList() {
  const [files, setFiles] = useState<FileItem[]>([]);
  const [loading, setLoading] = useState(true);
  // Which file the modal is asking about. null = modal closed. Store the id, not a copy.
  const [pendingId, setPendingId] = useState<string | null>(null);
  const [deleting, setDeleting] = useState(false);

  useEffect(() => {
    let cancelled = false;
    getFiles().then((data) => {
      if (cancelled) {
        return;
      }
      setFiles(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // Stable, because the modal's Escape effect depends on it.
  const handleCancel = useCallback(() => {
    setPendingId(null);
  }, []);

  function handleConfirm() {
    if (!pendingId) {
      return;
    }
    const id = pendingId;
    setDeleting(true);
    deleteFile(id).then(() => {
      setFiles((prev) => {
        return prev.filter((f) => {
          return f.id !== id;
        });
      });
      setDeleting(false);
      setPendingId(null);
    });
  }

  // derived
  const pendingFile = files.find((f) => {
    return f.id === pendingId;
  });

  let modal = null;
  if (pendingFile) {
    modal = (
      <ConfirmDeleteModal
        fileName={pendingFile.name}
        deleting={deleting}
        onCancel={handleCancel}
        onConfirm={() => { handleConfirm(); }}
      />
    );
  }

  if (loading) {
    return <p className="msg">Loading files…</p>;
  }

  let empty = null;
  if (files.length === 0) {
    empty = <p className="msg empty">No files left. Drop something here to upload.</p>;
  }

  return (
    <div>
      <ul className="file-list">
        {files.map((f) => {
          return <FileRow key={f.id} file={f} onDelete={(id) => { setPendingId(id); }} />;
        })}
      </ul>
      {empty}
      {modal}
    </div>
  );
}

export default function App() {
  return (
    <div className="layout">
      <Sidebar />
      <main className="main">
        <h1>My files</h1>

          <span>Modified</span>
          <span className="size">Size</span>
          <span />
        </div>
        <FileList />
      </main>
    </div>
  );
}

import React, { useEffect } from 'react';
import { Button } from './Button';

type ConfirmDeleteModalProps = {
  fileName: string;
  deleting: boolean;
  onCancel: () => void;
  onConfirm: () => void;
};

export function ConfirmDeleteModal({ fileName, deleting, onCancel, onConfirm }: ConfirmDeleteModalProps) {
  // Escape closes (but not halfway through a delete). Always remove the listener.
  useEffect(() => {
    function handleKey(e: KeyboardEvent) {
      if (e.key === 'Escape' && !deleting) {
        onCancel();
      }
    }
    document.addEventListener('keydown', handleKey);
    return () => {
      document.removeEventListener('keydown', handleKey);
    };
  }, [deleting, onCancel]);

  function handleBackdropClick(e: React.MouseEvent<HTMLDivElement>) {
    if (e.target === e.currentTarget && !deleting) {
      onCancel();
    }
  }

  // Focus starts on Cancel (autoFocus below): the safe choice for a destructive dialog.
  let deleteLabel = 'Delete';
  if (deleting) {
    deleteLabel = 'Deleting…';
  }

  return (
    <div className="backdrop" onClick={(e) => { handleBackdropClick(e); }}>
      <div className="modal" role="alertdialog" aria-modal="true" aria-labelledby="confirm-title" aria-describedby="confirm-desc">
        <div className="modal-icon">🗑</div>
        <h2 id="confirm-title" className="modal-title">Delete “{fileName}”?</h2>
        <p id="confirm-desc" className="modal-text">This file will be permanently deleted for everyone it’s shared with. You can’t undo this.</p>
        <div className="modal-actions">
          <Button autoFocus disabled={deleting} onClick={() => { onCancel(); }}>Cancel</Button>
          <Button variant="danger" disabled={deleting} onClick={() => { onConfirm(); }}>{deleteLabel}</Button>
        </div>
      </div>
    </div>
  );
}
