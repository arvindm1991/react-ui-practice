// A smooth-ish ridge line for one mountain layer, the same every time.
function ridge(seed: number, base: number, amp: number): string {
  const pts: string[] = ['0,800'];
  for (let x = 0; x <= 1200; x += 60) {
    const y = base - amp * (0.55 + 0.45 * Math.sin(x / 140 + seed) * Math.cos(x / 310 + seed * 1.7));
    pts.push(x + ',' + Math.round(y));
  }
  pts.push('1200,800');
  return pts.join(' ');
}

function landscape(p: Palette, seed: number): string {
  const svg =
    '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 1200 800">' +
    '<defs><linearGradient id="s" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stop-color="' + p[0] + '"/><stop offset="1" stop-color="' + p[1] + '"/></linearGradient></defs>' +
    '<rect width="1200" height="800" fill="url(#s)"/>' +
    '<circle cx="' + (300 + (seed * 137) % 600) + '" cy="300" r="70" fill="' + p[2] + '" opacity="0.9"/>' +
    '<polygon points="' + ridge(seed, 470, 180) + '" fill="' + p[3] + '"/>' +
    '<polygon points="' + ridge(seed + 2, 580, 150) + '" fill="' + p[4] + '"/>' +
    '<polygon points="' + ridge(seed + 4, 700, 110) + '" fill="' + p[5] + '"/>' +
    '</svg>';
  return 'data:image/svg+xml;charset=utf-8,' + encodeURIComponent(svg);
}

const PHOTOS: Photo[] = SHOTS.map((s, i) => {
  const url = landscape(s.palette, i + 1);
  return { id: 'p' + (i + 1), title: s.title, location: s.location, src: url, thumb: url };
});

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /albums/iceland-2025/photos -> Photo[]
export async function getPhotos(): Promise<Photo[]> {
  await randomDelay(400, 1000);
  return PHOTOS.map((p) => ({ ...p }));
}

// Wrap an index into [0, length): wrapIndex(12, 12) -> 0, wrapIndex(-1, 12) -> 11
export function wrapIndex(i: number, length: number): number {
  return ((i % length) + length) % length;
}

import React, { useEffect, useRef, useState } from 'react';
import { getPhotos } from './api';
import type { Photo } from './api';
import { wrapIndex } from './utils/wrap';

const SWIPE_THRESHOLD = 60; // px

type ThumbnailProps = {
  photo: Photo;
  onOpen: () => void;
  buttonRef: (el: HTMLButtonElement | null) => void;
};

function Thumbnail({ photo, onOpen, buttonRef }: ThumbnailProps) {
  return (
    <button type="button" className="thumb" aria-label={'Open ' + photo.title} ref={buttonRef} onClick={() => { onOpen(); }}>
      <img src={photo.thumb} alt="" loading="lazy" />
      <span className="thumb-caption">{photo.title}</span>
    </button>
  );
}

function Header({ count }: { count: number }) {
  return (
    <header className="header">
      <p className="kicker">Atlas · Field notes</p>
      <h1>Iceland, late summer</h1>
      <p className="meta">{count} photographs · Ring Road, 14 days</p>
    </header>
  );
}

type LightboxProps = {
  photos: Photo[];
  index: number;
  onChange: (index: number) => void;
  onClose: () => void;
};

function Lightbox({ photos, index, onChange, onClose }: LightboxProps) {
  const dialogRef = useRef<HTMLDivElement>(null);
  const closeRef = useRef<HTMLButtonElement>(null);
  const dragStartRef = useRef<number | null>(null);
  const [dragX, setDragX] = useState(0);

  const photo = photos[index];
  const total = photos.length;

  // Move focus into the dialog when it opens.
  useEffect(() => {
    if (closeRef.current) {
      closeRef.current.focus();
    }
  }, []);

  // Preload neighbours so ←/→ feel instant.
  useEffect(() => {
    [wrapIndex(index + 1, total), wrapIndex(index - 1, total)].forEach((i) => {
      const img = new Image();
      img.src = photos[i].src;
    });
  }, [index, photos, total]);

  function next() {
    onChange(wrapIndex(index + 1, total));
  }

  function prev() {
    onChange(wrapIndex(index - 1, total));
  }

  function handleKeyDown(e: React.KeyboardEvent<HTMLDivElement>) {
    if (e.key === 'ArrowRight') {
      next();
    } else if (e.key === 'ArrowLeft') {
      prev();
    } else if (e.key === 'Escape') {
      onClose();
    } else if (e.key === 'Tab' && dialogRef.current) {
      // Focus trap: wrap from the last control to the first and back.
      const els = dialogRef.current.querySelectorAll<HTMLElement>('button');
      const first = els[0];
      const last = els[els.length - 1];
      if (e.shiftKey && (e.target === first || e.target === dialogRef.current)) {
        e.preventDefault();
        last.focus();
      } else if (!e.shiftKey && e.target === last) {
        e.preventDefault();
        first.focus();
      }
    }
  }

  // ---- change: drag to swipe ----
  function handlePointerDown(e: React.PointerEvent<HTMLDivElement>) {
    if (e.button !== 0) {
      return;
    }
    e.currentTarget.setPointerCapture(e.pointerId);
    dragStartRef.current = e.clientX;
  }

  function handlePointerMove(e: React.PointerEvent<HTMLDivElement>) {
    if (dragStartRef.current === null) {
      return;
    }
    setDragX(e.clientX - dragStartRef.current);
  }

  function handlePointerUp(e: React.PointerEvent<HTMLDivElement>) {
    if (dragStartRef.current === null) {
      return;
    }
    const dx = e.clientX - dragStartRef.current;
    dragStartRef.current = null;
    setDragX(0);
    if (dx <= -SWIPE_THRESHOLD) {
      next();
    } else if (dx >= SWIPE_THRESHOLD) {
      prev();
    }
  }

  let imgCls = 'lightbox-img';
  if (dragX !== 0) {
    imgCls = 'lightbox-img is-dragging';
  }

  return (
    <div
      className="lightbox"
      role="dialog"
      aria-modal="true"
      aria-label={photo.title}
      tabIndex={-1}
      ref={dialogRef}
      onKeyDown={handleKeyDown}
    >
      <div className="lightbox-top">
        <span className="lightbox-count" aria-live="polite">{index + 1} of {total}</span>
        <button type="button" className="lb-btn lb-close" aria-label="Close" ref={closeRef} onClick={() => { onClose(); }}>✕</button>
      </div>
      <div
        className="lightbox-stage"
        onPointerDown={handlePointerDown}
        onPointerMove={handlePointerMove}
        onPointerUp={handlePointerUp}
        onPointerCancel={handlePointerUp}
      >
        <button type="button" className="lb-btn lb-nav lb-prev" aria-label="Previous photo" onClick={() => { prev(); }} onPointerDown={(e) => { e.stopPropagation(); }}>←</button>
        <img
          key={photo.id}
          className={imgCls}
          src={photo.src}
          alt={photo.title}
          draggable={false}
          style={{ transform: 'translateX(' + dragX + 'px)' }}
        />
        <button type="button" className="lb-btn lb-nav lb-next" aria-label="Next photo" onClick={() => { next(); }} onPointerDown={(e) => { e.stopPropagation(); }}>→</button>
      </div>
      <div className="lightbox-caption">
        <h2>{photo.title}</h2>
        <p>{photo.location}</p>
      </div>
    </div>
  );
}

export default function App() {
  const [photos, setPhotos] = useState<Photo[]>([]);
  const [loading, setLoading] = useState(true);
  const [openIndex, setOpenIndex] = useState<number | null>(null);
  // One DOM node per thumbnail, so focus can go back to the right one.
  const thumbRefs = useRef<(HTMLButtonElement | null)[]>([]);

  useEffect(() => {
    let cancelled = false;
    getPhotos().then((data) => {
      if (cancelled) {
        return;
      }
      setPhotos(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // Return focus to the thumbnail of the photo you were looking at.
  function handleClose() {
    const idx = openIndex;
    setOpenIndex(null);
    if (idx !== null) {
      const el = thumbRefs.current[idx];
      if (el) {
        el.focus();
      }
    }
  }

  let grid;
  if (loading) {
    grid = <p className="loading">Developing film…</p>;
  } else {
    grid = (
      <div className="grid">
        {photos.map((p, i) => {
          return (
            <Thumbnail
              key={p.id}
              photo={p}
              onOpen={() => { setOpenIndex(i); }}
              buttonRef={(el) => { thumbRefs.current[i] = el; }}
            />
          );
        })}
      </div>
    );
  }

  let lightbox = null;
  if (openIndex !== null) {
    lightbox = <Lightbox photos={photos} index={openIndex} onChange={setOpenIndex} onClose={handleClose} />;
  }

  return (
    <div className="app">
      <Header count={photos.length} />
      {grid}
      {lightbox}
    </div>
  );
}
