// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /events?year=2026&month=8 -> SchoolEvent[] for that month (month is 0-based, like Date)
export async function getEvents(year: number, month: number): Promise<SchoolEvent[]> {
  await randomDelay(400, 900);
  const prefix = year + '-' + String(month + 1).padStart(2, '0') + '-';
  return EVENTS.filter((e) => e.date.startsWith(prefix)).map((e) => ({ ...e }));
}

export const MONTH_NAMES = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October', 'November', 'December'];
export const WEEKDAYS = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
const LONG_WEEKDAYS = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];

// daysInMonth(2026, 8) -> 30 (September; month is 0-based)
export function daysInMonth(year: number, month: number): number {
  return new Date(year, month + 1, 0).getDate();
}

// firstWeekday(2026, 8) -> 2 (Sept 1, 2026 is a Tuesday; 0 = Sunday)
export function firstWeekday(year: number, month: number): number {
  return new Date(year, month, 1).getDay();
}

// toKey(2026, 8, 5) -> "2026-09-05" (same format as SchoolEvent.date)
export function toKey(year: number, month: number, day: number): string {
  return year + '-' + String(month + 1).padStart(2, '0') + '-' + String(day).padStart(2, '0');
}

// formatLongDate("2026-09-15") -> "Tuesday, September 15"
export function formatLongDate(key: string): string {
  const parts = key.split('-').map(Number);
  const d = new Date(parts[0], parts[1] - 1, parts[2]);
  return LONG_WEEKDAYS[d.getDay()] + ', ' + MONTH_NAMES[d.getMonth()] + ' ' + d.getDate();
}

import { getEvents } from './api';

import { MONTH_NAMES, WEEKDAYS, daysInMonth, firstWeekday, formatLongDate, toKey } from './utils/date';

// The app's "today". Passed down as a prop so the calendar never reads the real clock.
const TODAY = new Date(2026, 8, 15);

function SchoolHeader() {
  return (

type DayPanelProps = {
  dateKey: string;
  events: SchoolEvent[];
  onClose: () => void;
};

// Change: the side panel for the clicked day.
function DayPanel({ dateKey, events, onClose }: DayPanelProps) {
  let body = <p className="muted">No events this day.</p>;
  if (events.length > 0) {
    body = (
      <ul className="panel-list">
        {events.map((e) => {
          let time = 'All day';
          if (e.time) {
            time = e.time;
          }
          return (
            <li key={e.id} className="panel-item">
              <span className="panel-time">{time}</span>
              <EventChip event={e} />
            </li>
          );
        })}
      </ul>
    );
  }
  return (
    <aside className="day-panel">
      <div className="panel-head">
        <h2>{formatLongDate(dateKey)}</h2>
        <button type="button" className="panel-close" aria-label="Close" onClick={() => { onClose(); }}>✕</button>
      </div>
      {body}
    </aside>
  );
}

const MAX_CHIPS = 2;

function SchoolCalendar({ today }: { today: Date }) {
  // The visible month. Starts at "today" but moves independently of it.
  const [view, setView] = useState({ year: today.getFullYear(), month: today.getMonth() });
  const [events, setEvents] = useState<SchoolEvent[]>([]);
  const [loading, setLoading] = useState(true);
  const [selectedKey, setSelectedKey] = useState<string | null>(null);

  // Refetch whenever the visible month changes. The guard drops a slow
  // response for a month you've already clicked away from.
  useEffect(() => {
    let cancelled = false;
    setLoading(true);
    getEvents(view.year, view.month).then((data) => {
      if (cancelled) {
        return;
      }
      setEvents(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, [view.year, view.month]);

  function handleMonth(delta: number) {
    setView((prev) => {
      // new Date handles the year rollover (month 12 -> January next year).
      const d = new Date(prev.year, prev.month + delta, 1);
      return { year: d.getFullYear(), month: d.getMonth() };
    });
    setSelectedKey(null);
  }

  // derived: the cells of the grid. null = blank cell before the 1st / after the last day.
  const cells: (number | null)[] = [];
  for (let i = 0; i < firstWeekday(view.year, view.month); i++) {
    cells.push(null);
  }
  for (let d = 1; d <= daysInMonth(view.year, view.month); d++) {
    cells.push(d);
  }
  while (cells.length % 7 !== 0) {
    cells.push(null);
  }

  const todayKey = toKey(today.getFullYear(), today.getMonth(), today.getDate());

  function eventsOn(key: string): SchoolEvent[] {
    return events.filter((e) => {
      return e.date === key;
    });
  }

  let status = null;
  if (loading) {
    status = <span className="loading-dot">Loading…</span>;
  }

  let panel = null;
  let layoutCls = 'cal-layout';
  if (selectedKey) {
    layoutCls = 'cal-layout has-panel';
    panel = <DayPanel dateKey={selectedKey} events={eventsOn(selectedKey)} onClose={() => { setSelectedKey(null); }} />;
  }

  return (
    <section className="card">
      <div className="card-head">
        <h1 className="cal-title">{MONTH_NAMES[view.month]} {view.year}</h1>
        {status}
        <div className="cal-nav">
          <button type="button" className="nav-btn" aria-label="Previous month" onClick={() => { handleMonth(-1); }}>‹</button>
          <button type="button" className="nav-btn" aria-label="Next month" onClick={() => { handleMonth(1); }}>›</button>
        </div>
      </div>
      <div className={layoutCls}>
        <div className="cal-grid">
          {WEEKDAYS.map((w) => {
            return <div key={w} className="cal-weekday">{w}</div>;
          })}
          {cells.map((day, i) => {
            if (day === null) {
              return <div key={'blank-' + i} className="cal-cell is-blank" />;
            }
            const key = toKey(view.year, view.month, day);
            const dayEvents = eventsOn(key);
            let cls = 'cal-cell is-day';
            if (key === todayKey) {
              cls = cls + ' is-today';
            }
            if (key === selectedKey) {
              cls = cls + ' is-selected';
            }
            let more = null;
            if (dayEvents.length > MAX_CHIPS) {
              more = <span className="more">+{dayEvents.length - MAX_CHIPS} more</span>;
            }
            return (
              <button key={key} type="button" className={cls} onClick={() => { setSelectedKey(key); }}>
                <span className="day-num">{day}</span>
                {dayEvents.slice(0, MAX_CHIPS).map((e) => {
                  return <EventChip key={e.id} event={e} />;
                })}
                {more}
              </button>
            );
          })}
        </div>
        {panel}
      </div>
      <Legend />
    </section>
  );
}

export default function App() {
  return (
    <div className="app">
      <SchoolHeader />
      <main className="page">
        <SchoolCalendar today={TODAY} />
      </main>
    </div>
  );
}
