export type Availability = {
  cabinId: string;
  pricePerNight: number;
  unavailable: string[]; // ISO dates (nights) already booked
};

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /cabins/:id/availability
export async function getAvailability(cabinId: string): Promise<Availability> {
  await randomDelay(400, 1000);
  return {
    cabinId,
    pricePerNight: 185,
    unavailable: ['2026-03-12', '2026-03-13', '2026-03-20', '2026-03-21', '2026-03-22', '2026-04-03', '2026-04-04', '2026-04-17', '2026-05-08', '2026-05-09'],
  };
}

// "Today" is fixed so the calendar is the same every time you open it.
// Never use new Date() / Date.now() for "today" in this app: use TODAY.
export const TODAY = new Date(2026, 2, 3); // Tue, Mar 3 2026 (months are 0-based)

const MONTHS = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October', 'November', 'December'];

// Local date -> "2026-03-04". ISO strings compare correctly as plain strings: "2026-03-04" < "2026-03-10".
export function toISO(d: Date): string {
  const m = String(d.getMonth() + 1).padStart(2, '0');
  const day = String(d.getDate()).padStart(2, '0');
  return d.getFullYear() + '-' + m + '-' + day;
}

// "2026-03-04" -> local Date (midnight)
export function fromISO(iso: string): Date {
  const [y, m, d] = iso.split('-').map(Number);
  return new Date(y, m - 1, d);
}

export function addDays(d: Date, n: number): Date {
  return new Date(d.getFullYear(), d.getMonth(), d.getDate() + n);
}

// Always returns the 1st of the resulting month.
export function addMonths(d: Date, n: number): Date {
  return new Date(d.getFullYear(), d.getMonth() + n, 1);
}

export function startOfMonth(d: Date): Date {
  return new Date(d.getFullYear(), d.getMonth(), 1);
}

// "March 2026"
export function monthLabel(d: Date): string {
  return MONTHS[d.getMonth()] + ' ' + d.getFullYear();
}

// "2026-03-04" -> "Mar 4"
export function formatShort(iso: string): string {
  const d = fromISO(iso);
  return MONTHS[d.getMonth()].slice(0, 3) + ' ' + d.getDate();
}

// nightsBetween("2026-03-04", "2026-03-09") -> 5
export function nightsBetween(startIso: string, endIso: string): number {
  return Math.round((fromISO(endIso).getTime() - fromISO(startIso).getTime()) / 86400000);
}

// The cells of a month view, weeks starting on Sunday.
// Leading blanks are null: e.g. March 2026 starts on a Sunday, April 2026 -> [null, null, null, Apr 1, ...]
export function getMonthGrid(month: Date): (Date | null)[] {
  const first = startOfMonth(month);
  const cells: (Date | null)[] = [];
  for (let i = 0; i < first.getDay(); i++) {
    cells.push(null);
  }
  const days = new Date(first.getFullYear(), first.getMonth() + 1, 0).getDate();
  for (let d = 1; d <= days; d++) {
    cells.push(new Date(first.getFullYear(), first.getMonth(), d));
  }
  return cells;
}

import React, { useEffect, useState } from 'react';
import { getAvailability } from './api';
import type { Availability } from './api';
import { TODAY, toISO, fromISO, addDays, addMonths, startOfMonth, getMonthGrid, monthLabel, formatShort, nightsBetween } from './utils/date';
import { formatPrice } from './utils/money';

const WEEKDAYS = ['Su', 'Mo', 'Tu', 'We', 'Th', 'Fr', 'Sa'];
const TODAY_ISO = toISO(TODAY);
// CHANGE
const MIN_NIGHTS = 2;

// CORE: everything a day button needs, computed by App (NEXT: isPreviewEnd)
type DayInfo = {
  disabled: boolean;
  unavailable: boolean;
  isStart: boolean;
  isEnd: boolean;
  inRange: boolean;
  isPreviewEnd: boolean;
};

// CORE: MonthView stays dumb: it asks getInfo(iso) and reports clicks (NEXT: hover)
type MonthViewProps = {
  month: Date;
  getInfo: (iso: string) => DayInfo;
  onDayClick: (iso: string) => void;
  onDayHover: (iso: string | null) => void;
};

function MonthView({ month, getInfo, onDayClick, onDayHover }: MonthViewProps) {
  return (
    <div className="month">
      <h3 className="month-title">{monthLabel(month)}</h3>
      <div className="weekdays">
        {WEEKDAYS.map((w) => {
          return <span key={w}>{w}</span>;
        })}
      </div>
      <div className="days">
        {getMonthGrid(month).map((d, i) => {
          if (!d) {
            return <span key={'blank-' + i} />;
          }
          const iso = toISO(d);
          // CORE: class list from the info
          const info = getInfo(iso);
          let cls = 'day';
          if (info.unavailable) {
            cls += ' is-unavailable';
          }
          if (info.inRange) {
            cls += ' in-range';
          }
          if (info.isStart) {
            cls += ' is-start';
          }
          if (info.isEnd || info.isPreviewEnd) {
            cls += ' is-end';
          }
          if (iso === TODAY_ISO) {
            cls += ' is-today';
          }
          return (
            <button
              key={iso}
              type="button"
              className={cls}
              data-date={iso}
              disabled={info.disabled}
              aria-pressed={info.isStart || info.isEnd}
              aria-label={d.toDateString()}
              onClick={() => { onDayClick(iso); }}
              onMouseEnter={() => { onDayHover(iso); }}
            >
              {d.getDate()}
            </button>
          );
        })}
      </div>
    </div>
  );
}

export default function App() {
  const [availability, setAvailability] = useState<Availability | null>(null);
  // CORE: the left month, and the selection as ISO strings
  const [viewMonth, setViewMonth] = useState<Date>(startOfMonth(TODAY));
  const [start, setStart] = useState<string | null>(null);
  const [end, setEnd] = useState<string | null>(null);
  // NEXT: hovered day, for the preview
  const [hover, setHover] = useState<string | null>(null);

  useEffect(() => {
    let cancelled = false;
    getAvailability('alder').then((data) => {
      if (cancelled) {
        return;
      }
      setAvailability(data);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // CORE: derived
  const unavailable = new Set<string>();
  if (availability) {
    availability.unavailable.forEach((u) => {
      unavailable.add(u);
    });
  }
  const pickingEnd = start !== null && end === null;
  // NEXT: while picking checkout, nothing on or after the next booked night can be chosen.
  let limit: string | null = null;
  if (pickingEnd && start) {
    const s = start;
    const later = Array.from(unavailable).filter((u) => {
      return u > s;
    }).sort();
    if (later.length > 0) {
      limit = later[0];
    }
  }

  function isDisabled(iso: string): boolean {
    // CORE: past or booked
    if (iso < TODAY_ISO || unavailable.has(iso)) {
      return true;
    }
    if (pickingEnd && start && iso > start) {
      // NEXT: never cross a booked night
      if (limit !== null && iso >= limit) {
        return true;
      }
      // CHANGE: a checkout must be at least MIN_NIGHTS after check-in
      return nightsBetween(start, iso) < MIN_NIGHTS;
    }
    // CHANGE: as a check-in, the earliest possible checkout (MIN_NIGHTS later) must still be reachable:
    // none of the next MIN_NIGHTS days can be booked (same rule as the limit above).
    for (let i = 1; i <= MIN_NIGHTS; i++) {
      if (unavailable.has(toISO(addDays(fromISO(iso), i)))) {
        return true;
      }
    }
    // CORE
    return false;
  }

  // NEXT: the range end to draw: the real end, or the hovered day while picking.
  let rangeEnd = end;
  let previewing = false;
  if (pickingEnd && hover && start && hover > start && !isDisabled(hover)) {
    rangeEnd = hover;
    previewing = true;
  }

  // CORE
  function getInfo(iso: string): DayInfo {
    return {
      disabled: isDisabled(iso),
      unavailable: unavailable.has(iso),
      isStart: iso === start,
      isEnd: iso === end,
      inRange: start !== null && rangeEnd !== null && iso > start && iso < rangeEnd,
      isPreviewEnd: previewing && iso === rangeEnd,
    };
  }

  // CORE: click rules, in order
  function handleDayClick(iso: string) {
    if (isDisabled(iso)) {
      return;
    }
    if (!start || end) {
      setStart(iso); // first click, or starting over
      setEnd(null);
    } else if (iso <= start) {
      setStart(iso); // earlier day: move the check-in instead
    } else {
      setEnd(iso);
    }
  }

  // NEXT
  function handleClear() {
    setStart(null);
    setEnd(null);
    setHover(null);
  }

  // NEXT: derived summary
  let summary = 'Add your check-in date';
  let total = null;
  if (start && end && availability) {
    const nights = nightsBetween(start, end);
    let noun = 'nights';
    if (nights === 1) {
      noun = 'night';
    }
    summary = formatShort(start) + ' → ' + formatShort(end) + ' · ' + nights + ' ' + noun;
    total = <span className="total">{formatPrice(nights * availability.pricePerNight)} total</span>;
  } else if (start) {
    summary = formatShort(start) + ' → Add checkout';
    // CHANGE: say why the next night is greyed out
    summary += ' · ' + MIN_NIGHTS + '-night minimum';
  }

  let clearBtn = null;
  if (start) {
    clearBtn = <button type="button" className="clear-btn" onClick={() => { handleClear(); }}>Clear dates</button>;
  }

  // NEXT: can't page before the current month
  const canGoBack = viewMonth > startOfMonth(TODAY);

  // CORE
  let calendar;
  if (!availability) {
    calendar = <p className="loading">Checking availability…</p>;
  } else {
    calendar = (
      <div className="calendar" onMouseLeave={() => { setHover(null); }}>
        {/* NEXT: month paging */}
        <button type="button" className="nav-btn nav-prev" aria-label="Previous month" disabled={!canGoBack} onClick={() => { setViewMonth((m) => { return addMonths(m, -1); }); }}>‹</button>
        <button type="button" className="nav-btn nav-next" aria-label="Next month" onClick={() => { setViewMonth((m) => { return addMonths(m, 1); }); }}>›</button>
        {/* CORE: the same component twice */}
        <MonthView month={viewMonth} getInfo={getInfo} onDayClick={handleDayClick} onDayHover={setHover} />
        <MonthView month={addMonths(viewMonth, 1)} getInfo={getInfo} onDayClick={handleDayClick} onDayHover={setHover} />
      </div>
    );
  }

  let price = null;
  if (availability) {
    price = <span className="price"><b>{formatPrice(availability.pricePerNight)}</b> / night</span>;
  }

  return (
    <div className="page">
      <CabinHero />
      <section className="booking">
        <div className="booking-head">
          <h2>Select dates</h2>
          {price}
        </div>
        {calendar}
        {/* STRETCH: legend */}
        <div className="legend">
          <span className="legend-item"><span className="swatch swatch-booked" />Booked</span>
          <span className="legend-item"><span className="swatch swatch-range" />Your stay</span>
        </div>
        {/* NEXT: summary bar */}
        <div className="summary-bar">
          <div className="summary-text">
            <span className="summary" aria-live="polite">{summary}</span>
            {total}
          </div>
          {clearBtn}
          <button type="button" className="reserve-btn" disabled={!end}>Reserve</button>
        </div>
      </section>
    </div>
  );
}
