const PAGE_SIZE = 5;

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /activity?page=1 -> { items, nextPage }
// Pages start at 1. Each page has up to 5 items. nextPage is null on the last page.
export async function getActivity(page: number): Promise<ActivityPage> {
  await randomDelay(500, 1200);
  const start = (page - 1) * PAGE_SIZE;
  const items = ACTIVITY.slice(start, start + PAGE_SIZE).map((a) => ({ ...a }));
  let nextPage: number | null = null;
  if (start + PAGE_SIZE < ACTIVITY.length) {
    nextPage = page + 1;
  }
  return { items, nextPage };
}

// timeAgo(3) -> "3m ago", timeAgo(95) -> "1h ago", timeAgo(2900) -> "2d ago"
export function timeAgo(minutes: number): string {
  if (minutes < 1) {
    return 'just now';
  }
  if (minutes < 60) {
    return minutes + 'm ago';
  }
  if (minutes < 60 * 24) {
    return Math.floor(minutes / 60) + 'h ago';
  }
  return Math.floor(minutes / (60 * 24)) + 'd ago';
}

import React, { useEffect, useState } from 'react';
import { getActivity } from './api';
import type { Activity } from './api';
import ActivityItem from './components/ActivityItem';

const NAV = ['Overview', 'Deployments', 'Services', 'Incidents', 'Settings'];

function Sidebar() {
  return (
    <aside className="sidebar">
      <div className="brand"><span className="brand-mark">S</span> Shipyard</div>
      <nav className="nav">
        {NAV.map((label) => {
          let cls = 'nav-link';
          if (label === 'Overview') {
            cls = 'nav-link is-active';
          }
          return <a key={label} className={cls} href="#">{label}</a>;
        })}
      </nav>
    </aside>
  );
}

function StatCard({ label, value, note }: { label: string; value: string; note: string }) {
  return (
    <div className="stat">
      <span className="stat-label">{label}</span>
      <span className="stat-value">{value}</span>
      <span className="stat-note">{note}</span>
    </div>
  );
}

function ActivityFeed() {
  const [items, setItems] = useState<Activity[]>([]);
  // The API tells us the next page. null = no more pages.
  const [nextPage, setNextPage] = useState<number | null>(null);
  const [loading, setLoading] = useState(true);

  // First page on mount.
  useEffect(() => {
    let cancelled = false;
    getActivity(1).then((page) => {
      if (cancelled) {
        return;
      }
      setItems(page.items);
      setNextPage(page.nextPage);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  function handleLoadMore() {
    // Guard: a second click while loading must not fetch the same page twice.
    if (loading || nextPage === null) {
      return;
    }
    setLoading(true);
    getActivity(nextPage).then((page) => {
      // Append with the updater form: never replace what's already shown.
      setItems((prev) => {
        return [...prev, ...page.items];
      });
      setNextPage(page.nextPage);
      setLoading(false);
    });
  }

  // decisions
  let body;
  if (items.length === 0 && loading) {
    body = <p className="feed-message">Loading activity…</p>;
  } else {
    body = (
      <ul className="activity-list">
        {items.map((item) => {
          return <ActivityItem key={item.id} item={item} />;
        })}
      </ul>
    );
  }

  let footer = null;
  if (items.length > 0 && nextPage !== null) {
    let label = 'Load more';
    if (loading) {
      label = 'Loading…';
    }
    footer = (
      <div className="feed-footer">
        <button type="button" className="load-more" disabled={loading} onClick={() => { handleLoadMore(); }}>
          {label}
        </button>
      </div>
    );
  } else if (items.length > 0) {
    footer = <p className="feed-end">You're all caught up · {items.length} events</p>;
  }

  return (
    <section className="card">
      <div className="card-head">
        <h2>Recent activity</h2>
        <span className="card-sub">All services</span>
      </div>
      {body}
      {footer}
    </section>
  );
}

export default function App() {
  return (
    <div className="layout">
      <Sidebar />
      <main className="main">
        <header className="page-head">
          <h1>Overview</h1>
          <span className="env-pill">production</span>
        </header>
        <div className="stats">
          <StatCard label="Deploys today" value="12" note="+3 vs yesterday" />
          <StatCard label="Uptime (30d)" value="99.97%" note="SLO 99.9%" />
          <StatCard label="Open incidents" value="1" note="checkout-svc" />
        </div>
        <ActivityFeed />
      </main>
    </div>
  );
}
