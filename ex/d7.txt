import React, { useEffect, useId } from 'react';
import { createPortal } from 'react-dom';

export default function Modal({ open, title, children, onClose }) {
  const titleId = useId();

  useEffect(() => {
    if (!open) return undefined;
    function onKey(e) {
      if (e.key === 'Escape') onClose();
    }
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey); // cleanup!
  }, [open, onClose]);

  if (!open) return null;

  return createPortal(
    // Close only when the click started on the backdrop itself
    <div className="backdrop" onMouseDown={(e) => e.target === e.currentTarget && onClose()}>
      <div className="dialog" role="dialog" aria-modal="true" aria-labelledby={titleId}>
        <h2 id={titleId}>{title}</h2>
        {children}
      </div>
    </div>,
    document.body
  );
}

import React, { useEffect, useRef, useState } from 'react';
import Modal from './components/Modal';

const INITIAL = [
  { id: 1, name: 'Billing dashboard' },
  { id: 2, name: 'Docs search' },
  { id: 3, name: 'Prompt library' },
];

export default function App() {
  const [projects, setProjects] = useState(INITIAL);
  const [pendingId, setPendingId] = useState(null); // which project we're confirming
  const cancelRef = useRef(null);

  const pending = projects.find((p) => p.id === pendingId);
  const close = () => setPendingId(null);

  // Move focus into the dialog when it opens
  useEffect(() => {
    if (pending && cancelRef.current) cancelRef.current.focus();
  }, [pending]);

  function confirmDelete() {
    setProjects((prev) => prev.filter((p) => p.id !== pendingId));
    close();
  }

  return (
    <div className="page">
      <h1>Projects</h1>
      {projects.length === 0 ? (
        <div className="empty">No projects left.</div>
      ) : (
        <ul className="list">
          {projects.map((p) => (
            <li key={p.id} className="row">
              <span>{p.name}</span>
              <button className="btn" onClick={() => setPendingId(p.id)}>Delete</button>
            </li>
          ))}
        </ul>
      )}

      <Modal open={Boolean(pending)} title="Delete project?" onClose={close}>
        <p>
          <b>{pending && pending.name}</b> and its history will be removed. This can't be undone.
        </p>
        <div className="dialog-actions">
          <button className="btn" ref={cancelRef} onClick={close}>Cancel</button>
          <button className="btn btn-danger" onClick={confirmDelete}>Delete project</button>
        </div>
      </Modal>
    </div>
  );
}
