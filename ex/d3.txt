@@@ solution src/App.jsx
import React, { useState } from 'react';
import { people } from './data';

const TEAMS = ['All', 'Research', 'Platform', 'Applied'];

export default function App() {
  // State holds only what the user controls.
  const [query, setQuery] = useState('');
  const [team, setTeam] = useState('All');

  // Derived data: computed on every render, never stored in state.
  const q = query.trim().toLowerCase();
  const visible = people.filter((p) => {
    const matchesQuery =
      q === '' ||
      p.name.toLowerCase().includes(q) ||
      p.email.toLowerCase().includes(q);
    const matchesTeam = team === 'All' || p.team === team;
    return matchesQuery && matchesTeam;
  });

  function clearFilters() {
    setQuery('');
    setTeam('All');
  }

  return (
    <div className="page">
      <h1>Directory</h1>

      <div className="controls">
        <input
          className="search"
          placeholder="Search by name or email"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
        />
        <select className="select" value={team} onChange={(e) => setTeam(e.target.value)}>
          {TEAMS.map((t) => (
            <option key={t} value={t}>
              {t === 'All' ? 'All teams' : t}
            </option>
          ))}
        </select>
      </div>

      <p className="count">
        Showing {visible.length} of {people.length}
      </p>

      {visible.length === 0 ? (
        <div className="empty">
          <p>No people match your filters.</p>
          <button onClick={clearFilters}>Clear filters</button>
        </div>
      ) : (
        <ul className="list">
          {visible.map((p) => (
            <li key={p.id} className="row">
              <div className="name">{p.name}</div>
              <div className="email">{p.email}</div>
              <span className="team">{p.team}</span>
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}
