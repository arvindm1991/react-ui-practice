const BUDGET: Budget = {
  name: 'Q3 operating budget',
  months: ['Jul', 'Aug', 'Sep'],
  rows: [
    { id: 'r1', category: 'Rent', values: [4200, 4200, 4200] },

    { id: 'r3', category: 'Software', values: [1240, 1310, 1310] },

    { id: 'r5', category: 'Travel', values: [800, 0, 1650] },
  ],

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /budgets/q3 -> Budget
export async function getBudget(): Promise<Budget> {
  await randomDelay(400, 1000);
  return {
    name: BUDGET.name,
    months: [...BUDGET.months],
    rows: BUDGET.rows.map((r) => ({ ...r, values: [...r.values] })),
  };
}

// formatMoney(18500) -> "$18,500", formatMoney(-40) -> "-$40"
export function formatMoney(n: number): string {
  const abs = Math.abs(n).toLocaleString('en-US', { maximumFractionDigits: 2 });
  if (n < 0) {
    return '-$' + abs;
  }
  return '$' + abs;
}

// parseAmount("1,200") -> 1200, parseAmount("$50") -> 50, parseAmount("") -> 0,
// parseAmount("abc") -> null (not a number)
export function parseAmount(text: string): number | null {
  const cleaned = text.replace(/[$,\s]/g, '');
  if (cleaned === '') {
    return 0;
  }
  const n = Number(cleaned);
  if (!Number.isFinite(n)) {
    return null;
  }
  return n;
}

import React, { useEffect, useState } from 'react';
import { getBudget } from './api';
import { formatMoney, parseAmount } from './utils/money';
import { uid } from './utils/id';

// Cells are stored as the text the user typed, so an invalid entry can stay
// visible (and highlighted) instead of being silently thrown away.
type GridRow = { id: string; category: string; cells: string[] };
type CellPos = { row: number; col: number };

function AppHeader({ name }: { name: string }) {
  return (
    <header className="app-header">
      <div className="logo">ledgerly</div>
      <div className="crumbs">Budgets <span>/</span> <b>{name}</b></div>
      <span className="badge">Draft</span>
    </header>
  );
}

// Sum of the valid numbers in a list of cell strings (invalid cells count as 0).
function sumCells(cells: string[]): number {
  return cells.reduce((sum, text) => {
    const n = parseAmount(text);
    if (n === null) {
      return sum;
    }
    return sum + n;
  }, 0);
}

type BudgetTableProps = {
  months: string[];
  rows: GridRow[];
  onCellChange: (row: number, col: number, text: string) => void;
  onDeleteRow: (id: string) => void;
};

function BudgetTable({ months, rows, onCellChange, onDeleteRow }: BudgetTableProps) {
  const [editing, setEditing] = useState<CellPos | null>(null);
  const [draft, setDraft] = useState('');

  function startEdit(row: number, col: number) {
    setEditing({ row: row, col: col });
    setDraft(rows[row].cells[col]);
  }

  // Save the draft into the grid, then either move to `next` or stop editing.
  function commit(next: CellPos | null) {
    if (editing) {
      onCellChange(editing.row, editing.col, draft.trim());
    }
    if (next) {
      startEdit(next.row, next.col);
    } else {
      setEditing(null);
    }
  }

  function handleKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
    if (!editing) {
      return;
    }
    const last = months.length - 1;
    if (e.key === 'Enter') {
      e.preventDefault();
      let next: CellPos | null = null;
      if (editing.row + 1 < rows.length) {
        next = { row: editing.row + 1, col: editing.col };
      }
      commit(next);
    } else if (e.key === 'Tab') {
      e.preventDefault(); // keep focus inside the grid
      let row = editing.row;
      let col = editing.col;
      if (e.shiftKey) {
        col = col - 1;
        if (col < 0) {
          col = last;
          row = row - 1;
        }
      } else {
        col = col + 1;
        if (col > last) {
          col = 0;
          row = row + 1;
        }
      }
      let next: CellPos | null = null;
      if (row >= 0 && row < rows.length) {
        next = { row: row, col: col };
      }
      commit(next);
    } else if (e.key === 'Escape') {
      setEditing(null); // cancel: the draft is thrown away
    }
  }

  function handleBlur() {
    // A blur caused by Enter/Tab moving to another cell has already committed.
    if (editing) {
      commit(null);
    }
  }

  // derived: totals are computed every render, never stored
  const colTotals = months.map((m, col) => {
    return sumCells(rows.map((r) => {
      return r.cells[col];
    }));
  });
  const grandTotal = colTotals.reduce((sum, t) => {
    return sum + t;
  }, 0);

  let emptyRow = null;
  if (rows.length === 0) {
    emptyRow = (
      <tr>
        <td className="empty" colSpan={months.length + 3}>No categories. Add one below.</td>
      </tr>
    );
  }

  return (
    <table className="grid">
      <thead>
        <tr>
          <th className="col-category">Category</th>
          {months.map((m) => {
            return <th key={m} className="col-num">{m}</th>;
          })}
          <th className="col-num col-total">Total</th>
          <th className="col-actions"><span className="sr-only">Actions</span></th>
        </tr>
      </thead>
      <tbody>
        {rows.map((row, r) => {
          return (
            <tr key={row.id}>
              <th scope="row" className="col-category">{row.category}</th>
              {row.cells.map((text, c) => {
                const isEditing = editing !== null && editing.row === r && editing.col === c;
                if (isEditing) {
                  return (
                    <td key={c} className="cell is-editing">
                      <input
                        className="cell-input"
                        autoFocus
                        aria-label={row.category + ' ' + months[c]}
                        value={draft}
                        onChange={(e) => { setDraft(e.target.value); }}
                        onKeyDown={handleKeyDown}
                        onBlur={() => { handleBlur(); }}
                      />
                    </td>
                  );
                }
                const amount = parseAmount(text);
                let cls = 'cell';
                let shown = text;
                let title: string | undefined = undefined;
                if (amount === null) {
                  cls = 'cell is-invalid';
                  title = 'Not a number';
                } else {
                  shown = formatMoney(amount);
                }
                return (
                  <td key={c} className={cls} title={title}>
                    <button type="button" className="cell-btn" aria-label={row.category + ' ' + months[c] + ': ' + shown} onClick={() => { startEdit(r, c); }}>
                      {shown}
                    </button>
                  </td>
                );
              })}
              <td className="cell cell-total">{formatMoney(sumCells(row.cells))}</td>
              <td className="col-actions">
                <button type="button" className="row-delete" aria-label={'Delete ' + row.category} onClick={() => { setEditing(null); onDeleteRow(row.id); }}>×</button>
              </td>
            </tr>
          );
        })}
        {emptyRow}
      </tbody>
      <tfoot>
        <tr>
          <th scope="row" className="col-category">Total</th>
          {colTotals.map((t, i) => {
            return <td key={months[i]} className="cell cell-total">{formatMoney(t)}</td>;
          })}
          <td className="cell cell-grand">{formatMoney(grandTotal)}</td>
          <td className="col-actions" />
        </tr>
      </tfoot>
    </table>
  );
}

function AddRowForm({ onAdd }: { onAdd: (name: string) => void }) {
  const [name, setName] = useState('');

  function handleSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    if (name.trim() === '') {
      return;
    }
    onAdd(name.trim());
    setName('');
  }

  return (
    <form className="add-row" onSubmit={handleSubmit}>
      <input className="add-input" placeholder="New category" aria-label="New category name" value={name} onChange={(e) => { setName(e.target.value); }} />
      <button type="submit" className="add-btn" disabled={name.trim() === ''}>+ Add row</button>
    </form>
  );
}

export default function App() {
  const [name, setName] = useState('');
  const [months, setMonths] = useState<string[]>([]);
  const [rows, setRows] = useState<GridRow[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    getBudget().then((data) => {
      if (cancelled) {
        return;
      }
      setName(data.name);
      setMonths(data.months);
      setRows(data.rows.map((r) => {
        return { id: r.id, category: r.category, cells: r.values.map((v) => { return String(v); }) };
      }));
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  function handleCellChange(row: number, col: number, text: string) {
    setRows((prev) => {
      return prev.map((r, i) => {
        if (i !== row) {
          return r;
        }
        const cells = r.cells.map((t, j) => {
          if (j === col) {
            return text;
          }
          return t;
        });
        return { ...r, cells: cells };
      });
    });
  }

  function handleAddRow(category: string) {
    setRows((prev) => {
      return [...prev, { id: uid('row'), category: category, cells: months.map(() => { return ''; }) }];
    });
  }

  function handleDeleteRow(id: string) {
    setRows((prev) => {
      return prev.filter((r) => {
        return r.id !== id;
      });
    });
  }

  // derived
  let invalidCount = 0;
  rows.forEach((r) => {
    r.cells.forEach((t) => {
      if (parseAmount(t) === null) {
        invalidCount = invalidCount + 1;
      }
    });
  });

  let body;
  if (loading) {
    body = <p className="loading">Loading budget…</p>;
  } else {
    let warning = null;
    if (invalidCount > 0) {
      let noun = 'cells need';
      if (invalidCount === 1) {
        noun = 'cell needs';
      }
      warning = <p className="sheet-warning" role="alert">{invalidCount} {noun} a number. They are left out of the totals.</p>;
    }
    body = (
      <div>
        <BudgetTable months={months} rows={rows} onCellChange={handleCellChange} onDeleteRow={handleDeleteRow} />
        <div className="sheet-foot">
          <AddRowForm onAdd={handleAddRow} />
          {warning}
        </div>
      </div>
    );
  }

  return (
    <div className="app">
      <AppHeader name={name} />
      <main className="sheet">{body}</main>
    </div>
  );
}
