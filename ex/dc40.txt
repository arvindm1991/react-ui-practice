// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /workspace/users -> User[] (everyone you can @mention)

  return USERS.map((u) => ({ ...u }));

@@@ shared src/utils/mentions.ts
export type TextPart = { text: string; isMention: boolean };

// splitMentions("Thanks @Priya Shah!") ->
//   [{ text: "Thanks ", isMention: false }, { text: "@Priya Shah", isMention: true }, { text: "!", isMention: false }]
// A mention is "@" + one or two capitalized words.
export function splitMentions(text: string): TextPart[] {
  return text
    .split(/(@[A-Z][a-z]+(?: [A-Z][a-z]+)?)/)
    .filter((piece) => {
      return piece !== '';
    })
    .map((piece) => {
      return { text: piece, isMention: piece.startsWith('@') };
    });
}

import React, { useEffect, useRef, useState } from 'react';
import { getUsers } from '../api';
import type { User } from '../api';
import { Avatar } from './Avatar';

type CommentBoxProps = {
  onSubmit: (text: string) => void;
};

// CORE: where the "@fragment" being typed starts, and the letters after the "@".
type Mention = { start: number; query: string };

// Looks at the text BEFORE the caret: does it end in "@letters" (at the start or after a space)?
function findMention(value: string, caret: number): Mention | null {
  const before = value.slice(0, caret);
  const match = /(^|\s)@([a-zA-Z]*)$/.exec(before);
  if (!match) {
    return null;
  }
  return { start: caret - match[2].length - 1, query: match[2] };
}

const MAX_SUGGESTIONS = 5;

export function CommentBox({ onSubmit }: CommentBoxProps) {
  const [text, setText] = useState('');
  // The people you can mention. Loaded already, not shown anywhere yet.
  const [users, setUsers] = useState<User[]>([]);
  // CORE: the fragment being typed (null = no dropdown)
  const [mention, setMention] = useState<Mention | null>(null);
  // NEXT: highlighted row
  const [activeIndex, setActiveIndex] = useState(0);
  // CHANGE: to put the caret back
  const inputRef = useRef<HTMLTextAreaElement>(null);

  useEffect(() => {
    let cancelled = false;
    getUsers().then((data) => {
      if (cancelled) {
        return;
      }
      setUsers(data);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // CORE: derived suggestions for the current fragment (any word of the name starts with it)
  let suggestions: User[] = [];
  if (mention) {
    const q = mention.query.toLowerCase();
    suggestions = users
      .filter((u) => {
        return u.name.toLowerCase().split(' ').some((word) => {
          return word.startsWith(q);
        });
      })
      .slice(0, MAX_SUGGESTIONS);
  }
  const open = mention !== null && suggestions.length > 0;

  // CORE: re-check the fragment on every change
  function handleChange(e: React.ChangeEvent<HTMLTextAreaElement>) {
    const value = e.target.value;
    setText(value);
    // selectionStart = caret position right after this keystroke
    setMention(findMention(value, e.target.selectionStart));
    // NEXT: highlight resets when the letters change
    setActiveIndex(0);
  }

  // CORE: replace "@pa" with "@Paula Reyes "
  function insertMention(user: User) {
    if (!mention) {
      return;
    }
    const before = text.slice(0, mention.start);
    const after = text.slice(mention.start + 1 + mention.query.length);
    const inserted = '@' + user.name + ' ';
    const caret = before.length + inserted.length;
    setText(before + inserted + after);
    setMention(null);
    // CHANGE: React re-renders the value, which would put the caret at the end.
    // Put it back right after the inserted name once the DOM has updated.
    requestAnimationFrame(() => {
      const el = inputRef.current;
      if (el) {
        el.focus();
        el.setSelectionRange(caret, caret);
      }
    });
  }

  // CORE: Enter inserts instead of a newline. NEXT: arrows + Escape.
  function handleKeyDown(e: React.KeyboardEvent<HTMLTextAreaElement>) {
    if (!open) {
      return;
    }
    // NEXT
    if (e.key === 'ArrowDown') {
      e.preventDefault();
      setActiveIndex((prev) => {
        return (prev + 1) % suggestions.length;
      });
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      setActiveIndex((prev) => {
        if (prev === 0) {
          return suggestions.length - 1;
        }
        return prev - 1;
      });
    } else if (e.key === 'Enter') {
      // CORE (Core can insert suggestions[0]; keyboard nav, NEXT, uses activeIndex)
      e.preventDefault(); // no newline
      insertMention(suggestions[Math.min(activeIndex, suggestions.length - 1)]);
    } else if (e.key === 'Escape') {
      // NEXT
      e.preventDefault();
      setMention(null);
    }
  }

  // (starter code, plus closing the dropdown)
  function handleSubmit() {
    if (text.trim() === '') {
      return;
    }
    onSubmit(text.trim());
    setText('');
    setMention(null);
  }

  // CORE: dropdown
  let dropdown = null;
  if (open) {
    dropdown = (
      <ul className="mention-menu" role="listbox" aria-label="Mention someone">
        {suggestions.map((u, i) => {
          // NEXT: highlighted row
          let cls = 'mention-option';
          if (i === activeIndex) {
            cls = 'mention-option is-active';
          }
          // CORE (NEXT: aria-selected + onMouseEnter). onMouseDown preventDefault keeps focus, so blur doesn't eat the click.
          return (
            <li
              key={u.id}
              role="option"
              aria-selected={i === activeIndex}
              className={cls}
              onMouseEnter={() => { setActiveIndex(i); }}
              onMouseDown={(e) => { e.preventDefault(); }}
              onClick={() => { insertMention(u); }}
            >
              <Avatar name={u.name} color={u.color} size={26} />
              <span className="mention-name">{u.name}</span>
              <span className="mention-role">{u.role}</span>
            </li>
          );
        })}
      </ul>
    );
  }

  // CORE (CHANGE: ref. NEXT: onBlur)
  return (
    <div className="comment-box">
      <div className="textarea-wrap">
        <textarea
          ref={inputRef}
          className="comment-input"
          placeholder="Add a comment. Type @ to mention someone"
          value={text}
          onChange={handleChange}
          onKeyDown={handleKeyDown}
          onBlur={() => { setMention(null); }}
        />
        {dropdown}
      </div>
      <div className="comment-actions">
        <span className="hint">Visible to everyone on this report</span>
        <button type="button" className="btn-primary" disabled={text.trim() === ''} onClick={() => { handleSubmit(); }}>Comment</button>
      </div>
    </div>
  );
}
