// Prices are stored in cents to avoid floating point surprises.
// formatPrice(1200) -> "$12"   formatPrice(2417) -> "$24.17"   formatPrice(0) -> "Free"
export function formatPrice(cents: number): string {
  if (cents === 0) {
    return 'Free';
  }
  const dollars = cents / 100;
  if (Number.isInteger(dollars)) {
    return '$' + dollars;
  }
  return '$' + dollars.toFixed(2);
}

import React, { useState } from 'react';
import { PLANS } from './data/plans';
import type { Plan } from './data/plans';
import { formatPrice } from './utils/formatPrice';

const VISIBLE_FEATURES = 3;

type Billing = 'monthly' | 'yearly';

function TopBar() {
  return (
    <header className="topbar">
      <span className="logo">▶ framecut</span>
      <nav className="topnav">
        <a>Features</a>
        <a>Templates</a>
        <a className="is-active">Pricing</a>
      </nav>
    </header>
  );
}

type BillingToggleProps = {
  billing: Billing;
  onChange: (next: Billing) => void;
};

function BillingToggle({ billing, onChange }: BillingToggleProps) {
  let monthlyCls = 'billing-opt';
  let yearlyCls = 'billing-opt';
  if (billing === 'monthly') {
    monthlyCls = 'billing-opt is-active';
  } else {
    yearlyCls = 'billing-opt is-active';
  }
  return (
    <div className="billing" role="group" aria-label="Billing period">
      <button type="button" className={monthlyCls} aria-pressed={billing === 'monthly'} onClick={() => { onChange('monthly'); }}>Monthly</button>
      <button type="button" className={yearlyCls} aria-pressed={billing === 'yearly'} onClick={() => { onChange('yearly'); }}>
        Yearly <span className="save-pill">2 months free</span>
      </button>
    </div>
  );
}

type PlanCardProps = {
  plan: Plan;
  billing: Billing;
  selected: boolean;
  onChoose: (id: string) => void;
};

function PlanCard({ plan, billing, selected, onChoose }: PlanCardProps) {
  // Only this card cares whether its list is open, so the state lives here.
  const [expanded, setExpanded] = useState(false);

  let badge = null;
  if (plan.popular) {
    badge = <span className="popular">Most popular</span>;
  }

  // derived price: yearly = 10 months (2 months free)
  const isFree = plan.monthlyCents === 0;
  let amount = formatPrice(plan.monthlyCents);
  let period = null;
  let note = null;
  if (!isFree && billing === 'monthly') {
    period = <span className="period">/month</span>;
  }
  if (!isFree && billing === 'yearly') {
    const yearlyCents = plan.monthlyCents * 10;
    amount = formatPrice(yearlyCents);
    period = <span className="period">/year</span>;
    note = <p className="price-note">{formatPrice(Math.round(yearlyCents / 12))}/mo, billed yearly</p>;
  }

  let shown = plan.features.slice(0, VISIBLE_FEATURES);
  let moreLabel = 'Show all ' + plan.features.length + ' features';
  if (expanded) {
    shown = plan.features;
    moreLabel = 'Show less';
  }

  let cls = 'plan';
  let chooseLabel = 'Choose ' + plan.name;
  if (selected) {
    cls = 'plan is-selected';
    chooseLabel = 'Selected ✓';
  }

  return (
    <article className={cls}>
      <div className="plan-top">
        <h2>{plan.name}</h2>
        {badge}
      </div>
      <p className="tagline">{plan.tagline}</p>
      <p className="price">
        <span className="amount">{amount}</span>
        {period}
      </p>
      {note}
      <button type="button" className="choose-btn" onClick={() => { onChoose(plan.id); }}>{chooseLabel}</button>
      <ul className="features">
        {shown.map((f) => {
          return <li key={f}>{f}</li>;
        })}
      </ul>
      <button type="button" className="more-btn" aria-expanded={expanded} onClick={() => { setExpanded((prev) => { return !prev; }); }}>{moreLabel}</button>
    </article>
  );
}

export default function App() {
  // Shared by the toggle and every card, so it lives in the parent.
  const [billing, setBilling] = useState<Billing>('monthly');
  const [selectedId, setSelectedId] = useState<string | null>(null);

  return (
    <div className="page">
      <TopBar />
      <section className="hero">
        <h1>Simple pricing for every channel</h1>
        <p>Start free. Upgrade when your audience grows.</p>
        <BillingToggle billing={billing} onChange={(next) => { setBilling(next); }} />
      </section>
      <section className="plans">
        {PLANS.map((p) => {
          return (
            <PlanCard
              key={p.id}
              plan={p}
              billing={billing}
              selected={p.id === selectedId}
              onChoose={(id) => { setSelectedId(id); }}
            />
          );
        })}
      </section>
    </div>
  );
}
