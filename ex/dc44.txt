// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /projects/orbit/deployments -> Deployment[]
export async function getDeployments(): Promise<Deployment[]> {
  await randomDelay(300, 900);
  return DEPLOYMENTS.map((d) => ({ ...d }));
}

// fuzzyMatch('Redeploy production', 'rdp') -> true: every typed letter appears in order,
// case-insensitive, spaces in the query ignored. fuzzyMatch('Go to Settings', '') -> true.
export function fuzzyMatch(text: string, query: string): boolean {
  const t = text.toLowerCase();
  const q = query.toLowerCase().replace(/\s+/g, '');
  let pos = 0;
  for (const ch of q) {
    const found = t.indexOf(ch, pos);
    if (found === -1) {
      return false;
    }
    pos = found + 1;
  }
  return true;
}

type ModalProps = {
  label: string;            // accessible name of the dialog
  onClose: () => void;      // called when the dimmed backdrop is clicked
  children: React.ReactNode;
};

// Centered panel over a dimmed backdrop. Render it only while open.
export default function Modal({ label, onClose, children }: ModalProps) {
  function handleBackdropClick(e: React.MouseEvent<HTMLDivElement>) {
    if (e.target === e.currentTarget) {
      onClose(); // a click on the backdrop itself, not inside the panel
    }
  }
  return (
    <div className="palette-backdrop" onClick={handleBackdropClick}>
      <div className="palette" role="dialog" aria-modal="true" aria-label={label}>
        {children}
      </div>
    </div>
  );
}

// CORE: palette component
type PaletteProps = { commands: Command[]; onClose: () => void };

// Only rendered while open, so query and active index reset on every open.
function CommandPalette({ commands, onClose }: PaletteProps) {
  const [query, setQuery] = useState('');
  // NEXT: highlighted index
  const [active, setActive] = useState(0);

  // CORE: derived results with fuzzyMatch (utils/fuzzy); CHANGE: built in SECTIONS order
  // derived: filtered and put in section order, so the arrow keys follow what you see
  const results: Command[] = [];
  SECTIONS.forEach((section) => {
    commands.forEach((c) => {
      if (c.section === section && fuzzyMatch(c.label, query)) {
        results.push(c);
      }
    });
  });

  function handleQueryChange(value: string) {
    setQuery(value);
    setActive(0); // the list changed, so start from the top
  }

  // CORE: run + close
  function runCommand(cmd: Command) {
    onClose();
    cmd.run();
  }

  // NEXT: arrows wrap, Enter runs (CORE: the Escape branch)
  function handleKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
    if (e.key === 'ArrowDown') {
      e.preventDefault(); // don't move the text cursor
      if (results.length > 0) {
        setActive((i) => { return (i + 1) % results.length; });
      }
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      if (results.length > 0) {
        setActive((i) => { return (i - 1 + results.length) % results.length; });
      }
    } else if (e.key === 'Enter') {
      e.preventDefault();
      if (results[active]) {
        runCommand(results[active]);
      }
    } else if (e.key === 'Escape') {
      e.preventDefault();
      onClose();
    }
  }

  // NEXT: aria-activedescendant
  let activeId: string | undefined = undefined;
  if (results[active]) {
    activeId = 'cmd-' + results[active].id;
  }

  // CORE: empty state or list; CHANGE: grouped under section headers
  let body;
  if (results.length === 0) {
    body = <p className="palette-empty">No commands match "{query}"</p>;
  } else {
    body = (
      <div className="palette-list" id="palette-list" role="listbox" aria-label="Commands">
        {SECTIONS.map((section) => {
          const items = results.filter((c) => {
            return c.section === section;
          });
          if (items.length === 0) {
            return null;
          }
          return (
            <div key={section} role="group" aria-label={section}>
              <p className="palette-section" aria-hidden="true">{section}</p>
              {items.map((c) => {
                const index = results.indexOf(c);
                let cls = 'palette-item';
                if (index === active) {
                  cls = 'palette-item is-active';
                }
                return (
                  <div
                    key={c.id}
                    id={'cmd-' + c.id}
                    role="option"
                    aria-selected={index === active}
                    className={cls}
                    onMouseMove={() => { setActive(index); }}
                    onClick={() => { runCommand(c); }}
                  >
                    {c.label}
                  </div>
                );
              })}
            </div>
          );
        })}
      </div>
    );
  }

  // CORE: reuse Modal (backdrop click already calls onClose)
  return (
    <Modal label="Command palette" onClose={onClose}>
      <div className="palette-input-row">
        <span className="palette-icon" aria-hidden="true">⌕</span>
        <input
          className="palette-input"
          autoFocus
          placeholder="Type a command…"
          role="combobox"
          aria-expanded="true"
          aria-controls="palette-list"
          aria-activedescendant={activeId}
          value={query}
          onChange={(e) => { handleQueryChange(e.target.value); }}
          onKeyDown={handleKeyDown}
        />
        <Kbd keys={['esc']} />
      </div>
      {body}
      <div className="palette-foot"><Kbd keys={['↑', '↓']} /> to navigate <Kbd keys={['↵']} /> to run</div>
    </Modal>
  );
}

  useEffect(() => {
    function handleKey(e: KeyboardEvent) {
      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') {
        e.preventDefault(); // the browser uses Ctrl+K for its own search bar
        if (paletteOpen) {
          setPaletteOpen(false);
        } else {
          returnFocus.current = e.target as HTMLElement; // what had focus when K was pressed
          setPaletteOpen(true);

  let toastEl = null;
  if (toast) {
    toastEl = <div className="toast" role="status">{toast}</div>;
  }

  // CORE: only rendered while open
  let palette = null;
  if (paletteOpen) {
    palette = <CommandPalette commands={commands} onClose={closePalette} />;
  }

  const title = NAV.find((n) => { return n.id === page; })?.label || '';

  return (
    <div className={'app theme-' + theme}>
      <Sidebar page={page} collapsed={collapsed} onNavigate={setPage} />
      <main className="main">
        <TopBar title={title} onSearch={openPalette} />
        <section className="content">{content}</section>
      </main>
      {toastEl}
      {palette}
    </div>
  );
}
