// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /playlists/sunday -> Playlist
export async function getPlaylist(): Promise<Playlist> {
  await randomDelay(400, 1000);
  return { id: 'sunday', name: 'Sunday Slow Mornings', owner: 'Arvind', tracks: TRACKS.map((t) => ({ ...t })) };
}

// PUT /playlists/sunday/order { trackIds } -> { ok: true }
export async function savePlaylistOrder(trackIds: string[]): Promise<{ ok: true }> {
  console.log('PUT /playlists/sunday/order', trackIds.join(','));
  await randomDelay(600, 1200);
  TRACKS = trackIds.map((id) => {
    const t = TRACKS.find((x) => x.id === id);
    return { ...(t as Track) };
  });
  return { ok: true };
}

// formatDuration(214) -> "3:34"
export function formatDuration(seconds: number): string {
  const m = Math.floor(seconds / 60);
  const s = seconds % 60;
  return m + ':' + String(s).padStart(2, '0');
}

// formatTotal(1826) -> "30 min", formatTotal(3900) -> "1 hr 5 min"
export function formatTotal(seconds: number): string {
  const totalMin = Math.round(seconds / 60);
  const h = Math.floor(totalMin / 60);
  const m = totalMin % 60;
  if (h === 0) {
    return m + ' min';
  }
  return h + ' hr ' + m + ' min';
}

import React, { useEffect, useState } from 'react';
import { getPlaylist, savePlaylistOrder } from './api';
import type { Playlist, Track } from './api';
import { formatDuration, formatTotal } from './utils/time';
import Cover from './components/Cover';

type DropTarget = { id: string; position: 'before' | 'after' };

// Pure helper: returns a new array with the item at `from` moved to `to`.
function moveItem<T>(list: T[], from: number, to: number): T[] {
  const next = [...list];
  const [item] = next.splice(from, 1);
  next.splice(to, 0, item);
  return next;
}

function PlaylistHeader({ playlist, tracks }: { playlist: Playlist; tracks: Track[] }) {
  // derived: never stored in state
  const totalSeconds = tracks.reduce((sum, t) => {
    return sum + t.seconds;
  }, 0);
  return (
    <header className="pl-header">
      <Cover color="#E8664F" size={112} label="Playlist cover" />
      <div>
        <p className="pl-kicker">Playlist</p>
        <h1>{playlist.name}</h1>
        <p className="pl-meta">{playlist.owner} · {tracks.length} songs · <span className="pl-total">{formatTotal(totalSeconds)}</span></p>
      </div>
    </header>
  );
}

type TrackRowProps = {
  track: Track;
  index: number;
  count: number;
  startsAt: number;
  isDragging: boolean;
  dropPosition: 'before' | 'after' | null;
  onMove: (from: number, to: number) => void;
  onDragStart: (e: React.DragEvent<HTMLLIElement>, id: string) => void;
  onDragOver: (e: React.DragEvent<HTMLLIElement>, id: string) => void;
  onDrop: (e: React.DragEvent<HTMLLIElement>) => void;
  onDragEnd: () => void;
};

function TrackRow({ track, index, count, startsAt, isDragging, dropPosition, onMove, onDragStart, onDragOver, onDrop, onDragEnd }: TrackRowProps) {
  let cls = 'track';
  if (isDragging) {
    cls = cls + ' is-dragging';
  }
  if (dropPosition === 'before') {
    cls = cls + ' drop-before';
  }
  if (dropPosition === 'after') {
    cls = cls + ' drop-after';
  }

  return (
    <li
      className={cls}
      draggable={true}
      onDragStart={(e) => { onDragStart(e, track.id); }}
      onDragOver={(e) => { onDragOver(e, track.id); }}
      onDrop={(e) => { onDrop(e); }}
      onDragEnd={() => { onDragEnd(); }}
    >
      <span className="drag-handle" aria-hidden="true">⋮⋮</span>
      <span className="track-index">{index + 1}</span>
      <Cover color={track.color} size={40} label={track.title + ' cover'} />
      <div className="track-main">
        <p className="track-title">{track.title}</p>
        <p className="track-artist">{track.artist}</p>
      </div>
      <span className="track-start">starts {formatDuration(startsAt)}</span>
      <span className="track-time">{formatDuration(track.seconds)}</span>
      <div className="move-btns">
        <button type="button" className="move-btn" aria-label={'Move ' + track.title + ' up'} disabled={index === 0} onClick={() => { onMove(index, index - 1); }}>▲</button>
        <button type="button" className="move-btn" aria-label={'Move ' + track.title + ' down'} disabled={index === count - 1} onClick={() => { onMove(index, index + 1); }}>▼</button>
      </div>
    </li>
  );
}

export default function App() {
  const [playlist, setPlaylist] = useState<Playlist | null>(null);
  const [tracks, setTracks] = useState<Track[]>([]);
  const [savedIds, setSavedIds] = useState<string[]>([]); // order the server has
  const [saving, setSaving] = useState(false);
  const [dragId, setDragId] = useState<string | null>(null);
  const [dropTarget, setDropTarget] = useState<DropTarget | null>(null);
  const [announcement, setAnnouncement] = useState('');

  useEffect(() => {
    let cancelled = false;
    getPlaylist().then((data) => {
      if (cancelled) {
        return;
      }
      setPlaylist(data);
      setTracks(data.tracks);
      setSavedIds(data.tracks.map((t) => {
        return t.id;
      }));
    });
    return () => {
      cancelled = true;
    };
  }, []);

  function handleMove(from: number, to: number) {
    if (to < 0 || to >= tracks.length) {
      return;
    }
    const moved = tracks[from];
    setTracks((prev) => {
      return moveItem(prev, from, to);
    });
    setAnnouncement(moved.title + ' moved to position ' + (to + 1) + ' of ' + tracks.length);
  }

  function handleDragStart(e: React.DragEvent<HTMLLIElement>, id: string) {
    e.dataTransfer.effectAllowed = 'move';
    e.dataTransfer.setData('text/plain', id); // Firefox needs data to start a drag
    setDragId(id);
  }

  function handleDragOver(e: React.DragEvent<HTMLLIElement>, id: string) {
    e.preventDefault(); // required, or drop never fires
    const rect = e.currentTarget.getBoundingClientRect();
    let position: 'before' | 'after' = 'before';
    if (e.clientY > rect.top + rect.height / 2) {
      position = 'after';
    }
    if (!dropTarget || dropTarget.id !== id || dropTarget.position !== position) {
      setDropTarget({ id: id, position: position });
    }
  }

  function handleDrop(e: React.DragEvent<HTMLLIElement>) {
    e.preventDefault();
    if (dragId && dropTarget && dragId !== dropTarget.id) {
      const from = tracks.findIndex((t) => {
        return t.id === dragId;
      });
      let to = tracks.findIndex((t) => {
        return t.id === dropTarget.id;
      });
      if (dropTarget.position === 'after') {
        to = to + 1;
      }
      if (from < to) {
        to = to - 1; // removing the dragged item shifts everything after it up by one
      }
      handleMove(from, to);
    }
    handleDragEnd();
  }

  function handleDragEnd() {
    setDragId(null);
    setDropTarget(null);
  }

  function handleSave() {
    const ids = tracks.map((t) => {
      return t.id;
    });
    setSaving(true);
    savePlaylistOrder(ids).then(() => {
      setSavedIds(ids);
      setSaving(false);
      setAnnouncement('Playlist order saved');
    });
  }

  function handleDiscard() {
    setTracks((prev) => {
      return savedIds.map((id) => {
        return prev.find((t) => {
          return t.id === id;
        }) as Track;
      });
    });
  }

  if (!playlist) {
    return <div className="app"><p className="loading">Loading playlist…</p></div>;
  }

  // derived
  const currentIds = tracks.map((t) => {
    return t.id;
  });
  const dirty = currentIds.join(',') !== savedIds.join(',');

  let saveLabel = 'Save order';
  if (saving) {
    saveLabel = 'Saving…';
  }

  let statusEl = <span className="save-state is-saved">✓ All changes saved</span>;
  if (dirty) {
    statusEl = <span className="save-state is-dirty">● Unsaved changes</span>;
  }

  let discardBtn = null;
  if (dirty && !saving) {
    discardBtn = <button type="button" className="ghost-btn" onClick={() => { handleDiscard(); }}>Discard</button>;
  }

  // derived: each track's start time in the playlist (recalculates on every reorder)
  const starts: number[] = [];
  let running = 0;
  tracks.forEach((t) => {
    starts.push(running);
    running = running + t.seconds;
  });

  return (
    <div className="app">
      <PlaylistHeader playlist={playlist} tracks={tracks} />
      <div className="actions">
        <button type="button" className="play-btn" aria-label="Play">▶</button>
        <span className="actions-spacer" />
        {statusEl}
        {discardBtn}
        <button type="button" className="save-btn" disabled={!dirty || saving} onClick={() => { handleSave(); }}>{saveLabel}</button>
      </div>
      <ol className="tracks">
        {tracks.map((t, i) => {
          let dropPosition: 'before' | 'after' | null = null;
          if (dropTarget && dropTarget.id === t.id && dragId !== t.id) {
            dropPosition = dropTarget.position;
          }
          return (
            <TrackRow
              key={t.id}
              track={t}
              index={i}
              count={tracks.length}
              startsAt={starts[i]}
              isDragging={dragId === t.id}
              dropPosition={dropPosition}
              onMove={handleMove}
              onDragStart={handleDragStart}
              onDragOver={handleDragOver}
              onDrop={handleDrop}
              onDragEnd={handleDragEnd}
            />
          );
        })}
      </ol>
      <p className="sr-only" aria-live="polite">{announcement}</p>
    </div>
  );
}
