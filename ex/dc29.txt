// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /listings/l1 -> Listing
export async function getListing(): Promise<Listing> {
  await randomDelay(400, 900);
  return { ...LISTING, amenities: [...LISTING.amenities], photos: LISTING.photos.map((p) => ({ ...p })) };
}

import React, { useEffect, useState } from 'react';
import { getListing } from './api';
import type { Listing, Photo } from './api';
import PhotoSlide from './components/PhotoSlide';
import { formatPrice } from './utils/format';

const AUTOPLAY_MS = 4000;

function Gallery({ photos }: { photos: Photo[] }) {
  const [index, setIndex] = useState(0);
  const [paused, setPaused] = useState(false);
  const count = photos.length;

  // Autoplay. `index` is a dependency on purpose: any manual move restarts
  // the 4s countdown, so the slide never jumps right after a click.
  useEffect(() => {
    if (paused || count < 2) {
      return;
    }
    const id = setInterval(() => {
      setIndex((i) => {
        return (i + 1) % count;
      });
    }, AUTOPLAY_MS);
    return () => {
      clearInterval(id);
    };
  }, [paused, count, index]);

  function goNext() {
    setIndex((i) => {
      return (i + 1) % count;
    });
  }

  function goPrev() {
    // + count keeps it positive: (0 - 1 + 7) % 7 = 6
    setIndex((i) => {
      return (i - 1 + count) % count;
    });
  }

  function handleKeyDown(e: React.KeyboardEvent<HTMLDivElement>) {
    if (e.key === 'ArrowRight') {
      e.preventDefault();
      goNext();
    }
    if (e.key === 'ArrowLeft') {
      e.preventDefault();
      goPrev();
    }
  }

  const photo = photos[index];
  const position = index + 1; // derived: humans count from 1

  return (
    <div
      className="gallery"
      tabIndex={0}
      role="region"
      aria-roledescription="carousel"
      aria-label="Listing photos"
      onMouseEnter={() => { setPaused(true); }}
      onMouseLeave={() => { setPaused(false); }}
      onKeyDown={handleKeyDown}
    >
      <PhotoSlide photo={photo} />
      <button type="button" className="nav-btn nav-prev" aria-label="Previous photo" onClick={() => { goPrev(); }}>‹</button>
      <button type="button" className="nav-btn nav-next" aria-label="Next photo" onClick={() => { goNext(); }}>›</button>
      <span className="counter" aria-live="polite">{position} / {count}</span>
      <div className="dots">
        {photos.map((p, i) => {
          let cls = 'dot';
          if (i === index) {
            cls = 'dot is-active';
          }
          const label = 'Show photo ' + (i + 1);
          return (
            <button key={p.id} type="button" className={cls} aria-label={label} onClick={() => { setIndex(i); }} />
          );
        })}
      </div>
    </div>
  );
}
