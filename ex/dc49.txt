// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /countries -> Country[] (sorted by name)
export async function getCountries(): Promise<Country[]> {
  await randomDelay(400, 1000);
  return COUNTRIES.map((c) => ({ ...c }));

// Keep the props: App already passes the countries and receives the selection.
export default function CountryCombobox({ id, countries, loading, selected, onChange }: Props) {
  function isSelected(c: Country): boolean {
    return selected.some((s) => {
      return s.code === c.code;
    });
  }

  // Adds the country, or removes it if it is already selected.
  function toggle(c: Country) {
    if (isSelected(c)) {
      onChange(selected.filter((s) => {
        return s.code !== c.code;
      }));
    } else {
      onChange([...selected, c]);
    }
  }

  function removeChip(c: Country) {
    onChange(selected.filter((s) => {
      return s.code !== c.code;
    }));
  }

  function handleNativePick(code: string) {
    const c = countries.find((x) => {
      return x.code === code;
    });
    if (c) {
      toggle(c);
    }
  }

  let placeholder = 'Select a country';
  if (loading) {
    placeholder = 'Loading countries…';
  }

  return (
    <div className="cb">
      <div className="cb-control">
        <ul className="cb-chips" aria-label="Selected countries">
          {selected.map((c) => {
            return (
              <li key={c.code} className="cb-chip">
                <span aria-hidden="true">{c.flag}</span> {c.name}
                <button type="button" className="cb-chip-remove" aria-label={'Remove ' + c.name} onClick={() => { removeChip(c); }}>×</button>

import React, { useEffect, useRef, useState } from 'react';
import type { Country } from '../api';

type Props = {
  id: string;               // id of the input; the label from FormField points at it
  countries: Country[];
  loading: boolean;
  selected: Country[];
  onChange: (next: Country[]) => void;
};

// WAI-ARIA combobox pattern (search: "aria apg combobox listbox").
// Focus stays on the input the whole time; aria-activedescendant tells
// screen readers which option is "active".
export default function CountryCombobox({ id, countries, loading, selected, onChange }: Props) {
  // CORE: query, open, active index (focus never leaves the input)
  const [query, setQuery] = useState('');
  const [open, setOpen] = useState(false);
  const [active, setActive] = useState(0);
  // NEXT + CHANGE: the input ref is for giving focus back
  const inputRef = useRef<HTMLInputElement>(null);
  // NEXT: scroll the active option into view
  const listRef = useRef<HTMLUListElement>(null);
  const listId = id + '-listbox';

  // CORE: filtered options + clamped active index, derived
  // derived
  const q = query.trim().toLowerCase();
  const options = countries.filter((c) => {
    return c.name.toLowerCase().includes(q);
  });
  const activeIndex = Math.min(active, options.length - 1); // -1 when nothing matches

  // NEXT
  // Keep the active option visible while arrowing through a long list.
  useEffect(() => {
    if (!open || activeIndex < 0 || !listRef.current) {
      return;
    }
    const el = listRef.current.children[activeIndex] as HTMLElement | undefined;
    if (el) {
      el.scrollIntoView({ block: 'nearest' });
    }
  }, [open, activeIndex]);

  function isSelected(c: Country): boolean {
    return selected.some((s) => {
      return s.code === c.code;
    });
  }

  // Adds the country, or removes it if it is already selected.
  // CORE: clear the query after a pick
  function toggle(c: Country) {
    if (isSelected(c)) {
      onChange(selected.filter((s) => {
        return s.code !== c.code;
      }));
    } else {
      onChange([...selected, c]);
    }
    setQuery('');
    setActive(0);
  }

  // CHANGE: removing a chip puts focus back in the input
  function removeChip(c: Country) {
    onChange(selected.filter((s) => {
      return s.code !== c.code;
    }));
    if (inputRef.current) {
      inputRef.current.focus(); // the chip button is about to disappear
    }
  }

  // NEXT: clear-all button + click-to-focus
  // Clicking anywhere in the box (around the chips) focuses the input.
  function focusInput() {
    if (inputRef.current) {
      inputRef.current.focus();
    }
  }

  function clearAll() {
    onChange([]);
    setQuery('');
    setOpen(false);
    if (inputRef.current) {
      inputRef.current.focus();
    }
  }

  // CORE: typing opens the list, first option active
  function handleInput(value: string) {
    setQuery(value);
    setOpen(true);
    setActive(0);
  }

  // CORE: ↓ / ↑ (clamped), Enter, Escape. NEXT: Home / End, Tab, second Escape. CHANGE: Backspace
  function handleKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
    const last = options.length - 1;
    if (e.key === 'ArrowDown') {
      e.preventDefault();
      if (!open) {
        setOpen(true);
        setActive(0);
      } else {
        setActive(Math.min(activeIndex + 1, last));
      }
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      if (!open) {
        setOpen(true);
        setActive(last);
      } else {
        setActive(Math.max(activeIndex - 1, 0));
      }
    // NEXT
    } else if (e.key === 'Home' && open) {
      e.preventDefault();
      setActive(0);
    } else if (e.key === 'End' && open) {
      e.preventDefault();
      setActive(last);
    } else if (e.key === 'Enter') {
      if (open && activeIndex >= 0) {
        e.preventDefault(); // don't submit the form
        toggle(options[activeIndex]);
      }
    } else if (e.key === 'Escape') {
      if (open) {
        e.preventDefault();
        setOpen(false);
      } else if (query) {
        setQuery('');
      }
    } else if (e.key === 'Tab') {
      setOpen(false); // let focus move on normally
    // CHANGE
    } else if (e.key === 'Backspace') {
      if (query === '' && selected.length > 0) {
        onChange(selected.slice(0, -1));
      }
    }
  }

  // NEXT: aria-activedescendant only while open
  // decisions
  let activeId: string | undefined = undefined;
  if (open && activeIndex >= 0) {
    activeId = id + '-opt-' + options[activeIndex].code;
  }

  // CORE
  let placeholder = 'Search countries';
  if (loading) {
    placeholder = 'Loading countries…';
  }

  // NEXT
  let clearBtn = null;
  if (selected.length > 0 || query) {
    clearBtn = (
      <button type="button" className="cb-clear" aria-label="Clear all countries" onClick={() => { clearAll(); }}>✕</button>
    );
  }

  // CORE: options with role=option + ids; mousedown preventDefault keeps focus (NEXT: no-match row, ✓)
  let items;
  if (options.length === 0) {
    items = <li className="cb-empty" role="option" aria-disabled="true" aria-selected="false">No countries match "{query}"</li>;
  } else {
    items = options.map((c, i) => {
      const sel = isSelected(c);
      let cls = 'cb-option';
      if (i === activeIndex) {
        cls = cls + ' is-active';
      }
      if (sel) {
        cls = cls + ' is-selected';
      }
      return (
        <li
          key={c.code}
          id={id + '-opt-' + c.code}
          role="option"
          aria-selected={sel}
          className={cls}
          onMouseDown={(e) => { e.preventDefault(); }}
          onMouseMove={() => { setActive(i); }}
          onClick={() => { toggle(c); }}
        >
          <span className="cb-flag" aria-hidden="true">{c.flag}</span>
          <span className="cb-name">{c.name}</span>
          <span className="cb-check" aria-hidden="true">✓</span>
        </li>
      );
    });
  }

  let wrapCls = 'cb';
  if (open) {
    wrapCls = 'cb is-open';
  }

  return (
    <div className={wrapCls}>
      <div className="cb-control" onClick={() => { focusInput(); }}>
        <ul className="cb-chips" aria-label="Selected countries">
          {selected.map((c) => {
            return (
              <li key={c.code} className="cb-chip">
                <span aria-hidden="true">{c.flag}</span> {c.name}
                <button type="button" className="cb-chip-remove" aria-label={'Remove ' + c.name} onClick={(e) => { e.stopPropagation(); removeChip(c); }}>×</button>
              </li>
            );
          })}
        </ul>
        <input
          ref={inputRef}
          id={id}
          className="cb-input"
          type="text"
          role="combobox"
          autoComplete="off"
          aria-autocomplete="list"
          aria-expanded={open}
          aria-controls={listId}
          aria-activedescendant={activeId}
          aria-describedby={id + '-hint'}
          disabled={loading}
          placeholder={placeholder}
          value={query}
          onChange={(e) => { handleInput(e.target.value); }}
          onKeyDown={handleKeyDown}
          onClick={() => { setOpen(true); }}
          onBlur={() => { setOpen(false); }}
        />
        {clearBtn}
        <span className="cb-caret" aria-hidden="true">▾</span>
      </div>
      <ul ref={listRef} id={listId} role="listbox" aria-multiselectable="true" aria-label="Countries" className="cb-list" hidden={!open}>
        {items}
      </ul>
      {/* CHANGE: announce the count */}
      <p className="sr-only" aria-live="polite">{selected.length} countries selected</p>
    </div>
  );
}

          <FormField id="visited" label="Countries visited" hint="Start typing to search. Pick as many as apply.">
            <CountryCombobox id="visited" countries={countries} loading={loading} selected={visited} onChange={setVisited} />
          </FormField>
          <div className="actions">
            <button type="button" className="btn-secondary">Back</button>
            <button type="submit" className="btn-primary">Save and continue</button>
          </div>
          {savedEl}
        </form>
      </main>
    </div>
  );
}
