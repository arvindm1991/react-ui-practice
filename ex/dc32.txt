import type { OnboardingData } from './types';

export type Account = {
  id: string;
  name: string;
  planName: string;
};

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

const PLAN_NAMES = { lose: 'Lean & Light', strength: 'Iron Foundations', run: 'Road to 10K' };

// POST /accounts -> Account
export async function createAccount(data: OnboardingData): Promise<Account> {
  await randomDelay(700, 1300);
  let planName = 'Starter plan';
  if (data.goal !== '') {
    planName = PLAN_NAMES[data.goal];
  }
  return { id: 'acc_' + Math.floor(Math.random() * 100000), name: data.name.trim(), planName };
}

// Label + control + error message. Pass error to show it in red under the control.

type StepProps = {

export function ProfileStep({ values, errors, onChange }: StepProps) {
  return (
    <div className="step-body">
      <Field label="Full name" error={errors.name}>
        <input className="input" value={values.name} onChange={(e) => { onChange('name', e.target.value); }} placeholder="Jordan Rivera" />

        <input className="input" value={values.email} onChange={(e) => { onChange('email', e.target.value); }} placeholder="you@example.com" />

      <Field label="Age" error={errors.age}>
        <input className="input input-short" value={values.age} onChange={(e) => { onChange('age', e.target.value); }} inputMode="numeric" placeholder="30" />

    </div>

}

// Read-only summary for the last step.
export function ReviewStep({ values }: { values: OnboardingData }) {
  const goal = GOALS.find((g) => {
    return g.id === values.goal;
  });
  let goalLabel = '—';
  if (goal) {
    goalLabel = goal.label;
  }
  let unitsLabel = 'Metric';
  if (values.units === 'imperial') {
    unitsLabel = 'Imperial';
  }
  const rows = [
    ['Name', values.name.trim()],
    ['Email', values.email.trim()],
    ['Age', values.age],
    ['Goal', goalLabel],
    ['Workouts', values.daysPerWeek + ' days a week'],
    ['Units', unitsLabel],
  ];
  return (
    <dl className="review">
      {rows.map((row) => {
        return (
          <div key={row[0]} className="review-row">
            <dt>{row[0]}</dt>
            <dd>{row[1]}</dd>
          </div>
        );
      })}
    </dl>
  );
}

// isValidEmail('a@b.co') -> true, isValidEmail('a@b') -> false
export function isValidEmail(value: string): boolean {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value.trim());
}

// isIntInRange('34', 13, 99) -> true, isIntInRange('12', 13, 99) -> false, isIntInRange('abc', 13, 99) -> false
export function isIntInRange(value: string, min: number, max: number): boolean {
  const n = Number(value);
  return value.trim() !== '' && Number.isInteger(n) && n >= min && n <= max;
}

import React, { useState } from 'react';
import { EMPTY_FORM } from './types';
import type { OnboardingData } from './types';
import { ProfileStep, PreferencesStep, ReviewStep } from './components/Steps';
import type { StepErrors } from './components/Steps';
import { isIntInRange, isValidEmail } from './utils/validate';
import { createAccount } from './api';
import type { Account } from './api';

// CORE: step names + titles
const STEPS = ['Profile', 'Preferences', 'Review'];
const TITLES = ['Tell us about you', 'What are you training for?', 'Check your details'];

// CORE: validation. Pure function: which fields on this step are invalid?
function validateStep(step: number, form: OnboardingData): StepErrors {
  const errors: StepErrors = {};
  if (step === 0) {
    if (form.name.trim() === '') {
      errors.name = 'Enter your name';
    }
    if (!isValidEmail(form.email)) {
      errors.email = 'Enter a valid email';
    }
    if (!isIntInRange(form.age, 13, 99)) {
      errors.age = 'Age must be between 13 and 99';
    }
  }
  if (step === 1) {
    if (form.goal === '') {
      errors.goal = 'Pick a goal';
    }
  }
  return errors;
}

// STRETCH: step indicator
type IndicatorProps = {
  current: number;
  onJump: (index: number) => void;
};

// STRETCH: step indicator. Completed steps are buttons you can jump back to; current and future steps are disabled.
function StepIndicator({ current, onJump }: IndicatorProps) {
  return (
    <ol className="stepper">
      {STEPS.map((label, i) => {
        let cls = 'step';
        let mark = String(i + 1);
        if (i < current) {
          cls = 'step is-done';
          mark = '✓';
        } else if (i === current) {
          cls = 'step is-current';
        }
        return (
          <li key={label} className="stepper-item">
            <button type="button" className={cls} disabled={i >= current} onClick={() => { onJump(i); }}>
              <span className="step-dot">{mark}</span>
              <span className="step-label">{label}</span>
            </button>
          </li>
        );
      })}
    </ol>
  );
}

function OnboardingWizard() {
  const [form, setForm] = useState<OnboardingData>(EMPTY_FORM);
  // CORE: step + errors
  const [step, setStep] = useState(0);
  const [errors, setErrors] = useState<StepErrors>({});
  // NEXT: submit
  const [submitting, setSubmitting] = useState(false);
  const [account, setAccount] = useState<Account | null>(null);

  function handleChange(field: keyof OnboardingData, value: string) {
    setForm((prev) => {
      return { ...prev, [field]: value };
    });
    // CHANGE: clear that field's error as soon as they fix it.
    setErrors((prev) => {
      return { ...prev, [field]: undefined };
    });
  }

  // CORE: Continue validates, then advances
  function handleNext() {
    const found = validateStep(step, form);
    if (Object.keys(found).length > 0) {
      setErrors(found);
      return;
    }
    setErrors({});
    setStep((prev) => {
      return prev + 1;
    });
  }

  // CORE: Back (CHANGE: also clears errors)
  function handleBack() {
    setErrors({});
    setStep((prev) => {
      return prev - 1;
    });
  }

  // STRETCH: jump back from the indicator
  function handleJump(index: number) {
    if (index < step) {
      setErrors({});
      setStep(index);
    }
  }

  // NEXT: submit + success screen
  function handleSubmit() {
    setSubmitting(true);
    createAccount(form).then((result) => {
      setSubmitting(false);
      setAccount(result);
    });
  }

  if (account) {
    return (
      <section className="wizard wizard-done">
        <div className="done-badge">✓</div>
        <h1 className="wizard-title">You're all set, {account.name}!</h1>
        <p className="done-copy">Your plan <b>{account.planName}</b> starts tomorrow.</p>
      </section>
    );
  }

  // CORE: body per step
  let body;
  if (step === 0) {
    body = <ProfileStep values={form} errors={errors} onChange={handleChange} />;
  } else if (step === 1) {
    body = <PreferencesStep values={form} errors={errors} onChange={handleChange} />;
  } else {
    body = <ReviewStep values={form} />;
  }

  // CORE: footer buttons
  let backBtn = <span />;
  if (step > 0) {
    backBtn = <button type="button" className="btn btn-ghost" disabled={submitting} onClick={() => { handleBack(); }}>Back</button>;
  }

  let nextBtn = <button type="button" className="btn btn-primary" onClick={() => { handleNext(); }}>Continue</button>;
  // NEXT: last step submits
  if (step === STEPS.length - 1) {
    let label = 'Create my plan';
    if (submitting) {
      label = 'Creating…';
    }
    nextBtn = <button type="button" className="btn btn-primary" disabled={submitting} onClick={() => { handleSubmit(); }}>{label}</button>;
  }

  // CORE: render (STRETCH: StepIndicator)
  return (
    <section className="wizard">
      <StepIndicator current={step} onJump={handleJump} />
      <p className="eyebrow">Step {step + 1} of {STEPS.length}</p>
      <h1 className="wizard-title">{TITLES[step]}</h1>
      {body}
      <div className="wizard-foot">
        {backBtn}
        {nextBtn}
      </div>
    </section>
  );
}

export default function App() {
  return (
    <div className="app">
      <div className="shell">
        <SidePanel />
        <OnboardingWizard />
      </div>
    </div>
  );
}
