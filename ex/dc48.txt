// Fake API. Treat it like a real backend you can't change.
export type FieldType = 'text' | 'email' | 'select' | 'checkbox';

export type Field = {
  id: string;
  type: FieldType;
  label: string;
  required: boolean;
  options: string[]; // only used by "select"
};

export type FormDef = {
  name: string;
  fields: Field[];
};

const FORM: FormDef = {
  name: 'Beta waitlist',
  fields: [
    { id: 'f1', type: 'text', label: 'Full name', required: true, options: [] },
    { id: 'f2', type: 'email', label: 'Work email', required: true, options: [] },
  ],
};

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /forms/waitlist -> FormDef
export async function getForm(): Promise<FormDef> {
  await randomDelay(300, 800);
  return { name: FORM.name, fields: FORM.fields.map((f) => ({ ...f, options: [...f.options] })) };
}

// isEmail("a@b.co") -> true
export function isEmail(value: string): boolean {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value);
}

import React from 'react';
import type { Field } from '../api';

type PreviewFieldProps = {
  field: Field;
  value: string | boolean | undefined; // checkbox: boolean, everything else: string
  error?: string;                      // shown under the field when not empty
  onChange: (value: string | boolean) => void;
};

// One field of the live preview: a text / email input, a select or a checkbox,
// with a linked label, a red * when required, and the error message if given.
export default function PreviewField({ field, value, error, onChange }: PreviewFieldProps) {
  const inputId = 'pv-' + field.id;

  let cls = 'pv-field';
  let errorEl = null;
  let describedBy: string | undefined = undefined;
  if (error) {
    cls = 'pv-field has-error';
    errorEl = <span className="pv-error" id={inputId + '-err'}>{error}</span>;
    describedBy = inputId + '-err';
  }

  let star = null;
  if (field.required) {
    star = <span className="pv-star" aria-hidden="true">*</span>;
  }

  let text = '';
  if (typeof value === 'string') {
    text = value;
  }

  let control;
  if (field.type === 'checkbox') {
    control = (
      <label className="pv-check" htmlFor={inputId}>
        <input id={inputId} type="checkbox" aria-invalid={Boolean(error)} aria-describedby={describedBy} checked={value === true} onChange={(e) => { onChange(e.target.checked); }} />
        {field.label}{star}
      </label>
    );
  } else {
    let input;
    if (field.type === 'select') {
      // blank lines in the options editor are ignored
      const options = field.options.filter((o) => {
        return o.trim() !== '';
      });
      input = (
        <select id={inputId} className="pv-input" aria-invalid={Boolean(error)} aria-describedby={describedBy} value={text} onChange={(e) => { onChange(e.target.value); }}>
          <option value="">Choose…</option>
          {options.map((o, i) => {
            return <option key={i} value={o}>{o}</option>;
          })}
        </select>
      );
    } else {
      input = <input id={inputId} className="pv-input" type={field.type} aria-invalid={Boolean(error)} aria-describedby={describedBy} value={text} onChange={(e) => { onChange(e.target.value); }} />;
    }
    control = (
      <div>
        <label className="pv-label" htmlFor={inputId}>{field.label}{star}</label>
        {input}
      </div>
    );
  }

  return (
    <div className={cls}>
      {control}
      {errorEl}
    </div>
  );
}

type Values = Record<string, string | boolean>; // field id -> value
type Errors = Record<string, string>;           // field id -> message

// CORE: editable card (NEXT: onMove / onDelete)
type FieldEditorProps = {
  field: Field;
  index: number;
  count: number;
  onChange: (id: string, patch: Partial<Field>) => void;
  onMove: (from: number, to: number) => void;
  onDelete: (id: string) => void;
};

function FieldEditor({ field, index, count, onChange, onMove, onDelete }: FieldEditorProps) {
  // STRETCH: options textarea for dropdowns
  let optionsEditor = null;
  if (field.type === 'select') {
    optionsEditor = (
      <label className="editor-row">
        <span className="editor-label">Options (one per line)</span>
        <textarea
          className="editor-input options-input"
          rows={3}
          value={field.options.join('\n')}
          onChange={(e) => { onChange(field.id, { options: e.target.value.split('\n') }); }}
        />
      </label>
    );
  }

  // CORE: label input
  return (
    <li className="field-card is-editing">
      <div className="field-card-top">
        <span className="type-badge" title={typeInfo(field.type).label}>{typeInfo(field.type).icon}</span>
        <input
          className="editor-input label-input"
          aria-label="Field label"
          value={field.label}
          onChange={(e) => { onChange(field.id, { label: e.target.value }); }}
        />
        {/* NEXT: move up / down + delete */}
        <div className="card-actions">
          <button type="button" className="card-btn" aria-label={'Move ' + field.label + ' up'} disabled={index === 0} onClick={() => { onMove(index, index - 1); }}>↑</button>
          <button type="button" className="card-btn" aria-label={'Move ' + field.label + ' down'} disabled={index === count - 1} onClick={() => { onMove(index, index + 1); }}>↓</button>
          <button type="button" className="card-btn is-danger" aria-label={'Delete ' + field.label} onClick={() => { onDelete(field.id); }}>✕</button>
        </div>
      </div>
      {optionsEditor}
      {/* CORE: Required toggle (reuse Toggle) */}
      <div className="field-card-foot">
        <span className="muted-sm">{typeInfo(field.type).label}</span>
        <Toggle label="Required" checked={field.required} onChange={(checked) => { onChange(field.id, { required: checked }); }} />
      </div>
    </li>
  );
}

// NEXT: validation rules
// Returns an error message for one field, or '' when it is valid.
function validateField(field: Field, value: string | boolean | undefined): string {
  if (field.type === 'checkbox') {
    if (field.required && value !== true) {
      return 'Please check this box';
    }
    return '';
  }
  let text = '';
  if (typeof value === 'string') {
    text = value.trim();
  }
  if (field.required && text === '') {
    return 'This field is required';
  }
  if (field.type === 'email' && text !== '' && !isEmail(text)) {
    return 'Enter a valid email';
  }
  return '';
}

// CORE: live preview renders PreviewField per field
function PreviewForm({ name, fields }: { name: string; fields: Field[] }) {
  const [values, setValues] = useState<Values>({});
  // NEXT: validation state
  const [errors, setErrors] = useState<Errors>({});
  const [submitted, setSubmitted] = useState(false);

  // CORE: controlled preview values
  function setValue(id: string, value: string | boolean) {
    setValues((prev) => {
      return { ...prev, [id]: value };
    });
    setSubmitted(false);
    // NEXT
    // clear this field's error as soon as the user fixes it
    setErrors((prev) => {
      const next = { ...prev };
      delete next[id];
      return next;
    });
  }

  // NEXT: validate on submit
  function handleSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const nextErrors: Errors = {};
    fields.forEach((f) => {
      const msg = validateField(f, values[f.id]);
      if (msg) {
        nextErrors[f.id] = msg;
      }
    });
    setErrors(nextErrors);
    setSubmitted(Object.keys(nextErrors).length === 0);
  }

  let success = null;
  if (submitted) {
    success = <p className="form-success" role="status">✓ Submitted. All required fields are filled in.</p>;
  }

  // CORE
  let empty = null;
  if (fields.length === 0) {
    empty = <p className="muted">Add a field to see it here.</p>;
  }

  // CORE: a real form from the fields (NEXT: noValidate, Submit, success)
  return (
    <form className="preview-card" noValidate onSubmit={handleSubmit}>
      <h3 className="preview-name">{name}</h3>
      {empty}
      {fields.map((f) => {
        return <PreviewField key={f.id} field={f} value={values[f.id]} error={errors[f.id]} onChange={(v) => { setValue(f.id, v); }} />;
      })}
      <button type="submit" className="pv-submit">Submit</button>
      {success}
    </form>
  );
}

export default function App() {
  const [name, setName] = useState('');
  const [fields, setFields] = useState<Field[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    getForm().then((form) => {
      if (cancelled) {
        return;
      }
      setName(form.name);
      setFields(form.fields);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // CORE: add (dropdowns start with two options) + one merge-patch update
  function handleAdd(type: FieldType) {
    let options: string[] = [];
    if (type === 'select') {
      options = ['Option 1', 'Option 2'];
    }
    const field: Field = { id: uid('f'), type: type, label: 'Untitled ' + typeInfo(type).label.toLowerCase(), required: false, options: options };
    setFields((prev) => {
      return [...prev, field];
    });
  }

  function handleChange(id: string, patch: Partial<Field>) {
    setFields((prev) => {
      return prev.map((f) => {
        if (f.id !== id) {
          return f;
        }
        return { ...f, ...patch };
      });
    });
  }

  // NEXT: reorder + delete
  function handleMove(from: number, to: number) {
    setFields((prev) => {
      if (to < 0 || to >= prev.length) {
        return prev;
      }
      const next = [...prev];
      const [item] = next.splice(from, 1);
      next.splice(to, 0, item);
      return next;
    });
  }

  function handleDelete(id: string) {
    setFields((prev) => {
      return prev.filter((f) => {
        return f.id !== id;
      });
    });
  }

  // CHANGE: JSON export is derived, no state
  // derived: the exported definition is just the current state, serialized
  const json = JSON.stringify({ name: name, fields: fields }, null, 2);

  // CORE: a FieldEditor per field, empty state
  let list;
  if (loading) {
    list = <p className="muted">Loading form…</p>;
  } else if (fields.length === 0) {
    list = <p className="muted">No fields yet. Add one from the left.</p>;
  } else {
    list = (
      <ul className="field-list">
        {fields.map((f, i) => {
          return <FieldEditor key={f.id} field={f} index={i} count={fields.length} onChange={handleChange} onMove={handleMove} onDelete={handleDelete} />;
        })}
      </ul>
    );
  }

  // CORE: wire the panels (CHANGE: .right-col + JSON panel)
  return (
    <div className="app">
      <header className="topbar">
        <span className="logo">formcraft</span>
        <span className="form-name">{name}</span>
      </header>
      <div className="workspace">
        <AddFieldPanel onAdd={handleAdd} />
        <section className="panel builder-panel">
          <h2 className="panel-title">Fields · {fields.length}</h2>
          {list}
        </section>
        <div className="right-col">
          <section className="panel preview-panel">
            <h2 className="panel-title">Live preview</h2>
            <PreviewForm name={name} fields={fields} />
          </section>
          {/* CHANGE: JSON panel */}
          <section className="panel json-panel">
            <h2 className="panel-title">Form definition (JSON)</h2>
            <pre className="json" aria-label="Form definition JSON" tabIndex={0}>{json}</pre>
          </section>
        </div>
      </div>
    </div>
  );
}
