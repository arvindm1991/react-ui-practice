// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /directory/staff -> Staff[]
export async function getStaff(): Promise<Staff[]> {
  await randomDelay(400, 1300);
  return STAFF.map((s) => {
    return { ...s };
  });
}

import React from 'react';
import type { Staff } from '../api';

function initials(name: string): string {
  const parts = name.replace('Dr. ', '').split(' ');
  return parts[0][0] + parts[parts.length - 1][0];
}

// Already built: one person in the directory.
export function StaffCard({ person }: { person: Staff }) {
  let status = null;
  if (person.onCall) {
    status = <span className="on-call">On call</span>;
  }
  const avatarText = initials(person.name);
  return (
    <li className="staff-card">
      <div className="staff-avatar">{avatarText}</div>
      <div className="staff-info">
        <p className="staff-name">{person.name}</p>
        <p className="staff-role">{person.role} · {person.department}</p>
      </div>
      <div className="staff-side">
        {status}
        <span className="ext">ext. {person.ext}</span>
      </div>
    </li>
  );
}

import React, { useEffect, useState } from 'react';
import { getStaff } from './api';
import type { Staff } from './api';
import { StaffCard } from './components/StaffCard';

function TopBar() {
  return (
    <header className="topbar">
      <span className="cross">+</span>
      <span className="org">Cedar Health</span>
      <span className="unit">North Campus</span>
    </header>
  );
}

function Directory() {
  const [staff, setStaff] = useState<Staff[]>([]);
  const [loading, setLoading] = useState(true);
  const [query, setQuery] = useState('');
  const [onCallOnly, setOnCallOnly] = useState(false);

  useEffect(() => {
    let cancelled = false;
    getStaff().then((data) => {
      if (cancelled) {
        return;
      }
      setStaff(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  function handleClear() {
    setQuery('');
  }

  function handleToggleOnCall() {
    setOnCallOnly((prev) => {
      return !prev;
    });
  }

  // derived: never stored in state
  const q = query.trim().toLowerCase();
  const visible = staff.filter((person) => {
    if (onCallOnly && !person.onCall) {
      return false;
    }
    return person.name.toLowerCase().includes(q) || person.role.toLowerCase().includes(q);
  });

  let chipCls = 'chip';
  if (onCallOnly) {
    chipCls = 'chip is-on';
  }

  let body;
  if (loading) {
    body = <p className="message">Loading care team…</p>;
  } else if (visible.length === 0) {
    body = (
      <div className="message">
        <p>No matches for “{query}”.</p>
        <button type="button" className="link-btn" onClick={() => { handleClear(); }}>Clear search</button>
      </div>
    );
  } else {
    body = (
      <ul className="staff-list">
        {visible.map((person) => {
          return <StaffCard key={person.id} person={person} />;
        })}
      </ul>
    );
  }

  let count = null;
  if (!loading) {
    count = <p className="count">Showing {visible.length} of {staff.length}</p>;
  }

  return (
    <main className="directory">
      <div className="dir-head">
        <h1>Care team directory</h1>
        <p className="sub">Find a colleague by name or role.</p>
      </div>
      <div className="toolbar">
        <input
          className="search"
          type="search"
          placeholder="Search name or role"
          value={query}
          onChange={(e) => { setQuery(e.target.value); }}
        />
        <button type="button" className={chipCls} aria-pressed={onCallOnly} onClick={() => { handleToggleOnCall(); }}>
          On call now
        </button>
      </div>
      {count}
      {body}
    </main>
  );
}

export default function App() {
  return (
    <div className="app">
      <TopBar />
      <Directory />
    </div>
  );
}
