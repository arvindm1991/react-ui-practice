@@@ solution src/App.tsx
import React, { useState } from 'react';
import { FAQS } from './data/faqs';
import type { Faq } from './data/faqs';
import { ChevronIcon } from './components/ChevronIcon';

type Plan = {
  id: string;
  name: string;
  price: string;
  blurb: string;
  featured: boolean;
};

type Vote = 'yes' | 'no';

const PLANS: Plan[] = [
  { id: 'hobby', name: 'Hobby', price: '$0', blurb: 'Personal projects', featured: false },
  { id: 'pro', name: 'Pro', price: '$20', blurb: 'Small teams shipping daily', featured: true },
  { id: 'scale', name: 'Scale', price: '$90', blurb: 'SLAs and SSO', featured: false },
];

function Hero() {
  return (
    <header className="hero">
      <p className="kicker">Skyport pricing</p>
      <h1>Deploy on push. Pay for what you ship.</h1>
    </header>
  );
}

function PlanCards() {
  return (
    <div className="plans">
      {PLANS.map((plan) => {
        let cls = 'plan';
        if (plan.featured) {
          cls = 'plan is-featured';
        }
        return (
          <div key={plan.id} className={cls}>
            <p className="plan-name">{plan.name}</p>
            <p className="plan-price">{plan.price}<span>/mo</span></p>
            <p className="plan-blurb">{plan.blurb}</p>
          </div>
        );
      })}
    </div>
  );
}

type FaqItemProps = {
  faq: Faq;
  isOpen: boolean;
  vote: Vote | undefined;
  onToggle: (id: string) => void;
  onVote: (id: string, vote: Vote) => void;
};

function FaqItem({ faq, isOpen, vote, onToggle, onVote }: FaqItemProps) {
  let cls = 'faq-item';
  let answer = null;

  if (isOpen) {
    cls = 'faq-item is-open';

    // Change: once voted, the buttons are replaced by a thank-you line.
    let feedback = (
      <div className="feedback">
        <span>Was this helpful?</span>
        <button type="button" className="vote-btn" onClick={() => { onVote(faq.id, 'yes'); }}>Yes</button>
        <button type="button" className="vote-btn" onClick={() => { onVote(faq.id, 'no'); }}>No</button>
      </div>
    );
    if (vote) {
      feedback = <p className="feedback thanks">Thanks for the feedback!</p>;
    }

    answer = (
      <div className="faq-answer">
        <p>{faq.answer}</p>
        {feedback}
      </div>
    );
  }

  return (
    <li className={cls}>
      <button type="button" className="faq-question" aria-expanded={isOpen} onClick={() => { onToggle(faq.id); }}>
        <span>{faq.question}</span>
        <ChevronIcon />
      </button>
      {answer}
    </li>
  );
}

function FaqSection() {
  // One id (or null), not a boolean per item: that is what makes "one open at a time" automatic.
  const [openId, setOpenId] = useState<string | null>(null);
  // Change: votes by question id. Lives here so it survives closing an item.
  const [votes, setVotes] = useState<Record<string, Vote>>({});

  function handleToggle(id: string) {
    setOpenId((prev) => {
      if (prev === id) {
        return null;
      }
      return id;
    });
  }

  function handleVote(id: string, vote: Vote) {
    setVotes((prev) => {
      return { ...prev, [id]: vote };
    });
  }

  return (
    <section className="faq">
      <h2>Frequently asked questions</h2>
      <ul className="faq-list">
        {FAQS.map((faq) => {
          const isOpen = faq.id === openId;
          const vote = votes[faq.id];
          return (
            <FaqItem
              key={faq.id}
              faq={faq}
              isOpen={isOpen}
              vote={vote}
              onToggle={handleToggle}
              onVote={handleVote}
            />
          );
        })}
      </ul>
    </section>
  );
}

export default function App() {
  return (
    <div className="page">
      <Hero />
      <PlanCards />
      <FaqSection />
    </div>
  );
}
