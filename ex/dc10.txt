import React, { useState } from 'react';
import { ORDER, LABELS } from './data/order';
import type { Order } from './data/order';

// CORE
const STAR_VALUES = [1, 2, 3, 4, 5];
// STRETCH
const PROBLEMS = ['Late', 'Cold food', 'Missing item', 'Wrong order'];

function OrderSummary({ order }: { order: Order }) {
  return (
    <div className="order">
      <div className="resto-logo">LS</div>
      <div>
        <p className="resto">{order.restaurant}</p>
        <p className="delivered">{order.deliveredAt} · #{order.id}</p>
        <ul className="items">
          {order.items.map((item) => {
            return <li key={item.id}>{item.qty}× {item.name}</li>;
          })}
        </ul>
      </div>
    </div>
  );
}

// CORE: a controlled component: value + onChange
type StarRatingProps = {
  value: number;
  onChange: (value: number) => void;
};

// value lives in the form (it's submitted). hover is local: only the stars care.
function StarRating({ value, onChange }: StarRatingProps) {
  // NEXT: hover preview is local state
  const [hover, setHover] = useState(0);

  // CORE: what to show right now (NEXT: the hover override)
  let shown = value;
  if (hover > 0) {
    shown = hover;
  }

  // CORE: label word from LABELS
  let label = 'Tap a star to rate';
  let labelCls = 'rating-label is-empty';
  if (shown > 0) {
    label = LABELS[shown];
    labelCls = 'rating-label';
  }

  // NEXT: Clear (and the aria-labels below)
  let clearBtn = null;
  if (value > 0) {
    clearBtn = <button type="button" className="clear-btn" onClick={() => { onChange(0); }}>Clear</button>;
  }

  return (
    <div className="rating">
      <div className="stars" onMouseLeave={() => { setHover(0); }}>
        {STAR_VALUES.map((n) => {
          let cls = 'star';
          if (n <= shown) {
            cls = 'star is-on';
          }
          const aria = 'Rate ' + n + ' out of 5';
          return (
            <button
              key={n}
              type="button"
              className={cls}
              aria-label={aria}
              onMouseEnter={() => { setHover(n); }}
              onClick={() => { onChange(n); }}
            >
              ★
            </button>
          );
        })}
      </div>
      <p className={labelCls}>{label}</p>
      {clearBtn}
    </div>
  );
}

type ProblemChipsProps = {
  selected: string[];
  onToggle: (problem: string) => void;
};

// STRETCH: multi-select chips for low ratings.
function ProblemChips({ selected, onToggle }: ProblemChipsProps) {
  return (
    <div className="problems">
      <p className="label">What went wrong?</p>
      <div className="chips">
        {PROBLEMS.map((p) => {
          const isOn = selected.includes(p);
          let cls = 'chip';
          if (isOn) {
            cls = 'chip is-on';
          }
          return (
            <button key={p} type="button" className={cls} aria-pressed={isOn} onClick={() => { onToggle(p); }}>{p}</button>
          );
        })}
      </div>
    </div>
  );
}

// CHANGE: onSubmit now reports the rating up
function ReviewForm({ order, onSubmit }: { order: Order; onSubmit: (rating: number) => void }) {
  const [comment, setComment] = useState('');
  // CORE: the rating is submitted, so it lives in the form
  const [rating, setRating] = useState(0);
  // STRETCH
  const [problems, setProblems] = useState<string[]>([]);

  function handleSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    onSubmit(rating);
  }

  // STRETCH
  function handleToggleProblem(problem: string) {
    setProblems((prev) => {
      if (prev.includes(problem)) {
        return prev.filter((p) => {
          return p !== problem;
        });
      }
      return [...prev, problem];
    });
  }

  // CORE: submit gate, derived (STRETCH: chips required for 1-2 stars)
  const isLow = rating > 0 && rating <= 2;
  let canSubmit = rating > 0;
  if (isLow && problems.length === 0) {
    canSubmit = false;
  }

  let problemsBlock = null;
  if (isLow) {
    problemsBlock = <ProblemChips selected={problems} onToggle={handleToggleProblem} />;
  }

  return (
    <form className="card" onSubmit={handleSubmit}>
      <OrderSummary order={order} />
      <div className="rate-block">
        <h1>How was your order?</h1>
        <StarRating value={rating} onChange={setRating} />
      </div>
      {problemsBlock}
      <label className="label" htmlFor="comment">Tell us more (optional)</label>
      <textarea id="comment" className="comment" rows={3} placeholder="Food, packaging, delivery…" value={comment} onChange={(e) => { setComment(e.target.value); }} />
      <button type="submit" className="submit" disabled={!canSubmit}>Submit review</button>
    </form>
  );
}

// CHANGE: shows the rating that was submitted
function ThankYou({ rating }: { rating: number }) {
  const stars = '★'.repeat(rating);
  return (
    <div className="card thanks">
      <div className="thanks-icon">✓</div>
      <h1>Thanks for the feedback</h1>
      <p className="thanks-rating">You rated it {stars} {LABELS[rating]}</p>
      <p>It helps {ORDER.restaurant} and couriers like {ORDER.courier} do better.</p>
    </div>
  );
}

export default function App() {
  const [submitted, setSubmitted] = useState(false);
  // CHANGE: the parent keeps the submitted rating
  const [finalRating, setFinalRating] = useState(0);

  function handleSubmit(rating: number) {
    setFinalRating(rating);
    setSubmitted(true);
  }

  let content = <ReviewForm order={ORDER} onSubmit={handleSubmit} />;
  if (submitted) {
    content = <ThankYou rating={finalRating} />;
  }

  return (
    <div className="page">
      <header className="topbar">forkful</header>
      {content}
    </div>
  );
}
