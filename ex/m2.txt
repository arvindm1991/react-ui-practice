const wait = (ms) => new Promise((r) => setTimeout(r, ms));

// GET /members -> Member[]
// Member: { id, name, email, role: 'owner'|'admin'|'member'|'viewer',
//           status: 'active'|'invited', joinedAt: 'YYYY-MM-DD' | null }
export async function getMembers() {
  await wait(600);
  return members.map((m) => ({ ...m }));
}

// POST /invites -> the new Member (status 'invited', name null, joinedAt null)
// Rejects with an Error if the email is already on the team.
export async function inviteMember(email, role) {
  await wait(500);
  if (members.some((m) => m.email.toLowerCase() === email.toLowerCase())) {
    throw new Error('Someone with that email is already on the team.');
  }
  const member = { id: 'u' + Date.now(), name: null, email, role, status: 'invited', joinedAt: null };
  members = [...members, member];
  return { ...member };
}

// PATCH /members/:id
export async function updateRole(id, role) {
  await wait(300);
  members = members.map((m) => (m.id === id ? { ...m, role } : m));
}

// DELETE /members/:id (also cancels an invite)
export async function removeMember(id) {
  await wait(300);
  members = members.filter((m) => m.id !== id);
}

import React from 'react';

const COLORS = ['#3451D1', '#1C7A43', '#B54708', '#7A2E9E', '#0E7490', '#B42318'];

function initialsFor(name, email) {
  const source = name || email;
  const parts = source.split(/[\s@.]+/).filter(Boolean);
  return ((parts[0] || '')[0] + ((parts[1] || '')[0] || '')).toUpperCase();
}

export default function Avatar({ name, email, size = 32 }) {
  const key = (name || email).length % COLORS.length;
  return (
    <span
      className="avatar"
      style={{ width: size, height: size, fontSize: size * 0.4, background: COLORS[key] }}
      aria-hidden="true"
    >
      {initialsFor(name, email)}
    </span>
  );
}

export function formatDate(isoDate) {
  if (!isoDate) return '';
  return new Date(isoDate + 'T00:00:00').toLocaleDateString('en-US', {
    month: 'short',
    day: 'numeric',
    year: 'numeric',
  });
}

export function capitalize(s) {
  return s.charAt(0).toUpperCase() + s.slice(1);
}

import React, { useEffect, useState } from 'react';
import { getMembers, inviteMember, updateRole, removeMember } from '../api';
import Avatar from '../components/Avatar';
import Button from '../components/Button';
import { formatDate, capitalize } from '../utils/format';

const ROLES = ['admin', 'member', 'viewer'];
const FILTERS = ['all', ...ROLES];
const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

// Includes the mid-session change (pending invites).
export default function MembersPage() {
  const [members, setMembers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [filter, setFilter] = useState('all');

  useEffect(() => {
    getMembers().then((data) => {
      setMembers(data);
      setLoading(false);
    });
  }, []);

  // Optimistic updates: change the UI now, call the API in the background.
  function handleRoleChange(id, role) {
    setMembers((prev) => prev.map((m) => (m.id === id ? { ...m, role } : m)));
    updateRole(id, role);
  }

  function handleRemove(id) {
    setMembers((prev) => prev.filter((m) => m.id !== id));
    removeMember(id);
  }

  function handleInvited(member) {
    setMembers((prev) => [...prev, member]);
  }

  // ---- derived data ----
  const owner = members.find((m) => m.role === 'owner');
  const others = members.filter((m) => m.role !== 'owner');
  const ordered = owner ? [owner, ...others] : others;
  const visible = filter === 'all' ? ordered : ordered.filter((m) => m.role === filter);
  const countFor = (f) => (f === 'all' ? members.length : members.filter((m) => m.role === f).length);
  const pendingCount = members.filter((m) => m.status === 'invited').length;

  return (
    <div className="members">
      <header className="members-header">
        <h1>Members</h1>
        <p className="members-sub">
          {members.length} {members.length === 1 ? 'member' : 'members'}
          {pendingCount > 0 && ` · ${pendingCount} pending ${pendingCount === 1 ? 'invite' : 'invites'}`}
        </p>
      </header>

      <InviteForm onInvited={handleInvited} />

      <div className="chips" role="tablist">
        {FILTERS.map((f) => (
          <button
            key={f}
            role="tab"
            aria-selected={filter === f}
            className={'chip' + (filter === f ? ' is-active' : '')}
            onClick={() => setFilter(f)}
          >
            {f === 'all' ? 'All' : capitalize(f) + 's'}
            <span className="chip-count">{countFor(f)}</span>
          </button>
        ))}
      </div>

      <div className="table-card">
        <table className="table">
          <thead>
            <tr>
              <th>Member</th>
              <th>Role</th>
              <th>Joined</th>
              <th aria-label="Actions" />
            </tr>
          </thead>
          <tbody>
            {loading && (
              <tr>
                <td colSpan={4} className="table-message">Loading members…</td>
              </tr>
            )}
            {!loading && visible.length === 0 && (
              <tr>
                <td colSpan={4} className="table-message">No {filter}s yet.</td>
              </tr>
            )}
            {visible.map((m) => (
              <MemberRow key={m.id} member={m} onRoleChange={handleRoleChange} onRemove={handleRemove} />
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
}

function MemberRow({ member, onRoleChange, onRemove }) {
  const isOwner = member.role === 'owner';
  const isInvited = member.status === 'invited';

  return (
    <tr>
      <td>
        <div className="member-cell">
          <Avatar name={member.name} email={member.email} />
          <div className="member-text">
            <span className="member-name">{member.name || member.email}</span>
            {member.name && <span className="member-email">{member.email}</span>}
          </div>
        </div>
      </td>
      <td>
        {isOwner ? (
          <span className="role-static">Owner</span>
        ) : (
          <select
            className="role-select"
            value={member.role}
            onChange={(e) => onRoleChange(member.id, e.target.value)}
            aria-label={`Role for ${member.name || member.email}`}
          >
            {ROLES.map((r) => (
              <option key={r} value={r}>{capitalize(r)}</option>
            ))}
          </select>
        )}
      </td>
      <td className="muted">
        {isInvited ? <span className="badge-invited">Invited</span> : formatDate(member.joinedAt)}
      </td>
      <td className="actions">
        {!isOwner && (
          <Button variant="ghost" onClick={() => onRemove(member.id)}>
            {isInvited ? 'Cancel invite' : 'Remove'}
          </Button>
        )}
      </td>
    </tr>
  );
}

function InviteForm({ onInvited }) {
  const [email, setEmail] = useState('');
  const [role, setRole] = useState('member');
  const [error, setError] = useState('');
  const [submitting, setSubmitting] = useState(false);

  async function handleSubmit(e) {
    e.preventDefault();
    const value = email.trim();
    if (!EMAIL_RE.test(value)) {
      setError('Enter a valid email address.');
      return;
    }
    setSubmitting(true);
    try {
      const member = await inviteMember(value, role);
      onInvited(member);
      setEmail('');
      setRole('member');
    } catch (err) {
      setError(err.message);
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <form className="invite" onSubmit={handleSubmit} noValidate>
      <div className="invite-row">
        <input
          className={'input' + (error ? ' has-error' : '')}
          type="email"
          placeholder="name@company.com"
          value={email}
          onChange={(e) => {
            setEmail(e.target.value);
            setError('');
          }}
        />
        <select className="role-select" value={role} onChange={(e) => setRole(e.target.value)}>
          {ROLES.map((r) => (
            <option key={r} value={r}>{capitalize(r)}</option>
          ))}
        </select>
        <Button variant="primary" type="submit" disabled={submitting}>
          {submitting ? 'Inviting…' : 'Invite'}
        </Button>
      </div>
      {error && <p className="field-error">{error}</p>}
    </form>
  );
}
