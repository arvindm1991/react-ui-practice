// Fake API. Treat it like a real backend you can't change.
//
// getUsage(range) where range is '7d' | '30d'
// resolves to:
// {
//   range: '7d',
//   models: [{ id: 'opus', name: 'Claude Opus' }, ...],
//   days: [
//     {
//       date: '2026-09-28',
//       usage: {
//         opus:   { tokens: 812000, requests: 140 },
//         sonnet: { tokens: ...,    requests: ... },
//         haiku:  { tokens: ...,    requests: ... },
//       },
//     },
//     ... one entry per day, oldest first
//   ],
// }

const MODELS = [
  { id: 'opus', name: 'Claude Opus', base: 420000, perRequest: 5200 },
  { id: 'sonnet', name: 'Claude Sonnet', base: 980000, perRequest: 3100 },
  { id: 'haiku', name: 'Claude Haiku', base: 610000, perRequest: 1400 },
];

function seededRandom(seed) {
  let s = seed;
  return () => {
    s = (s * 16807) % 2147483647;
    return s / 2147483647;
  };
}

function isoDay(date) {
  return date.toISOString().slice(0, 10);
}

export function getUsage(range = '7d') {
  const count = range === '30d' ? 30 : 7;
  const rand = seededRandom(count === 30 ? 7 : 3);
  const today = new Date();
  const days = [];
  for (let i = count - 1; i >= 0; i--) {
    const d = new Date(today);
    d.setDate(today.getDate() - i);
    const weekend = d.getDay() === 0 || d.getDay() === 6;
    const usage = {};
    MODELS.forEach((m) => {
      const tokens = Math.round(m.base * (weekend ? 0.45 : 1) * (0.65 + rand() * 0.7));
      usage[m.id] = { tokens, requests: Math.max(1, Math.round(tokens / m.perRequest)) };
    });
    days.push({ date: isoDay(d), usage });
  }
  const result = { range, models: MODELS.map(({ id, name }) => ({ id, name })), days };
  return new Promise((resolve) => setTimeout(() => resolve(result), 650));
}
@@@ shared src/utils/format.js
const compact = new Intl.NumberFormat('en-US', { notation: 'compact', maximumFractionDigits: 1 });
const whole = new Intl.NumberFormat('en-US');

// 1234567 -> "1.2M"
export function formatCompact(n) {
  return compact.format(n);
}

// 1234567 -> "1,234,567"
export function formatNumber(n) {
  return whole.format(Math.round(n));
}

// '2026-09-28' -> "Sep 28"
export function formatDay(isoDay) {
  return new Date(isoDay + 'T00:00:00').toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
}

// 0.4567 -> "45.7%"
export function formatPercent(fraction) {
  return (fraction * 100).toFixed(1) + '%';
}

import React, { useEffect, useState } from 'react';
import { getUsage } from '../api';
import { formatCompact, formatNumber, formatDay, formatPercent } from '../utils/format';

const RANGES = [
  { id: '7d', label: '7 days' },
  { id: '30d', label: '30 days' },
];

// Includes the mid-session change (click a model to filter).
export default function UsagePage() {
  const [range, setRange] = useState('7d');
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [modelId, setModelId] = useState(null); // null = all models

  useEffect(() => {
    let cancelled = false;
    setLoading(true);
    getUsage(range).then((result) => {
      if (cancelled) return;
      setData(result);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, [range]);

  return (
    <main className="usage">
      <header className="usage-header">
        <div>
          <h1>Usage</h1>
          <p className="usage-sub">
            {modelId && data
              ? `Showing ${data.models.find((m) => m.id === modelId).name} only`
              : 'All models'}
          </p>
        </div>
        <div className="segmented" role="tablist" aria-label="Date range">
          {RANGES.map((r) => (
            <button
              key={r.id}
              role="tab"
              aria-selected={range === r.id}
              className={'segment' + (range === r.id ? ' is-active' : '')}
              onClick={() => setRange(r.id)}
            >
              {r.label}
            </button>
          ))}
        </div>
      </header>

      {!data ? (
        <div className="panel panel-loading">Loading usage…</div>
      ) : (
        <Dashboard
          data={data}
          modelId={modelId}
          onModelClick={(id) => setModelId((cur) => (cur === id ? null : id))}
          dimmed={loading}
        />
      )}
    </main>
  );
}

// Pure helpers keep the component body readable.
function totalsForDay(day, modelId) {
  const entries = Object.entries(day.usage).filter(([id]) => !modelId || id === modelId);
  return entries.reduce(
    (acc, [, u]) => ({ tokens: acc.tokens + u.tokens, requests: acc.requests + u.requests }),
    { tokens: 0, requests: 0 }
  );
}

function Dashboard({ data, modelId, onModelClick, dimmed }) {
  const daily = data.days.map((d) => ({ date: d.date, ...totalsForDay(d, modelId) }));
  const totalTokens = daily.reduce((s, d) => s + d.tokens, 0);
  const totalRequests = daily.reduce((s, d) => s + d.requests, 0);
  const maxTokens = Math.max(...daily.map((d) => d.tokens), 1);

  // Breakdown always shows all models so you can switch between them.
  const allTokens = data.days.reduce((s, d) => s + totalsForDay(d, null).tokens, 0);
  const byModel = data.models
    .map((m) => ({
      ...m,
      tokens: data.days.reduce((s, d) => s + d.usage[m.id].tokens, 0),
    }))
    .sort((a, b) => b.tokens - a.tokens);

  return (
    <div className={'dashboard' + (dimmed ? ' is-dimmed' : '')}>
      <section className="stats">
        <StatCard label="Total tokens" value={formatCompact(totalTokens)} detail={formatNumber(totalTokens)} />
        <StatCard label="Requests" value={formatNumber(totalRequests)} />
        <StatCard
          label="Avg tokens / request"
          value={formatNumber(totalRequests ? totalTokens / totalRequests : 0)}
        />
      </section>

      <section className="panel">
        <h2 className="panel-title">Daily tokens</h2>
        <div className="chart" role="img" aria-label="Daily token usage bar chart">
          {daily.map((d) => (
            <div key={d.date} className="bar-col">
              <div className="bar-track">
                <div
                  className="bar"
                  style={{ height: `${(d.tokens / maxTokens) * 100}%` }}
                  title={`${formatDay(d.date)}: ${formatNumber(d.tokens)} tokens`}
                />
              </div>
              {daily.length <= 7 && <span className="bar-label">{formatDay(d.date)}</span>}
            </div>
          ))}
        </div>
        {daily.length > 7 && (
          <div className="chart-range">
            <span>{formatDay(daily[0].date)}</span>
            <span>{formatDay(daily[daily.length - 1].date)}</span>
          </div>
        )}
      </section>

      <section className="panel">
        <h2 className="panel-title">By model</h2>
        <ul className="models">
          {byModel.map((m) => {
            const share = allTokens ? m.tokens / allTokens : 0;
            const active = modelId === m.id;
            return (
              <li key={m.id}>
                <button
                  className={'model-row' + (active ? ' is-active' : '') + (modelId && !active ? ' is-faded' : '')}
                  onClick={() => onModelClick(m.id)}
                  aria-pressed={active}
                >
                  <span className="model-name">{m.name}</span>
                  <span className="model-value">
                    {formatCompact(m.tokens)} · {formatPercent(share)}
                  </span>
                  <span className="model-track">
                    <span className="model-fill" style={{ width: formatPercent(share) }} />
                  </span>
                </button>
              </li>
            );
          })}
        </ul>
      </section>
    </div>
  );
}

function StatCard({ label, value, detail }) {
  return (
    <div className="stat">
      <span className="stat-label">{label}</span>
      <span className="stat-value" title={detail}>{value}</span>
    </div>
  );
}
