  { id: 'A-1044', customer: 'Kestrel & Co', amount: 87.25, status: 'refunded', createdAt: '2026-09-30T09:05:00Z' },
  { id: 'A-1045', customer: 'Atlas Freight', amount: 5120, status: 'paid', createdAt: '2026-10-01T16:22:00Z' },
  { id: 'A-1046', customer: 'Juniper Health', amount: 760, status: 'pending', createdAt: '2026-10-02T11:58:00Z' },
];

export function fetchOrders() {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (Math.random() < FAIL_RATE) {
        reject(new Error('Request failed: 503 Service Unavailable'));
      } else {
        resolve(RETURN_EMPTY ? [] : orders.map((o) => ({ ...o })));
      }
    }, DELAY_MS);
  });
}

import React, { useState, useEffect } from 'react';
import { fetchOrders } from './api';

const money = new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' });
const formatDate = (iso) =>
  new Date(iso).toLocaleDateString('en-US', { month: 'short', day: 'numeric' });

function SkeletonRows({ count = 3 }) {
  return Array.from({ length: count }, (_, i) => (
    <div key={i} className="skeleton-row">
      <div className="skeleton" style={{ width: '18%' }} />
      <div className="skeleton" style={{ width: '40%' }} />
      <div className="skeleton" style={{ width: '22%' }} />
    </div>
  ));
}

export default function App() {
  // One status value is easier to reason about than three booleans.
  const [status, setStatus] = useState('loading'); // 'loading' | 'error' | 'success'
  const [orders, setOrders] = useState([]);
  const [error, setError] = useState('');
  const [attempt, setAttempt] = useState(0); // bump to refetch

  useEffect(() => {
    let cancelled = false; // ignore late responses after unmount / refetch
    setStatus('loading');
    fetchOrders()
      .then((data) => {
        if (cancelled) return;
        setOrders(data);
        setStatus('success');
      })
      .catch((err) => {
        if (cancelled) return;
        setError(err.message);
        setStatus('error');
      });
    return () => {
      cancelled = true;
    };
  }, [attempt]);

  const total = orders.reduce((sum, o) => sum + o.amount, 0);

  let content;
  if (status === 'loading') {
    content = <SkeletonRows />;
  } else if (status === 'error') {
    content = (
      <div className="message error">
        <p>{error}</p>
        <button className="btn" onClick={() => setAttempt((n) => n + 1)}>Try again</button>
      </div>
    );
  } else if (orders.length === 0) {
    content = <div className="message">No orders yet.</div>;
  } else {
    content = (
      <table className="table">
        <thead>
          <tr>
            <th>Order</th>
            <th>Customer</th>
            <th>Date</th>
            <th>Status</th>
            <th className="num">Amount</th>
          </tr>
        </thead>
        <tbody>
          {orders.map((o) => (
            <tr key={o.id}>
              <td>{o.id}</td>
              <td>{o.customer}</td>
              <td>{formatDate(o.createdAt)}</td>
              <td>
                <span className={`status status-${o.status}`}>{o.status}</span>
              </td>
              <td className="num">{money.format(o.amount)}</td>
            </tr>
          ))}
        </tbody>
      </table>
    );
  }

  return (
    <div className="page">
      <header className="page-header">
        <h1>Recent orders</h1>
        {status === 'success' && orders.length > 0 && (
          <span className="total">Total {money.format(total)}</span>
        )}
      </header>
      <div className="card">{content}</div>
    </div>
  );
}
