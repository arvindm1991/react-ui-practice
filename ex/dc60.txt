// "Now" is fixed so the groups are the same every time you open the app.
// Use NOW, never new Date() / Date.now(), for anything relative.
export const NOW = new Date(2026, 5, 15, 14, 0); // Mon, Jun 15 2026, 2:00 pm (months are 0-based)

// Midnight at the start of the same calendar day.
export function startOfDay(d: Date): Date {
  return new Date(d.getFullYear(), d.getMonth(), d.getDate());
}

// Whole calendar days from `earlier` to `later`: daysBetween(Jun 14 23:00, Jun 15 01:00) -> 1
export function daysBetween(earlier: Date, later: Date): number {
  return Math.round((startOfDay(later).getTime() - startOfDay(earlier).getTime()) / 86400000);
}

// "2:05 pm" for today, "Jun 12" otherwise
export function formatWhen(iso: string): string {
  const d = new Date(iso);
  if (daysBetween(d, NOW) === 0) {
    return d.toLocaleTimeString('en-US', { hour: 'numeric', minute: '2-digit' }).toLowerCase();
  }
  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
}

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

function patch(id: string, changes: Partial<Conversation>): Conversation {
  const current = DB.find((c) => c.id === id);
  if (!current) {
    throw new Error('Conversation ' + id + ' not found');
  }
  const next = { ...current, ...changes };
  DB = DB.map((c) => (c.id === id ? next : c));
  return { ...next };
}

// GET /conversations -> Conversation[] (unsorted)
export async function getConversations(): Promise<Conversation[]> {
  await randomDelay(400, 1000);
  return DB.map((c) => ({ ...c }));
}

// POST /conversations -> the new, empty conversation
export async function createConversation(): Promise<Conversation> {
  await randomDelay(300, 700);
  const conv: Conversation = { id: 'c' + nextId, title: 'New chat', updatedAt: NOW.toISOString(), pinned: false };
  nextId += 1;
  DB = [conv, ...DB];
  return { ...conv };
}

// PATCH /conversations/:id { title }
export async function renameConversation(id: string, title: string): Promise<Conversation> {
  await randomDelay(300, 800);
  return patch(id, { title });
}

// PATCH /conversations/:id { pinned }  (does not change updatedAt)
export async function setPinned(id: string, pinned: boolean): Promise<Conversation> {
  await randomDelay(300, 800);
  return patch(id, { pinned });
}

// DELETE /conversations/:id   (permanent: there is no undo on the server)
export async function deleteConversation(id: string): Promise<void> {
  await randomDelay(300, 800);
  DB = DB.filter((c) => c.id !== id);
}

import React, { useEffect, useRef, useState } from 'react';
import { getConversations, createConversation, renameConversation, deleteConversation, setPinned } from './api';
import type { Conversation } from './api';
import { NOW, daysBetween, formatWhen } from './utils/date';
import ChatPane from './components/ChatPane';

const UNDO_MS = 5000;
const GROUP_ORDER = ['Pinned', 'Today', 'Yesterday', 'Previous 7 days', 'Older'];

function groupFor(c: Conversation): string {
  if (c.pinned) {
    return 'Pinned';
  }
  const days = daysBetween(new Date(c.updatedAt), NOW);
  if (days <= 0) {
    return 'Today';
  }
  if (days === 1) {
    return 'Yesterday';
  }
  if (days <= 7) {
    return 'Previous 7 days';
  }
  return 'Older';
}

type RowProps = {
  conversation: Conversation;
  selected: boolean;
  onSelect: (id: string) => void;
  onRename: (id: string, title: string) => void;
  onDelete: (conv: Conversation) => void;
  onTogglePin: (conv: Conversation) => void;
};

function ConversationRow({ conversation, selected, onSelect, onRename, onDelete, onTogglePin }: RowProps) {
  // Edit state is local: nothing outside the row needs it.
  const [editing, setEditing] = useState(false);
  const [draft, setDraft] = useState('');

  function startEdit() {
    setDraft(conversation.title);
    setEditing(true);
  }

  function commit() {
    setEditing(false);
    const title = draft.trim();
    if (title !== '' && title !== conversation.title) {
      onRename(conversation.id, title);
    }
  }

  function handleKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
    if (e.key === 'Enter') {
      commit();
    } else if (e.key === 'Escape') {
      setEditing(false);
    }
  }

  let cls = 'conv';
  if (selected) {
    cls += ' is-selected';
  }

  if (editing) {
    return (
      <li className={cls + ' is-editing'}>
        <input
          className="conv-input"
          aria-label="Chat title"
          autoFocus
          value={draft}
          onFocus={(e) => { e.target.select(); }}
          onChange={(e) => { setDraft(e.target.value); }}
          onKeyDown={handleKeyDown}
          onBlur={() => { commit(); }}
        />
      </li>
    );
  }

  let pinLabel = 'Pin';
  let pinCls = 'conv-action';
  if (conversation.pinned) {
    pinLabel = 'Unpin';
    pinCls = 'conv-action is-pinned';
  }

  return (
    <li className={cls}>
      <button
        type="button"
        className="conv-main"
        aria-current={selected}
        onClick={() => { onSelect(conversation.id); }}
        onDoubleClick={() => { startEdit(); }}
      >
        <span className="conv-title">{conversation.title}</span>
        <span className="conv-when">{formatWhen(conversation.updatedAt)}</span>
      </button>
      <div className="conv-actions">
        <button type="button" className={pinCls} aria-label={pinLabel + ' ' + conversation.title} title={pinLabel} onClick={() => { onTogglePin(conversation); }}>⚲</button>
        <button type="button" className="conv-action" aria-label={'Rename ' + conversation.title} title="Rename" onClick={() => { startEdit(); }}>✎</button>
        <button type="button" className="conv-action" aria-label={'Delete ' + conversation.title} title="Delete" onClick={() => { onDelete(conversation); }}>✕</button>
      </div>
    </li>
  );
}

type PendingDelete = { conv: Conversation; timer: ReturnType<typeof setTimeout> };

export default function App() {
  const [conversations, setConversations] = useState<Conversation[]>([]);
  const [loading, setLoading] = useState(true);
  const [selectedId, setSelectedId] = useState<string | null>(null);
  const [query, setQuery] = useState('');
  const [creating, setCreating] = useState(false);
  // The toast shows this; the ref holds the timer so Undo can cancel it.
  const [pending, setPending] = useState<Conversation | null>(null);
  const pendingRef = useRef<PendingDelete | null>(null);

  useEffect(() => {
    let cancelled = false;
    getConversations().then((data) => {
      if (cancelled) {
        return;
      }
      setConversations(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // Leaving the page with an undo window open: the user meant it, so delete for real.
  useEffect(() => {
    return () => {
      const p = pendingRef.current;
      if (p) {
        clearTimeout(p.timer);
        deleteConversation(p.conv.id);
      }
    };
  }, []);

  function commitPendingDelete() {
    const p = pendingRef.current;
    if (!p) {
      return;
    }
    clearTimeout(p.timer);
    pendingRef.current = null;
    setPending(null);
    deleteConversation(p.conv.id);
  }

  function handleDelete(conv: Conversation, nextSelectedId: string | null) {
    commitPendingDelete(); // one undo at a time: a second delete finalizes the first
    setConversations((prev) => {
      return prev.filter((c) => {
        return c.id !== conv.id;
      });
    });
    if (selectedId === conv.id) {
      setSelectedId(nextSelectedId);
    }
    // Deferred: the API delete only happens if nobody clicks Undo within 5 s.
    const timer = setTimeout(() => {
      commitPendingDelete();
    }, UNDO_MS);
    pendingRef.current = { conv, timer };
    setPending(conv);
  }

  function handleUndo() {
    const p = pendingRef.current;
    if (!p) {
      return;
    }
    clearTimeout(p.timer);
    pendingRef.current = null;
    setPending(null);
    setConversations((prev) => {
      return [...prev, p.conv]; // sorting puts it back in its old place
    });
  }

  function replaceConversation(next: Conversation) {
    setConversations((prev) => {
      return prev.map((c) => {
        if (c.id !== next.id) {
          return c;
        }
        return next;
      });
    });
  }

  function handleRename(id: string, title: string) {
    const before = conversations.find((c) => {
      return c.id === id;
    });
    if (!before) {
      return;
    }
    replaceConversation({ ...before, title }); // optimistic
    renameConversation(id, title).catch(() => {
      replaceConversation(before);
    });
  }

  // Change: pinning moves a chat to the Pinned section without touching its date.
  function handleTogglePin(conv: Conversation) {
    replaceConversation({ ...conv, pinned: !conv.pinned });
    setPinned(conv.id, !conv.pinned).catch(() => {
      replaceConversation(conv);
    });
  }

  function handleNewChat() {
    setCreating(true);
    createConversation().then((conv) => {
      setConversations((prev) => {
        return [conv, ...prev];
      });
      setSelectedId(conv.id);
      setQuery('');
      setCreating(false);
    });
  }

  // derived: search -> sort -> group (never stored)
  const q = query.trim().toLowerCase();
  const sorted = conversations
    .filter((c) => {
      return c.title.toLowerCase().includes(q);
    })
    .sort((a, b) => {
      return new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime();
    });
  const groups: Record<string, Conversation[]> = {};
  sorted.forEach((c) => {
    const g = groupFor(c);
    if (!groups[g]) {
      groups[g] = [];
    }
    groups[g].push(c);
  });
  // Display order, used to pick a neighbour when the selected chat is deleted.
  const ordered: Conversation[] = [];
  GROUP_ORDER.forEach((g) => {
    (groups[g] || []).forEach((c) => {
      ordered.push(c);
    });
  });

  function deleteWithNeighbour(conv: Conversation) {
    const i = ordered.findIndex((c) => {
      return c.id === conv.id;
    });
    const neighbour = ordered[i + 1] || ordered[i - 1] || null;
    let nextId: string | null = null;
    if (neighbour) {
      nextId = neighbour.id;
    }
    handleDelete(conv, nextId);
  }

  let list;
  if (loading) {
    list = <p className="side-msg">Loading chats…</p>;
  } else if (sorted.length === 0 && q !== '') {
    list = <p className="side-msg">No chats match “{query.trim()}”</p>;
  } else {
    list = GROUP_ORDER.map((g) => {
      const items = groups[g];
      if (!items) {
        return null;
      }
      return (
        <section key={g} className="conv-group" aria-label={g}>
          <h3 className="group-title">{g}</h3>
          <ul className="conv-list">
            {items.map((c) => {
              return (
                <ConversationRow
                  key={c.id}
                  conversation={c}
                  selected={c.id === selectedId}
                  onSelect={setSelectedId}
                  onRename={handleRename}
                  onDelete={deleteWithNeighbour}
                  onTogglePin={handleTogglePin}
                />
              );
            })}
          </ul>
        </section>
      );
    });
  }

  let toast = null;
  if (pending) {
    toast = (
      <div className="undo-toast" role="status" key={pending.id}>
        <span className="undo-text">Chat deleted</span>
        <button type="button" className="undo-btn" onClick={() => { handleUndo(); }}>Undo</button>
        <span className="undo-timer" aria-hidden="true" />
      </div>
    );
  }

  const selected = conversations.find((c) => {
    return c.id === selectedId;
  }) || null;

  let newLabel = '+ New chat';
  if (creating) {
    newLabel = 'Creating…';
  }

  return (
    <div className="app">
      <aside className="sidebar">
        <div className="side-top">
          <span className="brand">◎ Lumen</span>
          <button type="button" className="new-chat" disabled={creating} onClick={() => { handleNewChat(); }}>{newLabel}</button>
        </div>
        <input
          className="side-search"
          type="search"
          placeholder="Search chats"
          aria-label="Search chats"
          value={query}
          onChange={(e) => { setQuery(e.target.value); }}
        />
        <nav className="side-scroll" aria-label="Chats">
          {list}
        </nav>
        {toast}
      </aside>
      <ChatPane conversation={selected} />
    </div>
  );
}
