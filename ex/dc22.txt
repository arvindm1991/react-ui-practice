export const MAX_TAGS = 5;

// cleanTag("  #React  ") -> "React", cleanTag("design,") -> "design", cleanTag("   ") -> ""
// Trims, drops commas and a leading "#", and collapses inner spaces.
export function cleanTag(raw: string): string {
  return raw
    .replace(/,/g, '')
    .replace(/^#/, '')
    .replace(/\s+/g, ' ')
    .trim();
}

import React from 'react';

type ChipProps = {
  label: string;
  onRemove?: () => void; // when given, the chip shows a small × button
};

export default function Chip({ label, onRemove }: ChipProps) {
  let removeBtn = null;
  if (onRemove) {
    removeBtn = (
      <button type="button" className="chip-x" aria-label={'Remove ' + label} onClick={() => { onRemove(); }}>
        ×
      </button>
    );
  }
  return (
    <span className="chip">
      {label}
      {removeBtn}
    </span>
  );
}

import React, { useState } from 'react';
import Chip from './components/Chip';
import { cleanTag, MAX_TAGS } from './utils/tags';

function TopBar() {
  return (
    <header className="topbar">
      <span className="logo">Inkwell</span>
      <span className="draft-state">Draft · Saved</span>
      <button type="button" className="publish-btn">Publish</button>
    </header>
  );
}

type EditorProps = {
  title: string;
  body: string;
  onTitleChange: (value: string) => void;
  onBodyChange: (value: string) => void;
};

function Editor({ title, body, onTitleChange, onBodyChange }: EditorProps) {
  return (
    <article className="editor">
      <input
        className="title-input"
        placeholder="Title"
        value={title}
        onChange={(e) => { onTitleChange(e.target.value); }}
      />
      <textarea
        className="body-input"
        placeholder="Tell your story…"
        value={body}
        onChange={(e) => { onBodyChange(e.target.value); }}
      />
    </article>
  );
}

type TagFieldProps = {
  tags: string[];
  onChange: (tags: string[]) => void;
};

// The tags belong to the post (App). Only the half-typed text is local.
function TagField({ tags, onChange }: TagFieldProps) {
  const [draft, setDraft] = useState('');
  // NEXT: the error message (duplicates, limit)
  const [message, setMessage] = useState('');

  // CORE: clean the draft, ignore empty, append, clear the input
  function addTag() {
    const tag = cleanTag(draft);
    if (tag === '') {
      setDraft('');
      return;
    }
    // STRETCH: limit of 5
    if (tags.length >= MAX_TAGS) {
      setMessage('You can add up to ' + MAX_TAGS + ' tags.');
      return;
    }
    // NEXT: case-insensitive duplicates
    const exists = tags.some((t) => {
      return t.toLowerCase() === tag.toLowerCase();
    });
    if (exists) {
      setMessage('"' + tag + '" is already added.');
      setDraft('');
      return;
    }
    // CORE: add
    onChange([...tags, tag]);
    setDraft('');
    setMessage('');
  }

  // CORE: × removes one chip
  function removeTag(tag: string) {
    onChange(tags.filter((t) => {
      return t !== tag;
    }));
    setMessage('');
  }

  // CORE: Enter or comma adds
  function handleKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
    if (e.key === 'Enter' || e.key === ',') {
      e.preventDefault(); // stop the comma from being typed
      addTag();
      return;
    }
    // NEXT: Backspace in an empty input removes the last chip
    if (e.key === 'Backspace' && draft === '' && tags.length > 0) {
      onChange(tags.slice(0, -1));
      setMessage('');
    }
  }

  // STRETCH: counter help line (NEXT: the message replaces it)
  let help = <p className="field-help">Press Enter or comma to add · {tags.length} / {MAX_TAGS}</p>;
  if (message !== '') {
    help = <p className="field-help is-error" role="alert">{message}</p>;
  }

  // CORE: hide the placeholder once there are chips
  let placeholder = 'Add a tag…';
  if (tags.length > 0) {
    placeholder = '';
  }

  // CORE: chips before the input
  return (
    <div className="field">
      <label className="field-label" htmlFor="tags">Tags</label>
      <div className="tag-box">
        {tags.map((tag) => {
          // tags are unique (case-insensitive), so the text is a stable key
          return <Chip key={tag} label={tag} onRemove={() => { removeTag(tag); }} />;
        })}
        <input
          id="tags"
          className="tag-input"
          placeholder={placeholder}
          value={draft}
          onChange={(e) => { setDraft(e.target.value); }}
          onKeyDown={handleKeyDown}
        />
      </div>
      {help}
    </div>
  );
}

function PostSettings({ tags, onTagsChange }: { tags: string[]; onTagsChange: (tags: string[]) => void }) {
  return (
    <aside className="settings">
      <h2>Post settings</h2>
      <div className="field">
        <span className="field-label">Publication</span>
        <div><Chip label="Engineering Notes" /></div>
      </div>
      <TagField tags={tags} onChange={onTagsChange} />
      <div className="field">
        <span className="field-label">Visibility</span>
        <p className="field-value">Public · appears on your profile</p>
      </div>
    </aside>
  );
}

export default function App() {
  const [title, setTitle] = useState('What I learned shipping a design system');
  const [tags, setTags] = useState<string[]>([]);
  const [body, setBody] = useState('Six months ago our product had four different button styles and no shared tokens. This is the story of how we fixed it without stopping feature work.');

    <div className="app">
      <TopBar />
      <div className="workspace">
