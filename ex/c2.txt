import React, { useEffect, useRef, useState } from 'react';
import { agents } from './data';

// Each column knows how to read and compare its own values.
const COLUMNS = [
  { key: 'name', label: 'Name', compare: (a, b) => a.name.localeCompare(b.name, undefined, { sensitivity: 'base' }) },
  { key: 'owner', label: 'Owner', compare: (a, b) => a.owner.localeCompare(b.owner) },
  // Lookup: "javascript natural sort strings with numbers" -> localeCompare with { numeric: true }
  { key: 'version', label: 'Version', compare: (a, b) => a.version.localeCompare(b.version, undefined, { numeric: true }) },
  { key: 'lastRun', label: 'Last run', compare: (a, b) => new Date(a.lastRun) - new Date(b.lastRun) },
  { key: 'successRate', label: 'Success', numeric: true, compare: (a, b) => a.successRate - b.successRate },
];

const pct = new Intl.NumberFormat('en-US', { style: 'percent', minimumFractionDigits: 1 });
const when = (iso) => new Date(iso).toLocaleString('en-US', { month: 'short', day: 'numeric', hour: 'numeric', minute: '2-digit' });

// Lookup: "indeterminate checkbox react" -> it's a DOM property, not an attribute; set it via a ref
function IndeterminateCheckbox({ checked, indeterminate, ...props }) {
  const ref = useRef(null);
  useEffect(() => {
    if (ref.current) ref.current.indeterminate = indeterminate;
  }, [indeterminate]);
  return <input ref={ref} type="checkbox" checked={checked} {...props} />;
}

export default function App() {
  const [sort, setSort] = useState({ key: 'name', dir: 'asc' });
  const [selected, setSelected] = useState(() => new Set());

  // derived: sorted copy
  const column = COLUMNS.find((c) => c.key === sort.key);
  const rows = [...agents].sort((a, b) => (sort.dir === 'asc' ? 1 : -1) * column.compare(a, b));

  const allChecked = selected.size === agents.length;
  const someChecked = selected.size > 0 && !allChecked;

  function toggleSort(key) {
    setSort((s) => (s.key === key ? { key, dir: s.dir === 'asc' ? 'desc' : 'asc' } : { key, dir: 'asc' }));
  }

  // Sets are fine in state as long as you create a new Set on every change
  function toggleRow(id) {
    setSelected((prev) => {
      const next = new Set(prev);
      next.has(id) ? next.delete(id) : next.add(id);
      return next;
    });
  }

  function toggleAll() {
    setSelected(allChecked ? new Set() : new Set(agents.map((a) => a.id)));
  }

  return (
    <div className="page">
      <h1>Agents</h1>

      {selected.size > 0 && (
        <div className="selection-bar" role="status">
          <span>{selected.size} selected</span>
          <button className="link-btn" onClick={() => setSelected(new Set())}>Clear</button>
        </div>
      )}

      <div className="table-scroll">
        <table className="table">
          <thead>
            <tr>
              <th className="check-col">
                <IndeterminateCheckbox
                  checked={allChecked}
                  indeterminate={someChecked}
                  onChange={toggleAll}
                  aria-label="Select all"
                />
              </th>
              {COLUMNS.map((c) => {
                const active = sort.key === c.key;
                return (
                  // Lookup: "aria-sort mdn"
                  <th
                    key={c.key}
                    className={c.numeric ? 'num' : ''}
                    aria-sort={active ? (sort.dir === 'asc' ? 'ascending' : 'descending') : 'none'}
                  >
                    <button className={'sort-btn' + (active ? ' is-active' : '')} onClick={() => toggleSort(c.key)}>
                      {c.label}
                      <span className="sort-icon" aria-hidden="true">
                        {active ? (sort.dir === 'asc' ? '▲' : '▼') : '↕'}
                      </span>
                    </button>
                  </th>
                );
              })}
            </tr>
          </thead>
          <tbody>
            {rows.map((a) => {
              const isSelected = selected.has(a.id);
              return (
                <tr key={a.id} className={isSelected ? 'is-selected' : ''}>
                  <td className="check-col">
                    <input
                      type="checkbox"
                      checked={isSelected}
                      onChange={() => toggleRow(a.id)}
                      aria-label={'Select ' + a.name}
                    />
                  </td>
                  <td className="name-cell">{a.name}</td>
                  <td>{a.owner}</td>
                  <td><code>{a.version}</code></td>
                  <td>{when(a.lastRun)}</td>
                  <td className="num">{pct.format(a.successRate)}</td>
                </tr>
              );
            })}
          </tbody>
        </table>
      </div>
    </div>
  );
}
