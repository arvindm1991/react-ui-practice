// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /patients/today -> PatientSummary[]
export async function getPatients(): Promise<PatientSummary[]> {
  await randomDelay(300, 800);
  return PATIENTS.map((p) => ({ id: p.id, name: p.name, mrn: p.mrn, nextAppt: p.nextAppt, flag: p.flag }));
}

// GET /patients/:id -> PatientDetail
// Margaret's chart is archived and always loads slowly (1.2–1.5 s). The others take 0.3–0.7 s.
export async function getPatient(id: string): Promise<PatientDetail> {
  if (id === 'p1') {
    await randomDelay(1200, 1500);
  } else {
    await randomDelay(300, 700);
  }
  const found = PATIENTS.find((p) => p.id === id);
  if (!found) {
    throw new Error('Patient not found');
  }
  return {
    ...found,
    allergies: [...found.allergies],
    conditions: [...found.conditions],
    medications: found.medications.map((m) => ({ ...m })),
    vitals: { ...found.vitals },
  };
}

const COLORS = ['#3B82F6', '#14B8A6', '#8B5CF6', '#F59E0B', '#EC4899'];

// <Avatar name="Margaret Ellison" size={40} /> -> a colored circle with "ME"
export function Avatar({ name, size }: { name: string; size?: number }) {
  const initials = name
    .split(' ')
    .map((part) => {
      return part[0];
    })
    .join('')
    .slice(0, 2)
    .toUpperCase();
  const color = COLORS[name.length % COLORS.length];
  const px = size || 36;
  return (
    <span className="avatar" style={{ width: px, height: px, fontSize: px * 0.38, background: color }}>
      {initials}
    </span>
  );
}

// ageFrom("1951-03-12") -> 75 (as of today)
export function ageFrom(isoDate: string): number {
  const dob = new Date(isoDate + 'T00:00:00');
  const now = new Date();
  let age = now.getFullYear() - dob.getFullYear();
  const beforeBirthday = now.getMonth() < dob.getMonth() || (now.getMonth() === dob.getMonth() && now.getDate() < dob.getDate());
  if (beforeBirthday) {
    age = age - 1;
  }
  return age;
}

// CORE: the list reports clicks up and highlights the selected row
type PatientListProps = {
  selectedId: string | null;
  onSelect: (id: string) => void;
};

function PatientList({ selectedId, onSelect }: PatientListProps) {
  const [patients, setPatients] = useState<PatientSummary[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    getPatients().then((data) => {
      if (cancelled) {
        return;
      }
      setPatients(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  let body;
  if (loading) {
    body = <p className="list-msg">Loading schedule…</p>;
  } else {
    body = (
      <ul className="patient-list">
        {patients.map((p) => {
          let cls = 'patient-row';
          if (p.id === selectedId) {
            cls = 'patient-row is-selected';
          }
          return (
            <li key={p.id}>
              <button type="button" className={cls} aria-current={p.id === selectedId} onClick={() => { onSelect(p.id); }}>
                <Avatar name={p.name} />
                <span className="patient-main">
                  <span className="patient-name">{p.name}</span>
                  <span className="patient-meta">{p.nextAppt} · MRN {p.mrn}</span>
                </span>
                <FlagDot flag={p.flag} />
              </button>
            </li>
          );
        })}
      </ul>
    );
  }

  return (
    <aside className="list-pane">
      <h2 className="pane-title">Today’s patients</h2>
      {body}
    </aside>
  );
}

// CORE: fetch the selected patient's chart
function ChartPanel({ patientId }: { patientId: string | null }) {
  const [patient, setPatient] = useState<PatientDetail | null>(null);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    if (!patientId) {
      return;
    }
    // NEXT: the `cancelled` guard. A newer click already started another request: drop this stale answer.
    let cancelled = false;
    setLoading(true);
    getPatient(patientId).then((data) => {
      if (cancelled) {
        return;
      }
      setPatient(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, [patientId]);

  if (!patientId) {
    return (
      <section className="chart-pane">
        <div className="chart-empty">
          <span className="empty-icon">🩺</span>
          <p>Select a patient to view their chart.</p>
        </div>
      </section>
    );
  }

  // CORE: show the skeleton while loading, so the previous patient's chart never sits under a new name.
  if (loading || !patient) {
    return (
      <section className="chart-pane">
        <ChartSkeleton />
      </section>
    );
  }

  // CORE: the loaded chart (PatientChart already exists)
  return (
    <section className="chart-pane">
      <PatientChart patient={patient} />
    </section>
  );
}

export default function App() {
  // CORE: the list sets it, the chart reads it: so it lives in their parent. Store the id, not the patient.
  const [selectedId, setSelectedId] = useState<string | null>(null);

  return (
    <div className="app">
      <TopBar />
      <div className="panes">
        <PatientList selectedId={selectedId} onSelect={(id) => { setSelectedId(id); }} />
        <ChartPanel patientId={selectedId} />
      </div>
    </div>
  );
}

import React from 'react';
import type { PatientDetail } from '../api';
import { Avatar } from './Avatar';
import { FlagDot } from './FlagDot';
import { ageFrom } from '../utils/age';

// NEXT: an empty list gets a message instead of a blank area
function TagList({ items, empty }: { items: string[]; empty: string }) {
  if (items.length === 0) {
    return <p className="none">{empty}</p>;
  }
  return (
    <div className="tags">
      {items.map((item) => {
        return <span key={item} className="tag">{item}</span>;
      })}
    </div>
  );
}

// Renders one patient's full chart. Built for the printable view; give it a loaded PatientDetail.
export function PatientChart({ patient }: { patient: PatientDetail }) {
  // NEXT: age, not the raw date of birth (reuse utils/age)
  const age = ageFrom(patient.dob);
  return (
    <div className="chart">
      <header className="chart-head">
        <Avatar name={patient.name} size={52} />
        <div>
          <h2 className="chart-name">{patient.name}</h2>
          <p className="chart-meta">{age} y · {patient.sex} · MRN {patient.mrn} · {patient.phone}</p>
        </div>
        <FlagDot flag={patient.flag} />
      </header>
      <div className="vitals">
        <div className="vital"><span>Blood pressure</span><b>{patient.vitals.bp}</b></div>
        <div className="vital"><span>Heart rate</span><b>{patient.vitals.hr} bpm</b></div>
        <div className="vital"><span>Temperature</span><b>{patient.vitals.tempF} °F</b></div>
      </div>
      <div className="chart-grid">
        <div className="chart-section">
          <h3>Allergies</h3>
          <TagList items={patient.allergies} empty="No known allergies" />
        </div>
        <div className="chart-section">
          <h3>Conditions</h3>
          <TagList items={patient.conditions} empty="None recorded" />
        </div>
        <div className="chart-section wide">
          <h3>Medications</h3>
          <ul className="meds">
            {patient.medications.map((m) => {
              return <li key={m.name}><b>{m.name}</b><span>{m.dose}</span></li>;
            })}
          </ul>
        </div>
      </div>
      <p className="chart-foot">Last visit {patient.lastVisit} with {patient.provider}</p>
    </div>
  );
}
