// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /products -> Product[]
export async function getProducts(): Promise<Product[]> {
  await randomDelay(500, 1200);
  return PRODUCTS.map((p) => ({ ...p }));
}

// GET /categories -> Category[]  (no counts: the API doesn't send them)
export async function getCategories(): Promise<Category[]> {
  await randomDelay(300, 900);
  return CATEGORIES.map((c) => ({ ...c }));
}

import React from 'react';

type ChipProps = {
  label: string;
  count: number;
  active: boolean;
  disabled?: boolean;
  onClick: () => void;
};

// A pill button with a small count bubble: "Pots (4)". Styles: .chip in styles.css.
// Used by the Journal's tag filter too.
export function Chip({ label, count, active, disabled, onClick }: ChipProps) {
  let cls = 'chip';
  if (active) {
    cls = 'chip is-active';
  }
  return (
    <button type="button" className={cls} aria-pressed={active} disabled={disabled} onClick={() => { onClick(); }}>
      {label} <span className="chip-count">{count}</span>
    </button>
  );
}

@@@ solution src/App.tsx
import React, { useEffect, useState } from 'react';
import { getCategories, getProducts } from './api';
import type { Category, Product } from './api';
import { ProductCard } from './components/ProductCard';
import { Chip } from './components/Chip';

const ALL = 'all';

function Header() {
  return (
    <header className="header">
      <span className="logo">Fernwood</span>
      <nav className="nav">
        <a>Shop</a>
        <a>Journal</a>
        <a>Care guides</a>
      </nav>
      <span className="cart">Cart (0)</span>
    </header>
  );
}

// CORE: the grid becomes a plain component that just renders what it is given.
function ProductGrid({ products }: { products: Product[] }) {
  return (
    <div className="grid">
      {products.map((p) => {
        return <ProductCard key={p.id} product={p} />;
      })}
    </div>
  );
}

// CORE: products move up here. The chips need them for their counts and the grid needs them to render.
function Catalog() {
  const [products, setProducts] = useState<Product[]>([]);
  const [categories, setCategories] = useState<Category[]>([]);
  const [loading, setLoading] = useState(true);
  const [activeCategory, setActiveCategory] = useState(ALL);

  useEffect(() => {
    let cancelled = false;
    // NEXT: two independent requests, one loading state: wait for both with Promise.all
    Promise.all([getProducts(), getCategories()]).then(([productData, categoryData]) => {
      if (cancelled) {
        return;
      }
      setProducts(productData);
      setCategories(categoryData);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // CORE: derived. The API has no counts, so count the products we already have.
  function countFor(categoryId: string) {
    return products.filter((p) => {
      return p.categoryId === categoryId;
    }).length;
  }

  // CORE: derived visible list
  let visible = products;
  if (activeCategory !== ALL) {
    visible = products.filter((p) => {
      return p.categoryId === activeCategory;
    });
  }

  if (loading) {
    return <p className="loading">Loading the greenhouse…</p>;
  }

  // CORE: chips row (reuses components/Chip). NEXT: disabled when the count is 0
  return (
    <div>
      <div className="chips" role="group" aria-label="Filter by category">
        <Chip label="All" count={products.length} active={activeCategory === ALL} onClick={() => { setActiveCategory(ALL); }} />
        {categories.map((c) => {
          const count = countFor(c.id);
          return (
            <Chip
              key={c.id}
              label={c.name}
              count={count}
              active={activeCategory === c.id}
              disabled={count === 0}
              onClick={() => { setActiveCategory(c.id); }}
            />
          );
        })}
      </div>
      <ProductGrid products={visible} />
    </div>
  );
}

export default function App() {
  return (
    <div className="app">
      <Header />
      <main className="shop">
        <p className="eyebrow">The spring collection</p>
        <h1>Bring the outside in</h1>
        <Catalog />
      </main>
    </div>
  );
}
