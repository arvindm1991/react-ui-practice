// Fake API. Treat it like a real backend you can't change.
export type Level = 'DEBUG' | 'INFO' | 'WARN' | 'ERROR';

export type LogLine = {
  line: number;     // 1-based, unique
  time: string;     // "14:02:07.481"
  level: Level;
  service: string;
  message: string;
};

const SERVICES = ['api-gateway', 'auth', 'billing', 'search', 'worker', 'notifier'];
const MESSAGES: Record<Level, string[]> = {
  DEBUG: ['cache lookup key=user:{n}', 'span closed in {n}µs', 'retry budget remaining={n}', 'pool stats active={n}'],
  INFO: ['GET /v1/orders/{n} 200', 'job {n} completed', 'user {n} signed in', 'webhook delivered id={n}', 'POST /v1/carts 201'],
  WARN: ['slow query took {n}ms', 'rate limit at 80% for tenant {n}', 'deprecated header used by client {n}'],
  ERROR: ['payment {n} declined: card_expired', 'upstream timeout after {n}ms', 'failed to parse payload id={n}'],
};

// Deterministic pseudo-random numbers, so the same 10,000 lines every time.
let seed = 42;
function rand(): number {
  seed = (seed * 1664525 + 1013904223) % 4294967296;
  return seed / 4294967296;
}

function pickLevel(): Level {
  const r = rand();
  if (r < 0.25) {
    return 'DEBUG';
  }
  if (r < 0.8) {
    return 'INFO';
  }
  if (r < 0.94) {
    return 'WARN';
  }
  return 'ERROR';
}

const TOTAL = 10000;
const LOGS: LogLine[] = [];
let ms = (14 * 3600 + 2 * 60) * 1000;
for (let i = 1; i <= TOTAL; i++) {
  ms += Math.floor(rand() * 900);
  const level = pickLevel();
  const templates = MESSAGES[level];
  const template = templates[Math.floor(rand() * templates.length)];
  const d = new Date(ms);
  const time = String(d.getUTCHours()).padStart(2, '0') + ':' + String(d.getUTCMinutes()).padStart(2, '0') + ':' +
    String(d.getUTCSeconds()).padStart(2, '0') + '.' + String(d.getUTCMilliseconds()).padStart(3, '0');
  LOGS.push({
    line: i,
    time: time,
    level: level,
    service: SERVICES[Math.floor(rand() * SERVICES.length)],
    message: template.replace('{n}', String(Math.floor(rand() * 9000) + 100)),
  });
}

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /logs?since=1h -> LogLine[] (10,000 lines, oldest first)
export async function getLogs(): Promise<LogLine[]> {
  await randomDelay(800, 1500);
  return LOGS.map((l) => ({ ...l }));
}

import React, { useEffect, useState } from 'react';
import { getLogs } from './api';
import type { LogLine } from './api';
import LogRow, { LogHeader } from './components/LogRow';
import { formatNumber } from './utils/format';

const PREVIEW_LIMIT = 200;

function TopBar({ total }: { total: number }) {
  return (
    <header className="topbar">
      <span className="logo">▍tracewell</span>
      <span className="crumb">production / <b>all services</b> / last 1h</span>
      <span className="live-dot">● live</span>
      <span className="total">{formatNumber(total)} lines</span>
    </header>
  );
}

export default function App() {
  const [logs, setLogs] = useState<LogLine[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    getLogs().then((data) => {
      if (cancelled) {
        return;
      }
      setLogs(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // TODO: rendering all 10,000 rows freezes the tab, so for now only the first 200 render.
  const shown = logs.slice(0, PREVIEW_LIMIT);

  let body;
  if (loading) {
    body = <p className="log-message">Fetching logs…</p>;
  } else {
    body = (
      <div className="log-scroll">
        <LogHeader />
        {shown.map((l) => {
          return <LogRow key={l.line} log={l} />;
        })}
        <p className="log-message">Showing the first {PREVIEW_LIMIT} of {formatNumber(logs.length)} lines.</p>
      </div>
    );
  }

  return (
    <div className="app">
      <TopBar total={logs.length} />
      <div className="toolbar">
        <span className="toolbar-hint">Filters and jump-to-line coming soon</span>
      </div>
      <main className="log-panel">{body}</main>
    </div>
  );
}

import React, { useEffect, useMemo, useRef, useState } from 'react';
import { getLogs } from './api';
import type { Level, LogLine } from './api';
import LogRow, { LogHeader, ROW_HEIGHT } from './components/LogRow';
import { formatNumber } from './utils/format';

const OVERSCAN = 8;      // extra rows above and below the viewport, so fast scrolling doesn't flash blank
const HEADER_HEIGHT = 30;
const LEVELS: (Level | 'ALL')[] = ['ALL', 'DEBUG', 'INFO', 'WARN', 'ERROR'];

function TopBar({ total }: { total: number }) {
  return (
    <header className="topbar">
      <span className="logo">▍tracewell</span>
      <span className="crumb">production / <b>all services</b> / last 1h</span>
      <span className="live-dot">● live</span>
      <span className="total">{formatNumber(total)} lines</span>
    </header>
  );
}

type ViewerProps = {
  logs: LogLine[];
};

function LogViewer({ logs }: ViewerProps) {
  const [scrollTop, setScrollTop] = useState(0);
  const [viewportHeight, setViewportHeight] = useState(0);
  const [query, setQuery] = useState('');
  const [level, setLevel] = useState<Level | 'ALL'>('ALL');
  const [jumpValue, setJumpValue] = useState('');
  const [jumpError, setJumpError] = useState('');
  const [highlightLine, setHighlightLine] = useState<number | null>(null);
  const scrollRef = useRef<HTMLDivElement>(null);

  // Measure the scroll container (and re-measure when it resizes).
  useEffect(() => {
    const el = scrollRef.current;
    if (!el) {
      return;
    }
    function measure() {
      if (el) {
        setViewportHeight(el.clientHeight);
      }
    }
    measure();
    const ro = new ResizeObserver(measure);
    ro.observe(el);
    return () => {
      ro.disconnect();
    };
  }, []);

  // derived + memoized: filtering 10,000 rows on every scroll event would be wasteful
  const filtered = useMemo(() => {
    const q = query.trim().toLowerCase();
    return logs.filter((l) => {
      if (level !== 'ALL' && l.level !== level) {
        return false;
      }
      if (q === '') {
        return true;
      }
      return l.message.toLowerCase().includes(q) || l.service.includes(q);
    });
  }, [logs, query, level]);

  function scrollToTop() {
    if (scrollRef.current) {
      scrollRef.current.scrollTop = 0;
    }
    setScrollTop(0);
  }

  function handleQuery(value: string) {
    setQuery(value);
    scrollToTop(); // old scroll position means nothing in a new list
  }

  function handleLevel(value: Level | 'ALL') {
    setLevel(value);
    scrollToTop();
  }

  function handleScroll(e: React.UIEvent<HTMLDivElement>) {
    setScrollTop(e.currentTarget.scrollTop);
  }

  function handleJump(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const n = Number(jumpValue);
    if (!Number.isInteger(n) || n < 1 || n > logs.length) {
      setJumpError('Enter a line between 1 and ' + formatNumber(logs.length));
      return;
    }
    const index = filtered.findIndex((l) => {
      return l.line === n;
    });
    if (index === -1) {
      setJumpError('Line ' + n + ' is hidden by the current filter');
      return;
    }
    setJumpError('');
    setHighlightLine(n);
    // Put the row in the middle of the viewport. Pure math: no need to render it first.
    const rowsArea = viewportHeight - HEADER_HEIGHT;
    const top = Math.max(0, index * ROW_HEIGHT - rowsArea / 2 + ROW_HEIGHT / 2);
    if (scrollRef.current) {
      scrollRef.current.scrollTop = top;
    }
    setScrollTop(top);
  }

  // derived: the window of rows to render
  const totalHeight = filtered.length * ROW_HEIGHT;
  const start = Math.max(0, Math.floor(scrollTop / ROW_HEIGHT) - OVERSCAN);
  const end = Math.min(filtered.length, Math.ceil((scrollTop + viewportHeight) / ROW_HEIGHT) + OVERSCAN);
  const visible = filtered.slice(start, end);

  let empty = null;
  if (filtered.length === 0) {
    empty = <p className="log-message">No lines match the filter.</p>;
  }

  let jumpErrorEl = null;
  if (jumpError) {
    jumpErrorEl = <span className="jump-error" role="alert">{jumpError}</span>;
  }

  return (
    <div className="viewer">
      <div className="toolbar">
        <input className="tb-input filter-input" placeholder="Filter messages…" aria-label="Filter messages" value={query} onChange={(e) => { handleQuery(e.target.value); }} />
        <div className="level-tabs" role="group" aria-label="Level">
          {LEVELS.map((lv) => {
            let cls = 'level-tab';
            if (lv === level) {
              cls = 'level-tab is-active';
            }
            return <button key={lv} type="button" className={cls} aria-pressed={lv === level} onClick={() => { handleLevel(lv); }}>{lv}</button>;
          })}
        </div>
        <form className="jump" onSubmit={handleJump}>
          <input className="tb-input jump-input" inputMode="numeric" placeholder="Line #" aria-label="Jump to line" value={jumpValue} onChange={(e) => { setJumpValue(e.target.value); }} />
          <button type="submit" className="jump-btn">Go</button>
        </form>
        {jumpErrorEl}
        <span className="match-count">{formatNumber(filtered.length)} / {formatNumber(logs.length)}</span>
      </div>
      <div className="log-scroll" ref={scrollRef} onScroll={handleScroll}>
        <LogHeader />
        <div className="log-spacer" style={{ height: totalHeight }}>
          {visible.map((l, i) => {
            const index = start + i;
            return <LogRow key={l.line} log={l} style={{ top: index * ROW_HEIGHT }} highlighted={l.line === highlightLine} />;
          })}
        </div>
        {empty}
      </div>
    </div>
  );
}

export default function App() {
  const [logs, setLogs] = useState<LogLine[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    getLogs().then((data) => {
      if (cancelled) {
        return;
      }
      setLogs(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  let body;
  if (loading) {
    body = <p className="log-message">Fetching logs…</p>;
  } else {
    body = <LogViewer logs={logs} />;
  }

  return (
    <div className="app">
      <TopBar total={logs.length} />
      <main className="log-panel">{body}</main>
    </div>
  );
}
