  id: string;
  name: string;     // "Smileys"
  icon: string;     // emoji shown on the category tab

const CATEGORIES: EmojiCategory[] = [

];

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /emoji/categories -> EmojiCategory[]
export async function getEmojiCategories(): Promise<EmojiCategory[]> {
  await randomDelay(400, 1000);
  return CATEGORIES.map((c) => ({ ...c, emojis: [...c.emojis] }));
}

import React from 'react';

export type ChatMessage = { id: string; author: string; time: string; text: string };

const COLORS = ['#E01E5A', '#2EB67D', '#ECB22E', '#36C5F0', '#7C3AED'];

function colorFor(name: string): string {
  let sum = 0;
  for (let i = 0; i < name.length; i++) {
    sum += name.charCodeAt(i);
  }
  return COLORS[sum % COLORS.length];
}

export default function Message({ message }: { message: ChatMessage }) {
  return (
    <li className="message">
      <span className="msg-avatar" style={{ background: colorFor(message.author) }}>{message.author[0]}</span>
      <div>
        <p className="msg-meta"><b>{message.author}</b> <span>{message.time}</span></p>
        <p className="msg-text">{message.text}</p>
      </div>
    </li>
  );
}

import React, { useEffect, useRef, useState } from 'react';

const MAX_RECENT = 8;

const INITIAL: ChatMessage[] = [
  { id: 'm1', author: 'Rosa', time: '9:41 AM', text: 'Launch checklist is in the doc. Two items left on my side.' },

type EmojiPickerProps = {
  recent: string[];
  onPick: (emoji: string) => void;
};

// Only mounted while open, so it loads fresh each time it opens.
function EmojiPicker({ recent, onPick }: EmojiPickerProps) {
  const [categories, setCategories] = useState<EmojiCategory[]>([]);
  const [activeId, setActiveId] = useState('');
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    getEmojiCategories().then((data) => {
      if (cancelled) {
        return;
      }
      setCategories(data);
      setActiveId(data[0].id);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  function renderGrid(emojis: string[]) {
    return (
      <div className="emoji-grid">
        {emojis.map((emoji) => {
          return (
            <button key={emoji} type="button" className="emoji-btn" onClick={() => { onPick(emoji); }}>
              {emoji}
            </button>
          );
        })}
      </div>
    );
  }

  // derived
  const active = categories.find((c) => {
    return c.id === activeId;
  });

  let recentRow = null;
  if (recent.length > 0) {
    recentRow = (
      <div className="emoji-section recent-row">
        <p className="emoji-heading">Recently used</p>
        {renderGrid(recent)}
      </div>
    );
  }

  let body;
  if (loading) {
    body = <p className="emoji-loading">Loading emoji…</p>;
  } else {
    let grid = null;
    if (active) {
      grid = (
        <div className="emoji-section">
          <p className="emoji-heading">{active.name}</p>
          {renderGrid(active.emojis)}
        </div>
      );
    }
    body = (
      <div>
        <div className="emoji-tabs" role="tablist">
          {categories.map((c) => {
            const selected = c.id === activeId;
            let cls = 'emoji-tab';
            if (selected) {
              cls = 'emoji-tab is-active';
            }
            return (
              <button key={c.id} type="button" role="tab" aria-selected={selected} aria-label={c.name} className={cls} onClick={() => { setActiveId(c.id); }}>
                {c.icon}
              </button>
            );
          })}
        </div>
        {grid}
      </div>
    );
  }

  return (
    <div className="emoji-popover" role="dialog" aria-label="Emoji picker">
      {recentRow}
      {body}
    </div>
  );
}

function Composer({ onSend }: { onSend: (text: string) => void }) {
  const [text, setText] = useState('');
  const [pickerOpen, setPickerOpen] = useState(false);
  // Lives here (not in the picker) so it survives the picker closing.
  const [recent, setRecent] = useState<string[]>([]);
  const textareaRef = useRef<HTMLTextAreaElement>(null);
  const wrapRef = useRef<HTMLDivElement>(null);
  const caretRef = useRef<number | null>(null);

  // Outside click + Escape close the popover. Only listen while open.
  useEffect(() => {
    if (!pickerOpen) {
      return;
    }
    function handleMouseDown(e: MouseEvent) {
      if (wrapRef.current && !wrapRef.current.contains(e.target as Node)) {
        setPickerOpen(false);
      }
    }
    function handleKeyDown(e: KeyboardEvent) {
      if (e.key === 'Escape') {
        setPickerOpen(false);
      }
    }
    document.addEventListener('mousedown', handleMouseDown);
    document.addEventListener('keydown', handleKeyDown);
    return () => {
      document.removeEventListener('mousedown', handleMouseDown);
      document.removeEventListener('keydown', handleKeyDown);
    };
  }, [pickerOpen]);

  // After the text changes from an insert, put the caret right after the emoji.
  useEffect(() => {
    const el = textareaRef.current;
    if (caretRef.current === null || !el) {
      return;
    }
    el.focus();
    el.setSelectionRange(caretRef.current, caretRef.current);
    caretRef.current = null;
  }, [text]);

  function handleSend() {
    if (text.trim() === '') {
      return;
    }
    onSend(text.trim());
    setText('');
  }

  function togglePicker() {
    setPickerOpen((open) => {
      return !open;
    });
  }

  function handlePick(emoji: string) {
    const el = textareaRef.current;
    let start = text.length;
    let end = text.length;
    if (el) {
      // The textarea keeps its selection after it loses focus.
      start = el.selectionStart;
      end = el.selectionEnd;
    }
    setText(text.slice(0, start) + emoji + text.slice(end));
    caretRef.current = start + emoji.length;
    setRecent((prev) => {
      const rest = prev.filter((x) => {
        return x !== emoji;
      });
      return [emoji, ...rest].slice(0, MAX_RECENT);
    });
    setPickerOpen(false);
  }

  let picker = null;
  if (pickerOpen) {
    picker = <EmojiPicker recent={recent} onPick={handlePick} />;
  }

  let toolCls = 'tool-btn';
  if (pickerOpen) {
    toolCls = 'tool-btn is-open';
  }

  return (
    <div className="composer">
      <textarea
        ref={textareaRef}
        className="composer-input"
        placeholder="Message #launch-q4"
        value={text}
        onChange={(e) => { setText(e.target.value); }}
      />
      <div className="composer-bar">
        <div className="emoji-anchor" ref={wrapRef}>
          <button
            type="button"
            className={toolCls}
            aria-label="Add emoji"
            aria-expanded={pickerOpen}
            onClick={() => { togglePicker(); }}
          >
            🙂
          </button>
          {picker}
        </div>
        <button type="button" className="send-btn" disabled={text.trim() === ''} onClick={() => { handleSend(); }}>Send</button>
      </div>
    </div>
  );
}

export default function App() {
  const [messages, setMessages] = useState<ChatMessage[]>(INITIAL);

  function handleSend(text: string) {
    setMessages((prev) => {
      return [...prev, { id: 'm' + Date.now(), author: 'You', time: 'now', text }];
    });
  }

  return (
    <div className="app">
      <Sidebar />
      <main className="channel-view">
        <header className="channel-head"># launch-q4 <span>Ship it on the 14th</span></header>
        <ul className="messages">
          {messages.map((m) => {
            return <Message key={m.id} message={m} />;
          })}
        </ul>
        <Composer onSend={handleSend} />
      </main>
    </div>
  );
}
