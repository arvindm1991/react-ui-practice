// Fake API. Treat it like a real backend you can't change.
export type Deploy = {
  id: string;
  commit: string;
  env: 'staging' | 'production';
  author: string;
  when: string;
};

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// POST /services/:service/deploy -> Deploy
export async function deployToStaging(service: string): Promise<Deploy> {
  await randomDelay(300, 800);
  const commit = Math.random().toString(16).slice(2, 9);
  return { id: 'd_' + commit, commit: commit, env: 'staging', author: 'you', when: 'just now' };
}

// POST /services/:service/rollback
// Always rejects today: production is in a change freeze.
export async function rollbackProduction(service: string): Promise<void> {
  await randomDelay(300, 800);
  throw new Error('Rollback blocked: ' + service + ' is in a change freeze until 17:00 UTC');
}

type ActionsProps = {
  notify: (kind: ToastKind, message: string) => void;

function Actions({ notify }: ActionsProps) {
  const [busy, setBusy] = useState<string | null>(null);

  function handleDeploy() {
    setBusy('deploy');
    deployToStaging(SERVICE)
      .then((deploy) => {
        notify('success', 'Deployed ' + deploy.commit + ' to staging');
        setBusy(null);
      })
      .catch(() => {

    rollbackProduction(SERVICE)
      .then(() => {
        notify('success', 'Rolled back production');

        setBusy(null);
      });
  }

    notify('success', 'Cache cleared for ' + SERVICE);
  }

  let deployLabel = 'Deploy to staging';
  if (busy === 'deploy') {
    deployLabel = 'Deploying…';

  if (busy === 'rollback') {
    rollbackLabel = 'Rolling back…';
  }

  return (

      <h2>Actions</h2>
      <div className="actions">
        <Button variant="primary" disabled={busy !== null} onClick={() => { handleDeploy(); }}>{deployLabel}</Button>
        <Button variant="danger" disabled={busy !== null} onClick={() => { handleRollback(); }}>{rollbackLabel}</Button>
        <Button onClick={() => { handleClearCache(); }}>Clear cache</Button>
      </div>
    </section>
  );
}

export default function App() {
  const [toasts, setToasts] = useState<ToastData[]>([]);
  // A ref, not state: changing it should not re-render anything.
  const nextId = useRef(1);

  function notify(kind: ToastKind, message: string) {
    const toast = { id: nextId.current, kind: kind, message: message };
    nextId.current = nextId.current + 1;
    setToasts((prev) => {
      // Change: keep only the newest MAX_TOASTS (the oldest drops off the top).
      return [...prev, toast].slice(-MAX_TOASTS);
    });
  }

  // useCallback keeps the same function between renders. Each Toast's effect
  // depends on it, so a new function every render would restart every timer.
  const dismiss = useCallback((id: number) => {
    setToasts((prev) => {
      return prev.filter((t) => {
        return t.id !== id;
      });
    });
  }, []);

  return (
    <div className="layout">
      <Sidebar />
      <main className="main">
        <header className="page-head">
          <h1>{SERVICE}</h1>
          <span className="pill">● Running · 6 replicas</span>
        </header>
        <Actions notify={notify} />
        <DeployHistory />
      </main>
      <ToastStack toasts={toasts} onDismiss={dismiss} />
    </div>
  );
}

import React, { useEffect } from 'react';

export type ToastKind = 'success' | 'error';

export type ToastData = {
  id: number;
  kind: ToastKind;
  message: string;
};

const DISMISS_AFTER_MS = 3000;

type ToastProps = {
  toast: ToastData;
  onDismiss: (id: number) => void;
};

function Toast({ toast, onDismiss }: ToastProps) {
  // Each toast owns its own timer. The cleanup clears it if the toast is closed
  // early (or removed by the limit), so we never dismiss something twice.
  useEffect(() => {
    const timer = setTimeout(() => {
      onDismiss(toast.id);
    }, DISMISS_AFTER_MS);
    return () => {
      clearTimeout(timer);
    };
  }, [toast.id, onDismiss]);

  let cls = 'toast is-success';
  let icon = '✓';
  if (toast.kind === 'error') {
    cls = 'toast is-error';
    icon = '!';
  }

  return (
    <div className={cls} role="status">
      <span className="toast-icon">{icon}</span>
      <p className="toast-msg">{toast.message}</p>
      <button type="button" className="toast-close" aria-label="Dismiss" onClick={() => { onDismiss(toast.id); }}>✕</button>
    </div>
  );
}

type ToastStackProps = {
  toasts: ToastData[];
  onDismiss: (id: number) => void;
};

export function ToastStack({ toasts, onDismiss }: ToastStackProps) {
  return (
    <div className="toast-stack" aria-live="polite">
      {toasts.map((t) => {
        return <Toast key={t.id} toast={t} onDismiss={onDismiss} />;
      })}
    </div>
  );
}
