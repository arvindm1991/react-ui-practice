// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /threads/lisbon/messages -> Message[] (oldest first)
export async function getMessages(): Promise<Message[]> {
  await randomDelay(400, 900);
  return MESSAGES.map((m) => ({ ...m }));
}

// The server's clock for new messages (fixed so the demo is stable).
let clock = new Date('2026-10-10T15:00:00').getTime();
let nextId = 100;
const failedOnce = new Set<string>();

// POST /threads/lisbon/messages { text } -> Message
// Simulated flaky network: a message containing "fail" fails the FIRST time it's sent
// (throws), and succeeds when the same text is retried.
export async function sendMessage(text: string): Promise<Message> {
  await randomDelay(600, 1100);
  if (text.toLowerCase().includes('fail') && !failedOnce.has(text)) {
    failedOnce.add(text);
    throw new Error('Network error');
  }
  clock += 60000;
  nextId += 1;
  const d = new Date(clock);
  const pad = (n: number) => String(n).padStart(2, '0');
  const sentAt = d.getFullYear() + '-' + pad(d.getMonth() + 1) + '-' + pad(d.getDate()) + 'T' + pad(d.getHours()) + ':' + pad(d.getMinutes()) + ':00';
  const message: Message = { id: 'm' + nextId, author: 'me', text, sentAt };
  MESSAGES.push(message);
  return { ...message };
}

// The app's "now". Fixed so the demo data always reads the same.
export const NOW_ISO = '2026-10-10T15:00:00';
const NOW = new Date(NOW_ISO);
const DAYS = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

// formatTime("2026-10-10T09:12:00") -> "9:12 AM"
export function formatTime(iso: string): string {
  const d = new Date(iso);
  let h = d.getHours() % 12;
  if (h === 0) {
    h = 12;
  }
  let suffix = 'AM';
  if (d.getHours() >= 12) {
    suffix = 'PM';
  }
  return h + ':' + String(d.getMinutes()).padStart(2, '0') + ' ' + suffix;
}

function startOfDay(x: Date): number {
  return new Date(x.getFullYear(), x.getMonth(), x.getDate()).getTime();
}

// dayLabel("2026-10-10T09:12:00") -> "Today", "2026-10-09..." -> "Yesterday", "2026-10-08..." -> "Thu, Oct 8"
export function dayLabel(iso: string): string {
  const d = new Date(iso);
  const diffDays = Math.round((startOfDay(NOW) - startOfDay(d)) / 86400000);
  if (diffDays === 0) {
    return 'Today';
  }
  if (diffDays === 1) {
    return 'Yesterday';
  }
  return DAYS[d.getDay()] + ', ' + MONTHS[d.getMonth()] + ' ' + d.getDate();
}

import React, { useEffect, useRef, useState } from 'react';
import { getMessages, sendMessage } from './api';
import type { Message } from './api';
import { Avatar } from './components/Avatar';
import { NOW_ISO, dayLabel, formatTime } from './utils/time';

function ChatHeader() {
  return (
    <header className="chat-header">
      <Avatar initials="LC" online />
      <div>
        <p className="chat-name">Lena Costa</p>
        <p className="chat-sub">Wayfarer concierge · replies in ~5 min</p>
      </div>
    </header>
  );
}

// What the UI shows: a server message plus where it is in its journey.
type ChatMessage = Message & { status: 'sent' | 'sending' | 'failed' };

type BubbleProps = {
  message: ChatMessage;
  onRetry: (id: string) => void;
};

function MessageBubble({ message, onRetry }: BubbleProps) {
  let cls = 'bubble-row';
  if (message.author === 'me') {
    cls = 'bubble-row is-mine';
  }
  let meta = <span className="bubble-meta">{formatTime(message.sentAt)}</span>;
  if (message.status === 'sending') {
    cls = cls + ' is-sending';
    meta = <span className="bubble-meta">Sending…</span>;
  }
  if (message.status === 'failed') {
    cls = cls + ' is-failed';
    meta = (
      <span className="bubble-meta bubble-failed">
        Failed ·
        <button type="button" className="retry-btn" onClick={() => { onRetry(message.id); }}>Retry</button>
      </span>
    );
  }
  return (
    <div className={cls}>
      <div className="bubble">
        <p className="bubble-text">{message.text}</p>
      </div>
      {meta}
    </div>
  );
}

function ChatThread() {
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [loading, setLoading] = useState(true);
  const [draft, setDraft] = useState('');
  const listRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    let cancelled = false;
    getMessages().then((data) => {
      if (cancelled) {
        return;
      }
      setMessages(data.map((m): ChatMessage => {
        return { ...m, status: 'sent' };
      }));
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // Auto-scroll: runs after the DOM updates whenever a message is added (or the first load lands).
  useEffect(() => {
    const el = listRef.current;
    if (el) {
      el.scrollTop = el.scrollHeight;
    }
  }, [messages.length]);

  function setStatus(id: string, status: ChatMessage['status']) {
    setMessages((prev) => {
      return prev.map((m) => {
        if (m.id === id) {
          return { ...m, status };
        }
        return m;
      });
    });
  }

  // Shared by first send and retry: swap the temp message for the server's on success.
  function deliver(tempId: string, text: string) {
    sendMessage(text)
      .then((saved) => {
        setMessages((prev) => {
          return prev.map((m): ChatMessage => {
            if (m.id === tempId) {
              return { ...saved, status: 'sent' };
            }
            return m;
          });
        });
      })
      .catch(() => {
        setStatus(tempId, 'failed');
      });
  }

  function handleSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const text = draft.trim();
    if (text === '') {
      return;
    }
    const tempId = 'temp-' + Date.now();
    const temp: ChatMessage = { id: tempId, author: 'me', text, sentAt: NOW_ISO, status: 'sending' };
    setMessages((prev) => {
      return [...prev, temp];
    });
    setDraft('');
    deliver(tempId, text);
  }

  function handleRetry(id: string) {
    const failed = messages.find((m) => {
      return m.id === id;
    });
    if (!failed) {
      return;
    }
    setStatus(id, 'sending');
    deliver(id, failed.text);
  }

  // Change: a separator row whenever the day changes from the previous message.
  let body;
  if (loading) {
    body = <p className="chat-empty">Loading conversation…</p>;
  } else {
    const rows: React.ReactNode[] = [];
    let lastDay = '';
    messages.forEach((m) => {
      const day = dayLabel(m.sentAt);
      if (day !== lastDay) {
        rows.push(<div key={'day-' + day} className="day-sep"><span>{day}</span></div>);
        lastDay = day;
      }
      rows.push(<MessageBubble key={m.id} message={m} onRetry={handleRetry} />);
    });
    body = rows;
  }

  return (
    <section className="chat">
      <ChatHeader />
      <div className="messages" ref={listRef}>{body}</div>
      <form className="composer" onSubmit={handleSubmit}>
        <input className="composer-input" placeholder="Message Lena…" value={draft} onChange={(e) => { setDraft(e.target.value); }} />
        <button type="submit" className="send-btn" disabled={draft.trim() === ''}>Send</button>
      </form>
    </section>
  );
}

function TripCard() {
  return (
    <aside className="trip">
      <p className="trip-eyebrow">Upcoming trip</p>
      <h2 className="trip-title">Lisbon, Portugal</h2>
      <p className="trip-dates">Oct 14 – Oct 19 · 2 travellers</p>
      <ul className="trip-list">
        <li><span>✈︎</span> TP 236 · SFO → LIS</li>
        <li><span>🏨</span> Memmo Alfama · 5 nights</li>
        <li><span>🍷</span> Clube de Fado · Fri 8:30 PM</li>
      </ul>
    </aside>
  );
}

export default function App() {
  return (
    <div className="app">
      <ChatThread />
      <TripCard />
    </div>
  );
}
