// Fake API. Treat it like a real backend you can't change.
export type Project = {
  id: string;
  name: string;
  framework: 'Next.js' | 'Vite' | 'Astro' | 'Remix';
  branch: string;
  status: 'ready' | 'building' | 'error';
  deployedMinutesAgo: number;
};

const PROJECTS: Project[] = [
  { id: 'prj_1', name: 'Marketing site', framework: 'Next.js', branch: 'main', status: 'ready', deployedMinutesAgo: 12 },
  { id: 'prj_2', name: 'Docs', framework: 'Astro', branch: 'main', status: 'ready', deployedMinutesAgo: 95 },
  { id: 'prj_3', name: 'Dashboard', framework: 'Vite', branch: 'release/2.4', status: 'building', deployedMinutesAgo: 1 },
  { id: 'prj_4', name: 'Checkout', framework: 'Remix', branch: 'main', status: 'error', deployedMinutesAgo: 340 },
  { id: 'prj_5', name: 'Status page', framework: 'Next.js', branch: 'main', status: 'ready', deployedMinutesAgo: 2880 },
];

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /projects -> Project[]
export async function getProjects(): Promise<Project[]> {
  await randomDelay(400, 1000);
  return PROJECTS.map((p) => ({ ...p }));
}

// PATCH /projects/:id { name } -> Project
// Rejects (throws) if another project already has that name (case-insensitive).
export async function renameProject(id: string, name: string): Promise<Project> {
  await randomDelay(500, 1000);
  const clash = PROJECTS.find((p) => p.id !== id && p.name.toLowerCase() === name.toLowerCase());
  if (clash) {
    throw new Error('A project named "' + clash.name + '" already exists');
  }
  const project = PROJECTS.find((p) => p.id === id);
  if (!project) {
    throw new Error('Project not found');
  }
  project.name = name;
  return { ...project };
}

// formatAgo(12) -> "12m ago", formatAgo(95) -> "1h ago", formatAgo(2880) -> "2d ago"
export function formatAgo(minutes: number): string {
  if (minutes < 60) {
    return minutes + 'm ago';
  }
  if (minutes < 1440) {
    return Math.floor(minutes / 60) + 'h ago';
  }
  return Math.floor(minutes / 1440) + 'd ago';
}

import React from 'react';

type NameInputProps = {
  value: string;
  error?: string;
  disabled?: boolean;
  onChange: (value: string) => void;
  onKeyDown: (e: React.KeyboardEvent<HTMLInputElement>) => void;
};

// Inline rename field from the design system. It focuses itself and selects its text when it mounts.
// Pass error to get a red border and the message under it.
export function NameInput({ value, error, disabled, onChange, onKeyDown }: NameInputProps) {
  let cls = 'name-input';
  let message = null;
  if (error) {
    cls = 'name-input has-error';
    message = <span className="name-error">{error}</span>;
  }
  return (
    <span className="name-edit">
      <input
        className={cls}
        aria-label="Project name"
        value={value}
        disabled={disabled}
        autoFocus
        onFocus={(e) => { e.target.select(); }}
        onChange={(e) => { onChange(e.target.value); }}
        onKeyDown={onKeyDown}
      />
      {message}
    </span>
  );
}

// CORE: the row is presentational; the list owns the editing state
// CORE: the list decides WHICH row is editing; the row owns its draft.
type RowProps = {
  project: Project;
  isEditing: boolean;
  onStartEdit: () => void;
  onStopEdit: () => void;
  onRenamed: (updated: Project) => void;
  // STRETCH: another row is saving
  locked: boolean;
  onBusyChange: (busy: boolean) => void;
};

function ProjectRow({ project, isEditing, onStartEdit, onStopEdit, onRenamed, locked, onBusyChange }: RowProps) {
  // CORE: the draft is separate from the saved name, so Cancel can throw it away
  const [draft, setDraft] = useState('');
  // NEXT: saving / CHANGE: error message
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState('');

  // CORE: start editing from the current name
  function handleStartEdit() {
    setDraft(project.name);
    onStartEdit();
    // CHANGE: forget an old error
    setError('');
  }

  // NEXT: typing clears the error (Core can pass setDraft straight to NameInput)
  function handleDraftChange(value: string) {
    setDraft(value);
    setError('');
  }

  // CORE: save through the API, then hand the updated project to the list
  function handleSave() {
    const name = draft.trim();
    // NEXT: empty is rejected, unchanged just closes
    if (name === '') {
      setError("Name can't be empty");
      return;
    }
    if (name === project.name) {
      onStopEdit();
      return;
    }
    // NEXT: saving flag (STRETCH: tell the list)
    setSaving(true);
    onBusyChange(true);
    // CORE
    renameProject(project.id, name)
      .then((updated) => {
        setSaving(false);
        onBusyChange(false);
        onRenamed(updated);
      })
      // CHANGE: the API rejected (e.g. duplicate name): stay in edit mode and show why
      .catch((err: Error) => {
        setSaving(false);
        onBusyChange(false);
        setError(err.message);
      });
  }

  // CORE: Enter saves, Escape cancels
  function handleKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
    if (e.key === 'Enter') {
      handleSave();
    }
    if (e.key === 'Escape') {
      onStopEdit();
    }
  }

  // CORE: name or input, Edit or Cancel/Save
  let nameEl = (
    <span className="project-name" onDoubleClick={() => { handleStartEdit(); }}>{project.name}</span>
  );
  let actions = <button type="button" className="btn-small" disabled={locked} onClick={() => { handleStartEdit(); }}>Edit</button>;
  if (isEditing) {
    // NEXT: saving label
    let saveLabel = 'Save';
    if (saving) {
      saveLabel = 'Saving…';
    }
    // CORE: NameInput already autofocuses and selects (NEXT/CHANGE: error, saving)
    nameEl = <NameInput value={draft} error={error} disabled={saving} onChange={handleDraftChange} onKeyDown={handleKeyDown} />;
    actions = (
      <>
        <button type="button" className="btn-small" disabled={saving} onClick={() => { onStopEdit(); }}>Cancel</button>
        <button type="button" className="btn-small btn-dark" disabled={saving} onClick={() => { handleSave(); }}>{saveLabel}</button>
      </>
    );
  }

  // NEXT: tinted row while editing
  let rowCls = 'project-row';
  if (isEditing) {
    rowCls = 'project-row is-editing';
  }

  // CORE
  return (
    <li className={rowCls}>
      <div className="project-main">
        {nameEl}
        <span className="project-sub">{project.framework} · <code>{project.branch}</code></span>
      </div>
      <StatusDot status={project.status} />
      <span className="project-time">{formatAgo(project.deployedMinutesAgo)}</span>
      <div className="row-actions">
        {actions}
      </div>
    </li>
  );
}

function ProjectList() {
  const [projects, setProjects] = useState<Project[]>([]);
  const [loading, setLoading] = useState(true);
  // CORE: one editingId for the whole list, so only one row can ever be open (STRETCH comes for free)
  const [editingId, setEditingId] = useState<string | null>(null);
  // STRETCH: a save is in flight somewhere
  const [busy, setBusy] = useState(false);

  useEffect(() => {
    let cancelled = false;
    getProjects().then((data) => {
      if (cancelled) {
        return;
      }
      setProjects(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // CORE: replace the renamed project and close the editor
  function handleRenamed(updated: Project) {
    setProjects((prev) => {
      return prev.map((p) => {
        if (p.id === updated.id) {
          return updated;
        }
        return p;
      });
    });
    setEditingId(null);
  }

  if (loading) {
    return <p className="empty">Loading projects…</p>;
  }

  // CORE: pass the editing state down
  return (
    <ul className="project-list">
      {projects.map((p) => {
        return (
          <ProjectRow
            key={p.id}
            project={p}
            isEditing={p.id === editingId}
            onStartEdit={() => { setEditingId(p.id); }}
            onStopEdit={() => { setEditingId(null); }}
            onRenamed={handleRenamed}
            locked={busy}
            onBusyChange={setBusy}
          />
        );
      })}
    </ul>
  );
}

export default function App() {
  return (
    <div className="app">
      <Sidebar />
      <main className="main">
        <div className="page-head">
          <div>
            <h1>Projects</h1>
            <p className="page-sub">Double-click a name to rename it.</p>
          </div>
          <button type="button" className="btn-primary">New project</button>
        </div>
        <ProjectList />
      </main>
    </div>
  );
}
