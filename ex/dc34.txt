// Fake API. Treat it like a real backend you can't change.
export type Project = {
  id: string;
  name: string;
  framework: 'Next.js' | 'Vite' | 'Astro' | 'Remix';
  branch: string;
  status: 'ready' | 'building' | 'error';
  deployedMinutesAgo: number;
};

const PROJECTS: Project[] = [
  { id: 'prj_1', name: 'Marketing site', framework: 'Next.js', branch: 'main', status: 'ready', deployedMinutesAgo: 12 },
  { id: 'prj_2', name: 'Docs', framework: 'Astro', branch: 'main', status: 'ready', deployedMinutesAgo: 95 },
  { id: 'prj_3', name: 'Dashboard', framework: 'Vite', branch: 'release/2.4', status: 'building', deployedMinutesAgo: 1 },
  { id: 'prj_4', name: 'Checkout', framework: 'Remix', branch: 'main', status: 'error', deployedMinutesAgo: 340 },
  { id: 'prj_5', name: 'Status page', framework: 'Next.js', branch: 'main', status: 'ready', deployedMinutesAgo: 2880 },
];

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /projects -> Project[]
export async function getProjects(): Promise<Project[]> {
  await randomDelay(400, 1000);
  return PROJECTS.map((p) => ({ ...p }));
}

// PATCH /projects/:id { name } -> Project
// Rejects (throws) if another project already has that name (case-insensitive).
export async function renameProject(id: string, name: string): Promise<Project> {
  await randomDelay(500, 1000);
  const clash = PROJECTS.find((p) => p.id !== id && p.name.toLowerCase() === name.toLowerCase());
  if (clash) {
    throw new Error('A project named "' + clash.name + '" already exists');
  }
  const project = PROJECTS.find((p) => p.id === id);
  if (!project) {
    throw new Error('Project not found');
  }
  project.name = name;
  return { ...project };
}

// formatAgo(12) -> "12m ago", formatAgo(95) -> "1h ago", formatAgo(2880) -> "2d ago"
export function formatAgo(minutes: number): string {
  if (minutes < 60) {
    return minutes + 'm ago';
  }
  if (minutes < 1440) {
    return Math.floor(minutes / 60) + 'h ago';
  }
  return Math.floor(minutes / 1440) + 'd ago';
}

import React, { useEffect, useState } from 'react';
import { getProjects, renameProject } from './api';
import type { Project } from './api';
import { StatusDot } from './components/StatusDot';
import { formatAgo } from './utils/format';

type RowProps = {
  project: Project;
  isEditing: boolean;
  draft: string;
  error: string;
  saving: boolean;
  onStartEdit: (project: Project) => void;
  onDraftChange: (value: string) => void;
  onSave: () => void;
  onCancel: () => void;
};

// The row owns no state: the list decides which row is being edited.
function ProjectRow({ project, isEditing, draft, error, saving, onStartEdit, onDraftChange, onSave, onCancel }: RowProps) {
  function handleKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
    if (e.key === 'Enter') {
      onSave();
    }
    if (e.key === 'Escape') {
      onCancel();
    }
  }

  let nameEl = (
    <span className="project-name" onDoubleClick={() => { onStartEdit(project); }}>{project.name}</span>
  );
  let actions = <button type="button" className="btn-small" disabled={saving} onClick={() => { onStartEdit(project); }}>Edit</button>;
  if (isEditing) {
    let inputCls = 'name-input';
    if (error) {
      inputCls = 'name-input has-error';
    }
    let message = null;
    if (error) {
      message = <span className="name-error">{error}</span>;
    }
    let saveLabel = 'Save';
    if (saving) {
      saveLabel = 'Saving…';
    }
    nameEl = (
      <span className="name-edit">
        <input
          className={inputCls}
          aria-label="Project name"
          value={draft}
          disabled={saving}
          autoFocus
          onFocus={(e) => { e.target.select(); }}
          onChange={(e) => { onDraftChange(e.target.value); }}
          onKeyDown={handleKeyDown}
        />
        {message}
      </span>
    );
    actions = (
      <>
        <button type="button" className="btn-small" disabled={saving} onClick={() => { onCancel(); }}>Cancel</button>
        <button type="button" className="btn-small btn-dark" disabled={saving} onClick={() => { onSave(); }}>{saveLabel}</button>
      </>
    );
  }

  let rowCls = 'project-row';
  if (isEditing) {
    rowCls = 'project-row is-editing';
  }

  return (
    <li className={rowCls}>
      <div className="project-main">
        {nameEl}
        <span className="project-sub">{project.framework} · <code>{project.branch}</code></span>
      </div>
      <StatusDot status={project.status} />
      <span className="project-time">{formatAgo(project.deployedMinutesAgo)}</span>
      <div className="row-actions">
        {actions}
      </div>
    </li>
  );
}

function ProjectList() {
  const [projects, setProjects] = useState<Project[]>([]);
  const [loading, setLoading] = useState(true);
  // Change: one editingId for the whole list means only one row can be open at a time.
  const [editingId, setEditingId] = useState<string | null>(null);
  const [draft, setDraft] = useState('');
  const [error, setError] = useState('');
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    let cancelled = false;
    getProjects().then((data) => {
      if (cancelled) {
        return;
      }
      setProjects(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  function handleStartEdit(project: Project) {
    if (saving) {
      return;
    }
    // Opening another row simply replaces the old draft (the old row cancels).
    setEditingId(project.id);
    setDraft(project.name);
    setError('');
  }

  function handleCancel() {
    setEditingId(null);
    setError('');
  }

  function handleDraftChange(value: string) {
    setDraft(value);
    setError('');
  }

  function handleSave() {
    const id = editingId;
    const name = draft.trim();
    const current = projects.find((p) => {
      return p.id === id;
    });
    if (!id || !current) {
      return;
    }
    if (name === '') {
      setError("Name can't be empty");
      return;
    }
    if (name === current.name) {
      handleCancel();
      return;
    }
    setSaving(true);
    renameProject(id, name)
      .then((updated) => {
        setProjects((prev) => {
          return prev.map((p) => {
            if (p.id === updated.id) {
              return updated;
            }
            return p;
          });
        });
        setSaving(false);
        setEditingId(null);
      })
      .catch((err: Error) => {
        // Server said no (e.g. duplicate name): stay in edit mode and show why.
        setSaving(false);
        setError(err.message);
      });
  }

  if (loading) {
    return <p className="empty">Loading projects…</p>;
  }

  return (
    <ul className="project-list">
      {projects.map((p) => {
        return (
          <ProjectRow
            key={p.id}
            project={p}
            isEditing={p.id === editingId}
            draft={draft}
            error={error}
            saving={saving}
            onStartEdit={handleStartEdit}
            onDraftChange={handleDraftChange}
            onSave={handleSave}
            onCancel={handleCancel}
          />
        );
      })}
    </ul>
  );
}

export default function App() {
  return (
    <div className="app">
      <Sidebar />
      <main className="main">
        <div className="page-head">
          <div>
            <h1>Projects</h1>
            <p className="page-sub">Double-click a name to rename it.</p>
          </div>
          <button type="button" className="btn-primary">New project</button>
        </div>
        <ProjectList />
      </main>
    </div>
  );
}
