// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /stays?region=west -> Stay[]
export async function getStays(): Promise<Stay[]> {
  await randomDelay(300, 1000);
  return STAYS.map((s) => {
    return { ...s };
  });
}

}

import React, { useEffect, useRef, useState } from 'react';
import { getStays } from './api';
import type { Stay } from './api';
import { StayCard } from './components/StayCard';

type SortKey = 'recommended' | 'price-asc' | 'price-desc' | 'rating';

const SORT_OPTIONS: { key: SortKey; label: string }[] = [
  { key: 'recommended', label: 'Recommended' },
  { key: 'price-asc', label: 'Price: low to high' },
  { key: 'price-desc', label: 'Price: high to low' },
  { key: 'rating', label: 'Top rated' },
];

function Header() {
  return (
    <header className="header">
      <span className="logo">staybird</span>
      <span className="search-pill">West Coast · Nov 20–23 · 2 guests</span>
    </header>
  );
}

type SortMenuProps = {
  value: SortKey;
  onChange: (key: SortKey) => void;
};

function SortMenu({ value, onChange }: SortMenuProps) {
  // CORE: open is local UI state: nothing else on the page cares whether the menu is open.
  const [open, setOpen] = useState(false);
  // NEXT: a ref on the wrapper for the outside-click check
  const wrapRef = useRef<HTMLDivElement>(null);

  // NEXT: close on outside click (CHANGE: and Escape). Only listen while open.
  useEffect(() => {
    if (!open) {
      return;
    }
    function handleMouseDown(e: MouseEvent) {
      if (wrapRef.current && !wrapRef.current.contains(e.target as Node)) {
        setOpen(false);
      }
    }
    // CHANGE: Escape (plus the keydown add/remove lines below)
    function handleKeyDown(e: KeyboardEvent) {
      if (e.key === 'Escape') {
        setOpen(false);
      }
    }
    document.addEventListener('mousedown', handleMouseDown);
    document.addEventListener('keydown', handleKeyDown);
    return () => {
      document.removeEventListener('mousedown', handleMouseDown);
      document.removeEventListener('keydown', handleKeyDown);
    };
  }, [open]);

  // CORE
  function handleToggle() {
    setOpen((prev) => {
      return !prev;
    });
  }

  function handleSelect(key: SortKey) {
    onChange(key);
    setOpen(false);
  }

  const current = SORT_OPTIONS.find((o) => {
    return o.key === value;
  });
  let currentLabel = 'Recommended';
  if (current) {
    currentLabel = current.label;
  }

  // CORE: the menu (STRETCH: is-selected + ✓ on the current option)
  let menu = null;
  if (open) {
    menu = (
      <ul className="sort-menu" role="listbox">
        {SORT_OPTIONS.map((o) => {
          const selected = o.key === value;
          let cls = 'sort-option';
          let check = null;
          if (selected) {
            cls = 'sort-option is-selected';
            check = <span className="check">✓</span>;
          }
          return (
            <li key={o.key} role="option" aria-selected={selected}>
              <button type="button" className={cls} onClick={() => { handleSelect(o.key); }}>
                {o.label}
                {check}
              </button>
            </li>
          );
        })}
      </ul>
    );
  }

  return (
    <div className="sort" ref={wrapRef}>
      <button type="button" className="sort-btn" aria-haspopup="listbox" aria-expanded={open} onClick={() => { handleToggle(); }}>
        Sort by: <b>{currentLabel}</b> <span className="caret">▾</span>
      </button>
      {menu}
    </div>
  );
}

// NEXT: pure, returns a NEW array. .sort() mutates, so copy first.
function sortStays(stays: Stay[], key: SortKey): Stay[] {
  if (key === 'recommended') {
    return stays;
  }
  const copy = [...stays];
  copy.sort((a, b) => {
    if (key === 'price-asc') {
      return a.pricePerNight - b.pricePerNight;
    }
    if (key === 'price-desc') {
      return b.pricePerNight - a.pricePerNight;
    }
    return b.rating - a.rating;
  });
  return copy;
}

function StaysPage() {
  const [stays, setStays] = useState<Stay[]>([]);
  const [loading, setLoading] = useState(true);
  // Lives here because the grid will need it.
  const [sortKey, setSortKey] = useState<SortKey>('recommended');

  useEffect(() => {
    let cancelled = false;
    getStays().then((data) => {
      if (cancelled) {
        return;
      }
      setStays(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // NEXT: derived, never stored in state (stays keeps the API's recommended order)
  const sorted = sortStays(stays, sortKey);

  let body = <p className="loading">Finding stays…</p>;
  if (!loading) {
    body = (
      <ul className="stay-grid">
        {sorted.map((stay) => {
          return <StayCard key={stay.id} stay={stay} />;
        })}
      </ul>
    );
  }

  return (
    <main className="page">
      <div className="toolbar">
        <h1>{stays.length} stays</h1>
        <SortMenu value={sortKey} onChange={setSortKey} />
      </div>
      {body}
    </main>
  );
}

export default function App() {
  return (
    <div className="app">
      <Header />
      <StaysPage />
    </div>
  );
}
