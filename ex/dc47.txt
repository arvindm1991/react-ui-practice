function copyComment(c: Comment): Comment {
  return { ...c, replies: c.replies.map(copyComment) };
}

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /threads/812 -> Thread
export async function getThread(): Promise<Thread> {
  await randomDelay(400, 1000);
  return { ...THREAD, comments: THREAD.comments.map(copyComment) };
}

// POST /comments/:id/upvote -> { ok: true }
// shouldFail = true simulates being offline: the request rejects.
export async function upvoteComment(id: string, shouldFail: boolean): Promise<{ ok: true }> {
  console.log('POST /comments/' + id + '/upvote');
  await randomDelay(400, 900);
  if (shouldFail) {
    throw new Error('Network error');
  }
  return { ok: true };
}

// timeAgo(8) -> "8m ago", timeAgo(190) -> "3h ago", timeAgo(3000) -> "2d ago"
export function timeAgo(minutes: number): string {
  if (minutes < 60) {
    return minutes + 'm ago';
  }
  if (minutes < 60 * 24) {
    return Math.floor(minutes / 60) + 'h ago';
  }
  return Math.floor(minutes / (60 * 24)) + 'd ago';
}

const COLORS = ['#F97316', '#0EA5E9', '#22C55E', '#A855F7', '#EF4444', '#14B8A6'];

// Colored initial. The color is stable per name.
export default function Avatar({ name }: { name: string }) {
  let hash = 0;
  for (let i = 0; i < name.length; i++) {
    hash = hash + name.charCodeAt(i);
  }
  const style = { background: COLORS[hash % COLORS.length] };
  return <span className="avatar" style={style} aria-hidden="true">{name.charAt(0).toUpperCase()}</span>;
}

import React, { useEffect, useState } from 'react';
import { getThread, upvoteComment } from './api';
import type { Comment, Thread } from './api';
import { timeAgo } from './utils/time';
import Avatar from './components/Avatar';

// CHANGE: sort mode
type SortMode = 'top' | 'new';
// NEXT: votes map lives in App
type Votes = Record<string, boolean>; // comment id -> I upvoted it

// CORE: recursive count for "4 replies hidden"
// Every reply below this comment, at any depth.
function countDescendants(c: Comment): number {
  return c.replies.reduce((sum, r) => {
    return sum + 1 + countDescendants(r);
  }, 0);
}

// NEXT: shown score is derived
// The score shown = server score + my (optimistic) vote. Derived, never stored.
function displayScore(c: Comment, votes: Votes): number {
  if (votes[c.id]) {
    return c.score + 1;
  }
  return c.score;
}

// CHANGE: sort a copy, by score or by time
// Sort a copy. Never .sort() the array from state.
function sortComments(list: Comment[], mode: SortMode, votes: Votes): Comment[] {
  return [...list].sort((a, b) => {
    if (mode === 'new') {
      return a.minutesAgo - b.minutesAgo;
    }
    return displayScore(b, votes) - displayScore(a, votes);
  });
}

function PostHeader({ thread }: { thread: Thread }) {
  return (
    <header className="post">
      <div className="post-votes"><span className="post-arrow">▲</span><b>{thread.points}</b></div>
      <div>
        <h1>{thread.title}</h1>
        <p className="post-meta">posted by {thread.author} · {thread.comments.length} threads</p>
      </div>
    </header>
  );
}

// CORE: recursive node (CHANGE: sort prop; NEXT: votes + onVote)
type CommentNodeProps = {
  comment: Comment;
  sort: SortMode;
  votes: Votes;
  onVote: (c: Comment) => void;
};

// Recursive: a comment renders its replies with CommentNode.
function CommentNode({ comment, sort, votes, onVote }: CommentNodeProps) {
  // CORE: collapse is local to each node
  // Collapse state is local: it only affects this subtree.
  const [collapsed, setCollapsed] = useState(false);

  // NEXT
  const voted = votes[comment.id] === true;
  const hiddenCount = countDescendants(comment);

  // CORE: [–] / [+] and the hidden count
  let toggleLabel = '[–]';
  if (collapsed) {
    toggleLabel = '[+]';
  }

  let body = null;
  if (collapsed) {
    let noun = 'replies';
    if (hiddenCount === 1) {
      noun = 'reply';
    }
    let hiddenEl = null;
    if (hiddenCount > 0) {
      hiddenEl = <span className="hidden-count">{hiddenCount} {noun} hidden</span>;
    }
    body = hiddenEl;
  } else {
    // NEXT: voted state of the button
    let voteCls = 'vote-btn';
    if (voted) {
      voteCls = 'vote-btn is-voted';
    }
    // CORE: recursion; replies indented with a thread line
    let replies = null;
    if (comment.replies.length > 0) {
      replies = (
        <div className="replies">
          {sortComments(comment.replies, sort, votes).map((r) => {
            return <CommentNode key={r.id} comment={r} sort={sort} votes={votes} onVote={onVote} />;
          })}
        </div>
      );
    }
    body = (
      <div>
        <p className="comment-text">{comment.text}</p>
        <div className="comment-actions">
          <button type="button" className={voteCls} aria-label={'Upvote ' + comment.author} aria-pressed={voted} disabled={voted} onClick={() => { onVote(comment); }}>▲</button>
          <span className="score">{displayScore(comment, votes)}</span>
        </div>
        {replies}
      </div>
    );
  }

  let cls = 'comment';
  if (collapsed) {
    cls = 'comment is-collapsed';
  }

  return (
    <div className={cls} data-id={comment.id}>
      <div className="comment-head">
        <button type="button" className="collapse-btn" aria-expanded={!collapsed} aria-label={'Toggle comment by ' + comment.author} onClick={() => { setCollapsed(!collapsed); }}>{toggleLabel}</button>
        <Avatar name={comment.author} />
        <span className="comment-author">{comment.author}</span>
        <span className="comment-time">{timeAgo(comment.minutesAgo)}</span>
      </div>
      {body}
    </div>
  );
}

export default function App() {
  const [thread, setThread] = useState<Thread | null>(null);
  const [offline, setOffline] = useState(false);
  // NEXT
  const [votes, setVotes] = useState<Votes>({});
  // CHANGE
  const [sort, setSort] = useState<SortMode>('top');
  // NEXT
  const [error, setError] = useState('');

  useEffect(() => {
    let cancelled = false;
    getThread().then((data) => {
      if (cancelled) {
        return;
      }
      setThread(data);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // NEXT: optimistic vote + rollback
  // Optimistic: count the vote now, roll it back if the request fails.
  function handleVote(c: Comment) {
    setError('');
    setVotes((prev) => {
      return { ...prev, [c.id]: true };
    });
    upvoteComment(c.id, offline).catch(() => {
      setVotes((prev) => {
        const next = { ...prev };
        delete next[c.id];
        return next;
      });
      setError('Your vote on ' + c.author + '’s comment could not be saved. Try again.');
    });
  }

  if (!thread) {
    return <div className="app"><p className="loading">Loading discussion…</p></div>;
  }

  // NEXT
  let errorEl = null;
  if (error) {
    errorEl = <div className="vote-error" role="alert">{error}</div>;
  }

  // CHANGE
  const topLevel = sortComments(thread.comments, sort, votes);

  // CORE: top level renders CommentNode (NEXT: {errorEl}; CHANGE: Sort by bar + sorted list)
  return (
    <div className="app">
      <nav className="topnav">
        <span className="brand">threadline</span>
        <label className="offline-toggle">
          <input type="checkbox" checked={offline} onChange={(e) => { setOffline(e.target.checked); }} />
          Simulate offline (votes fail)
        </label>
      </nav>
      <PostHeader thread={thread} />
      {errorEl}
      <section className="comments">
        {/* CHANGE: Sort by select */}
        <div className="comments-bar">
          <span className="comments-title">Comments</span>
          <label className="sort-label">
            Sort by
            <select className="sort-select" value={sort} onChange={(e) => { setSort(e.target.value as SortMode); }}>
              <option value="top">Top</option>
              <option value="new">New</option>
            </select>
          </label>
        </div>
        {topLevel.map((c) => {
          return <CommentNode key={c.id} comment={c} sort={sort} votes={votes} onVote={handleVote} />;
        })}
      </section>
    </div>
  );
}
