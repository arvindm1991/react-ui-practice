let nextId = 100;

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /recipes/lemon-risotto/comments -> Comment[]
export async function getComments(): Promise<Comment[]> {
  await randomDelay(400, 900);
  return COMMENTS.map((c) => ({ ...c, replies: c.replies.map((r) => ({ ...r })) }));
}

// POST /comments/:id/replies { text } -> the created Reply
export async function postReply(commentId: string, text: string): Promise<Reply> {
  await randomDelay(700, 1200);
  nextId += 1;
  return { id: 'r' + nextId, author: 'You', text, minutesAgo: 0 };
}

// timeAgo(0) -> "just now", timeAgo(30) -> "30m ago", timeAgo(180) -> "3h ago"
export function timeAgo(minutes: number): string {
  if (minutes < 1) {
    return 'just now';
  }
  if (minutes < 60) {
    return minutes + 'm ago';
  }
  return Math.floor(minutes / 60) + 'h ago';
}

import React from 'react';

const COLORS = ['#7C9A5B', '#C2410C', '#B45309', '#0F766E', '#9D174D'];

export default function Avatar({ name, small }: { name: string; small?: boolean }) {
  let sum = 0;
  for (let i = 0; i < name.length; i++) {
    sum += name.charCodeAt(i);
  }
  let cls = 'avatar';
  if (small) {
    cls = 'avatar avatar-sm';
  }
  return <span className={cls} style={{ background: COLORS[sum % COLORS.length] }}>{name[0]}</span>;
}

const MAX_REPLY = 280;

type ReplyBoxProps = {
  // NEXT: whose comment (for the placeholder)
  author: string;
  onSubmit: (text: string) => void;
  onCancel: () => void;
};

// Built by a teammate (starter). Owns its own draft + pending state.
// The parent unmounts it when the post succeeds.
function ReplyBox({ author, onSubmit, onCancel }: ReplyBoxProps) {
  const [text, setText] = useState('');
  // NEXT: pending flag
  const [pending, setPending] = useState(false);

  // NEXT: never post twice (CHANGE: or when too long)
  function handleSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    // NEXT: blank, too long or already posting: do nothing
    if (text.trim() === '' || pending || text.length > MAX_REPLY) {
      return;
    }
    setPending(true);
    onSubmit(text.trim());
  }

  // CHANGE: characters left (derived), red once it goes below 0
  const left = MAX_REPLY - text.length;
  let counterCls = 'char-count';
  if (left < 0) {
    counterCls = 'char-count is-over';
  }

  // NEXT: disabled + "Posting…" while pending (CHANGE: and when too long)
  const blocked = pending || text.trim() === '' || left < 0;
  let submitLabel = 'Post reply';
  if (pending) {
    submitLabel = 'Posting…';
  }

  // NEXT: autoFocus, placeholder, disabled (CHANGE: the counter span)
  return (
    <form className="reply-box" onSubmit={handleSubmit}>
      <textarea
        className="reply-input"
        placeholder={'Reply to ' + author + '…'}
        autoFocus
        disabled={pending}
        value={text}
        onChange={(e) => { setText(e.target.value); }}
      />
      <div className="reply-actions">
        <span className={counterCls}>{left} left</span>
        <button type="button" className="ghost-btn" disabled={pending} onClick={() => { onCancel(); }}>Cancel</button>
        <button type="submit" className="post-btn" disabled={blocked}>{submitLabel}</button>
      </div>
    </form>
  );
}

// CORE: which comment is being replied to comes from the parent
type CommentItemProps = {
  comment: Comment;
  replying: boolean;
  onReplyClick: (id: string) => void;
  onPostReply: (id: string, text: string) => void;
  onCancelReply: () => void;
  // STRETCH: collapse / expand
  expanded: boolean;
  onToggleReplies: (id: string) => void;
};

function CommentItem({ comment, replying, expanded, onReplyClick, onToggleReplies, onPostReply, onCancelReply }: CommentItemProps) {
  // STRETCH: "Show 3 replies" / "Hide replies" link
  const count = comment.replies.length;

  let toggle = null;
  if (count > 0) {
    let label = 'Hide replies';
    if (!expanded) {
      let noun = 'replies';
      if (count === 1) {
        noun = 'reply';
      }
      label = 'Show ' + count + ' ' + noun;
    }
    toggle = <button type="button" className="link-btn muted-link" onClick={() => { onToggleReplies(comment.id); }}>{label}</button>;
  }

  // STRETCH: replies only while expanded
  let replies = null;
  if (expanded && count > 0) {
    replies = (
      <ul className="replies">
        {comment.replies.map((r) => {
          return <ReplyItem key={r.id} reply={r} />;
        })}
      </ul>
    );
  }

  // CORE: the reply box only under the comment being replied to
  let box = null;
  if (replying) {
    box = (
      <ReplyBox
        author={comment.author}
        onSubmit={(text) => { onPostReply(comment.id, text); }}
        onCancel={onCancelReply}
      />
    );
  }

  // CORE: render (STRETCH: toggle + collapsible replies)
  return (
    <li className="comment">
      <Avatar name={comment.author} />
      <div className="comment-body">
        <p className="meta"><b>{comment.author}</b> · {timeAgo(comment.minutesAgo)}</p>
        <p className="text">{comment.text}</p>
        <div className="comment-actions">
          <button type="button" className="link-btn" onClick={() => { onReplyClick(comment.id); }}>Reply</button>
          {toggle}
        </div>
        {replies}
        {box}
      </div>
    </li>
  );
}

function CommentSection() {
  const [comments, setComments] = useState<Comment[]>([]);
  const [loading, setLoading] = useState(true);
  // CORE: only one reply box on the page: store WHICH comment, not a boolean per comment.
  const [replyingTo, setReplyingTo] = useState<string | null>(null);
  // STRETCH: which threads are expanded, by comment id
  const [expanded, setExpanded] = useState<Record<string, boolean>>({});

  useEffect(() => {
    let cancelled = false;
    getComments().then((data) => {
      if (cancelled) {
        return;
      }
      setComments(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // STRETCH: toggle one thread
  function handleToggleReplies(id: string) {
    setExpanded((prev) => {
      return { ...prev, [id]: !prev[id] };
    });
  }

  // CORE: post, then append the reply immutably and close the box
  function handlePostReply(commentId: string, text: string) {
    return postReply(commentId, text).then((reply) => {
      setComments((prev) => {
        return prev.map((c) => {
          if (c.id !== commentId) {
            return c;
          }
          return { ...c, replies: [...c.replies, reply] };
        });
      });
      setReplyingTo(null);
      // STRETCH: show the thread so the new reply is visible.
      setExpanded((prev) => {
        return { ...prev, [commentId]: true };
      });
    });
  }

  let body;
  if (loading) {
    body = <p className="muted">Loading comments…</p>;
  } else {
    body = (
      <ul className="comments">
        {comments.map((c) => {
          // CORE: derive each comment's flags
          const replying = replyingTo === c.id;
          const isExpanded = expanded[c.id] === true;
          return (
            <CommentItem
              key={c.id}
              comment={c}
              replying={replying}
              expanded={isExpanded}
              onReplyClick={setReplyingTo}
              onToggleReplies={handleToggleReplies}
              onPostReply={handlePostReply}
              onCancelReply={() => { setReplyingTo(null); }}
            />
          );
        })}
      </ul>
    );
  }

  return (
    <section className="discussion">
      <h2>Comments</h2>
      {body}
    </section>
  );
}

      <main className="page">
        <RecipeHero />
        <CommentSection />
      </main>
    </div>
  );
}
