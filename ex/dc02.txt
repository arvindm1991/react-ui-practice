import React from 'react';

type FieldProps = {
  label: string;
  htmlFor: string;
  hint?: string;
  children: React.ReactNode;
};

// Label + control + optional hint, with consistent spacing.
export function Field({ label, htmlFor, hint, children }: FieldProps) {
  let hintEl = null;
  if (hint) {
    hintEl = <p className="field-hint">{hint}</p>;
  }
  return (
    <div className="field">
      <label className="field-label" htmlFor={htmlFor}>{label}</label>
      {children}
      {hintEl}
    </div>
  );
}

import React, { useState } from 'react';
import { Field } from './components/Field';

const BIO_LIMIT = 160;
const WARN_AT = 20;

function Sidebar() {
  return (
    <nav className="sidebar">
      <div className="logo"><span className="logo-mark">&gt;_</span> forge</div>
      <a className="nav-item is-active">Profile</a>
      <a className="nav-item">Account</a>
      <a className="nav-item">SSH keys</a>
      <a className="nav-item">Access tokens</a>
    </nav>
  );
}

type BioMeterProps = {
  length: number;
};

// Counter + progress bar. Everything here is derived from the length.
function BioMeter({ length }: BioMeterProps) {
  const remaining = BIO_LIMIT - length;

  let status = 'ok';
  if (remaining < 0) {
    status = 'over';
  } else if (remaining < WARN_AT) {
    status = 'warn';
  }

  let text = remaining + ' characters left';
  if (remaining === 1) {
    text = '1 character left';
  } else if (remaining === -1) {
    text = '1 character over';
  } else if (remaining < 0) {
    text = -remaining + ' characters over';
  }

  // Change: bar width as a percentage, capped at 100.
  const percent = Math.min(100, (length / BIO_LIMIT) * 100);
  const fillStyle = { width: percent + '%' };
  const meterCls = 'bio-meter is-' + status;

  return (
    <div className={meterCls}>
      <div className="bio-bar">
        <div className="bio-bar-fill" style={fillStyle} />
      </div>
      <span className="bio-count">{text}</span>
    </div>
  );
}

function ProfileForm() {
  const [name, setName] = useState('Sam Okafor');
  const [bio, setBio] = useState('Backend engineer. I like boring tech, fast builds and good logs.');
  const [savedAt, setSavedAt] = useState('');

  function handleSave() {
    setSavedAt(new Date().toLocaleTimeString());
  }

  // derived: never stored in state
  const isOver = bio.length > BIO_LIMIT;

  let savedNote = null;
  if (savedAt) {
    savedNote = <span className="saved-note">Saved at {savedAt}</span>;
  }

  let bioCls = 'input bio-input';
  if (isOver) {
    bioCls = 'input bio-input is-over';
  }

  return (
    <section className="panel">
      <h1>Public profile</h1>
      <p className="sub">This is how other developers see you on forge.</p>

      <Field label="Display name" htmlFor="name">
        <input id="name" className="input" value={name} onChange={(e) => { setName(e.target.value); }} />
      </Field>

      <Field label="Bio" htmlFor="bio">
        <textarea id="bio" className={bioCls} rows={4} value={bio} onChange={(e) => { setBio(e.target.value); }} />
        <BioMeter length={bio.length} />
      </Field>

      <div className="form-footer">
        {savedNote}
        <button type="button" className="btn-primary" disabled={isOver} onClick={() => { handleSave(); }}>Save profile</button>
      </div>
    </section>
  );
}

export default function App() {
  return (
    <div className="layout">
      <Sidebar />
      <main className="main">
        <ProfileForm />
      </main>
    </div>
  );
}
