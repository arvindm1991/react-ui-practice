import React, { useState } from 'react';
import { plans, faqs } from './data';

function Tabs({ items }) {
  const [activeIndex, setActiveIndex] = useState(0);
  const active = items[activeIndex];

  function onKeyDown(e) {
    if (e.key !== 'ArrowRight' && e.key !== 'ArrowLeft') return;
    const dir = e.key === 'ArrowRight' ? 1 : -1;
    const next = (activeIndex + dir + items.length) % items.length;
    setActiveIndex(next);
    // move focus to the newly active tab
    const tabs = e.currentTarget.querySelectorAll('[role="tab"]');
    if (tabs[next]) tabs[next].focus();
  }

  return (
    <div>
      <div className="tablist" role="tablist" onKeyDown={onKeyDown}>
        {items.map((item, i) => (
          <button
            key={item.id}
            id={`tab-${item.id}`}
            role="tab"
            className="tab"
            aria-selected={i === activeIndex}
            aria-controls={`panel-${item.id}`}
            tabIndex={i === activeIndex ? 0 : -1}
            onClick={() => setActiveIndex(i)}
          >
            {item.label}
          </button>
        ))}
      </div>
      <div className="panel" role="tabpanel" id={`panel-${active.id}`} aria-labelledby={`tab-${active.id}`}>
        <p className="price">{active.price}</p>
        <ul className="features">
          {active.features.map((f) => (
            <li key={f}>{f}</li>
          ))}
        </ul>
      </div>
    </div>
  );
}

function Accordion({ items }) {
  const [openId, setOpenId] = useState(null); // single value = only one open

  return (
    <div className="accordion">
      {items.map((item) => {
        const open = item.id === openId;
        return (
          <div key={item.id} className="acc-item">
            <button
              className="acc-trigger"
              aria-expanded={open}
              aria-controls={`acc-${item.id}`}
              onClick={() => setOpenId(open ? null : item.id)}
            >
              {item.q}
              <span className="chevron" aria-hidden="true">▾</span>
            </button>
            {open && (
              <p className="acc-body" id={`acc-${item.id}`}>
                {item.a}
              </p>
            )}
          </div>
        );
      })}
    </div>
  );
}

export default function App() {
  return (
    <div className="page">
      <section>
        <h2>Plans</h2>
        <Tabs items={plans} />
      </section>
      <section>
        <h2>FAQ</h2>
        <Accordion items={faqs} />
      </section>
    </div>
  );
}
