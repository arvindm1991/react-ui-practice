// Fake search API. Treat it like a real backend you can't change.
import { PAGES } from './pages';
import type { Page } from './pages';

export type DocHit = Page;

function abortError(): DOMException {
  return new DOMException('The request was aborted.', 'AbortError');
}

// Like fetch: waits `ms`, but rejects with an AbortError as soon as `signal` aborts.
function abortableDelay(ms: number, signal: AbortSignal): Promise<void> {
  return new Promise((resolve, reject) => {
    if (signal.aborted) {
      reject(abortError());
      return;
    }
    function onAbort() {
      clearTimeout(timer);
      reject(abortError());
    }
    const timer = setTimeout(() => {
      signal.removeEventListener('abort', onAbort);
      resolve();
    }, ms);
    signal.addEventListener('abort', onAbort, { once: true });
  });
}

// GET /search?q=...
// Latency: queries of 1-3 characters scan the whole index and take 0.6-1.1 s.
// Longer queries hit the prefix index and take 0.12-0.24 s.
// Pass shouldFail = true to get a rejected promise (for the error state).
export async function searchDocs(query: string, signal: AbortSignal, shouldFail = false): Promise<DocHit[]> {
  const q = query.trim().toLowerCase();
  const ms = q.length <= 3 ? 600 + Math.random() * 500 : 120 + Math.random() * 120;
  await abortableDelay(ms, signal);
  if (shouldFail) {
    throw new Error('Search is temporarily unavailable.');
  }
  return PAGES
    .filter((p) => p.title.toLowerCase().includes(q) || p.section.toLowerCase().includes(q))
    .slice(0, 8)
    .map((p) => ({ ...p }));
}

// Escape a string so it can be used inside new RegExp(...)
// escapeRegExp('a.b(c)') -> 'a\\.b\\(c\\)'
export function escapeRegExp(s: string): string {
  return s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
}

type SidebarProps = {
  currentId: string;
  onNavigate: (page: Page) => void;
};

export default function Sidebar({ currentId, onNavigate }: SidebarProps) {
  const sections: string[] = [];
  PAGES.forEach((p) => {
    if (!sections.includes(p.section)) {
      sections.push(p.section);
    }
  });

  return (
    <nav className="sidebar" aria-label="Docs">
      {sections.map((section) => {
        return (
          <div key={section} className="nav-section">
            <h4>{section}</h4>
            {PAGES.filter((p) => p.section === section).map((p) => {
              let cls = 'nav-link';
              if (p.id === currentId) {
                cls = 'nav-link is-current';
              }
              return (
                <button key={p.id} type="button" className={cls} onClick={() => { onNavigate(p); }}>
                  {p.title}
                </button>
              );
            })}
          </div>
        );
      })}
    </nav>
  );
}

@@@ solution src/hooks/useDebouncedValue.ts
import { useEffect, useState } from 'react';

// Returns `value`, but only after it has stopped changing for `delay` ms.
export function useDebouncedValue<T>(value: T, delay: number): T {
  const [debounced, setDebounced] = useState(value);
  useEffect(() => {
    const t = setTimeout(() => {
      setDebounced(value);
    }, delay);
    return () => {
      clearTimeout(t);
    };
  }, [value, delay]);
  return debounced;
}

import React, { useEffect, useRef, useState } from 'react';
import { PAGES } from './pages';
import type { Page } from './pages';
import { searchDocs } from './api';
import type { DocHit } from './api';
import Sidebar from './components/Sidebar';

import { useDebouncedValue } from './hooks/useDebouncedValue';

type Status = 'idle' | 'loading' | 'done' | 'error';

function Highlight({ text, query }: { text: string; query: string }) {
  if (query === '') {
    return <span>{text}</span>;
  }
  // A capture group keeps the matches in the split result, at the odd indexes.
  const parts = text.split(new RegExp('(' + escapeRegExp(query) + ')', 'ig'));
  return (
    <span>
      {parts.map((part, i) => {
        if (i % 2 === 1) {
          return <mark key={i}>{part}</mark>;
        }
        return <span key={i}>{part}</span>;
      })}
    </span>
  );
}

type SearchBoxProps = {
  shouldFail: boolean;
  onNavigate: (page: Page) => void;
};

function SearchBox({ shouldFail, onNavigate }: SearchBoxProps) {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState<DocHit[]>([]);
  const [status, setStatus] = useState<Status>('idle');
  const [slow, setSlow] = useState(false);
  const [activeIndex, setActiveIndex] = useState(0);
  const [open, setOpen] = useState(false);
  const [retry, setRetry] = useState(0);
  // Cache lives in a ref: it never needs to cause a render by itself.
  const cacheRef = useRef(new Map<string, DocHit[]>());
  const inputRef = useRef<HTMLInputElement>(null);

  const debounced = useDebouncedValue(query.trim().toLowerCase(), 250);

  useEffect(() => {
    if (debounced === '') {
      setResults([]);
      setStatus('idle');
      setSlow(false);
      return;
    }
    const cached = cacheRef.current.get(debounced);
    if (cached) {
      setResults(cached);
      setStatus('done');
      setSlow(false);
      setActiveIndex(0);
      return;
    }
    const controller = new AbortController();
    setStatus('loading');
    // Change: only admit to "Searching…" if it takes longer than 300 ms.
    const slowTimer = setTimeout(() => {
      setSlow(true);
    }, 300);

    searchDocs(debounced, controller.signal, shouldFail)
      .then((hits) => {
        cacheRef.current.set(debounced, hits);
        setResults(hits);
        setStatus('done');
        setActiveIndex(0);
        setSlow(false);
        clearTimeout(slowTimer);
      })
      .catch((err) => {
        if (err.name === 'AbortError') {
          return; // a newer query replaced this one: not an error
        }
        setStatus('error');
        setSlow(false);
        clearTimeout(slowTimer);
      });

    // A newer query (or unmount) cancels the request in flight.
    return () => {
      controller.abort();
      clearTimeout(slowTimer);
    };
  }, [debounced, shouldFail, retry]);

  // "/" anywhere on the page focuses search (unless you're typing in a field).
  useEffect(() => {
    function handleKey(e: KeyboardEvent) {
      const tag = (e.target as HTMLElement).tagName;
      if (e.key === '/' && tag !== 'INPUT' && tag !== 'TEXTAREA') {
        e.preventDefault();
        if (inputRef.current) {
          inputRef.current.focus();
        }
      }
    }
    document.addEventListener('keydown', handleKey);
    return () => {
      document.removeEventListener('keydown', handleKey);
    };
  }, []);

  function choose(hit: DocHit) {
    onNavigate(hit);
    setQuery('');
    setOpen(false);
  }

  function handleKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
    const count = results.length;
    if (e.key === 'ArrowDown' && count > 0) {
      e.preventDefault(); // stop the caret jumping to the end
      setOpen(true);
      setActiveIndex((i) => {
        return (i + 1) % count;
      });
    } else if (e.key === 'ArrowUp' && count > 0) {
      e.preventDefault();
      setActiveIndex((i) => {
        return (i - 1 + count) % count;
      });
    } else if (e.key === 'Enter') {
      const hit = results[activeIndex];
      if (open && hit && status !== 'error') {
        choose(hit);
      }
    } else if (e.key === 'Escape') {
      if (open) {
        setOpen(false);
      } else {
        setQuery('');
      }
    }
  }

  // derived
  const showPanel = open && query.trim() !== '' && status !== 'idle';
  const showResults = results.length > 0 && status !== 'error';

  let statusEl = null;
  if (status === 'loading' && slow) {
    statusEl = <p className="search-status"><span className="spinner" aria-hidden="true" />Searching…</p>;
  }

  let body = null;
  if (status === 'error') {
    body = (
      <div className="search-msg is-error">
        <span>Search is temporarily unavailable.</span>
        <button type="button" className="retry-btn" onMouseDown={(e) => { e.preventDefault(); }} onClick={() => { setRetry((n) => { return n + 1; }); }}>Retry</button>
      </div>
    );
  } else if (status === 'done' && results.length === 0) {
    body = <p className="search-msg">No results for “{query.trim()}”</p>;
  } else if (showResults) {
    body = (
      <ul className="results" role="listbox" id="search-results" aria-label="Search results">
        {results.map((hit, i) => {
          let cls = 'result';
          if (i === activeIndex) {
            cls = 'result is-active';
          }
          return (
            <li
              key={hit.id}
              id={'hit-' + hit.id}
              role="option"
              aria-selected={i === activeIndex}
              className={cls}
              onMouseDown={(e) => { e.preventDefault(); }}
              onMouseEnter={() => { setActiveIndex(i); }}
              onClick={() => { choose(hit); }}
            >
              <span className="result-section">{hit.section}</span>
              <span className="result-title"><Highlight text={hit.title} query={debounced} /></span>
              <span className="result-snippet">{hit.snippet}</span>
            </li>
          );
        })}
      </ul>
    );
  }

  let panel = null;
  if (showPanel && (statusEl || body)) {
    panel = (
      <div className="search-panel">
        {statusEl}
        {body}
      </div>
    );
  }

  let activeDescendant: string | undefined = undefined;
  if (showPanel && showResults && results[activeIndex]) {
    activeDescendant = 'hit-' + results[activeIndex].id;
  }

  return (
    <div className="search">
      <span className="search-icon" aria-hidden="true">⌕</span>
      <input
        ref={inputRef}
        className="search-input"
        type="text"
        placeholder="Search docs"
        aria-label="Search docs"
        role="combobox"
        aria-expanded={panel !== null}
        aria-controls="search-results"
        aria-autocomplete="list"
        aria-activedescendant={activeDescendant}
        value={query}
        onChange={(e) => { setQuery(e.target.value); setOpen(true); }}
        onKeyDown={handleKeyDown}
        onFocus={() => { setOpen(true); }}
        onBlur={() => { setOpen(false); }}
      />
      <kbd className="search-kbd">/</kbd>
      {panel}
    </div>
  );
}
