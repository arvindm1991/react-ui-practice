// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

function copy(node: Scope): Scope {
  if (!node.children) {
    return { ...node };
  }
  return { ...node, children: node.children.map(copy) };
}

// GET /tokens/scopes -> Scope[] (a tree)
export async function getScopes(): Promise<Scope[]> {
  await randomDelay(400, 1200);
  return SCOPES.map(copy);
}

import type { Scope } from '../api';

export function isLeaf(node: Scope): boolean {
  return !node.children || node.children.length === 0;
}

// All leaf ids under a node. A leaf returns just itself.
// getLeafIds(repoNode) -> ["repo:contents:read", "repo:contents:write", ...]
export function getLeafIds(node: Scope): string[] {
  if (isLeaf(node)) {
    return [node.id];
  }
  let ids: string[] = [];
  (node.children || []).forEach((child) => {
    ids = ids.concat(getLeafIds(child));
  });
  return ids;
}

// Leaf count for a whole forest: countLeaves(scopes) -> 12
export function countLeaves(nodes: Scope[]): number {
  let total = 0;
  nodes.forEach((node) => {
    total += getLeafIds(node).length;
  });
  return total;
}

type TriCheckboxProps = {
  id: string;
  checked: boolean;
  indeterminate: boolean;
  onToggle: () => void;
};

// "indeterminate" is a DOM property, not an attribute, so React can't set it.
function TriCheckbox({ id, checked, indeterminate, onToggle }: TriCheckboxProps) {
  const ref = useRef<HTMLInputElement>(null);

  // No deps on purpose: the browser clears .indeterminate on every click,
  // so re-apply the derived value after every render.
  useEffect(() => {
    if (ref.current) {
      ref.current.indeterminate = indeterminate;
    }
  });

  return (
    <input
      ref={ref}
      id={id}
      type="checkbox"
      className="tri-check"
      checked={checked}
      onChange={() => { onToggle(); }}
    />
  );
}

function matches(node: Scope, q: string): boolean {
  return node.label.toLowerCase().includes(q) || node.id.toLowerCase().includes(q);
}

function addSubtree(node: Scope, out: Set<string>) {
  out.add(node.id);
  (node.children || []).forEach((child) => {
    addSubtree(child, out);
  });
}

// Fills `out` with ids to show: matches, everything under a matching group,
// and the ancestors of any match. Returns true if anything in `nodes` is visible.
function collectVisible(nodes: Scope[], q: string, out: Set<string>): boolean {
  let any = false;
  nodes.forEach((node) => {
    if (matches(node, q)) {
      addSubtree(node, out);
      any = true;
      return;
    }
    if (node.children && collectVisible(node.children, q, out)) {
      out.add(node.id);
      any = true;
    }
  });
  return any;
}

type ScopeNodeProps = {
  node: Scope;
  selected: Set<string>;
  visible: Set<string> | null;
  onToggle: (node: Scope) => void;
};

function ScopeNode({ node, selected, visible, onToggle }: ScopeNodeProps) {
  // derived: a group's state comes from its leaves, never stored
  const leafIds = getLeafIds(node);
  const onCount = leafIds.filter((id) => {
    return selected.has(id);
  }).length;
  const checked = onCount === leafIds.length;
  const indeterminate = onCount > 0 && !checked;

  let children = null;
  if (!isLeaf(node)) {
    children = (
      <ul className="scope-children" role="group">
        {(node.children || []).map((child) => {
          if (visible && !visible.has(child.id)) {
            return null;
          }
          return <ScopeNode key={child.id} node={child} selected={selected} visible={visible} onToggle={onToggle} />;
        })}
      </ul>
    );
  }

  let rowCls = 'scope-row';
  if (checked) {
    rowCls = 'scope-row is-on';
  } else if (indeterminate) {
    rowCls = 'scope-row is-mixed';
  }

  let count = null;
  if (!isLeaf(node)) {
    count = <span className="scope-count">{onCount}/{leafIds.length}</span>;
  }

  const inputId = 'scope-' + node.id;
  return (
    <li className="scope">
      <div className={rowCls}>
        <TriCheckbox id={inputId} checked={checked} indeterminate={indeterminate} onToggle={() => { onToggle(node); }} />
        <label className="scope-text" htmlFor={inputId}>
          <span className="scope-label">{node.label}</span>
          <span className="scope-desc">{node.description}</span>
        </label>
        {count}
        <code className="scope-code">{node.id}</code>
      </div>
      {children}
    </li>
  );
}

type PermissionsCardProps = {
  selected: Set<string>;
  onToggle: (node: Scope) => void;
  onClear: () => void;
};

function PermissionsCard({ selected, onToggle, onClear }: PermissionsCardProps) {
  const [scopes, setScopes] = useState<Scope[]>([]);
  const [loading, setLoading] = useState(true);
  const [query, setQuery] = useState('');

  useEffect(() => {
    let cancelled = false;
    getScopes().then((data) => {
      if (cancelled) {
        return;
      }
      setScopes(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // derived
  const total = countLeaves(scopes);
  const q = query.trim().toLowerCase();
  let visible: Set<string> | null = null;
  if (q !== '') {
    visible = new Set<string>();
    collectVisible(scopes, q, visible);
  }

  let body;
  if (loading) {
    body = <p className="muted-msg">Loading scopes…</p>;
  } else if (visible && visible.size === 0) {
    body = <p className="muted-msg">No scopes match “{query}”.</p>;
  } else {
    body = (
      <ul className="scope-tree">
        {scopes.map((node) => {
          if (visible && !visible.has(node.id)) {
            return null;
          }
          return <ScopeNode key={node.id} node={node} selected={selected} visible={visible} onToggle={onToggle} />;
        })}
      </ul>
    );
  }

  let clearBtn = null;
  if (selected.size > 0) {
    clearBtn = <button type="button" className="link-btn" onClick={() => { onClear(); }}>Clear</button>;
  }

  return (
    <section className="card">
      <div className="card-head">
        <div>
          <h2>Permissions</h2>
          <p>Grant only what this token needs.</p>
        </div>
        <div className="selection-summary">
          <span className="selected-count" aria-live="polite">{selected.size} of {total} selected</span>
          {clearBtn}
        </div>
      </div>
      <div className="scope-search-wrap">
        <input
          className="input scope-search"
          type="search"
          placeholder="Filter scopes"
          aria-label="Filter scopes"
          value={query}
          onChange={(e) => { setQuery(e.target.value); }}
        />
      </div>
      {body}
    </section>
  );
}

export default function App() {
  const [name, setName] = useState('');
  const [expiry, setExpiry] = useState('30');
  // Only LEAF ids live in state. Group checkboxes are derived from them.
  const [selected, setSelected] = useState<Set<string>>(new Set());
  const [created, setCreated] = useState('');

  // Checking a group selects every leaf under it (even hidden ones).
  // An indeterminate group also selects all: only a fully checked group clears.
  function handleToggle(node: Scope) {
    const ids = getLeafIds(node);
    setSelected((prev) => {
      const allOn = ids.every((id) => {
        return prev.has(id);
      });
      const next = new Set(prev);
      ids.forEach((id) => {
        if (allOn) {
          next.delete(id);
        } else {
          next.add(id);
        }
      });
      return next;
    });
  }

  function handleGenerate() {
    let noun = 'scopes';
    if (selected.size === 1) {
      noun = 'scope';
    }
    setCreated('Token “' + name.trim() + '” created with ' + selected.size + ' ' + noun + '.');
  }

  const canGenerate = name.trim() !== '' && selected.size > 0;

  let banner = null;
  if (created) {
    banner = <p className="success-banner" role="status">{created}</p>;
  }

  return (
    <div className="page">
      <TopNav />
      <main className="content">
        <h1>New fine-grained token</h1>
        {banner}
        <section className="card form-card">
          <Field label="Token name" hint="What's this token for?">
            <input className="input" value={name} placeholder="ci-deploy" onChange={(e) => { setName(e.target.value); }} />
          </Field>
          <Field label="Expiration">
            <select className="input" value={expiry} onChange={(e) => { setExpiry(e.target.value); }}>
              <option value="7">7 days</option>
              <option value="30">30 days</option>
              <option value="90">90 days</option>
            </select>
          </Field>
        </section>
        <PermissionsCard selected={selected} onToggle={handleToggle} onClear={() => { setSelected(new Set()); }} />
        <div className="actions">
          <button type="button" className="btn-primary" disabled={!canGenerate} onClick={() => { handleGenerate(); }}>Generate token</button>
        </div>
      </main>
    </div>
  );
}
