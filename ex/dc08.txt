// formatPrice(4800) -> "$48.00", formatPrice(105000) -> "$1,050.00". Prices are always in cents.
export function formatPrice(cents: number): string {
  return (cents / 100).toLocaleString('en-US', { style: 'currency', currency: 'USD' });
}

import React, { useState } from 'react';
import { EVENT } from './data/event';
import type { GigEvent } from './data/event';
import { formatPrice } from './utils/money';

const MIN_QTY = 1;
const MAX_QTY = 8;

function EventHero({ event }: { event: GigEvent }) {
  return (
    <header className="hero">
      <p className="tour">{event.tour}</p>
      <h1>{event.artist}</h1>
      <p className="when">{event.date}</p>
      <p className="where">{event.venue}</p>
    </header>
  );
}

function TicketPanel({ event }: { event: GigEvent }) {
  // CORE: one number in state
  const [qty, setQty] = useState(MIN_QTY);

  // CORE: clamp inside the updater so fast clicks can never go out of range.
  function handleDecrease() {
    setQty((prev) => {
      return Math.max(MIN_QTY, prev - 1);
    });
  }

  function handleIncrease() {
    setQty((prev) => {
      return Math.min(MAX_QTY, prev + 1);
    });
  }

  // CORE: derived, never stored in state (math in cents, format at the end)
  const subtotal = event.priceCents * qty;
  const subtotalLabel = 'Subtotal (' + qty + ' × ' + formatPrice(event.priceCents) + ')';

  // CHANGE: fees and total feed the button label (STRETCH: their own rows)
  const fees = event.feeCents * qty;
  const total = subtotal + fees;
  const checkoutLabel = 'Checkout · ' + formatPrice(total);

  // NEXT: bounds for disabled + the max note (and aria-labels in the JSX)
  const atMin = qty === MIN_QTY;
  const atMax = qty === MAX_QTY;
  let maxNote = null;
  if (atMax) {
    maxNote = <p className="max-note">Max {MAX_QTY} tickets per order</p>;
  }

  return (
    <section className="ticket-panel">
      <div className="tier-row">
        <div>
          <p className="tier">{event.tier}</p>
          <p className="tier-price">{formatPrice(event.priceCents)} each</p>
        </div>
        <div className="stepper">
          <button type="button" className="step-btn" aria-label="Decrease quantity" disabled={atMin} onClick={() => { handleDecrease(); }}>−</button>
          <span className="qty" aria-live="polite">{qty}</span>
          <button type="button" className="step-btn" aria-label="Increase quantity" disabled={atMax} onClick={() => { handleIncrease(); }}>+</button>
        </div>
      </div>
      {maxNote}

      <div className="summary-row">
        <span>{subtotalLabel}</span>
        <span className="subtotal">{formatPrice(subtotal)}</span>
      </div>
      <div className="summary-row is-muted">
        <span>Service fees</span>
        <span className="fees">{formatPrice(fees)}</span>
      </div>
      <div className="summary-row is-total">
        <span>Total</span>
        <span className="total">{formatPrice(total)}</span>
      </div>

      <button type="button" className="checkout">{checkoutLabel}</button>
      <p className="fine">Tickets are delivered to your phone. All sales final.</p>
    </section>
  );
}

export default function App() {
  return (
    <div className="page">
      <div className="logo">gigbox</div>
      <div className="layout">
        <EventHero event={EVENT} />
        <TicketPanel event={EVENT} />
      </div>
    </div>
  );
}
