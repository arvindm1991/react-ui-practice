// Fake API. Treat it like a real backend you can't change.
export type Account = {
  id: string;
  email: string;
};

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// POST /accounts -> Account
// Rejects with a readable message if the email is already registered.
// (Try taken@example.com.)
export async function createAccount(email: string, password: string): Promise<Account> {
  await randomDelay(600, 1200);
  if (email.trim().toLowerCase() === 'taken@example.com') {
    throw new Error('An account with this email already exists. Try logging in instead.');
  }
  if (password.length < 8) {
    throw new Error('Password is too short.');
  }
  return { id: 'acc_' + Math.random().toString(36).slice(2, 8), email: email.trim() };
}

import React from 'react';

type FieldProps = {
  id: string;
  label: string;
  error?: string | null; // pass null/undefined to show no error
  children: React.ReactNode;
};

// Label + input slot + error line. The input itself is passed as children.
export function Field({ id, label, error, children }: FieldProps) {
  let errorLine = null;
  let cls = 'field';
  if (error) {
    cls = 'field has-error';
    errorLine = <p className="field-error" id={id + '-error'}>{error}</p>;
  }
  return (
    <div className={cls}>
      <label htmlFor={id}>{label}</label>
      {children}
      {errorLine}
    </div>
  );
}

// isValidEmail("a@b.co") -> true. Good enough for a form; the server double-checks.
export function isValidEmail(email: string): boolean {
  return /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(email.trim());
}

export type PasswordRule = {
  id: string;
  label: string;
  test: (password: string) => boolean;
};

export const PASSWORD_RULES: PasswordRule[] = [
  { id: 'length', label: 'At least 8 characters', test: (pw) => pw.length >= 8 },
  { id: 'number', label: 'Contains a number', test: (pw) => /\d/.test(pw) },
  { id: 'upper', label: 'Contains an uppercase letter', test: (pw) => /[A-Z]/.test(pw) },
];

import React, { useState } from 'react';
import { Field } from './components/Field';
import { createAccount } from './api';
import { isValidEmail, PASSWORD_RULES } from './utils/validate';

type FieldName = 'email' | 'password' | 'confirm';
type Status = 'idle' | 'submitting' | 'success';

function BrandPanel() {
  return (
    <aside className="brand-panel">
      <div className="logo">✦ Northstar</div>
      <div>
        <p className="quote">“It turned my messy notes into a launch post in thirty seconds.”</p>
        <p className="quote-by">Dana Whitfield, founder at Kiln</p>
      </div>
      <p className="fine">Trusted by 40,000 writers and teams.</p>
    </aside>
  );
}

function PasswordChecklist({ password }: { password: string }) {
  return (
    <ul className="rules" aria-label="Password requirements">
      {PASSWORD_RULES.map((rule) => {
        let cls = 'rule';
        let icon = '○';
        if (rule.test(password)) {
          cls = 'rule is-met';
          icon = '✓';
        }
        return (
          <li key={rule.id} className={cls}>
            <span className="rule-icon">{icon}</span> {rule.label}
          </li>
        );
      })}
    </ul>
  );
}

function SuccessPanel({ email }: { email: string }) {
  return (
    <div className="success">
      <div className="success-icon">✓</div>
      <h1>Check your inbox</h1>
      <p className="sub">We sent a confirmation link to <b>{email}</b>.</p>
    </div>
  );
}

function SignupForm() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [confirm, setConfirm] = useState('');
  // Which fields the user has left at least once. Errors wait until then (or until submit).
  const [touched, setTouched] = useState<Record<FieldName, boolean>>({ email: false, password: false, confirm: false });
  const [submitted, setSubmitted] = useState(false);
  const [status, setStatus] = useState<Status>('idle');
  const [apiError, setApiError] = useState<string | null>(null);

  function handleBlur(field: FieldName) {
    setTouched((prev) => {
      return { ...prev, [field]: true };
    });
  }

  // derived: errors are computed from the values every render, never stored
  let emailError: string | null = null;
  if (email.trim() === '') {
    emailError = 'Email is required';
  } else if (!isValidEmail(email)) {
    emailError = 'Enter a valid email address';
  }

  const allRulesMet = PASSWORD_RULES.every((rule) => {
    return rule.test(password);
  });
  let passwordError: string | null = null;
  if (!allRulesMet) {
    passwordError = 'Password doesn’t meet all the requirements';
  }

  let confirmError: string | null = null;
  if (confirm === '') {
    confirmError = 'Confirm your password';
  } else if (confirm !== password) {
    confirmError = 'Passwords don’t match';
  }

  const isValid = !emailError && !passwordError && !confirmError;

  function visibleError(field: FieldName, error: string | null) {
    if (touched[field] || submitted) {
      return error;
    }
    return null;
  }

  function handleSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setSubmitted(true);
    if (!isValid || status === 'submitting') {
      return;
    }
    setStatus('submitting');
    setApiError(null);
    createAccount(email, password)
      .then(() => {
        setStatus('success');
      })
      .catch((err: Error) => {
        setApiError(err.message);
        setStatus('idle');
      });
  }

  if (status === 'success') {
    return <SuccessPanel email={email.trim()} />;
  }

  let banner = null;
  if (apiError) {
    banner = <div className="api-error" role="alert">{apiError}</div>;
  }

  const submitting = status === 'submitting';
  let submitLabel = 'Create account';
  if (submitting) {
    submitLabel = 'Creating…';
  }

  return (
    <form className="form" onSubmit={(e) => { handleSubmit(e); }} noValidate>
      <h1>Create your account</h1>
      <p className="sub">Start writing with AI in under a minute. No card required.</p>
      {banner}
      <Field id="email" label="Work email" error={visibleError('email', emailError)}>
        <input id="email" type="email" className="input" placeholder="you@company.com" value={email} onChange={(e) => { setEmail(e.target.value); }} onBlur={() => { handleBlur('email'); }} />
      </Field>
      <Field id="password" label="Password" error={visibleError('password', passwordError)}>
        <input id="password" type="password" className="input" value={password} onChange={(e) => { setPassword(e.target.value); }} onBlur={() => { handleBlur('password'); }} />
        <PasswordChecklist password={password} />
      </Field>
      <Field id="confirm" label="Confirm password" error={visibleError('confirm', confirmError)}>
        <input id="confirm" type="password" className="input" value={confirm} onChange={(e) => { setConfirm(e.target.value); }} onBlur={() => { handleBlur('confirm'); }} />
      </Field>
      <button type="submit" className="submit" disabled={submitting}>{submitLabel}</button>
      <p className="switch">Already have an account? <a>Log in</a></p>
    </form>
  );
}

export default function App() {
  return (
    <div className="split">
      <BrandPanel />
      <main className="form-side">
        <SignupForm />
      </main>
    </div>
  );
}
