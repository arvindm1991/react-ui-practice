import React from 'react';

type ToggleProps = {
  checked: boolean;
  onChange: (next: boolean) => void;
  label: string; // read by screen readers
};

// A controlled switch. It owns no state: the parent passes `checked`
// and is told the new value through `onChange`.
export function Toggle({ checked, onChange, label }: ToggleProps) {
  let cls = 'toggle';
  if (checked) {
    cls = 'toggle is-on';
  }
  return (
    <button
      type="button"
      role="switch"
      aria-checked={checked}
      aria-label={label}
      className={cls}
      onClick={() => { onChange(!checked); }}
    >
      <span className="toggle-knob" />
    </button>
  );
}

import React, { useState } from 'react';
import { Toggle } from './components/Toggle';
import { DEFAULT_SETTINGS } from './data/notifications';
import type { NotificationSetting } from './data/notifications';

function Sidebar() {
  const items = ['Profile', 'Notifications', 'Appearance', 'Security', 'Billing'];
  return (
    <nav className="sidebar">
      <div className="brand"><span className="brand-mark">t</span> Tandem</div>
      <p className="sidebar-label">Settings</p>
      {items.map((item) => {
        let cls = 'nav-item';
        if (item === 'Notifications') {
          cls = 'nav-item is-active';
        }
        return <a key={item} className={cls}>{item}</a>;
      })}
    </nav>
  );
}

// Already works: an example of using <Toggle /> with state.
function DoNotDisturbCard() {
  const [paused, setPaused] = useState(false);

  let status = 'Notifications are on';
  if (paused) {
    status = 'Paused until tomorrow, 9:00 AM';
  }

  return (
    <section className="card dnd">
      <div>
        <h3>Do not disturb</h3>
        <p className="muted">{status}</p>
      </div>
      <Toggle checked={paused} onChange={(next) => { setPaused(next); }} label="Do not disturb" />
    </section>
  );
}

function NotificationsCard() {
  // CORE: the list becomes state. It is the single source of truth. Copy the defaults in once.
  const [settings, setSettings] = useState<NotificationSetting[]>(DEFAULT_SETTINGS);

  // CORE: flip one row (map returns a new object only for the matching id)
  function handleToggle(id: string, next: boolean) {
    setSettings((prev) => {
      return prev.map((s) => {
        if (s.id !== id) {
          return s;
        }
        return { ...s, enabled: next };
      });
    });
  }

  // NEXT: bulk on/off
  function setAll(enabled: boolean) {
    setSettings((prev) => {
      return prev.map((s) => {
        return { ...s, enabled: enabled };
      });
    });
  }

  // CORE: derived count, never stored in state
  const enabledCount = settings.filter((s) => {
    return s.enabled;
  }).length;

  // CHANGE: when everything is off, the same button turns everything back on.
  let bulkLabel = 'Turn all off';
  let bulkNext = false;
  if (enabledCount === 0) {
    bulkLabel = 'Turn all on';
    bulkNext = true;
  }

  return (
    <section className="card">
      <header className="card-head">
        <div>
          <h3>Notify me about</h3>
          <p className="muted enabled-count">{enabledCount} of {settings.length} enabled</p>
        </div>
        <button type="button" className="link-btn" onClick={() => { setAll(bulkNext); }}>{bulkLabel}</button>
      </header>
      <ul className="setting-list">
        {settings.map((s) => {
          // NEXT: muted label for disabled rows
          let cls = 'setting-row';
          if (!s.enabled) {
            cls = 'setting-row is-off';
          }
          return (
            <li key={s.id} className={cls}>
              <div>
                <p className="setting-label">{s.label}</p>
                <p className="muted">{s.description}</p>
              </div>
              <Toggle checked={s.enabled} onChange={(next) => { handleToggle(s.id, next); }} label={s.label} />
            </li>
          );
        })}
      </ul>
    </section>
  );
}

export default function App() {
  return (
    <div className="layout">
      <Sidebar />
      <main className="main">
        <h1>Notifications</h1>
        <p className="muted lead">Choose what Tandem pings you about on desktop and mobile.</p>
        <DoNotDisturbCard />
        <NotificationsCard />
      </main>
    </div>
  );
}
