const CITIES = [
  'Austin', 'Boston', 'Chicago', 'Dallas', 'Denver', 'Detroit', 'Miami', 'Nashville',
  'New Orleans', 'New York', 'Newark', 'Oakland', 'Orlando', 'Phoenix', 'Portland',
  'Sacramento', 'Salt Lake City', 'San Diego', 'San Francisco', 'San Jose', 'Santa Fe',
  'Santa Monica', 'Savannah', 'Seattle',
];

// GET /cities?q=  -> string[]
// Short queries are slow, long queries are fast, so responses can arrive
// out of order. Pass { signal } from an AbortController to cancel a request.
export function searchCities(query, { signal } = {}) {
  console.log('GET /cities?q=' + query);
  return new Promise((resolve, reject) => {
    const delay = 250 + Math.max(0, 4 - query.length) * 300 + Math.random() * 150;
    const timer = setTimeout(() => {
      const q = query.trim().toLowerCase();
      resolve(CITIES.filter((c) => c.toLowerCase().includes(q)).slice(0, 8));
    }, delay);
    if (signal) {
      signal.addEventListener('abort', () => {
        clearTimeout(timer);
        reject(new DOMException('Request aborted', 'AbortError'));
      });
    }
  });
}

      />
      <ul className="results">
        {results.map((city) => (
          <li key={city} className="result">{city}</li>
        ))}
      </ul>
    </div>
  );
}
@@@ solution src/hooks/useDebounce.js
import { useEffect, useState } from 'react';

// Returns `value`, but only after it has stopped changing for `delay` ms.
export function useDebounce(value, delay = 300) {
  const [debounced, setDebounced] = useState(value);

  useEffect(() => {
    const id = setTimeout(() => setDebounced(value), delay);
    return () => clearTimeout(id); // a new keystroke cancels the pending update
  }, [value, delay]);

  return debounced;
}
@@@ solution src/hooks/useCitySearch.js
import { useEffect, useState } from 'react';
import { searchCities } from '../api';

export function useCitySearch(query) {
  const [state, setState] = useState({ data: [], loading: false, error: null });

  useEffect(() => {
    const q = query.trim();
    if (!q) {
      setState({ data: [], loading: false, error: null });
      return undefined;
    }

    const controller = new AbortController();
    setState((s) => ({ ...s, loading: true, error: null }));

    searchCities(q, { signal: controller.signal })
      .then((data) => setState({ data, loading: false, error: null }))
      .catch((err) => {
        if (err.name === 'AbortError') return; // superseded by a newer query
        setState({ data: [], loading: false, error: err.message });
      });

    // Runs before the next effect (new query) and on unmount
    return () => controller.abort();
  }, [query]);

  return state;
}

import React, { useState } from 'react';
import { useDebounce } from './hooks/useDebounce';
import { useCitySearch } from './hooks/useCitySearch';

// Wrap the first case-insensitive match in <mark>
function Highlight({ text, query }) {
  const q = query.trim();
  const i = q ? text.toLowerCase().indexOf(q.toLowerCase()) : -1;
  if (i === -1) return text;
  return (
    <>
      {text.slice(0, i)}
      <mark>{text.slice(i, i + q.length)}</mark>
      {text.slice(i + q.length)}
    </>
  );
}

export default function App() {
  const [query, setQuery] = useState('');
  const debounced = useDebounce(query, 300);
  const { data, loading, error } = useCitySearch(debounced);

  const typing = query.trim() !== debounced.trim();
  const showResults = debounced.trim() !== '';

  let status = '';
  if (error) status = error;
  else if (loading || typing) status = query.trim() ? 'Searching…' : '';
  else if (showResults) status = `${data.length} ${data.length === 1 ? 'result' : 'results'}`;

  return (
    <div className="page">
      <h1>Find a city</h1>
      <input
        className="search"
        placeholder="Start typing…"
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        aria-describedby="search-status"
      />
      <p className="status" id="search-status" aria-live="polite">{status}</p>
      {showResults && !loading && data.length === 0 && !error && (
        <div className="empty">No cities match “{debounced}”.</div>
      )}
      {showResults && data.length > 0 && (
        <ul className="results">
          {data.map((city) => (
            <li key={city} className="result">
              <Highlight text={city} query={debounced} />
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}
