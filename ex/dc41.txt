  notes: Note[];

  title: 'Sprint 42 retro',
  notes: [
    { id: 'n1', text: 'Release checklist saved us twice', color: 'green' },
    { id: 'n2', text: 'Too many meetings on Wednesday', color: 'pink' },
    { id: 'n3', text: 'Try pairing on the billing bugs', color: 'blue' },
    { id: 'n4', text: 'Flaky e2e tests block merges', color: 'pink' },
    { id: 'n5', text: 'Kudos to Priya for the on-call docs', color: 'yellow' },
  ],
};

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /boards/retro-42 -> Board
export async function getBoard(): Promise<Board> {
  await randomDelay(300, 900);
  return { title: BOARD.title, notes: BOARD.notes.map((n) => ({ ...n })) };
}

type IconButtonProps = {
  label: string;          // accessible name, also the tooltip
  icon: string;
  text?: string;          // optional visible text next to the icon
  disabled?: boolean;
  onClick: () => void;
};

// Small toolbar button. Use it for any new toolbar actions.
export default function IconButton({ label, icon, text, disabled, onClick }: IconButtonProps) {
  let textEl = null;
  if (text) {
    textEl = <span className="icon-btn-text">{text}</span>;
  }
  return (
    <button
      type="button"
      className="icon-btn"
      aria-label={label}
      title={label}
      disabled={disabled}
      onClick={() => { onClick(); }}
    >
      <span className="icon-btn-icon" aria-hidden="true">{icon}</span>
      {textEl}
    </button>
  );
}

type NoteCardProps = {
  note: Note;
  onChange: (id: string, text: string) => void;
  onDelete: (id: string) => void;
};

function NoteCard({ note, onChange, onDelete }: NoteCardProps) {
  return (
    <div className={'note note-' + note.color}>
      <button type="button" className="note-delete" aria-label="Delete note" onClick={() => { onDelete(note.id); }}>×</button>
      <textarea
        className="note-text"
        aria-label="Note text"

export default function App() {
  const [title, setTitle] = useState('');
  const [notes, setNotes] = useState<Note[]>([]);
  const [loading, setLoading] = useState(true);
  const [color, setColor] = useState<NoteColor>('yellow');

  useEffect(() => {
    let cancelled = false;
    getBoard().then((board) => {
      if (cancelled) {
        return;
      }
      setTitle(board.title);
      setNotes(board.notes);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  function handleAdd() {
    setNotes((prev) => {
      return [...prev, { id: uid(), text: '', color: color }];
    });
  }

  function handleChange(id: string, text: string) {
    setNotes((prev) => {
      return prev.map((n) => {
        if (n.id !== id) {
          return n;
        }
        return { ...n, text: text };
      });
    });
  }

  function handleDelete(id: string) {
    setNotes((prev) => {
      return prev.filter((n) => {
        return n.id !== id;
      });
    });
  }

  let content;
  if (loading) {
    content = <p className="board-message">Loading board…</p>;
  } else if (notes.length === 0) {
    content = <p className="board-message">No notes yet. Add the first one.</p>;
  } else {
    content = (
      <div className="board">
        {notes.map((n) => {
          return <NoteCard key={n.id} note={n} onChange={handleChange} onDelete={handleDelete} />;
        })}
      </div>
    );
  }

  return (
    <div className="app">
      <Header title={title} />
      <Toolbar color={color} count={notes.length} onColorChange={setColor} onAdd={handleAdd} />
      <main className="canvas">{content}</main>
    </div>
  );
}

import React, { useEffect, useRef, useState } from 'react';
import { getBoard } from './api';
import type { Note, NoteColor } from './api';
import { uid } from './utils/id';
import IconButton from './components/IconButton';

const COLORS: NoteColor[] = ['yellow', 'pink', 'blue', 'green'];
const MAX_HISTORY = 20;

// One state object, so past/present/future always change together.
type History = {
  past: Note[][];
  present: Note[];
  future: Note[][];
};

// Pure helpers: given the old history, return the new one.
function commitHistory(h: History, next: Note[]): History {
  // keep only the newest MAX_HISTORY snapshots
  return { past: [...h.past, h.present].slice(-MAX_HISTORY), present: next, future: [] };
}

function undoHistory(h: History): History {
  if (h.past.length === 0) {
    return h;
  }
  const previous = h.past[h.past.length - 1];
  return { past: h.past.slice(0, -1), present: previous, future: [h.present, ...h.future] };
}

function redoHistory(h: History): History {
  if (h.future.length === 0) {
    return h;
  }
  const next = h.future[0];
  return { past: [...h.past, h.present], present: next, future: h.future.slice(1) };
}

function Header({ title }: { title: string }) {
  return (

}

type NoteCardProps = {
  note: Note;
  onChange: (id: string, text: string) => void;
  onDelete: (id: string) => void;
  onEditStart: () => void;
  onEditEnd: () => void;
};

function NoteCard({ note, onChange, onDelete, onEditStart, onEditEnd }: NoteCardProps) {
  return (
    <div className={'note note-' + note.color}>
      <button type="button" className="note-delete" aria-label="Delete note" onClick={() => { onDelete(note.id); }}>×</button>
      <textarea
        className="note-text"
        aria-label="Note text"
        placeholder="Write something…"
        value={note.text}
        onChange={(e) => { onChange(note.id, e.target.value); }}
        onFocus={() => { onEditStart(); }}
        onBlur={() => { onEditEnd(); }}
      />
    </div>
  );
}

export default function App() {
  const [title, setTitle] = useState('');
  const [history, setHistory] = useState<History>({ past: [], present: [], future: [] });
  const [loading, setLoading] = useState(true);
  const [color, setColor] = useState<NoteColor>('yellow');
  // Snapshot taken when a note gets focus, committed on blur (one undo = one whole edit).
  const editStart = useRef<Note[] | null>(null);

  useEffect(() => {
    let cancelled = false;
    getBoard().then((board) => {
      if (cancelled) {
        return;
      }
      setTitle(board.title);
      // Loading is not an undoable step.
      setHistory({ past: [], present: board.notes, future: [] });
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // Ctrl/Cmd+Z and Shift+Ctrl/Cmd+Z. The pure helpers don't close over state, so [] deps is safe.
  useEffect(() => {
    function handleKey(e: KeyboardEvent) {
      const target = e.target as HTMLElement;
      // While typing in a note, let the browser undo the text itself.
      if (target.tagName === 'TEXTAREA' || target.tagName === 'INPUT') {
        return;
      }
      if (!(e.metaKey || e.ctrlKey) || e.key.toLowerCase() !== 'z') {
        return;
      }
      e.preventDefault();
      if (e.shiftKey) {
        setHistory(redoHistory);
      } else {
        setHistory(undoHistory);
      }
    }
    document.addEventListener('keydown', handleKey);
    return () => {
      document.removeEventListener('keydown', handleKey);
    };
  }, []);

  function handleAdd() {
    setHistory((h) => {
      return commitHistory(h, [...h.present, { id: uid(), text: '', color: color }]);
    });
  }

  function handleDelete(id: string) {
    setHistory((h) => {
      const next = h.present.filter((n) => {
        return n.id !== id;
      });
      return commitHistory(h, next);
    });
  }

  // Typing replaces `present` without adding history entries.
  function handleChange(id: string, text: string) {
    setHistory((h) => {
      const next = h.present.map((n) => {
        if (n.id !== id) {
          return n;
        }
        return { ...n, text: text };
      });
      return { ...h, present: next };
    });
  }

  function handleEditStart() {
    editStart.current = history.present;
  }

  function handleEditEnd() {
    const before = editStart.current;
    editStart.current = null;
    if (!before) {
      return;
    }
    setHistory((h) => {
      if (h.present === before) {
        return h; // focused but nothing changed
      }
      return { past: [...h.past, before].slice(-MAX_HISTORY), present: h.present, future: [] };
    });
  }

  function handleUndo() {
    setHistory(undoHistory);
  }

  function handleRedo() {
    setHistory(redoHistory);
  }

  // derived
  const notes = history.present;

  let content;
  if (loading) {
    content = <p className="board-message">Loading board…</p>;
  } else if (notes.length === 0) {
    content = <p className="board-message">No notes yet. Add the first one.</p>;
  } else {
    content = (
      <div className="board">
        {notes.map((n) => {
          return (
            <NoteCard
              key={n.id}
              note={n}
              onChange={handleChange}
              onDelete={handleDelete}
              onEditStart={handleEditStart}
              onEditEnd={handleEditEnd}
            />
          );
        })}
      </div>
    );
  }

  return (
    <div className="app">
      <Header title={title} />
      <Toolbar
        color={color}
        count={notes.length}
        canUndo={history.past.length > 0}
        canRedo={history.future.length > 0}
        undoSteps={history.past.length}
        onColorChange={setColor}
        onAdd={handleAdd}
        onUndo={handleUndo}
        onRedo={handleRedo}
      />
      <main className="canvas">{content}</main>
    </div>
  );
}
