// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /notifications -> Notification[] (newest first)
export async function getNotifications(): Promise<Notification[]> {
  await randomDelay(400, 1000);
  return NOTIFICATIONS.map((n) => ({ ...n }));
}

// POST /notifications/:id/read
export async function markRead(id: string): Promise<{ ok: true }> {
  await randomDelay(300, 700);
  const n = NOTIFICATIONS.find((x) => x.id === id);
  if (n) {
    n.read = true;
  }
  return { ok: true };
}

// POST /notifications/read-all
export async function markAllRead(): Promise<{ ok: true }> {
  await randomDelay(300, 700);
  NOTIFICATIONS.forEach((n) => {
    n.read = true;
  });
  return { ok: true };
}

import React from 'react';

const COLORS = ['#6366F1', '#0EA5E9', '#F59E0B', '#10B981', '#EC4899', '#8B5CF6'];

// Initials in a colored circle. The color is stable per name.
export function Avatar({ name, size = 32 }: { name: string; size?: number }) {
  const initials = name
    .split(' ')
    .map((part) => {
      return part[0];
    })
    .join('')
    .slice(0, 2);
  let sum = 0;
  for (let i = 0; i < name.length; i++) {
    sum += name.charCodeAt(i);
  }
  const style = { width: size, height: size, fontSize: size * 0.4, background: COLORS[sum % COLORS.length] };
  return <span className="avatar" style={style}>{initials}</span>;
}

// The app's "now". Fixed so the demo data always reads the same.
export const NOW = new Date('2026-10-10T15:00:00');

// timeAgo('2026-10-10T14:42:00') -> "18m ago", a day earlier -> "1d ago"
export function timeAgo(iso: string): string {
  const minutes = Math.round((NOW.getTime() - new Date(iso).getTime()) / 60000);
  if (minutes < 1) {
    return 'just now';
  }
  if (minutes < 60) {
    return minutes + 'm ago';
  }
  const hours = Math.floor(minutes / 60);
  if (hours < 24) {
    return hours + 'h ago';
  }
  return Math.floor(hours / 24) + 'd ago';
}

// isToday('2026-10-10T09:20:00') -> true (same calendar day as NOW)
export function isToday(iso: string): boolean {
  const d = new Date(iso);
  return d.getFullYear() === NOW.getFullYear() && d.getMonth() === NOW.getMonth() && d.getDate() === NOW.getDate();
}

import React, { useEffect, useRef, useState } from 'react';
import { getNotifications, markRead, markAllRead } from '../api';
import type { Notification } from '../api';
import { BellIcon } from './Icons';
import { NotificationItem } from './NotificationItem';
import { isToday } from '../utils/time';

// STRETCH: Today / Earlier groups. One heading + list per group; an empty group renders nothing.
// (Core is simply one <ul className="notif-list"> mapping items to <NotificationItem>.)
type GroupProps = {
  title: string;
  items: Notification[];
  onItemClick: (id: string) => void;
};

function NotificationGroup({ title, items, onItemClick }: GroupProps) {
  if (items.length === 0) {
    return null;
  }
  return (
    <section className="notif-group">
      <h3 className="notif-group-title">{title}</h3>
      <ul className="notif-list">
        {items.map((item) => {
          return <NotificationItem key={item.id} item={item} onClick={onItemClick} />;
        })}
      </ul>
    </section>
  );
}

export function NotificationsMenu() {
  // CORE: items, loading, open
  const [items, setItems] = useState<Notification[]>([]);
  const [loading, setLoading] = useState(true);
  const [open, setOpen] = useState(false);
  // STRETCH: ref for click-outside
  const wrapRef = useRef<HTMLDivElement>(null);

  // CORE: load on mount, not on open: the badge needs the count before the menu is ever opened.
  useEffect(() => {
    let cancelled = false;
    getNotifications().then((data) => {
      if (cancelled) {
        return;
      }
      setItems(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // NEXT: outside click + Escape. Only listen while the menu is open.
  useEffect(() => {
    if (!open) {
      return;
    }
    function handleMouseDown(e: MouseEvent) {
      if (wrapRef.current && !wrapRef.current.contains(e.target as Node)) {
        setOpen(false);
      }
    }
    function handleKey(e: KeyboardEvent) {
      if (e.key === 'Escape') {
        setOpen(false);
      }
    }
    document.addEventListener('mousedown', handleMouseDown);
    document.addEventListener('keydown', handleKey);
    return () => {
      document.removeEventListener('mousedown', handleMouseDown);
      document.removeEventListener('keydown', handleKey);
    };
  }, [open]);

  // CORE: toggle
  function handleToggle() {
    setOpen((prev) => {
      return !prev;
    });
  }

  // NEXT: optimistic mark-read. Update the UI now, tell the server in the background.
  function handleItemClick(id: string) {
    setItems((prev) => {
      return prev.map((n) => {
        if (n.id === id) {
          return { ...n, read: true };
        }
        return n;
      });
    });
    markRead(id);
  }

  // CHANGE: mark all as read
  function handleMarkAll() {
    setItems((prev) => {
      return prev.map((n) => {
        return { ...n, read: true };
      });
    });
    markAllRead();
  }

  // CORE: derived, never stored in state
  const unreadCount = items.filter((n) => {
    return !n.read;
  }).length;
  // STRETCH: the two groups
  const todayItems = items.filter((n) => {
    return isToday(n.createdAt);
  });
  const earlierItems = items.filter((n) => {
    return !isToday(n.createdAt);
  });

  // CHANGE: header button (disabled when nothing is unread)
  const markAllBtn = (
    <button type="button" className="link-btn" disabled={unreadCount === 0} onClick={() => { handleMarkAll(); }}>
      Mark all as read
    </button>
  );

  // CORE: badge + dropdown decisions
  let badge = null;
  if (unreadCount > 0) {
    badge = <span className="bell-badge">{unreadCount}</span>;
  }

  let dropdown = null;
  if (open) {
    let body;
    if (loading) {
      body = <p className="notif-empty">Loading notifications…</p>;
    } else if (items.length === 0) {
      body = <p className="notif-empty">You're all caught up.</p>;
    } else {
      body = (
        <div className="notif-scroll">
          <NotificationGroup title="Today" items={todayItems} onItemClick={handleItemClick} />
          <NotificationGroup title="Earlier" items={earlierItems} onItemClick={handleItemClick} />
        </div>
      );
    }
    dropdown = (
      <div className="notif-menu" role="dialog" aria-label="Notifications">
        <div className="notif-head">
          <h2>Notifications</h2>
          {markAllBtn}
        </div>
        {body}
      </div>
    );
  }

  // CORE: render
  return (
    <div className="notif-wrap" ref={wrapRef}>
      <button type="button" className="icon-btn bell-btn" aria-label="Notifications" aria-expanded={open} onClick={() => { handleToggle(); }}>
        <BellIcon />
        {badge}
      </button>
      {dropdown}
    </div>
  );
}
