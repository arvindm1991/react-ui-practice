// Fake API. Treat it like a real backend you can't change.
// Every task has a `version`. Writes must send the version you last saw;
// if someone else saved since then, the server refuses with a ConflictError.
export type Task = {
  id: string;
  title: string;
  done: boolean;
  version: number;
  updatedBy: string;
};

export class ConflictError extends Error {
  taskId: string;
  constructor(taskId: string) {
    super('Version mismatch on ' + taskId);
    this.name = 'ConflictError';
    this.taskId = taskId;
  }
}

let DB: Task[] = [
  { id: 't1', title: 'Book the main stage and PA system', done: true, version: 4, updatedBy: 'Marco' },
  { id: 't2', title: 'Send vendor contracts for signature', done: false, version: 2, updatedBy: 'You' },
  { id: 't3', title: 'Confirm food truck lineup', done: false, version: 1, updatedBy: 'Priya' },
  { id: 't4', title: 'Order 2,000 wristbands', done: true, version: 3, updatedBy: 'Lena' },
  { id: 't5', title: 'Schedule volunteer shifts', done: false, version: 1, updatedBy: 'Marco' },
  { id: 't6', title: 'Arrange waste and recycling stations', done: false, version: 1, updatedBy: 'You' },
];
let priyaEdited = false;

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

function find(id: string): Task {
  const t = DB.find((x) => x.id === id);
  if (!t) {
    throw new Error('Task ' + id + ' not found');
  }
  return t;
}

// GET /events/harborfest/tasks -> Task[]
export async function getTasks(): Promise<Task[]> {
  await randomDelay(400, 900);
  const snapshot = DB.map((t) => ({ ...t }));
  // Simulation: right after you load, Priya edits task #3 in another tab.
  if (!priyaEdited) {
    priyaEdited = true;
    DB = DB.map((t) => (t.id === 't3' ? { ...t, title: 'Confirm food truck lineup (8 trucks, 2 vegan)', version: 2, updatedBy: 'Priya' } : t));
  }
  return snapshot;
}

// GET /tasks/:id -> the latest version
export async function getTask(id: string): Promise<Task> {
  await randomDelay(300, 700);
  return { ...find(id) };
}

// PATCH /tasks/:id  { title?, done?, version } -> the saved task (with a new version)
export async function updateTask(id: string, patch: { title?: string; done?: boolean }, version: number): Promise<Task> {
  await randomDelay(400, 900);
  const current = find(id);
  if (current.version !== version) {
    throw new ConflictError(id);
  }
  const saved: Task = { ...current, ...patch, version: current.version + 1, updatedBy: 'You' };
  DB = DB.map((t) => (t.id === id ? saved : t));
  return { ...saved };
}

// DELETE /tasks/:id?version=N
export async function deleteTask(id: string, version: number): Promise<void> {
  await randomDelay(400, 900);
  const current = find(id);
  if (current.version !== version) {
    throw new ConflictError(id);
  }
  DB = DB.filter((t) => t.id !== id);
}

// Small pure helpers for immutable state updates. They never mutate their input.

// Replace the item with the same id. Use inside a state updater:
// setTasks((prev) => { return replaceById(prev, saved); });
export function replaceById<T extends { id: string }>(list: T[], next: T): T[] {
  return list.map((item) => {
    if (item.id !== next.id) {
      return item;
    }
    return next;
  });
}

// A copy of a per-id flag map with one id turned on or off (off removes the key).
// withFlag({ a: true }, 'b', true) -> { a: true, b: true }
// withFlag({ a: true }, 'a', false) -> {}
export function withFlag(flags: Record<string, boolean>, id: string, on: boolean): Record<string, boolean> {
  const next = { ...flags };
  if (on) {
    next[id] = true;
  } else {
    delete next[id];
  }
  return next;
}

import { getTasks, getTask, updateTask, deleteTask, ConflictError } from './api';
import type { Task } from './api';
import Presence from './components/Presence';
import { replaceById, withFlag } from './utils/state';

// CORE
type Patch = { title?: string; done?: boolean };

function ProgressHeader({ tasks }: { tasks: Task[] }) {
  const done = tasks.filter((t) => {
    return t.done;
  }).length;
  let pct = 0;
  if (tasks.length > 0) {
    pct = Math.round((done / tasks.length) * 100);
  }
  return (
    <div className="progress">
      <div className="progress-text">
        <b>{done} of {tasks.length}</b> done
      </div>
      <div className="progress-bar"><span style={{ width: pct + '%' }} /></div>

// CORE: saving/conflict flags + callbacks (NEXT: onRename, onDelete)
type TaskRowProps = {
  task: Task;
  saving: boolean;
  conflict: boolean;
  onToggle: (task: Task) => void;
  onRename: (task: Task, title: string) => void;
  onDelete: (task: Task) => void;
  onReload: (id: string) => void;
};

function TaskRow({ task, saving, conflict, onToggle, onRename, onDelete, onReload }: TaskRowProps) {
  // NEXT: inline rename is local to the row
  const [editing, setEditing] = useState(false);
  const [draft, setDraft] = useState(task.title);

  function startEdit() {
    setDraft(task.title);
    setEditing(true);
  }

  function commit() {
    setEditing(false);
    const title = draft.trim();
    if (title !== '' && title !== task.title) {
      onRename(task, title);
    }
  }

  function handleKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
    if (e.key === 'Enter') {
      commit();
    } else if (e.key === 'Escape') {
      setEditing(false); // discard the draft
    }
  }

  // CORE
  let cls = 'task';
  if (task.done) {
    cls += ' is-done';
  }
  if (saving) {
    cls += ' is-saving';
  }
  if (conflict) {
    cls += ' has-conflict';
  }

  // NEXT: title becomes an input while editing
  let title = (
    <button type="button" className="task-title" disabled={saving} onClick={() => { startEdit(); }} title="Click to rename">
      {task.title}
    </button>
  );
  if (editing) {
    title = (
      <input
        className="task-input"
        autoFocus
        aria-label="Task name"
        value={draft}
        onChange={(e) => { setDraft(e.target.value); }}
        onKeyDown={handleKeyDown}
        onBlur={() => { commit(); }}
      />
    );
  }

  // CORE: Saving… / conflict message
  let meta = <span className="task-meta">Last edited by {task.updatedBy}</span>;
  if (saving) {
    meta = <span className="task-meta">Saving…</span>;
  }

  let conflictEl = null;
  if (conflict) {
    conflictEl = (
      <p className="conflict" role="alert">
        Someone else changed this ·
        <button type="button" className="conflict-btn" onClick={() => { onReload(task.id); }}>Reload</button>
      </p>
    );
  }

  return (
    <li className={cls}>
      <input
        type="checkbox"
        className="task-check"
        checked={task.done}
        disabled={saving}
        aria-label={'Done: ' + task.title}
        onChange={() => { onToggle(task); }}
      />
      <div className="task-main">
        {title}
        {meta}
        {conflictEl}
      </div>
      <button type="button" className="icon-btn" disabled={saving} aria-label={'Delete ' + task.title} onClick={() => { onDelete(task); }}>✕</button>
    </li>
  );
}

export default function App() {
  const [tasks, setTasks] = useState<Task[]>([]);
  const [loading, setLoading] = useState(true);
  // CORE: per-task flags, keyed by id. Kept out of Task because they're client-only.
  const [saving, setSaving] = useState<Record<string, boolean>>({});
  const [conflicts, setConflicts] = useState<Record<string, boolean>>({});
  // CHANGE
  const [summary, setSummary] = useState('');

  useEffect(() => {
    let cancelled = false;
    getTasks().then((data) => {
      if (cancelled) {
        return;
      }
      setTasks(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // CORE: small wrappers so every write path reads the same (pure helpers in utils/state.ts)
  function replaceTask(next: Task) {
    setTasks((prev) => {
      return replaceById(prev, next);
    });
  }

  function setFlag(setter: typeof setSaving, id: string, on: boolean) {
    setter((prev) => {
      return withFlag(prev, id, on);
    });
  }

  // CORE: optimistic write. Resolves true/false and never rejects, so callers can batch it.
  function saveTask(task: Task, patch: Patch): Promise<boolean> {
    replaceTask({ ...task, ...patch });
    setFlag(setSaving, task.id, true);
    setFlag(setConflicts, task.id, false);
    return updateTask(task.id, patch, task.version)
      .then((saved) => {
        replaceTask(saved); // the server's copy has the new version
        return true;
      })
      .catch((err) => {
        replaceTask(task); // roll back just this task
        if (err instanceof ConflictError) {
          setFlag(setConflicts, task.id, true);
        }
        return false;
      })
      .finally(() => {
        setFlag(setSaving, task.id, false);
      });
  }

  function handleToggle(task: Task) {
    saveTask(task, { done: !task.done });
  }

  // NEXT
  function handleRename(task: Task, title: string) {
    saveTask(task, { title });
  }

  // NEXT: delete, re-insert at the old index on failure
  function handleDelete(task: Task) {
    const index = tasks.findIndex((t) => {
      return t.id === task.id;
    });
    setTasks((prev) => {
      return prev.filter((t) => {
        return t.id !== task.id;
      });
    });
    setFlag(setConflicts, task.id, false);
    deleteTask(task.id, task.version).catch((err) => {
      // put it back where it was
      setTasks((prev) => {
        const next = [...prev];
        next.splice(Math.min(index, next.length), 0, task);
        return next;
      });
      if (err instanceof ConflictError) {
        setFlag(setConflicts, task.id, true);
      }
    });
  }

  // CORE: fetch the latest copy and clear the flag
  function handleReload(id: string) {
    getTask(id).then((latest) => {
      replaceTask(latest);
      setFlag(setConflicts, id, false);
    });
  }

  // CHANGE: one optimistic save per open task; each one rolls back on its own.
  function handleMarkAllDone() {
    const targets = tasks.filter((t) => {
      return !t.done && !saving[t.id];
    });
    // saveTask resolves true/false and never rejects, so Promise.all waits for every one
    Promise.all(targets.map((t) => {
      return saveTask(t, { done: true });
    })).then((results) => {
      const ok = results.filter(Boolean).length;
      setSummary('Marked ' + ok + ' of ' + targets.length + ' tasks done.');
    });
  }

  // CHANGE
  const openCount = tasks.filter((t) => {
    return !t.done;
  }).length;

  // CORE
  let list;
  if (loading) {
    list = <p className="loading">Loading checklist…</p>;
  } else {
    list = (
      <ul className="task-list">
        {tasks.map((t) => {
          return (
            <TaskRow
              key={t.id}
              task={t}
              saving={Boolean(saving[t.id])}
              conflict={Boolean(conflicts[t.id])}
              onToggle={handleToggle}
              onRename={handleRename}
              onDelete={handleDelete}
              onReload={handleReload}
            />
          );
        })}
      </ul>
    );
  }

  // CHANGE
  let summaryEl = null;
  if (summary) {
    summaryEl = <p className="batch-summary" role="status">{summary}</p>;
  }

  return (
    <div className="app">
      <header className="header">
        <div>
          <p className="eyebrow">HarborFest 2026 · Jun 20</p>
          <h1>Vendor checklist</h1>
        </div>
        <Presence />
      </header>
      <section className="card">
        <div className="card-top">
          <ProgressHeader tasks={tasks} />
          {/* CHANGE */}
          <button type="button" className="btn-secondary" disabled={loading || openCount === 0} onClick={() => { handleMarkAllDone(); }}>Mark all done</button>
        </div>
        {summaryEl}
        {list}
      </section>
    </div>
  );
}
