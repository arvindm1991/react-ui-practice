// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /hashtags/trending -> Hashtag[]
export async function getHashtags(): Promise<Hashtag[]> {
  await randomDelay(400, 1400);
  return HASHTAGS.map((h) => ({ ...h }));
}

// formatCount(18400) -> "18.4K", formatCount(310200) -> "310.2K", formatCount(950) -> "950"
export function formatCount(n: number): string {
  if (n >= 1000000) {
    return (n / 1000000).toFixed(1) + 'M';
  }
  if (n >= 1000) {
    return (n / 1000).toFixed(1) + 'K';
  }
  return String(n);
}

import React, { useEffect, useState } from 'react';
import { getHashtags } from './api';
import type { Hashtag } from './api';
import { formatCount } from './utils/format';

const MAX_CHARS = 280;

function Header() {
  return (
    <header className="header">
      <span className="logo">◆</span>
      <h1>Home</h1>
    </header>
  );
}

function CharCount({ count }: { count: number }) {
  let cls = 'char-count';
  if (count > MAX_CHARS) {
    cls = 'char-count is-over';
  }
  return <span className={cls}>{count} / {MAX_CHARS}</span>;
}

// 1. The button only reports the click. The Composer owns the state.
function HashtagButton({ onClick }: { onClick: () => void }) {
  return (
    <button type="button" className="hashtag-btn" aria-label="Add hashtag" onClick={() => { onClick(); }}>
      #
    </button>
  );
}

type HashtagModalProps = {
  usedTags: string[];
  onPick: (tag: string) => void;
  onClose: () => void;
};

// 2. The modal loads its own data. It is only rendered while open,
//    so every open starts fresh (and refetches).
function HashtagModal({ usedTags, onPick, onClose }: HashtagModalProps) {
  const [hashtags, setHashtags] = useState<Hashtag[]>([]);
  const [loading, setLoading] = useState(true);
  const [query, setQuery] = useState('');

  useEffect(() => {
    let cancelled = false;
    getHashtags().then((data) => {
      if (cancelled) {
        return;
      }
      setHashtags(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // Escape closes
  useEffect(() => {
    function handleKey(e: KeyboardEvent) {
      if (e.key === 'Escape') {
        onClose();
      }
    }
    document.addEventListener('keydown', handleKey);
    return () => {
      document.removeEventListener('keydown', handleKey);
    };
  }, [onClose]);

  // Clicking the dark backdrop closes; clicks inside the panel don't.
  function handleBackdropClick(e: React.MouseEvent<HTMLDivElement>) {
    if (e.target === e.currentTarget) {
      onClose();
    }
  }

  // derived
  const q = query.trim().toLowerCase().replace(/^#/, '');
  const visible = hashtags.filter((h) => {
    return h.tag.includes(q);
  });

  // decisions
  let body;
  if (loading) {
    body = <p className="modal-message">Loading hashtags…</p>;
  } else if (visible.length === 0) {
    body = <p className="modal-message">No hashtags match "{query}".</p>;
  } else {
    body = (
      <ul className="tag-list">
        {visible.map((h) => {
          const used = usedTags.includes(h.tag);
          let cls = 'tag-row';
          if (used) {
            cls = 'tag-row is-used';
          }
          return (
            <li key={h.id}>
              <button type="button" className={cls} disabled={used} onClick={() => { onPick(h.tag); }}>
                <span className="tag-name">#{h.tag}</span>
                <span className="tag-count">{formatCount(h.posts)} posts</span>
              </button>
            </li>
          );
        })}
      </ul>
    );
  }

  return (
    <div className="backdrop" onClick={handleBackdropClick}>
      <div className="modal" role="dialog" aria-label="Add a hashtag">
        <div className="modal-head">
          <h2>Trending hashtags</h2>
          <button type="button" className="icon-btn" aria-label="Close" onClick={() => { onClose(); }}>✕</button>
        </div>
        <input
          className="modal-search"
          placeholder="Search hashtags"
          autoFocus
          value={query}
          onChange={(e) => { setQuery(e.target.value); }}
        />
        {body}
      </div>
    </div>
  );
}

function Composer() {
  const [text, setText] = useState('');
  const [pickerOpen, setPickerOpen] = useState(false);

  // 3. Add "#tag" to the end of the text, with a space before it if needed.
  function handlePick(tag: string) {
    setText((prev) => {
      if (prev === '' || prev.endsWith(' ')) {
        return prev + '#' + tag + ' ';
      }
      return prev + ' #' + tag + ' ';
    });
    setPickerOpen(false);
  }

  // Tags already in the text, so the picker can disable them.
  const usedTags = text
    .split(/\s+/)
    .filter((word) => {
      return word.startsWith('#');
    })
    .map((word) => {
      return word.slice(1).toLowerCase();
    });

  let modal = null;
  if (pickerOpen) {
    modal = (
      <HashtagModal
        usedTags={usedTags}
        onPick={handlePick}
        onClose={() => { setPickerOpen(false); }}
      />
    );
  }

  return (
    <div className="composer">
      <div className="avatar">A</div>
      <div className="composer-body">
        <div className="input-wrap">
          <textarea
            className="post-input"
            placeholder="What's happening?"
            value={text}
            onChange={(e) => { setText(e.target.value); }}
          />
          <HashtagButton onClick={() => { setPickerOpen(true); }} />
        </div>
        <div className="composer-footer">
          <CharCount count={text.length} />
          <button type="button" className="post-btn" disabled={text.trim() === ''}>Post</button>
        </div>
      </div>
      {modal}
    </div>
  );
}

function Feed() {
  return (
    <div className="feed">
      <article className="post">
        <div className="avatar">G</div>
        <div>
          <p className="post-meta"><b>Grace</b> · 2h</p>
          <p>Shipped the new onboarding flow today. #buildinpublic</p>
        </div>
      </article>
      <article className="post">
        <div className="avatar">K</div>
        <div>
          <p className="post-meta"><b>Kat</b> · 5h</p>
          <p>Hot take: most modals should be pages.</p>
        </div>
      </article>
    </div>
  );
}

export default function App() {
  return (
    <div className="app">
      <Header />
      <Composer />
      <Feed />
    </div>
  );
}
