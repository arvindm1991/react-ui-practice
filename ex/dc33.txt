// Fake API. Treat it like a real backend you can't change.
export type PollOption = {
  id: string;
  label: string;
  votes: number;
};

export type Poll = {
  id: string;
  question: string;
  options: PollOption[];
  myVote: string | null;   // option id this user voted for, or null
  closesIn: string;
};

const POLL: Poll = {
  id: 'p1',
  question: 'Which map should we run in Saturday’s scrim?',
  options: [
    { id: 'ascent', label: 'Ascent', votes: 42 },
    { id: 'bind', label: 'Bind', votes: 31 },
    { id: 'haven', label: 'Haven', votes: 18 },
    { id: 'lotus', label: 'Lotus', votes: 9 },
  ],
  myVote: null,
  closesIn: '2 days',
};

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

function copyPoll(): Poll {
  return { ...POLL, options: POLL.options.map((o) => ({ ...o })) };
}

// GET /polls/p1 -> Poll
export async function getPoll(): Promise<Poll> {
  await randomDelay(500, 1100);
  return copyPoll();
}

// POST /polls/p1/vote { optionId } -> Poll
// One vote per user. Voting again moves your vote from the old option to the new one.
export async function vote(optionId: string): Promise<Poll> {
  await randomDelay(600, 1200);
  POLL.options.forEach((o) => {
    if (o.id === POLL.myVote) {
      o.votes -= 1;
    }
  });
  POLL.options.forEach((o) => {
    if (o.id === optionId) {
      o.votes += 1;
    }
  });
  POLL.myVote = optionId;
  return copyPoll();
}

// percent(9, 101) -> 9, percent(0, 0) -> 0 (whole numbers, safe when nobody has voted)
export function percent(part: number, total: number): number {
  if (total === 0) {
    return 0;
  }
  return Math.round((part / total) * 100);
}

// formatVotes(1) -> "1 vote", formatVotes(101) -> "101 votes"
export function formatVotes(n: number): string {
  if (n === 1) {
    return '1 vote';
  }
  return n + ' votes';
}

import React, { useEffect, useState } from 'react';
import { getPoll, vote } from './api';
import type { Poll } from './api';
import { Avatar } from './components/Avatar';
import { formatVotes, percent } from './utils/poll';

function TopBar() {
  return (
    <header className="topbar">
      <span className="logo">⬢ Guildhall</span>
      <nav className="tabs">
        <button type="button" className="tab is-active">#scrims</button>
        <button type="button" className="tab">#clips</button>
        <button type="button" className="tab">#lfg</button>
      </nav>
    </header>
  );
}

function PollWidget() {
  const [poll, setPoll] = useState<Poll | null>(null);
  const [saving, setSaving] = useState(false);
  const [changing, setChanging] = useState(false);

  useEffect(() => {
    let cancelled = false;
    getPoll().then((data) => {
      if (cancelled) {
        return;
      }
      setPoll(data);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  function handleVote(optionId: string) {
    if (!poll) {
      return;
    }
    if (poll.myVote === optionId) {
      // Picked the same option again while changing: nothing to send.
      setChanging(false);
      return;
    }
    // Optimistic: move the vote locally now so the bars animate immediately.
    setPoll((prev) => {
      if (!prev) {
        return prev;
      }
      const options = prev.options.map((o) => {
        let votes = o.votes;
        if (o.id === prev.myVote) {
          votes = votes - 1;
        }
        if (o.id === optionId) {
          votes = votes + 1;
        }
        return { ...o, votes };
      });
      return { ...prev, options, myVote: optionId };
    });
    setChanging(false);
    setSaving(true);
    // Then trust the server's numbers (other people may have voted meanwhile).
    vote(optionId).then((fresh) => {
      setPoll(fresh);
      setSaving(false);
    });
  }

  if (!poll) {
    return <div className="poll poll-loading">Loading poll…</div>;
  }

  // derived
  const total = poll.options.reduce((sum, o) => {
    return sum + o.votes;
  }, 0);
  const showResults = poll.myVote !== null && !changing;

  let meta = <p className="poll-meta">Closes in {poll.closesIn}</p>;
  if (showResults) {
    let status = 'Closes in ' + poll.closesIn;
    if (saving) {
      status = 'Saving your vote…';
    }
    meta = (
      <p className="poll-meta">
        <span className="poll-total">{formatVotes(total)}</span>
        <span>·</span>
        <span>{status}</span>
        <button type="button" className="poll-link" disabled={saving} onClick={() => { setChanging(true); }}>Change vote</button>
      </p>
    );
  } else if (changing) {
    meta = (
      <p className="poll-meta">
        <span>Pick a new option</span>
        <button type="button" className="poll-link" onClick={() => { setChanging(false); }}>Cancel</button>
      </p>
    );
  }

  return (
    <div className="poll">
      <p className="poll-question">{poll.question}</p>
      <ul className="poll-options">
        {poll.options.map((o) => {
          const pct = percent(o.votes, total);
          let cls = 'poll-option';
          let width = '0%';
          let pctEl = null;
          let check = null;
          if (showResults) {
            cls = 'poll-option is-result';
            width = pct + '%';
            pctEl = <span className="poll-pct">{pct}%</span>;
          }
          if (o.id === poll.myVote) {
            cls = cls + ' is-mine';
            check = <span className="poll-check" aria-label="Your vote">✓</span>;
          }
          return (
            <li key={o.id}>
              <button type="button" className={cls} disabled={showResults || saving} onClick={() => { handleVote(o.id); }}>
                <span className="poll-bar" style={{ width }} />
                <span className="poll-label">{o.label}</span>
                {check}
                {pctEl}
              </button>
            </li>
          );
        })}
      </ul>
      {meta}
    </div>
  );
}

function Post() {
  return (
    <article className="post">
      <div className="post-head">
        <Avatar name="Vex" color="#8B5CF6" />
        <div>
          <p className="post-author">Vex <span className="role">Captain</span></p>
          <p className="post-time">Today at 6:12 PM</p>
        </div>
      </div>
      <p className="post-text">Scrim vs. Northwind is locked for Saturday 8 PM. Vote for the map. Majority wins, no complaining after. 🎯</p>
      <PollWidget />
      <div className="post-actions">
        <button type="button" className="ghost">💬 24 replies</button>
        <button type="button" className="ghost">🔁 Share</button>
      </div>
    </article>
  );
}

export default function App() {
  return (
    <div className="app">
      <TopBar />
      <main className="feed">
        <Post />
      </main>
    </div>
  );
}
