export const FOCUS_SECONDS = 25 * 60;
export const BREAK_SECONDS = 5 * 60;

// formatTime(1500) -> "25:00", formatTime(65) -> "01:05", formatTime(0) -> "00:00"
export function formatTime(totalSeconds: number): string {
  const m = Math.floor(totalSeconds / 60);
  const s = totalSeconds % 60;
  return String(m).padStart(2, '0') + ':' + String(s).padStart(2, '0');
}

import React, { useState } from 'react';

type Task = { id: string; title: string; done: boolean };

const INITIAL: Task[] = [
  { id: 't1', title: 'Draft Q4 hiring plan', done: true },
  { id: 't2', title: 'Review onboarding PR', done: false },
  { id: 't3', title: 'Write release notes', done: false },
];

export default function TaskList() {
  const [tasks, setTasks] = useState<Task[]>(INITIAL);

  function toggle(id: string) {
    setTasks((prev) => {
      return prev.map((t) => {
        if (t.id !== id) {
          return t;
        }
        return { ...t, done: !t.done };
      });
    });
  }

  return (
    <section className="panel">
      <h2>Today</h2>
      <ul className="tasks">
        {tasks.map((t) => {
          let cls = 'task';
          if (t.done) {
            cls = 'task is-done';
          }
          return (
            <li key={t.id} className={cls}>
              <label>
                <input type="checkbox" checked={t.done} onChange={() => { toggle(t.id); }} />
                {t.title}
              </label>
            </li>
          );
        })}
      </ul>
    </section>
  );
}

// CORE: hooks
import React, { useEffect, useState } from 'react';
import TaskList from './components/TaskList';
// NEXT: BREAK_SECONDS + Mode are for the break switch
import { formatTime, FOCUS_SECONDS, BREAK_SECONDS } from './utils/time';

const DAILY_GOAL = 4;

type Mode = 'focus' | 'break';

function Header() {
  return (
    <header className="header">
      <span className="logo">● Ripe</span>
      <span className="header-note">Deep work, one tomato at a time</span>
    </header>
  );
}

function SessionDots({ completed }: { completed: number }) {
  const dots = [];
  for (let i = 0; i < DAILY_GOAL; i++) {
    let cls = 'dot';
    if (i < completed) {
      cls = 'dot is-done';
    }
    dots.push(<span key={i} className={cls} />);
  }
  return (
    <div className="sessions">
      <div className="dots">{dots}</div>
      <span className="sessions-label">{completed} of {DAILY_GOAL} sessions</span>
    </div>
  );
}

// NEXT: onFocusComplete prop + mode state
function Timer({ onFocusComplete }: { onFocusComplete: () => void }) {
  const [mode, setMode] = useState<Mode>('focus');
  // CORE: time left + running flag
  const [secondsLeft, setSecondsLeft] = useState(FOCUS_SECONDS);
  const [running, setRunning] = useState(false);

  // CORE: one interval while running. The cleanup clears it when we pause,
  // reset or unmount, so there can never be two intervals at once.
  useEffect(() => {
    if (!running) {
      return;
    }
    const id = setInterval(() => {
      // updater form: the interval callback would otherwise see a stale value
      setSecondsLeft((s) => {
        return s - 1;
      });
    }, 1000);
    return () => {
      clearInterval(id);
    };
  }, [running]);

  // NEXT: reaching 00:00 switches the mode.
  useEffect(() => {
    if (secondsLeft > 0) {
      return;
    }
    if (mode === 'focus') {
      setMode('break');
      setSecondsLeft(BREAK_SECONDS);
      onFocusComplete();
    } else {
      setMode('focus');
      setSecondsLeft(FOCUS_SECONDS);
      setRunning(false);
    }
  }, [secondsLeft, mode]);

  // CORE: Start / Pause toggles running
  function handleStartPause() {
    setRunning((r) => {
      return !r;
    });
  }

  // NEXT: Reset
  function handleReset() {
    setRunning(false);
    setMode('focus');
    setSecondsLeft(FOCUS_SECONDS);
  }

  // NEXT: label + class for Break mode
  let total = FOCUS_SECONDS;
  let modeLabel = 'Focus';
  let cls = 'timer mode-focus';
  if (mode === 'break') {
    total = BREAK_SECONDS;
    modeLabel = 'Break';
    cls = 'timer mode-break';
  }
  // STRETCH: progress is derived from secondsLeft
  const progress = ((total - secondsLeft) / total) * 100;

  // CORE: button label
  let startLabel = 'Start';
  if (running) {
    startLabel = 'Pause';
  }

  // CORE: render the live time (NEXT: mode pill/class, STRETCH: progress bar)
  return (
    <section className={cls}>
      <span className="mode-pill">{modeLabel}</span>
      <div className="time">{formatTime(secondsLeft)}</div>
      <div className="progress" aria-hidden="true">
        <div className="progress-fill" style={{ width: progress + '%' }} />
      </div>
      <div className="controls">
        <button type="button" className="btn btn-primary" onClick={() => { handleStartPause(); }}>{startLabel}</button>
        <button type="button" className="btn btn-ghost" onClick={() => { handleReset(); }}>Reset</button>
      </div>
    </section>
  );
}

// NEXT: count completed focus sessions
export default function App() {
  const [completed, setCompleted] = useState(0);

  function handleFocusComplete() {
    setCompleted((c) => {
      return c + 1;
    });
  }

  return (
    <div className="app">
      <Header />
      <main className="main">
        <Timer onFocusComplete={handleFocusComplete} />
        <SessionDots completed={completed} />
        <TaskList />
      </main>
    </div>
  );
}
