@@@ shared src/utils/format.js
// formatCurrency(18400) -> "$18,400"
export function formatCurrency(n) {
  return '$' + n.toLocaleString('en-US');
}

// formatDate('2023-02-11') -> "Feb 11, 2023"
export function formatDate(isoDate) {
  return new Date(isoDate + 'T00:00:00').toLocaleDateString('en-US', {
    month: 'short',
    day: 'numeric',
    year: 'numeric',
  });
}

import React, { useState } from 'react';
import { CUSTOMERS } from '../data/customers';
import SearchInput from '../components/SearchInput';
import CustomerTable from '../components/CustomerTable';
import CustomerDrawer from '../components/CustomerDrawer';

const STATUS_OPTIONS = [
  ['all', 'All statuses'],
  ['active', 'Active'],
  ['trial', 'Trial'],
  ['churned', 'Churned'],
];

export default function CustomersPage() {
  // ---- state ----
  const [query, setQuery] = useState('');
  const [status, setStatus] = useState('all');
  const [sortKey, setSortKey] = useState(null); // null | 'revenue' | 'name'
  const [sortDir, setSortDir] = useState('desc');
  const [selectedId, setSelectedId] = useState(null); // mid-session change

  // ---- handlers ----
  function handleSort(key) {
    if (sortKey === key) {
      // same column: flip direction
      if (sortDir === 'desc') {
        setSortDir('asc');
      } else {
        setSortDir('desc');
      }
      return;
    }
    // new column: revenue starts high→low, name starts A→Z
    setSortKey(key);
    if (key === 'revenue') {
      setSortDir('desc');
    } else {
      setSortDir('asc');
    }
  }

  function handleClear() {
    setQuery('');
    setStatus('all');
  }

  // ---- derived: source → search → status → sort ----
  const q = query.trim().toLowerCase();
  const searched = CUSTOMERS.filter((c) => {
    return c.name.toLowerCase().includes(q) || c.company.toLowerCase().includes(q);
  });

  let filtered = searched;
  if (status !== 'all') {
    filtered = searched.filter((c) => {
      return c.status === status;
    });
  }

  let visible = filtered;
  if (sortKey === 'revenue') {
    visible = [...filtered].sort((a, b) => {
      return a.revenue - b.revenue;
    });
  }
  if (sortKey === 'name') {
    visible = [...filtered].sort((a, b) => {
      return a.name.localeCompare(b.name);
    });
  }
  if (sortKey && sortDir === 'desc') {
    visible = [...visible].reverse();
  }

  const selected = CUSTOMERS.find((c) => {
    return c.id === selectedId;
  });

  // ---- decisions ----
  let drawer = null;
  if (selected) {
    drawer = <CustomerDrawer customer={selected} onClose={() => { setSelectedId(null); }} />;
  }

  return (
    <div className="page">
      <header className="page-header">
        <h1>Customers</h1>
        <p className="page-sub">{visible.length} of {CUSTOMERS.length} customers</p>
      </header>

      <div className="toolbar">
        <SearchInput value={query} onChange={setQuery} placeholder="Search customers…" />
        <select
          className="select"
          value={status}
          onChange={(e) => { setStatus(e.target.value); }}
          aria-label="Filter by status"
        >
          {STATUS_OPTIONS.map((opt) => {
            return <option key={opt[0]} value={opt[0]}>{opt[1]}</option>;
          })}
        </select>
      </div>

      <CustomerTable
        customers={visible}
        sortKey={sortKey}
        sortDir={sortDir}
        onSort={handleSort}
        onClear={handleClear}
        selectedId={selectedId}
        onSelect={(id) => { setSelectedId(id); }}
      />

      {drawer}
    </div>
  );
}

import React from 'react';
import StatusBadge from './StatusBadge';
import { formatCurrency, formatDate } from '../utils/format';

function SortHeader({ label, column, sortKey, sortDir, onSort, className }) {
  let arrow = '↕';
  let arrowClass = 'sort-arrow';
  if (sortKey === column) {
    arrowClass = 'sort-arrow is-active';
    if (sortDir === 'desc') {
      arrow = '▼';
    } else {
      arrow = '▲';
    }
  }
  return (
    <th className={className}>
      <button className="sort-btn" onClick={() => { onSort(column); }}>
        {label}
        <span className={arrowClass}>{arrow}</span>
      </button>
    </th>
  );
}

export default function CustomerTable({ customers, sortKey, sortDir, onSort, onClear, selectedId, onSelect }) {
  let rows;
  if (customers.length === 0) {
    rows = (
      <tr>
        <td colSpan={4} className="empty">
          <p>No customers match your filters.</p>
          <button className="link-btn" onClick={() => { onClear(); }}>Clear filters</button>
        </td>
      </tr>
    );
  } else {
    rows = customers.map((c) => {
      let rowClass = 'row';
      if (c.id === selectedId) {
        rowClass = 'row is-selected';
      }
      return (
        <tr key={c.id} className={rowClass} onClick={() => { onSelect(c.id); }}>
          <td>
            <div className="cell-name">{c.name}</div>
            <div className="cell-sub">{c.company}</div>
          </td>
          <td><StatusBadge status={c.status} /></td>
          <td className="num">{formatCurrency(c.revenue)}</td>
          <td className="muted">{formatDate(c.joinedAt)}</td>
        </tr>
      );
    });
  }

  return (
    <div className="table-card">
      <table className="table">
        <thead>
          <tr>
            <SortHeader label="Name" column="name" sortKey={sortKey} sortDir={sortDir} onSort={onSort} />
            <th>Status</th>
            <SortHeader label="Revenue" column="revenue" sortKey={sortKey} sortDir={sortDir} onSort={onSort} className="num" />
            <th>Joined</th>
          </tr>
        </thead>
        <tbody>{rows}</tbody>
      </table>
    </div>
  );
}
