// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /cities?q=lo -> up to 6 cities whose "Name, Country" contains q (case-insensitive).
// Names that START with q come first. Note: responses can arrive out of order.
export async function searchCities(query: string): Promise<City[]> {
  await randomDelay(250, 700);
  const q = query.trim().toLowerCase();
  if (q === '') {
    return [];
  }
  const starts = CITIES.filter((c) => c.name.toLowerCase().startsWith(q));
  const contains = CITIES.filter((c) => !c.name.toLowerCase().startsWith(q) && (c.name + ', ' + c.country).toLowerCase().includes(q));
  return starts.concat(contains).slice(0, 6).map((c) => ({ ...c }));
}

// GET /me/recent-searches -> City[] (most recent first)
export async function getRecentSearches(): Promise<City[]> {
  await randomDelay(200, 500);
  return RECENT.map((id) => ({ ...(CITIES.find((c) => c.id === id) as City) }));
}

import React, { useEffect, useState } from 'react';
import { getRecentSearches, searchCities } from '../api';
import type { City } from '../api';
import { formatCity } from '../utils/format';

type CityInputProps = {
  label: string;
  placeholder: string;
};

// CHANGE: wraps the part of `text` that matches `query` in <mark>.
function Highlight({ text, query }: { text: string; query: string }) {
  const q = query.trim().toLowerCase();
  const start = text.toLowerCase().indexOf(q);
  if (q === '' || start === -1) {
    return <>{text}</>;
  }
  const end = start + q.length;
  return (
    <>
      {text.slice(0, start)}
      <mark>{text.slice(start, end)}</mark>
      {text.slice(end)}
    </>
  );
}

export function CityInput({ label, placeholder }: CityInputProps) {
  const [query, setQuery] = useState('');
  // CORE: results, loading, open
  const [results, setResults] = useState<City[]>([]);
  const [loading, setLoading] = useState(false);
  const [open, setOpen] = useState(false);
  // NEXT: the one highlighted row (keyboard + mouse) and the chosen city (code badge)
  const [activeIndex, setActiveIndex] = useState(-1);
  const [selected, setSelected] = useState<City | null>(null);
  // STRETCH: recent searches
  const [recent, setRecent] = useState<City[]>([]);

  // CORE: search on every query change. The cancelled flag throws away responses
  // for older queries that arrive late (the API's delay is random).
  useEffect(() => {
    if (query.trim() === '') {
      setResults([]);
      setLoading(false);
      return;
    }
    let cancelled = false;
    setLoading(true);
    searchCities(query).then((data) => {
      if (cancelled) {
        return;
      }
      setResults(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, [query]);

  // STRETCH: recent searches, loaded once.
  useEffect(() => {
    let cancelled = false;
    getRecentSearches().then((data) => {
      if (cancelled) {
        return;
      }
      setRecent(data);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // STRETCH: derived, the list the keyboard moves through (Core just uses `results`)
  const showingRecent = query.trim() === '';
  let options = results;
  if (showingRecent) {
    options = recent;
  }

  // CORE: pick a city
  function choose(city: City) {
    setQuery(formatCity(city));
    setOpen(false);
    // NEXT
    setSelected(city);
    setActiveIndex(-1);
    // STRETCH: move it to the top of recents (no duplicates, max 5)
    setRecent((prev) => {
      const rest = prev.filter((c) => {
        return c.id !== city.id;
      });
      return [city, ...rest].slice(0, 5);
    });
  }

  // CORE: typing opens the dropdown
  function handleChange(value: string) {
    setQuery(value);
    setOpen(true);
    // NEXT
    setSelected(null);
    setActiveIndex(-1);
  }

  // NEXT: keyboard
  function handleKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
    if (e.key === 'ArrowDown') {
      e.preventDefault(); // stop the cursor jumping to the end
      setOpen(true);
      if (options.length > 0) {
        setActiveIndex((prev) => {
          return (prev + 1) % options.length;
        });
      }
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      if (options.length > 0) {
        setActiveIndex((prev) => {
          if (prev <= 0) {
            return options.length - 1;
          }
          return prev - 1;
        });
      }
    } else if (e.key === 'Enter') {
      if (open && activeIndex >= 0 && activeIndex < options.length) {
        e.preventDefault(); // don't submit the search form
        choose(options[activeIndex]);
      }
    } else if (e.key === 'Escape') {
      setOpen(false);
      setActiveIndex(-1);
    }
  }

  // CORE: dropdown decisions
  let dropdown = null;
  if (open) {
    let content = null;
    if (showingRecent) {
      // STRETCH
      if (recent.length > 0) {
        content = <p className="ac-heading">Recent searches</p>;
      }
      // NEXT: "Searching…" / no matches
    } else if (results.length === 0 && loading) {
      content = <p className="ac-message">Searching…</p>;
    } else if (results.length === 0) {
      content = <p className="ac-message">No cities match "{query.trim()}"</p>;
    }

    // CORE: the list
    let list = null;
    if (options.length > 0) {
      list = (
        <ul className="ac-list" role="listbox" id="city-options">
          {options.map((city, i) => {
            // NEXT: highlighted row
            let cls = 'ac-option';
            if (i === activeIndex) {
              cls = 'ac-option is-active';
            }
            // STRETCH: clock icon for recents
            let icon = '📍';
            if (showingRecent) {
              icon = '🕘';
            }
            // CORE (NEXT: id/role/aria-selected/onMouseEnter. CHANGE: <Highlight>)
            // onMouseDown + preventDefault keeps focus in the input, so blur doesn't close the list before the click lands.
            return (
              <li
                key={city.id}
                id={'city-' + city.id}
                role="option"
                aria-selected={i === activeIndex}
                className={cls}
                onMouseEnter={() => { setActiveIndex(i); }}
                onMouseDown={(e) => { e.preventDefault(); }}
                onClick={() => { choose(city); }}
              >
                <span className="ac-icon">{icon}</span>
                <span className="ac-name"><Highlight text={formatCity(city)} query={query} /></span>
                <span className="ac-code">{city.code}</span>
              </li>
            );
          })}
        </ul>
      );
    }

    // CORE
    if (content || list) {
      dropdown = (
        <div className="ac-dropdown">
          {content}
          {list}
        </div>
      );
    }
  }

  // NEXT: ARIA active descendant
  let activeId: string | undefined = undefined;
  if (open && activeIndex >= 0 && options[activeIndex]) {
    activeId = 'city-' + options[activeIndex].id;
  }

  // NEXT: code badge
  let codeBadge = null;
  if (selected) {
    codeBadge = <span className="code-badge">{selected.code}</span>;
  }

  // CORE: render (NEXT: role/aria-*, onKeyDown, codeBadge)
  return (
    <div className="field city-field">
      <span className="field-label">{label}</span>
      <div className="city-input-row">
        <input
          className="field-input"
          placeholder={placeholder}
          value={query}
          role="combobox"
          aria-expanded={open}
          aria-controls="city-options"
          aria-activedescendant={activeId}
          autoComplete="off"
          onChange={(e) => { handleChange(e.target.value); }}
          onFocus={() => { setOpen(true); }}
          onBlur={() => { setOpen(false); }}
          onKeyDown={handleKeyDown}
        />
        {codeBadge}
      </div>
      {dropdown}
    </div>
  );
}
