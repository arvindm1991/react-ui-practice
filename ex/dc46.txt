// Fake API. Treat it like a real backend you can't change.
export type Tier = 'Premium' | 'Standard' | 'Balcony';

export type Seat = {
  id: string;      // "C5"
  row: string;     // "C"
  number: number;  // 5
  tier: Tier;
  price: number;
  taken: boolean;
};

export type SeatMap = {
  show: string;
  when: string;
  rows: string[];
  seatsPerRow: number;
  seats: Seat[];   // row by row, left to right
};

const ROWS = ['A', 'B', 'C', 'D', 'E', 'F'];
const PER_ROW = 10;
const TAKEN = new Set(['A3', 'A4', 'B7', 'C1', 'C2', 'D5', 'D6', 'E9', 'F1', 'F2', 'F3', 'F10']);

function tierFor(row: string): { tier: Tier; price: number } {
  if (row === 'A' || row === 'B') {
    return { tier: 'Premium', price: 95 };
  }
  if (row === 'C' || row === 'D') {
    return { tier: 'Standard', price: 65 };
  }
  return { tier: 'Balcony', price: 40 };
}

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /shows/hamlet/seats -> SeatMap
export async function getSeatMap(): Promise<SeatMap> {
  await randomDelay(400, 1000);
  const seats: Seat[] = [];
  ROWS.forEach((row) => {
    for (let n = 1; n <= PER_ROW; n++) {
      const id = row + n;
      const t = tierFor(row);
      seats.push({ id: id, row: row, number: n, tier: t.tier, price: t.price, taken: TAKEN.has(id) });
    }
  });
  return { show: 'Hamlet', when: 'Fri 14 Nov · 7:30 PM', rows: [...ROWS], seatsPerRow: PER_ROW, seats: seats };
}

// POST /shows/hamlet/bookings { seatIds } -> { confirmation }
// Simulated race: someone else always grabs seat C5 while you check out.
// Any booking that includes C5 is rejected (and C5 becomes taken).
export async function bookSeats(seatIds: string[]): Promise<{ confirmation: string }> {
  console.log('POST /bookings', seatIds.join(','));
  await randomDelay(800, 1400);
  if (seatIds.includes('C5')) {
    TAKEN.add('C5');
    throw new Error('Seat C5 was just taken by someone else.');
  }
  seatIds.forEach((id) => {
    TAKEN.add(id);
  });
  return { confirmation: 'LMN-' + (4800 + TAKEN.size) };
}

// formatPrice(65) -> "$65", formatPrice(130.5) -> "$130.50"
export function formatPrice(n: number): string {
  if (Number.isInteger(n)) {
    return '$' + n;
  }
  return '$' + n.toFixed(2);
}

import React, { useEffect, useRef, useState } from 'react';
import { bookSeats, getSeatMap } from './api';
import type { Seat, SeatMap } from './api';
import { formatPrice } from './utils/money';

const MAX_SEATS = 4;

function ShowHeader({ show, when }: { show: string; when: string }) {
  return (
    <header className="show-header">
      <p className="venue">Lumen Playhouse</p>
      <h1>{show}</h1>
      <p className="when">{when}</p>
    </header>
  );
}

function Legend() {
  return (
    <ul className="legend">
      <li><span className="dot tier-premium" /> Premium · $95</li>
      <li><span className="dot tier-standard" /> Standard · $65</li>
      <li><span className="dot tier-balcony" /> Balcony · $40</li>
      <li><span className="dot is-selected" /> Selected</li>
      <li><span className="dot is-taken" /> Taken</li>
    </ul>
  );
}

type SeatGridProps = {
  map: SeatMap;
  selected: string[];
  onToggle: (seat: Seat) => void;
};

function SeatGrid({ map, selected, onToggle }: SeatGridProps) {
  // Roving tabindex: only one seat is in the Tab order; arrow keys move it.
  const [focusId, setFocusId] = useState(map.seats[0].id);
  const seatRefs = useRef<Record<string, HTMLButtonElement | null>>({});

  function moveFocus(seat: Seat, dRow: number, dCol: number) {
    const rowIndex = map.rows.indexOf(seat.row) + dRow;
    const number = seat.number + dCol;
    if (rowIndex < 0 || rowIndex >= map.rows.length || number < 1 || number > map.seatsPerRow) {
      return; // edge of the map: stay put
    }
    const id = map.rows[rowIndex] + number;
    setFocusId(id);
    const el = seatRefs.current[id];
    if (el) {
      el.focus();
    }
  }

  function handleKeyDown(e: React.KeyboardEvent<HTMLButtonElement>, seat: Seat) {
    if (e.key === 'ArrowRight') {
      e.preventDefault();
      moveFocus(seat, 0, 1);
    } else if (e.key === 'ArrowLeft') {
      e.preventDefault();
      moveFocus(seat, 0, -1);
    } else if (e.key === 'ArrowDown') {
      e.preventDefault();
      moveFocus(seat, 1, 0);
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      moveFocus(seat, -1, 0);
    }
    // Enter and Space already "click" a <button>, so they toggle for free.
  }

  return (
    <div className="seat-map" role="group" aria-label="Seat map. Use arrow keys to move between seats.">
      <div className="stage">Stage</div>
      {map.rows.map((row) => {
        const seats = map.seats.filter((s) => {
          return s.row === row;
        });
        return (
          <div key={row} className="seat-row">
            <span className="row-label">{row}</span>
            {seats.map((s) => {
              const isSelected = selected.includes(s.id);
              let cls = 'seat tier-' + s.tier.toLowerCase();
              if (s.taken) {
                cls = 'seat is-taken';
              } else if (isSelected) {
                cls = cls + ' is-selected';
              }
              let label = 'Row ' + s.row + ', seat ' + s.number + ', ' + s.tier + ' ' + formatPrice(s.price);
              if (s.taken) {
                label = label + ', taken';
              }
              let tabIndex = -1;
              if (s.id === focusId) {
                tabIndex = 0;
              }
              // aria-disabled instead of disabled, so arrow keys can still move across taken seats
              return (
                <button
                  key={s.id}
                  ref={(el) => { seatRefs.current[s.id] = el; }}
                  type="button"
                  className={cls}
                  tabIndex={tabIndex}
                  aria-label={label}
                  aria-pressed={isSelected}
                  aria-disabled={s.taken}
                  onClick={() => { onToggle(s); }}
                  onKeyDown={(e) => { handleKeyDown(e, s); }}
                  onFocus={() => { setFocusId(s.id); }}
                >
                  {s.number}
                </button>
              );
            })}
            <span className="row-label">{row}</span>
          </div>
        );
      })}
    </div>
  );
}

type SummaryProps = {
  seats: Seat[];
  booking: boolean;
  onRemove: (seat: Seat) => void;
  onBook: () => void;
};

function Summary({ seats, booking, onRemove, onBook }: SummaryProps) {
  // derived
  const total = seats.reduce((sum, s) => {
    return sum + s.price;
  }, 0);

  let list = <p className="summary-empty">Pick up to {MAX_SEATS} seats on the map.</p>;
  if (seats.length > 0) {
    list = (
      <ul className="picked">
        {seats.map((s) => {
          return (
            <li key={s.id} className="picked-row">
              <span className={'dot tier-' + s.tier.toLowerCase()} />
              <span className="picked-seat">{s.id}</span>
              <span className="picked-tier">{s.tier}</span>
              <span className="picked-price">{formatPrice(s.price)}</span>
              <button type="button" className="picked-remove" aria-label={'Remove seat ' + s.id} onClick={() => { onRemove(s); }}>×</button>
            </li>
          );
        })}
      </ul>
    );
  }

  let bookLabel = 'Book seats';
  if (seats.length > 0) {
    let noun = 'seats';
    if (seats.length === 1) {
      noun = 'seat';
    }
    bookLabel = 'Book ' + seats.length + ' ' + noun + ' · ' + formatPrice(total);
  }
  if (booking) {
    bookLabel = 'Booking…';
  }

  return (
    <aside className="summary">
      <h2>Your seats <span className="picked-count">{seats.length} / {MAX_SEATS}</span></h2>
      {list}
      <div className="summary-total"><span>Total</span><b>{formatPrice(total)}</b></div>
      <button type="button" className="book-btn" disabled={seats.length === 0 || booking} onClick={() => { onBook(); }}>{bookLabel}</button>
    </aside>
  );
}

export default function App() {
  const [map, setMap] = useState<SeatMap | null>(null);
  const [reloadKey, setReloadKey] = useState(0); // bump to refetch the map
  const [selected, setSelected] = useState<string[]>([]);
  const [booking, setBooking] = useState(false);
  const [error, setError] = useState('');
  const [notice, setNotice] = useState('');
  const [confirmation, setConfirmation] = useState('');

  useEffect(() => {
    let cancelled = false;
    getSeatMap().then((data) => {
      if (cancelled) {
        return;
      }
      setMap(data);
      // Drop any selected seat that is now taken.
      setSelected((prev) => {
        return prev.filter((id) => {
          const seat = data.seats.find((s) => {
            return s.id === id;
          });
          return seat !== undefined && !seat.taken;
        });
      });
    });
    return () => {
      cancelled = true;
    };
  }, [reloadKey]);

  function handleToggle(seat: Seat) {
    if (seat.taken || booking) {
      return;
    }
    setConfirmation('');
    if (selected.includes(seat.id)) {
      setNotice('');
      setSelected((prev) => {
        return prev.filter((id) => {
          return id !== seat.id;
        });
      });
      return;
    }
    if (selected.length >= MAX_SEATS) {
      setNotice('You can select up to ' + MAX_SEATS + ' seats.');
      return;
    }
    setSelected((prev) => {
      return [...prev, seat.id];
    });
  }

  function handleBook() {
    const ids = selected;
    setBooking(true);
    setError('');
    setNotice('');
    bookSeats(ids)
      .then((res) => {
        setConfirmation('Booked! Confirmation ' + res.confirmation + ' · seats ' + ids.join(', '));
        setSelected([]);
      })
      .catch((err: Error) => {
        setError(err.message + ' We refreshed the map. Please choose another seat.');
      })
      .finally(() => {
        setBooking(false);
        setReloadKey((k) => {
          return k + 1;
        });
      });
  }

  if (!map) {
    return <div className="app"><p className="loading">Loading seat map…</p></div>;
  }

  // derived: the full seat objects, in the order they were picked
  const selectedSeats: Seat[] = [];
  selected.forEach((id) => {
    const seat = map.seats.find((s) => {
      return s.id === id;
    });
    if (seat) {
      selectedSeats.push(seat);
    }
  });

  let banner = null;
  if (error) {
    banner = <div className="banner is-error" role="alert">{error}</div>;
  } else if (confirmation) {
    banner = <div className="banner is-success" role="status">{confirmation}</div>;
  } else if (notice) {
    banner = <div className="banner is-notice" role="status">{notice}</div>;
  }

  return (
    <div className="app">
      <ShowHeader show={map.show} when={map.when} />
      {banner}
      <div className="layout">
        <section className="map-panel">
          <SeatGrid map={map} selected={selected} onToggle={handleToggle} />
          <Legend />
        </section>
        <Summary seats={selectedSeats} booking={booking} onRemove={handleToggle} onBook={handleBook} />
      </div>
    </div>
  );
}
