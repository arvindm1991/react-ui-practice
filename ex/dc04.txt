// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /products/:id -> Product
export async function getProduct(id: string): Promise<Product> {
  await randomDelay(400, 1200);
  return {
    ...PRODUCT,
    id,
    specs: PRODUCT.specs.map((s) => {
      return { ...s };
    }),
    reviews: PRODUCT.reviews.map((r) => {
      return { ...r };
    }),
  };
}

export function DescriptionPanel({ paragraphs }: { paragraphs: string[] }) {
  return (
    <div className="panel-text">
      {paragraphs.map((p) => {
        return <p key={p}>{p}</p>;
      })}
    </div>
  );
}

export function SpecsTable({ specs }: { specs: Spec[] }) {
  return (
    <table className="specs">
      <tbody>
        {specs.map((s) => {
          return (
            <tr key={s.label}>
              <th>{s.label}</th>
              <td>{s.value}</td>
            </tr>
          );
        })}
      </tbody>
    </table>
  );
}

export function ReviewList({ reviews }: { reviews: Review[] }) {
  return (
    <ul className="reviews">
      {reviews.map((r) => {
        const stars = '★'.repeat(r.rating) + '☆'.repeat(5 - r.rating);
        return (
          <li key={r.id} className="review">
            <p className="review-stars">{stars}</p>
            <p className="review-title">{r.title}</p>
            <p className="review-body">{r.body}</p>
            <p className="review-author">{r.author}</p>
          </li>
        );
      })}
    </ul>
  );
}

@@@ solution src/App.tsx
import React, { useEffect, useState } from 'react';
import { getProduct } from './api';
import type { Product } from './api';
import { formatPrice } from './utils/format';
import { DescriptionPanel, SpecsTable, ReviewList } from './components/Panels';

type TabId = 'description' | 'specs' | 'reviews';

function Nav() {
  return (
    <header className="nav">
      <span className="wordmark">Kettle &amp; Co.</span>
      <span className="nav-links">Brew · Grind · Serve</span>
      <span className="cart">Cart (0)</span>
    </header>
  );
}

type ProductSummaryProps = {
  product: Product;
  onReviewsClick: () => void;
};

function ProductSummary({ product, onReviewsClick }: ProductSummaryProps) {
  return (
    <section className="summary">
      <div className="gallery" aria-hidden="true">
        <div className="kettle" />
      </div>
      <div className="info">
        <p className="eyebrow">Electric kettles</p>
        <h1>{product.name}</h1>
        <p className="tagline">{product.tagline}</p>
        <button type="button" className="rating-link" onClick={() => { onReviewsClick(); }}>
          ★ {product.rating} · {product.reviews.length} reviews
        </button>
        <p className="price">{formatPrice(product.price)}</p>
        <button type="button" className="btn-cart">Add to cart</button>
      </div>
    </section>
  );
}

type ProductTabsProps = {
  product: Product;
  activeTab: TabId;
  onChange: (tab: TabId) => void;
};

function ProductTabs({ product, activeTab, onChange }: ProductTabsProps) {
  const tabs: { id: TabId; label: string }[] = [
    { id: 'description', label: 'Description' },
    { id: 'specs', label: 'Specs' },
    { id: 'reviews', label: 'Reviews (' + product.reviews.length + ')' },
  ];

  let panel = <DescriptionPanel paragraphs={product.description} />;
  if (activeTab === 'specs') {
    panel = <SpecsTable specs={product.specs} />;
  } else if (activeTab === 'reviews') {
    panel = <ReviewList reviews={product.reviews} />;
  }

  return (
    <section className="details">
      <div className="tabs" role="tablist">
        {tabs.map((t) => {
          const selected = t.id === activeTab;
          let cls = 'tab';
          if (selected) {
            cls = 'tab is-active';
          }
          return (
            <button key={t.id} type="button" role="tab" aria-selected={selected} className={cls} onClick={() => { onChange(t.id); }}>
              {t.label}
            </button>
          );
        })}
      </div>
      <div className="tab-panel" role="tabpanel">{panel}</div>
    </section>
  );
}

function ProductPage() {
  const [product, setProduct] = useState<Product | null>(null);
  // Lives here (not in ProductTabs) because the rating link in the summary changes it too.
  const [activeTab, setActiveTab] = useState<TabId>('description');

  useEffect(() => {
    let cancelled = false;
    getProduct('kettle-pour-1').then((data) => {
      if (cancelled) {
        return;
      }
      setProduct(data);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  function handleReviewsClick() {
    setActiveTab('reviews');
  }

  if (!product) {
    return <p className="loading">Loading product…</p>;
  }

  return (
    <main className="page">
      <ProductSummary product={product} onReviewsClick={handleReviewsClick} />
      <ProductTabs product={product} activeTab={activeTab} onChange={setActiveTab} />
    </main>
  );
}

export default function App() {
  return (
    <div className="app">
      <Nav />
      <ProductPage />
    </div>
  );
}
