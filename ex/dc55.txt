// Fake API. Treat it like a real backend you can't change.
export type Campaign = {
  id: string;
  name: string;
  channel: 'Search' | 'Social' | 'Display' | 'Email' | 'Video';
  status: 'Active' | 'Paused' | 'Ended';
  impressions: number;
  clicks: number;
  ctr: number;          // 0.0312 = 3.12%
  spend: number;        // dollars
  conversions: number;
  cpa: number;          // cost per acquisition, dollars
  roas: number;         // return on ad spend, 3.4 = 3.4x
};

const NAMES = [
  'Spring Sale Brand', 'Retargeting Cart Abandoners', 'Lookalike 1% Purchasers', 'Newsletter Winback', 'Q2 Webinar Promo',
  'Competitor Keywords', 'Product Launch Teaser', 'App Install iOS', 'App Install Android', 'Holiday Gift Guide',
  'Free Trial Nurture', 'Brand Awareness CTV', 'Pricing Page Retarget', 'Partner Co-marketing', 'Black Friday Early Access',
  'Customer Stories Video', 'Enterprise ABM', 'SMB Self-serve', 'Generic Keywords US', 'Generic Keywords EU',
  'Referral Program', 'Onboarding Drip', 'Summer Clearance', 'Podcast Sponsorship', 'Influencer Seeding',
  'Feature Announcement', 'Churn Prevention', 'Event Follow-up', 'Student Discount', 'Year in Review',
];
const CHANNELS: Campaign['channel'][] = ['Search', 'Social', 'Display', 'Email', 'Video'];
const STATUSES: Campaign['status'][] = ['Active', 'Active', 'Paused', 'Active', 'Ended'];

// Deterministic pseudo-random numbers so the data is the same on every load.
let seed = 42;
function rand(): number {
  seed = (seed * 16807) % 2147483647;
  return (seed - 1) / 2147483646;
}

const CAMPAIGNS: Campaign[] = NAMES.map((name, i) => {
  const impressions = Math.round(20000 + rand() * 900000);
  const ctr = 0.004 + rand() * 0.05;
  const clicks = Math.round(impressions * ctr);
  const spend = Math.round(clicks * (0.4 + rand() * 2.6) * 100) / 100;
  const conversions = Math.max(1, Math.round(clicks * (0.01 + rand() * 0.08)));
  return {
    id: 'cmp-' + (i + 1),
    name,
    channel: CHANNELS[i % CHANNELS.length],
    status: STATUSES[(i * 7) % STATUSES.length],
    impressions,
    clicks,
    ctr: clicks / impressions,
    spend,
    conversions,
    cpa: Math.round((spend / conversions) * 100) / 100,
    roas: Math.round((0.6 + rand() * 6) * 10) / 10,
  };
});

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /reports/campaigns?range=30d -> Campaign[]
export async function getCampaigns(): Promise<Campaign[]> {
  await randomDelay(400, 1100);
  return CAMPAIGNS.map((c) => ({ ...c }));
}

// formatNumber(1234567) -> "1,234,567"
export function formatNumber(n: number): string {
  return n.toLocaleString('en-US');
}

// formatCurrency(1234.5) -> "$1,234.50"
export function formatCurrency(n: number): string {
  return n.toLocaleString('en-US', { style: 'currency', currency: 'USD' });
}

// formatPercent(0.0312) -> "3.12%"
export function formatPercent(n: number): string {
  return (n * 100).toFixed(2) + '%';
}

import React, { useEffect, useState } from 'react';
import { getCampaigns } from './api';
import type { Campaign } from './api';
import { COLUMNS } from './columns';
import type { Column } from './columns';
import StatusPill from './components/StatusPill';

function Cell({ column, row }: { column: Column; row: Campaign }) {
  if (column.key === 'status') {
    return <td><StatusPill status={row.status} /></td>;
  }
  let cls = '';
  if (column.align === 'right') {
    cls = 'num';
  }
  return <td className={cls}>{column.render(row)}</td>;
}

function CampaignTable({ rows }: { rows: Campaign[] }) {
  return (
    <div className="grid-scroll">
      <table className="grid">
        <thead>
          <tr>
            {COLUMNS.map((col) => {
              let cls = '';
              if (col.align === 'right') {
                cls = 'num';
              }
              return <th key={col.key} className={cls}>{col.label}</th>;
            })}
          </tr>
        </thead>
        <tbody>
          {rows.map((row) => {
            return (
              <tr key={row.id}>
                {COLUMNS.map((col) => {
                  return <Cell key={col.key} column={col} row={row} />;
                })}
              </tr>
            );
          })}
        </tbody>
      </table>
    </div>
  );
}

    };

  let body;
  if (loading) {
    body = <div className="grid-loading">Loading report…</div>;
  } else {
    body = <CampaignTable rows={rows} />;
  }

  return (
    <div className="app">
      <header className="header">
        <div>
          <p className="eyebrow">Reports · Last 30 days</p>
          <h1>Campaign performance</h1>
        </div>
        <span className="row-count">{rows.length} campaigns</span>
      </header>
      {body}
    </div>
  );
}

import React, { useEffect, useRef, useState } from 'react';
import { getCampaigns } from './api';
import type { Campaign } from './api';
import { COLUMNS } from './columns';
import type { Column, ColumnKey } from './columns';
import StatusPill from './components/StatusPill';

const MIN_WIDTH = 70;
const STORAGE_KEY = 'campaigns.columnWidths.v1';

type Widths = Record<string, number>;
type Sort = { key: ColumnKey; dir: 'asc' | 'desc' } | null;

function defaultWidths(): Widths {
  const w: Widths = {};
  COLUMNS.forEach((col) => {
    w[col.key] = col.width;
  });
  return w;
}

// Change: storage can be missing, full, blocked or hold garbage. Never trust it.
function loadWidths(): Widths {
  const widths = defaultWidths();
  try {
    const saved = JSON.parse(localStorage.getItem(STORAGE_KEY) || '{}');
    COLUMNS.forEach((col) => {
      const v = saved[col.key];
      if (typeof v === 'number' && v >= MIN_WIDTH && v < 2000) {
        widths[col.key] = v;
      }
    });
  } catch (e) {
    // corrupted JSON or storage disabled: fall back to defaults
  }
  return widths;
}

function saveWidths(widths: Widths) {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(widths));
  } catch (e) {
    // quota exceeded or private mode: resizing still works, it just won't persist
  }
}

function compareRows(a: Campaign, b: Campaign, key: ColumnKey): number {
  const x = a[key];
  const y = b[key];
  if (typeof x === 'number' && typeof y === 'number') {
    return x - y;
  }
  return String(x).localeCompare(String(y));
}

function Cell({ column, row }: { column: Column; row: Campaign }) {
  let cls = '';
  if (column.align === 'right') {
    cls = 'num';
  }
  if (column.key === 'name') {
    cls = 'sticky-col';
  }
  if (column.key === 'status') {
    return <td className={cls}><StatusPill status={row.status} /></td>;
  }
  return <td className={cls} title={column.render(row)}>{column.render(row)}</td>;
}

type HeaderCellProps = {
  column: Column;
  sort: Sort;
  resizing: boolean;
  onSort: (key: ColumnKey) => void;
  onResizeStart: (e: React.PointerEvent<HTMLSpanElement>, key: ColumnKey) => void;
  onResizeMove: (e: React.PointerEvent<HTMLSpanElement>) => void;
  onResizeEnd: () => void;
  onResetWidth: (key: ColumnKey) => void;
};

function HeaderCell({ column, sort, resizing, onSort, onResizeStart, onResizeMove, onResizeEnd, onResetWidth }: HeaderCellProps) {
  let ariaSort: 'ascending' | 'descending' | 'none' = 'none';
  let arrow = '↕';
  if (sort && sort.key === column.key) {
    ariaSort = 'ascending';
    arrow = '▲';
    if (sort.dir === 'desc') {
      ariaSort = 'descending';
      arrow = '▼';
    }
  }

  let cls = 'th';
  if (column.align === 'right') {
    cls += ' num';
  }
  if (column.key === 'name') {
    cls += ' sticky-col';
  }
  if (ariaSort !== 'none') {
    cls += ' is-sorted';
  }

  let resizerCls = 'resizer';
  if (resizing) {
    resizerCls = 'resizer is-active';
  }

  return (
    <th className={cls} aria-sort={ariaSort}>
      <button type="button" className="sort-btn" onClick={() => { onSort(column.key); }}>
        <span className="th-label">{column.label}</span>
        <span className="sort-arrow" aria-hidden="true">{arrow}</span>
      </button>
      <span
        className={resizerCls}
        role="separator"
        aria-orientation="vertical"
        aria-label={'Resize ' + column.label}
        onPointerDown={(e) => { onResizeStart(e, column.key); }}
        onPointerMove={(e) => { onResizeMove(e); }}
        onPointerUp={() => { onResizeEnd(); }}
        onPointerCancel={() => { onResizeEnd(); }}
        onDoubleClick={() => { onResetWidth(column.key); }}
      />
    </th>
  );
}

function CampaignTable({ rows }: { rows: Campaign[] }) {
  const [widths, setWidths] = useState<Widths>(loadWidths); // lazy init: read storage once
  const [sort, setSort] = useState<Sort>(null);
  const [resizingKey, setResizingKey] = useState<ColumnKey | null>(null);
  // Drag bookkeeping changes on every pointermove and never renders: a ref.
  const dragRef = useRef<{ key: ColumnKey; startX: number; startWidth: number } | null>(null);

  // Persist once a drag ends, not on every pointermove.
  useEffect(() => {
    if (resizingKey === null) {
      saveWidths(widths);
    }
  }, [widths, resizingKey]);

  function handleSort(key: ColumnKey) {
    setSort((prev) => {
      if (prev && prev.key === key && prev.dir === 'asc') {
        return { key, dir: 'desc' };
      }
      return { key, dir: 'asc' };
    });
  }

  function handleResizeStart(e: React.PointerEvent<HTMLSpanElement>, key: ColumnKey) {
    e.preventDefault(); // no text selection while dragging
    // Capture: keep getting pointermove even when the pointer leaves the 6px handle.
    e.currentTarget.setPointerCapture(e.pointerId);
    dragRef.current = { key, startX: e.clientX, startWidth: widths[key] };
    setResizingKey(key);
  }

  function handleResizeMove(e: React.PointerEvent<HTMLSpanElement>) {
    const drag = dragRef.current;
    if (!drag) {
      return;
    }
    const next = Math.max(MIN_WIDTH, Math.round(drag.startWidth + e.clientX - drag.startX));
    setWidths((prev) => {
      return { ...prev, [drag.key]: next };
    });
  }

  function handleResizeEnd() {
    dragRef.current = null;
    setResizingKey(null);
  }

  function handleResetWidth(key: ColumnKey) {
    const col = COLUMNS.find((c) => { return c.key === key; });
    if (col) {
      setWidths((prev) => {
        return { ...prev, [key]: col.width };
      });
    }
  }

  // derived: sorted copy, never sort state in place
  let sorted = rows;
  if (sort) {
    const s = sort;
    sorted = [...rows].sort((a, b) => {
      const c = compareRows(a, b, s.key);
      if (s.dir === 'asc') {
        return c;
      }
      return -c;
    });
  }

  let total = 0;
  COLUMNS.forEach((col) => {
    total += widths[col.key];
  });

  let scrollCls = 'grid-scroll';
  if (resizingKey) {
    scrollCls = 'grid-scroll is-resizing';
  }

  return (
    <div className={scrollCls}>
      <table className="grid" style={{ width: total }}>
        <colgroup>
          {COLUMNS.map((col) => {
            return <col key={col.key} style={{ width: widths[col.key] }} />;
          })}
        </colgroup>
        <thead>
          <tr>
            {COLUMNS.map((col) => {
              return (
                <HeaderCell
                  key={col.key}
                  column={col}
                  sort={sort}
                  resizing={resizingKey === col.key}
                  onSort={handleSort}
                  onResizeStart={handleResizeStart}
                  onResizeMove={handleResizeMove}
                  onResizeEnd={handleResizeEnd}
                  onResetWidth={handleResetWidth}
                />
              );
            })}
          </tr>
        </thead>
        <tbody>
          {sorted.map((row) => {
            return (
              <tr key={row.id}>
                {COLUMNS.map((col) => {
                  return <Cell key={col.key} column={col} row={row} />;
                })}
              </tr>
            );
          })}
        </tbody>
      </table>
    </div>
  );
}

export default function App() {
  const [rows, setRows] = useState<Campaign[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    getCampaigns().then((data) => {
      if (cancelled) {
        return;
      }
      setRows(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  let body;
  if (loading) {
    body = <div className="grid-loading">Loading report…</div>;
  } else {
    body = <CampaignTable rows={rows} />;
  }

  return (
    <div className="app">
      <header className="header">
        <div>
          <p className="eyebrow">Reports · Last 30 days</p>
          <h1>Campaign performance</h1>
        </div>
        <span className="row-count">{rows.length} campaigns</span>
      </header>
      {body}
    </div>
  );
}
