export type Product = {
  id: string;
  name: string;
  price: number; // in CENTS: 2800 = $28.00
  glyph: string;
  tint: string;
};

export type CouponResult =
  | { ok: true; code: string; percentOff: number }
  | { ok: false; message: string };

const PRODUCTS: Product[] = [
  { id: 'p1', name: 'Gooseneck kettle', price: 4900, glyph: '🫖', tint: '#F1E4D3' },
  { id: 'p2', name: 'Ceramic dripper', price: 2800, glyph: '☕', tint: '#E7EDE4' },
  { id: 'p3', name: 'Paper filters (100)', price: 850, glyph: '📄', tint: '#F4EFE6' },
  { id: 'p4', name: 'Hand burr grinder', price: 12900, glyph: '⚙️', tint: '#E4E7EE' },
  { id: 'p5', name: 'Single-origin beans', price: 1950, glyph: '🫘', tint: '#EFE1D6' },
  { id: 'p6', name: 'Brew scale', price: 3400, glyph: '⚖️', tint: '#E6EEF0' },
];

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /products -> Product[]
export async function getProducts(): Promise<Product[]> {
  await randomDelay(400, 900);
  return PRODUCTS.map((p) => ({ ...p }));
}

// POST /coupons/validate { code } -> CouponResult
// Only "SAVE10" (any case) is valid: 10% off. Anything else returns ok: false.
export async function validateCoupon(code: string): Promise<CouponResult> {
  await randomDelay(500, 1000);
  const clean = code.trim().toUpperCase();
  if (clean === 'SAVE10') {
    return { ok: true, code: clean, percentOff: 10 };
  }
  return { ok: false, message: 'Code "' + clean + '" isn\'t valid.' };
}

// All money is integer CENTS. Only format at the very end.
export const TAX_RATE = 0.0825;

// formatMoney(2800) -> "$28.00", formatMoney(-840) -> "-$8.40"
export function formatMoney(cents: number): string {
  let sign = '';
  if (cents < 0) {
    sign = '-';
  }
  return sign + '$' + (Math.abs(cents) / 100).toFixed(2);
}

// calcTax(7560) -> 624 (8.25%, rounded to the nearest cent)
export function calcTax(cents: number): number {
  return Math.round(cents * TAX_RATE);
}

// percentOf(5600, 10) -> 560
export function percentOf(cents: number, percent: number): number {
  return Math.round((cents * percent) / 100);
}

function Header({ count, onOpenCart }: { count: number; onOpenCart: () => void }) {
  return (

function ProductCard({ product, onAdd }: { product: Product; onAdd: (id: string) => void }) {
  return (
    <li className="product">
      <div className="product-img" style={{ background: product.tint }}>{product.glyph}</div>

        <span className="product-name">{product.name}</span>

export default function App() {
  const [products, setProducts] = useState<Product[]>([]);
  const [loading, setLoading] = useState(true);
  // The cart stores ids + quantities only. Names and prices come from products.
  const [cart, setCart] = useState<CartLine[]>([]);
  const [drawerOpen, setDrawerOpen] = useState(false);
  // In App (not the drawer) so closing the drawer keeps the coupon.
  const [coupon, setCoupon] = useState<Coupon | null>(null);

  useEffect(() => {
    let cancelled = false;
    getProducts().then((data) => {
      if (cancelled) {
        return;
      }
      setProducts(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  function addToCart(productId: string) {
    setCart((prev) => {
      const existing = prev.find((line) => {
        return line.productId === productId;
      });
      if (existing) {
        return prev.map((line) => {
          if (line.productId !== productId) {
            return line;
          }
          return { ...line, qty: line.qty + 1 };
        });
      }
      return [...prev, { productId, qty: 1 }];
    });
  }

  function changeQty(productId: string, delta: number) {
    setCart((prev) => {
      return prev.map((line) => {
        if (line.productId !== productId) {
          return line;
        }
        return { ...line, qty: Math.max(1, line.qty + delta) };
      });
    });
  }

  function removeLine(productId: string) {
    setCart((prev) => {
      return prev.filter((line) => {
        return line.productId !== productId;
      });
    });
  }

  // derived
  const itemCount = cart.reduce((sum, line) => {
    return sum + line.qty;
  }, 0);

  let catalog;
  if (loading) {
    catalog = <p className="loading">Loading the shop…</p>;
  } else {
    catalog = (
      <ul className="grid">
        {products.map((p) => {
          return <ProductCard key={p.id} product={p} onAdd={addToCart} />;
        })}
      </ul>
    );
  }

  let drawer = null;
  if (drawerOpen) {
    drawer = (
      <CartDrawer
        cart={cart}
        products={products}
        coupon={coupon}
        onCouponChange={setCoupon}
        onChangeQty={changeQty}
        onRemove={removeLine}
        onClose={() => { setDrawerOpen(false); }}
      />
    );
  }

  return (
    <div className="app">
      <Header count={itemCount} onOpenCart={() => { setDrawerOpen(true); }} />
      <section className="banner">
        <h1>Slow coffee, done right.</h1>
        <p>Pour-over gear we actually use. Free shipping over $75.</p>
      </section>
      <main className="shop">{catalog}</main>
      {drawer}
    </div>
  );
}

import React, { useState } from 'react';
import { validateCoupon } from '../api';
import type { Product } from '../api';
import { formatMoney, calcTax, percentOf } from '../utils/money';

export type CartLine = { productId: string; qty: number };
export type Coupon = { code: string; percentOff: number };

const FREE_SHIPPING_CENTS = 7500;

type CartDrawerProps = {
  cart: CartLine[];
  products: Product[];
  coupon: Coupon | null;
  onCouponChange: (coupon: Coupon | null) => void;
  onChangeQty: (productId: string, delta: number) => void;
  onRemove: (productId: string) => void;
  onClose: () => void;
};

function CouponForm({ onApply }: { onApply: (coupon: Coupon) => void }) {
  const [code, setCode] = useState('');
  const [checking, setChecking] = useState(false);
  const [error, setError] = useState('');

  function handleSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    if (code.trim() === '' || checking) {
      return;
    }
    setChecking(true);
    setError('');
    validateCoupon(code).then((result) => {
      setChecking(false);
      if (result.ok) {
        onApply({ code: result.code, percentOff: result.percentOff });
      } else {
        setError(result.message);
      }
    });
  }

  let applyLabel = 'Apply';
  if (checking) {
    applyLabel = 'Checking…';
  }
  let errorEl = null;
  if (error !== '') {
    errorEl = <p className="coupon-error" role="alert">{error}</p>;
  }

  return (
    <form className="coupon" onSubmit={handleSubmit}>
      <div className="coupon-row">
        <input
          className="coupon-input"
          placeholder="Coupon code"
          aria-label="Coupon code"
          value={code}
          onChange={(e) => { setCode(e.target.value); }}
        />
        <button type="submit" className="coupon-btn" disabled={checking}>{applyLabel}</button>
      </div>
      {errorEl}
    </form>
  );
}

function ShippingProgress({ subtotal }: { subtotal: number }) {
  const remaining = FREE_SHIPPING_CENTS - subtotal;
  const percent = Math.min(100, Math.round((subtotal / FREE_SHIPPING_CENTS) * 100));

  let message = <span>Add <b>{formatMoney(remaining)}</b> more for free shipping</span>;
  let cls = 'ship';
  if (remaining <= 0) {
    message = <span>You've unlocked <b>free shipping</b> 🎉</span>;
    cls = 'ship is-unlocked';
  }

  return (
    <div className={cls}>
      <p className="ship-text">{message}</p>
      <div className="ship-track" role="progressbar" aria-valuemin={0} aria-valuemax={100} aria-valuenow={percent}>
        <div className="ship-fill" style={{ width: percent + '%' }} />
      </div>
    </div>
  );
}

export default function CartDrawer({ cart, products, coupon, onCouponChange, onChangeQty, onRemove, onClose }: CartDrawerProps) {
  // Join cart lines with product data. Skip lines whose product is unknown.
  const lines: { productId: string; qty: number; product: Product }[] = [];
  cart.forEach((line) => {
    const product = products.find((p) => {
      return p.id === line.productId;
    });
    if (!product) {
      return;
    }
    lines.push({ ...line, product });
  });

  // derived totals, all in cents
  const subtotal = lines.reduce((sum, line) => {
    return sum + line.product.price * line.qty;
  }, 0);
  let discount = 0;
  if (coupon) {
    discount = percentOf(subtotal, coupon.percentOff);
  }
  const tax = calcTax(subtotal - discount);
  const total = subtotal - discount + tax;

  function handleBackdropClick(e: React.MouseEvent<HTMLDivElement>) {
    if (e.target === e.currentTarget) {
      onClose();
    }
  }

  let body;
  if (lines.length === 0) {
    body = <p className="cart-empty">Your cart is empty.</p>;
  } else {
    let couponArea = <CouponForm onApply={onCouponChange} />;
    let discountRow = null;
    if (coupon) {
      couponArea = (
        <div className="coupon-applied">
          <span><b>{coupon.code}</b> applied · {coupon.percentOff}% off</span>
          <button type="button" className="text-btn" onClick={() => { onCouponChange(null); }}>Remove</button>
        </div>
      );
      discountRow = (
        <div className="sum-row discount"><span>Discount</span><span>{formatMoney(-discount)}</span></div>
      );
    }

    body = (
      <div className="cart-body">
        <ShippingProgress subtotal={subtotal} />
        <ul className="lines">
          {lines.map((line) => {
            const atMin = line.qty <= 1;
            const lineTotal = line.product.price * line.qty;
            return (
              <li key={line.productId} className="line">
                <span className="line-img" style={{ background: line.product.tint }}>{line.product.glyph}</span>
                <div className="line-main">
                  <span className="line-name">{line.product.name}</span>
                  <div className="qty">
                    <button type="button" className="qty-btn" aria-label="Decrease quantity" disabled={atMin} onClick={() => { onChangeQty(line.productId, -1); }}>−</button>
                    <span className="qty-value">{line.qty}</span>
                    <button type="button" className="qty-btn" aria-label="Increase quantity" onClick={() => { onChangeQty(line.productId, 1); }}>+</button>
                    <button type="button" className="text-btn" onClick={() => { onRemove(line.productId); }}>Remove</button>
                  </div>
                </div>
                <span className="line-total">{formatMoney(lineTotal)}</span>
              </li>
            );
          })}
        </ul>
        {couponArea}
        <div className="summary">
          <div className="sum-row"><span>Subtotal</span><span className="subtotal">{formatMoney(subtotal)}</span></div>
          {discountRow}
          <div className="sum-row"><span>Tax (8.25%)</span><span className="tax">{formatMoney(tax)}</span></div>
          <div className="sum-row total"><span>Total</span><span className="total-value">{formatMoney(total)}</span></div>
        </div>
        <button type="button" className="checkout">Checkout</button>
      </div>
    );
  }

  return (
    <div className="drawer-backdrop" onClick={handleBackdropClick}>
      <aside className="drawer" role="dialog" aria-label="Your cart">
        <div className="drawer-head">
          <h2>Your cart</h2>
          <button type="button" className="close-btn" aria-label="Close cart" onClick={() => { onClose(); }}>✕</button>
        </div>
        {body}
      </aside>
    </div>
  );
}
