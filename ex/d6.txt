import React, { useState } from 'react';

const STARTING_ITEMS = [
  { id: 1, title: 'Fix truncation on long project names', priority: 'high', done: false },
  { id: 2, title: 'Add empty state to the usage chart', priority: 'medium', done: true },
];

let nextId = 3; // simple id source; fine for an interview

export default function App() {
  const [items, setItems] = useState(STARTING_ITEMS);
  const [title, setTitle] = useState('');
  const [priority, setPriority] = useState('medium');
  const [error, setError] = useState('');

  const doneCount = items.filter((i) => i.done).length;
  const openCount = items.length - doneCount;

  function handleSubmit(e) {
    e.preventDefault(); // stop the browser from reloading the page
    const trimmed = title.trim();
    if (!trimmed) {
      setError('Add a title');
      return;
    }
    const item = { id: nextId++, title: trimmed, priority, done: false };
    setItems((prev) => [item, ...prev]); // new array, new item first
    setTitle('');
    setPriority('medium');
  }

  function toggle(id) {
    setItems((prev) => prev.map((i) => (i.id === id ? { ...i, done: !i.done } : i)));
  }

  function remove(id) {
    setItems((prev) => prev.filter((i) => i.id !== id));
  }

  function clearCompleted() {
    setItems((prev) => prev.filter((i) => !i.done));
  }

  return (
    <div className="page">
      <h1>Feedback</h1>
      <p className="summary">
        {openCount} open · {doneCount} done
      </p>

      <form className="form" onSubmit={handleSubmit} noValidate>
        <div className="field">
          <input
            className={'input' + (error ? ' has-error' : '')}
            placeholder="What needs fixing?"
            value={title}
            onChange={(e) => {
              setTitle(e.target.value);
              if (error) setError('');
            }}
            aria-invalid={Boolean(error)}
          />
          {error && <span className="error-text">{error}</span>}
        </div>
        <select className="select" value={priority} onChange={(e) => setPriority(e.target.value)}>
          <option value="high">High</option>
          <option value="medium">Medium</option>
          <option value="low">Low</option>
        </select>
        <button className="btn-primary" type="submit">Add</button>
      </form>

      {items.length === 0 ? (
        <div className="empty">Nothing here. Add the first item above.</div>
      ) : (
        <ul className="list">
          {items.map((item) => (
            <li key={item.id} className={'item' + (item.done ? ' is-done' : '')}>
              <input type="checkbox" checked={item.done} onChange={() => toggle(item.id)} />
              <span className="item-title">{item.title}</span>
              <span className={'priority priority-' + item.priority}>{item.priority}</span>
              <button className="icon-btn" aria-label="Delete" onClick={() => remove(item.id)}>
                ×
              </button>
            </li>
          ))}
        </ul>
      )}

      {doneCount > 0 && (
        <div className="footer">
          <button className="link-btn" onClick={clearCompleted}>Clear completed</button>
        </div>
      )}
    </div>
  );
}
