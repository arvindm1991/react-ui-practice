@@@ shared src/utils/format.ts
// pluralize(1, 'like') -> "1 like", pluralize(1240, 'like') -> "1,240 likes"
export function pluralize(n: number, word: string): string {
  let label = word + 's';
  if (n === 1) {
    label = word;
  }
  return n.toLocaleString('en-US') + ' ' + label;
}

import React, { useState } from 'react';
import { HeartIcon, CommentIcon, ShareIcon } from './components/Icons';
import { pluralize } from './utils/format';

type Post = {
  id: string;
  author: string;
  initials: string;
  location: string;
  caption: string;
  likes: number; // likes from OTHER people (the server count before you)
  comments: number;
  postedAgo: string;
};

const POST: Post = {
  id: 'p1',
  author: 'mira.lens',
  initials: 'ML',
  location: 'Big Sur, California',
  caption: 'Fog rolled in ten minutes after this. Worth the 5am alarm.',
  likes: 241,
  comments: 18,
  postedAgo: '2 hours ago',
};

function TopBar() {
  return (
    <header className="topbar">
      <span className="brand">lumen</span>
    </header>
  );
}

function PostHeader({ post }: { post: Post }) {
  return (
    <div className="post-head">
      <div className="avatar">{post.initials}</div>
      <div>
        <p className="author">{post.author}</p>
        <p className="location">{post.location}</p>
      </div>
    </div>
  );
}

// CHANGE: the photo reports a double-click up. It doesn't know about "liked".
function PostPhoto({ onDoubleClick }: { onDoubleClick: () => void }) {
  return (
    <div className="photo" role="img" aria-label="Coastline at sunrise" onDoubleClick={() => { onDoubleClick(); }}>
      <div className="sun" />
    </div>
  );
}

// CORE: LikeButton gets liked + onToggle as props
type LikeButtonProps = {
  liked: boolean;
  onToggle: () => void;
};

// The button is "dumb": it shows liked and reports clicks. PostCard owns the state.
function LikeButton({ liked, onToggle }: LikeButtonProps) {
  // CORE: class decided above the return (NEXT: the Like / Unlike label)
  let cls = 'icon-btn like-btn';
  let label = 'Like';
  if (liked) {
    cls = 'icon-btn like-btn is-liked';
    label = 'Unlike';
  }
  // NEXT: aria-label + aria-pressed on the button. CORE: filled={liked}
  return (
    <button type="button" className={cls} aria-label={label} aria-pressed={liked} onClick={() => { onToggle(); }}>
      <HeartIcon filled={liked} />
    </button>
  );
}

function PostCard({ post }: { post: Post }) {
  // CORE: liked lives here because the button, the count and the photo all need it.
  const [liked, setLiked] = useState(false);

  // CORE: toggle with the updater form
  function handleToggle() {
    setLiked((prev) => {
      return !prev;
    });
  }

  // CHANGE: double-click only ever likes. It never unlikes.
  function handlePhotoDoubleClick() {
    setLiked(true);
  }

  // CORE: derived, never stored in state
  let likeCount = post.likes;
  if (liked) {
    likeCount = post.likes + 1;
  }

  return (
    <article className="card">
      <PostHeader post={post} />
      <PostPhoto onDoubleClick={handlePhotoDoubleClick} />
      <div className="actions">
        <LikeButton liked={liked} onToggle={handleToggle} />
        <button type="button" className="icon-btn" aria-label="Comment"><CommentIcon /></button>
        <button type="button" className="icon-btn" aria-label="Share"><ShareIcon /></button>
      </div>
      <div className="post-body">
        <p className="likes">{pluralize(likeCount, 'like')}</p>
        <p className="caption"><b>{post.author}</b> {post.caption}</p>
        <p className="meta">View all {post.comments} comments · {post.postedAgo}</p>
      </div>
    </article>
  );
}

export default function App() {
  return (
    <div className="app">
      <TopBar />
      <main className="feed">
        <PostCard post={POST} />
      </main>
    </div>
  );
}
