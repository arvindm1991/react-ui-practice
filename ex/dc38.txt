// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /repo/tree -> FileNode[] (nested; folders have children)
export async function getTree(): Promise<FileNode[]> {
  await randomDelay(400, 900);
  return JSON.parse(JSON.stringify(TREE));
}

// GET /repo/files/:id -> FileContent
export async function getFile(id: string): Promise<FileContent> {
  await randomDelay(300, 800);
  const content = CONTENTS[id] || '';
  return { id, content, size: content.length };
}

// fileIcon("Button.tsx") -> "TSX", fileIcon("README.md") -> "MD" (short badge text by extension)
export function fileIcon(name: string): string {
  const ext = name.split('.').pop() || '';
  const map: Record<string, string> = { tsx: 'TSX', ts: 'TS', json: '{}', md: 'MD', svg: 'SVG', txt: 'TXT' };
  return map[ext] || 'FILE';
}

// formatBytes(412) -> "412 B", formatBytes(2048) -> "2.0 KB"
export function formatBytes(n: number): string {
  if (n < 1024) {
    return n + ' B';
  }
  return (n / 1024).toFixed(1) + ' KB';
}

type TreeRowProps = {
  node: FileNode;
  depth: number;          // 0 for the roots; each level indents 14px
  isOpen: boolean;        // folders: rotates the chevron
  isSelected: boolean;    // tinted row + purple left edge
  onClick: () => void;
  extra?: React.ReactNode; // anything shown at the right end of the row
};

// One row of the explorer (just the row, not its children).
export function TreeRow({ node, depth, isOpen, isSelected, onClick, extra }: TreeRowProps) {
  let cls = 'tree-row';
  if (isSelected) {
    cls = 'tree-row is-selected';
  }
  let icon = <span className="file-icon">{fileIcon(node.name)}</span>;
  if (node.type === 'folder') {
    let chevCls = 'folder-icon';
    if (isOpen) {
      chevCls = 'folder-icon is-open';
    }
    icon = <span className={chevCls}>▸</span>;
  }
  return (
    <button type="button" className={cls} style={{ paddingLeft: 10 + depth * 14 }} onClick={() => { onClick(); }}>
      {icon}
      <span className="tree-name">{node.name}</span>
      {extra}
    </button>
  );
}

// CORE: what gets selected: which file (NEXT: and the names from the root down to it, for the breadcrumb).
type Selection = { id: string; path: string[] };

// CHANGE: files anywhere below this folder (recursive).
function countFiles(node: FileNode): number {
  if (node.type === 'file') {
    return 1;
  }
  let total = 0;
  (node.children || []).forEach((child) => {
    total = total + countFiles(child);
  });
  return total;
}

function TitleBar() {
  return (
    <header className="titlebar">
      <span className="dots"><i /><i /><i /></span>
      <span className="title">codebox-web</span>
      <span className="branch">⎇ main</span>
    </header>
  );
}

// CORE: one row, and (if it's an open folder) itself again for each child.
type TreeNodeProps = {
  node: FileNode;
  depth: number;
  parentPath: string[];
  expanded: Set<string>;
  selectedId: string | null;
  onToggle: (id: string) => void;
  onSelect: (selection: Selection) => void;
};

function TreeNode({ node, depth, parentPath, expanded, selectedId, onToggle, onSelect }: TreeNodeProps) {
  const path = [...parentPath, node.name];
  const isFolder = node.type === 'folder';
  const isOpen = expanded.has(node.id);

  function handleClick() {
    if (isFolder) {
      onToggle(node.id);
    } else {
      onSelect({ id: node.id, path });
    }
  }

  // CHANGE: file count pill for folders
  let count = null;
  if (isFolder) {
    count = <span className="file-count">{countFiles(node)}</span>;
  }

  // CORE: recursion
  let children = null;
  if (isFolder && isOpen) {
    children = (
      <ul className="tree" role="group">
        {(node.children || []).map((child) => {
          return (
            <TreeNode
              key={child.id}
              node={child}
              depth={depth + 1}
              parentPath={path}
              expanded={expanded}
              selectedId={selectedId}
              onToggle={onToggle}
              onSelect={onSelect}
            />
          );
        })}
      </ul>
    );
  }

  // STRETCH: ARIA
  let ariaExpanded: boolean | undefined = undefined;
  if (isFolder) {
    ariaExpanded = isOpen;
  }

  // CORE: TreeRow does the indent (depth * 14px) and the selected style
  return (
    <li role="treeitem" aria-expanded={ariaExpanded}>
      <TreeRow node={node} depth={depth} isOpen={isOpen} isSelected={node.id === selectedId} onClick={handleClick} extra={count} />
      {children}
    </li>
  );
}

// CORE: the explorer reports clicks up; App owns the selection
type ExplorerProps = {
  selectedId: string | null;
  onSelect: (selection: Selection) => void;
};

function Explorer({ selectedId, onSelect }: ExplorerProps) {
  const [tree, setTree] = useState<FileNode[]>([]);
  const [loading, setLoading] = useState(true);
  // CORE: one Set for the whole tree (not state per folder), so NEXT "Collapse all" can clear it.
  const [expanded, setExpanded] = useState<Set<string>>(new Set(['src']));

  useEffect(() => {
    let cancelled = false;
    getTree().then((data) => {
      if (cancelled) {
        return;
      }
      setTree(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // CORE: toggle a folder
  function handleToggle(id: string) {
    setExpanded((prev) => {
      const next = new Set(prev); // copy: never mutate state
      if (next.has(id)) {
        next.delete(id);
      } else {
        next.add(id);
      }
      return next;
    });
  }

  // NEXT: collapse all
  function handleCollapseAll() {
    setExpanded(new Set());
  }

  // CORE: the roots
  let body;
  if (loading) {
    body = <p className="side-msg">Loading files…</p>;
  } else {
    body = (
      <ul className="tree" role="tree">
        {tree.map((node) => {
          return (
            <TreeNode
              key={node.id}
              node={node}
              depth={0}
              parentPath={[]}
              expanded={expanded}
              selectedId={selectedId}
              onToggle={handleToggle}
              onSelect={onSelect}
            />
          );
        })}
      </ul>
    );
  }

  // NEXT: Collapse all button
  return (
    <aside className="explorer">
      <div className="explorer-head">
        <span>Explorer</span>
        <button type="button" className="head-btn" disabled={expanded.size === 0} onClick={() => { handleCollapseAll(); }}>Collapse all</button>
      </div>
      {body}
    </aside>
  );
}

  if (!fileId) {
    return (

        </div>
        <div className="preview-empty">
          <p className="empty-mark">{'</>'}</p>
          <p>Select a file to preview it</p>
        </div>
      </section>
    );
  }

  // NEXT: breadcrumb, last part bold
  const crumbs = ['codebox-web', ...path];
  let body = <p className="preview-loading">Loading…</p>;
  let size = null;
  if (!loading && file) {
    body = <CodeView code={file.content} />;
    size = <span className="file-size">{formatBytes(file.size)}</span>;
  }

  return (
    <section className="preview">
      <div className="preview-head">
        <nav className="crumbs" aria-label="Breadcrumb">
          {crumbs.map((name, i) => {
            let cls = 'crumb';
            let sep = null;
            if (i === crumbs.length - 1) {
              cls = 'crumb is-current';
            }
            if (i > 0) {
              sep = <span className="crumb-sep">›</span>;
            }
            return (
              <span key={i} className="crumb-wrap">
                {sep}
                <span className={cls}>{name}</span>
              </span>
            );
          })}
        </nav>
        {size}
      </div>
      {body}
    </section>
  );
}
