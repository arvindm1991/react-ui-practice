// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /accounts/checking/transactions -> Transaction[]
export async function getTransactions(): Promise<Transaction[]> {
  await randomDelay(300, 1000);
  return TRANSACTIONS.map((t) => ({ ...t }));
}

// formatMoney(-4500) -> "−$45.00"   formatMoney(1284000) -> "+$12,840.00"
export function formatMoney(cents: number): string {
  const abs = Math.abs(cents) / 100;
  const text = abs.toLocaleString('en-US', { style: 'currency', currency: 'USD' });
  if (cents < 0) {
    return '−' + text;
  }
  return '+' + text;
}

// formatDate("2026-09-14") -> "Sep 14, 2026"
export function formatDate(iso: string): string {
  const d = new Date(iso + 'T00:00:00');
  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
}

type SortKey = 'date' | 'description' | 'category' | 'amountCents';
type SortDir = 'asc' | 'desc';

type SortHeaderProps = {
  label: string;
  column: SortKey;
  sortKey: SortKey | null;
  sortDir: SortDir;
  onSort: (column: SortKey) => void;
  numeric?: boolean;
};

function SortHeader({ label, column, sortKey, sortDir, onSort, numeric }: SortHeaderProps) {
  const active = sortKey === column;
  let arrow = '↕';
  let ariaSort: 'ascending' | 'descending' | 'none' = 'none';
  let btnCls = 'sort-btn';
  if (active && sortDir === 'asc') {
    arrow = '▲';
    ariaSort = 'ascending';
    btnCls = 'sort-btn is-active';
  }
  if (active && sortDir === 'desc') {
    arrow = '▼';
    ariaSort = 'descending';
    btnCls = 'sort-btn is-active';
  }
  let thCls = '';
  if (numeric) {
    thCls = 'num';
  }
  return (
    <th className={thCls} aria-sort={ariaSort}>
      <button type="button" className={btnCls} onClick={() => { onSort(column); }}>
        {label} <span className="arrow">{arrow}</span>
      </button>
    </th>
  );
}

// Returns a NEW sorted array. .sort() sorts in place, so copy first.
function sortTransactions(list: Transaction[], key: SortKey, dir: SortDir): Transaction[] {
  const sorted = [...list].sort((a, b) => {
    if (key === 'amountCents') {
      return a.amountCents - b.amountCents;
    }
    if (key === 'date') {
      return new Date(a.date).getTime() - new Date(b.date).getTime();
    }
    return a[key].localeCompare(b[key]);
  });
  if (dir === 'desc') {
    sorted.reverse(); // fine: `sorted` is already our own copy
  }
  return sorted;
}

function TransactionsTable() {
  const [transactions, setTransactions] = useState<Transaction[]>([]);
  const [loading, setLoading] = useState(true);
  const [sortKey, setSortKey] = useState<SortKey | null>(null);
  const [sortDir, setSortDir] = useState<SortDir>('asc');

  useEffect(() => {
    let cancelled = false;
    getTransactions().then((data) => {
      if (cancelled) {
        return;
      }
      setTransactions(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // New column: asc. Same column: asc -> desc -> off (back to the bank's order).
  function handleSort(column: SortKey) {
    if (column !== sortKey) {
      setSortKey(column);
      setSortDir('asc');
      return;
    }
    if (sortDir === 'asc') {
      setSortDir('desc');
      return;
    }
    setSortKey(null);
    setSortDir('asc');
  }

  // derived: the original list is never touched, so "off" can show it again
  let visible = transactions;
  if (sortKey) {
    visible = sortTransactions(transactions, sortKey, sortDir);
  }

  let body;
  if (loading) {
    body = <tr><td colSpan={4} className="table-msg">Loading transactions…</td></tr>;
  } else {
    body = visible.map((tx) => {
      return <TransactionRow key={tx.id} tx={tx} />;
    });
  }

  const headerProps = { sortKey: sortKey, sortDir: sortDir, onSort: handleSort };

  return (
    <table className="table">
      <thead>
        <tr>
          <SortHeader label="Date" column="date" {...headerProps} />
          <SortHeader label="Description" column="description" {...headerProps} />
          <SortHeader label="Category" column="category" {...headerProps} />
          <SortHeader label="Amount" column="amountCents" numeric {...headerProps} />
        </tr>
      </thead>
      <tbody>{body}</tbody>
    </table>
  );
}

export default function App() {
  return (
    <div className="layout">
      <Sidebar />
      <main className="main">
        <h1>Transactions</h1>
        <BalanceStrip />
        <div className="table-card">
          <TransactionsTable />
        </div>
      </main>
    </div>
  );
}
