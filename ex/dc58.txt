// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

  await randomDelay(400, 1000);

// Add the value if it's missing, remove it if it's there. Returns a new array.
// toggleInList(['a'], 'b') -> ['a', 'b'];  toggleInList(['a', 'b'], 'a') -> ['b']
export function toggleInList(list: string[], value: string): string[] {
  if (list.includes(value)) {
    return list.filter((v) => v !== value);
  }
  return [...list, value];
}

// Sorted unique values: uniqueSorted(['b', 'a', 'b']) -> ['a', 'b']
export function uniqueSorted(values: string[]): string[] {
  return Array.from(new Set(values)).sort((a, b) => a.localeCompare(b));
}

import { getProjects, STATUSES } from './api';

import FilterPopover from './components/FilterPopover';

import { uniqueSorted } from './utils/list';

// CORE: empty group = "any"
const EMPTY: Filters = { statuses: [], owners: [] };

// CORE: OR within a group, AND across groups (CHANGE reuses it for the preview count)
function matchesFilters(p: Project, f: Filters): boolean {
  const statusOk = f.statuses.length === 0 || f.statuses.includes(p.status);
  const ownerOk = f.owners.length === 0 || f.owners.includes(p.owner);
  return statusOk && ownerOk;
}

function ProjectList({ projects }: { projects: Project[] }) {
  return (
    <ul className="project-list">
      {projects.map((p) => {
        return (
          <li key={p.id} className="project">
            <div className="project-main">
              <span className="project-name">{p.name}</span>
              <span className="project-client">{p.client} · {p.area.toLocaleString('en-US')} m²</span>
            </div>
            <StatusBadge status={p.status} />
            <span className="project-owner"><Initials name={p.owner} />{p.owner}</span>
          </li>
        );
      })}
    </ul>
  );
}

// NEXT: removable chips
type ChipsProps = {
  filters: Filters;
  onRemove: (kind: 'statuses' | 'owners', value: string) => void;
  onClearAll: () => void;
};

function FilterChips({ filters, onRemove, onClearAll }: ChipsProps) {
  const chips: { kind: 'statuses' | 'owners'; label: string; value: string }[] = [];
  filters.statuses.forEach((s) => {
    chips.push({ kind: 'statuses', label: 'Status', value: s });
  });
  filters.owners.forEach((o) => {
    chips.push({ kind: 'owners', label: 'Owner', value: o });
  });
  if (chips.length === 0) {
    return null;
  }
  return (
    <div className="chips">
      {chips.map((c) => {
        return (
          <span key={c.kind + ':' + c.value} className="chip">
            <span className="chip-label">{c.label}:</span> {c.value}
            <button type="button" className="chip-x" aria-label={'Remove ' + c.label + ' ' + c.value} onClick={() => { onRemove(c.kind, c.value); }}>✕</button>
          </span>
        );
      })}
      <button type="button" className="clear-all" onClick={() => { onClearAll(); }}>Clear all</button>
    </div>
  );
}

export default function App() {
  const [projects, setProjects] = useState<Project[]>([]);
  const [loading, setLoading] = useState(true);
  // CORE: applied filters live here (list, chips and badge read them); the draft lives in the popover
  const [applied, setApplied] = useState<Filters>(EMPTY);
  const [open, setOpen] = useState(false);
  // NEXT: focus goes back to the button
  const filterBtnRef = useRef<HTMLButtonElement>(null);

  useEffect(() => {
    let cancelled = false;
    getProjects().then((data) => {
      if (cancelled) {
        return;
      }
      setProjects(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // CORE: close (NEXT: useCallback, because the popover's document listeners depend on it,
  // and return focus to the Filter button)
  const closePopover = useCallback(() => {
    setOpen(false);
    if (filterBtnRef.current) {
      filterBtnRef.current.focus();
    }
  }, []);

  // CORE
  function handleApply(next: Filters) {
    setApplied(next);
    closePopover();
  }

  // NEXT: remove one chip
  function handleRemove(kind: 'statuses' | 'owners', value: string) {
    setApplied((prev) => {
      return {
        ...prev,
        [kind]: prev[kind].filter((v) => {
          return v !== value;
        }),
      };
    });
  }

  // CHANGE: how many projects a draft would show
  function countMatches(f: Filters): number {
    return projects.filter((p) => {
      return matchesFilters(p, f);
    }).length;
  }

  // CORE: derived, never stored
  // NEXT: owner options from the data
  const ownerOptions = uniqueSorted(projects.map((p) => {
    return p.owner;
  }));
  // CORE
  const visible = projects.filter((p) => {
    return matchesFilters(p, applied);
  });
  // NEXT: the badge and "5 of 18" read this
  const activeCount = applied.statuses.length + applied.owners.length;

  // NEXT: count badge
  let badge = null;
  if (activeCount > 0) {
    badge = <span className="count-badge">{activeCount}</span>;
  }

  let btnCls = 'filter-btn';
  if (open || activeCount > 0) {
    btnCls = 'filter-btn is-active';
  }

  // CORE: mounted only while open, so every open starts a fresh draft
  let popover = null;
  if (open) {
    popover = (
      <FilterPopover
        applied={applied}
        statusOptions={STATUSES}
        ownerOptions={ownerOptions}
        anchorRef={filterBtnRef}
        countMatches={countMatches}
        onApply={handleApply}
        onCancel={closePopover}
      />
    );
  }

  let body;
  if (loading) {
    body = <p className="muted-msg">Loading projects…</p>;
  } else if (visible.length === 0) {
    // NEXT: empty state
    body = (
      <div className="empty">
        <p>No projects match these filters.</p>
        <button type="button" className="clear-all" onClick={() => { setApplied(EMPTY); }}>Clear all filters</button>
      </div>
    );
  } else {
    // CORE
    body = <ProjectList projects={visible} />;
  }

  // NEXT: "5 of 18 projects"
  let countText = projects.length + ' projects';
  if (activeCount > 0) {
    countText = visible.length + ' of ' + projects.length + ' projects';
  }

  // CORE: anchor wrapper + toggle button (NEXT: ref, aria-haspopup, aria-expanded)
  return (
    <div className="app">
      <header className="header">
        <span className="studio">Studio Varga</span>
        <h1>Projects</h1>
      </header>
      <div className="toolbar">
        <div className="filter-anchor">
          <button
            type="button"
            className={btnCls}
            ref={filterBtnRef}
            aria-haspopup="dialog"
            aria-expanded={open}
            onClick={() => { setOpen((o) => { return !o; }); }}
          >
            <span aria-hidden="true">☰</span> Filter
            {badge}
          </button>
          {popover}
        </div>
        <span className="result-count">{countText}</span>
      </div>
      {/* NEXT */}
      <FilterChips filters={applied} onRemove={handleRemove} onClearAll={() => { setApplied(EMPTY); }} />
      {body}
    </div>
  );
}

import React, { useEffect, useRef, useState } from 'react';
import { toggleInList } from '../utils/list';
import CheckList from './CheckList';

// CORE
export type Filters = { statuses: string[]; owners: string[] };

// STRETCH: Select all / Clear per section
type SectionHeadProps = {
  title: string;
  onSelectAll: () => void;
  onClear: () => void;
};

function SectionHead({ title, onSelectAll, onClear }: SectionHeadProps) {
  return (
    <div className="pop-section-head">
      <h3>{title}</h3>
      <button type="button" className="pop-link" onClick={() => { onSelectAll(); }}>Select all</button>
      <button type="button" className="pop-link" onClick={() => { onClear(); }}>Clear</button>
    </div>
  );
}

// CORE: props (NEXT: ownerOptions, anchorRef; CHANGE: countMatches)
type FilterPopoverProps = {
  applied: Filters;
  statusOptions: string[];
  ownerOptions: string[];
  anchorRef: React.RefObject<HTMLButtonElement>;
  countMatches: (filters: Filters) => number;
  onApply: (filters: Filters) => void;
  onCancel: () => void;
};

// Mounted only while open, so the draft starts as a fresh copy of `applied`
// every time, and Cancel is just "unmount without applying".
export default function FilterPopover({ applied, statusOptions, ownerOptions, anchorRef, countMatches, onApply, onCancel }: FilterPopoverProps) {
  // CORE: the draft is a copy; nothing outside changes until Apply
  const [draft, setDraft] = useState<Filters>(applied);
  // STRETCH
  const [ownerQuery, setOwnerQuery] = useState('');
  // NEXT
  const popRef = useRef<HTMLDivElement>(null);

  // NEXT: Escape or a click outside (but not on the Filter button itself) cancels.
  useEffect(() => {
    function handleKey(e: KeyboardEvent) {
      if (e.key === 'Escape') {
        onCancel();
      }
    }
    function handleMouseDown(e: MouseEvent) {
      const target = e.target as Node;
      if (popRef.current && popRef.current.contains(target)) {
        return;
      }
      if (anchorRef.current && anchorRef.current.contains(target)) {
        return;
      }
      onCancel();
    }
    document.addEventListener('keydown', handleKey);
    document.addEventListener('mousedown', handleMouseDown);
    return () => {
      document.removeEventListener('keydown', handleKey);
      document.removeEventListener('mousedown', handleMouseDown);
    };
  }, [onCancel, anchorRef]);

  // NEXT: move focus into the popover when it opens.
  useEffect(() => {
    if (popRef.current) {
      const first = popRef.current.querySelector<HTMLElement>('input');
      if (first) {
        first.focus();
      }
    }
  }, []);

  // STRETCH: derived: owners matching the search box
  const q = ownerQuery.trim().toLowerCase();
  const visibleOwners = ownerOptions.filter((o) => {
    return o.toLowerCase().includes(q);
  });

  // CORE: toggles only touch the draft
  function toggleStatus(value: string) {
    setDraft((prev) => {
      return { ...prev, statuses: toggleInList(prev.statuses, value) };
    });
  }

  // NEXT: the Owner section
  function toggleOwner(value: string) {
    setDraft((prev) => {
      return { ...prev, owners: toggleInList(prev.owners, value) };
    });
  }

  // STRETCH: "Select all" / "Clear" act on what you can see.
  function selectVisibleOwners() {
    setDraft((prev) => {
      const extra = visibleOwners.filter((o) => {
        return !prev.owners.includes(o);
      });
      return { ...prev, owners: [...prev.owners, ...extra] };
    });
  }

  function clearVisibleOwners() {
    setDraft((prev) => {
      return {
        ...prev,
        owners: prev.owners.filter((o) => {
          return !visibleOwners.includes(o);
        }),
      };
    });
  }

  // NEXT (STRETCH: the no-match message)
  let ownerList = <CheckList options={visibleOwners} selected={draft.owners} onToggle={toggleOwner} />;
  if (visibleOwners.length === 0) {
    ownerList = <p className="pop-empty">No one matches “{ownerQuery}”</p>;
  }

  // CHANGE: live preview on the Apply button (derived from the draft on every render)
  const matchCount = countMatches(draft);
  let applyLabel = 'Show ' + matchCount + ' projects';
  if (matchCount === 1) {
    applyLabel = 'Show 1 project';
  }

  // CORE: Status section + Cancel / Apply
  return (
    <div className="popover" role="dialog" aria-label="Filter projects" ref={popRef}>
      <section className="pop-section">
        {/* STRETCH: Select all / Clear (Core: a plain <h3>Status</h3>) */}
        <SectionHead
          title="Status"
          onSelectAll={() => { setDraft((prev) => { return { ...prev, statuses: [...statusOptions] }; }); }}
          onClear={() => { setDraft((prev) => { return { ...prev, statuses: [] }; }); }}
        />
        {/* CORE */}
        <CheckList options={statusOptions} selected={draft.statuses} onToggle={toggleStatus} />
      </section>
      {/* NEXT: Owner section (STRETCH: SectionHead + search) */}
      <section className="pop-section">
        <SectionHead title="Owner" onSelectAll={selectVisibleOwners} onClear={clearVisibleOwners} />
        <input
          className="pop-search"
          type="search"
          placeholder="Search people"
          aria-label="Search owners"
          value={ownerQuery}
          onChange={(e) => { setOwnerQuery(e.target.value); }}
        />
        {ownerList}
      </section>
      {/* CORE */}
      <footer className="pop-footer">
        <button type="button" className="btn-ghost" onClick={() => { onCancel(); }}>Cancel</button>
        <button type="button" className="btn-dark" onClick={() => { onApply(draft); }}>{applyLabel}</button>
      </footer>
    </div>
  );
}
