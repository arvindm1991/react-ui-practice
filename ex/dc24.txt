import React from 'react';

type Option = { value: string; label: string };

type SegmentedControlProps = {
  label: string;           // accessible name for the group
  options: Option[];
  value: string;
  onChange: (value: string) => void;
};

// A row of buttons where exactly one is selected. Fully controlled.
export default function SegmentedControl({ label, options, value, onChange }: SegmentedControlProps) {
  return (
    <div className="segmented" role="group" aria-label={label}>
      {options.map((opt) => {
        const selected = opt.value === value;
        let cls = 'segment';
        if (selected) {
          cls = 'segment is-selected';
        }
        return (
          <button
            key={opt.value}
            type="button"
            className={cls}
            aria-pressed={selected}
            onClick={() => { onChange(opt.value); }}
          >
            {opt.label}
          </button>
        );
      })}
    </div>
  );
}

import React, { useEffect, useState } from 'react';
import SegmentedControl from './components/SegmentedControl';
import { KPIS, BARS, INVOICES } from './data';

const RANGES = [
  { value: '7d', label: '7d' },
  { value: '30d', label: '30d' },
  { value: '12m', label: '12m' },
];

type ThemeChoice = 'light' | 'dark' | 'system';
type Theme = 'light' | 'dark';

const THEME_KEY = 'northwind-theme';
const THEME_OPTIONS = [
  { value: 'light', label: 'Light' },
  { value: 'dark', label: 'Dark' },
  { value: 'system', label: 'System' },
];
const DARK_QUERY = '(prefers-color-scheme: dark)';

// localStorage can throw (private mode, blocked storage), so wrap every access.
function readStoredChoice(): ThemeChoice {
  try {
    const saved = localStorage.getItem(THEME_KEY);
    if (saved === 'light' || saved === 'dark' || saved === 'system') {
      return saved;
    }
  } catch (e) {
    // ignore: fall back to the default
  }
  return 'system';
}

function saveChoice(choice: ThemeChoice) {
  try {
    localStorage.setItem(THEME_KEY, choice);
  } catch (e) {
    // ignore: the theme still works for this visit
  }
}

function getSystemTheme(): Theme {
  if (window.matchMedia(DARK_QUERY).matches) {
    return 'dark';
  }
  return 'light';
}

function KpiCard({ label, value, delta }: { label: string; value: string; delta: string }) {
  return (
    <div className="card kpi">
      <span className="kpi-label">{label}</span>
      <span className="kpi-value">{value}</span>
      <span className="kpi-delta">{delta}</span>
    </div>
  );
}

function RevenueChart() {
  return (
    <div className="card chart-card">
      <h2>Revenue</h2>
      <div className="bars">
        {BARS.map((h, i) => {
          return <div key={i} className="bar" style={{ height: h + '%' }} />;
        })}
      </div>
    </div>
  );
}

function InvoiceTable() {
  return (
    <div className="card">
      <h2>Recent invoices</h2>
      <table className="table">
        <tbody>
          {INVOICES.map((inv) => {
            return (
              <tr key={inv.id}>
                <td className="mono">{inv.id}</td>
                <td>{inv.customer}</td>
                <td className="num">{inv.amount}</td>
                <td><span className={'status status-' + inv.status.toLowerCase()}>{inv.status}</span></td>
              </tr>
            );
          })}
        </tbody>
      </table>
    </div>
  );
}

export default function App() {
  const [range, setRange] = useState('30d');
  // Lazy initial state: read storage once, not on every render.
  const [choice, setChoice] = useState<ThemeChoice>(() => {
    return readStoredChoice();
  });
  const [systemTheme, setSystemTheme] = useState<Theme>(() => {
    return getSystemTheme();
  });

  // Follow the OS setting live while "System" could be in use.
  useEffect(() => {
    const mq = window.matchMedia(DARK_QUERY);
    function handleChange() {
      setSystemTheme(getSystemTheme());
    }
    mq.addEventListener('change', handleChange);
    return () => {
      mq.removeEventListener('change', handleChange);
    };
  }, []);

  function handleThemeChange(value: string) {
    const next = value as ThemeChoice;
    setChoice(next);
    saveChoice(next);
  }

  // derived: the theme actually applied. Never stored in state.
  let theme: Theme = systemTheme;
  if (choice !== 'system') {
    theme = choice;
  }

  let systemNote = null;
  if (choice === 'system') {
    systemNote = <span className="theme-note">Using {theme}</span>;
  }

  return (
    <div className="app" data-theme={theme}>
      <header className="topbar">
        <span className="logo">◐ Northwind Metrics</span>
        <div className="topbar-right">
          <SegmentedControl label="Date range" options={RANGES} value={range} onChange={setRange} />
          <div className="theme-picker">
            {systemNote}
            <SegmentedControl label="Theme" options={THEME_OPTIONS} value={choice} onChange={handleThemeChange} />
          </div>
        </div>
      </header>
      <main className="content">
        <div className="kpis">
          {KPIS.map((k) => {
            return <KpiCard key={k.label} label={k.label} value={k.value} delta={k.delta} />;
          })}
        </div>
        <RevenueChart />
        <InvoiceTable />
      </main>
    </div>
  );
}
