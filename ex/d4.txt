import React from 'react';

export default function MessageList({ messages, selectedId, onSelect }) {
  return (
    <ul className="msg-list">
      {messages.map((m) => {
        let className = 'msg';
        if (m.id === selectedId) className += ' is-selected';
        if (!m.read) className += ' is-unread';
        return (
          <li key={m.id}>
            <button className={className} onClick={() => onSelect && onSelect(m.id)}>
              <span className="msg-from">{m.from}</span>
              <span className="msg-time">{m.time}</span>
              <span className="msg-subject">{m.subject}</span>
            </button>
          </li>
        );
      })}
    </ul>
  );
}

import React from 'react';

export default function MessageDetail({ message, onMarkUnread }) {
  if (!message) {
    return <section className="detail detail-empty">Select a message to read it.</section>;
  }
  return (
    <section className="detail">
      <header className="detail-header">
        <div>
          <h2 className="detail-subject">{message.subject}</h2>
          <p className="detail-from">{message.from} · {message.time}</p>
        </div>
        <button className="btn" onClick={() => onMarkUnread && onMarkUnread(message.id)}>
          Mark as unread
        </button>
      </header>
      <p className="detail-body">{message.body}</p>
    </section>
  );
}

@@@ solution src/App.jsx
import React, { useState } from 'react';
import { initialMessages } from './data';
import MessageList from './components/MessageList';
import MessageDetail from './components/MessageDetail';

export default function App() {
  // State lives in the closest common parent of the list and the detail.
  const [messages, setMessages] = useState(initialMessages);
  const [selectedId, setSelectedId] = useState(null);

  // Derived values
  const selected = messages.find((m) => m.id === selectedId) || null;
  const unreadCount = messages.filter((m) => !m.read).length;

  // Immutable update helper: never mutate the object in state
  function setRead(id, read) {
    setMessages((prev) => prev.map((m) => (m.id === id ? { ...m, read } : m)));
  }

  function handleSelect(id) {
    setSelectedId(id);
    setRead(id, true);
  }

  function handleMarkUnread(id) {
    setRead(id, false);
    setSelectedId(null); // like most mail apps: close it after marking unread
  }

  return (
    <div className="app">
      <header className="app-header">
        <h1>Inbox</h1>
        {unreadCount > 0 && <span className="unread-count">{unreadCount} unread</span>}
      </header>
      <div className="inbox">
        <MessageList messages={messages} selectedId={selectedId} onSelect={handleSelect} />
        <MessageDetail message={selected} onMarkUnread={handleMarkUnread} />
      </div>
    </div>
  );
}
