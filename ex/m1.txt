@@@ shared src/utils/dates.js
// Shared date helpers used across the app.
export const GROUP_ORDER = ['Today', 'Yesterday', 'Previous 7 days', 'Older'];

const DAY_MS = 24 * 60 * 60 * 1000;

function startOfDay(value) {
  const d = new Date(value);
  d.setHours(0, 0, 0, 0);
  return d;
}

// Returns one of GROUP_ORDER for a date (ISO string or Date).
export function getDateGroup(value, now = new Date()) {
  const days = Math.round((startOfDay(now) - startOfDay(value)) / DAY_MS);
  if (days <= 0) return 'Today';
  if (days === 1) return 'Yesterday';
  if (days <= 7) return 'Previous 7 days';
  return 'Older';
}

@@@ solution src/components/Sidebar.jsx
import React, { useEffect, useState } from 'react';
import { getConversations } from '../api';
import { getDateGroup, GROUP_ORDER } from '../utils/dates';
import { PlusIcon, SearchIcon } from './Icon';
import ConversationItem from './ConversationItem';

// Includes the mid-session change (Starred section).
export default function Sidebar({ activeId, onSelect }) {
  const [conversations, setConversations] = useState([]);
  const [loading, setLoading] = useState(true);
  const [query, setQuery] = useState('');

  useEffect(() => {
    let cancelled = false;
    getConversations().then((data) => {
      if (cancelled) return;
      setConversations(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  function toggleStar(id) {
    setConversations((prev) =>
      prev.map((c) => (c.id === id ? { ...c, starred: !c.starred } : c))
    );
  }

  function startNewChat() {
    const chat = { id: 'new-' + Date.now(), title: 'New chat', updatedAt: new Date().toISOString(), starred: false };
    setConversations((prev) => [chat, ...prev]);
    setQuery('');
    onSelect(chat.id);
  }

  // ---- derived data ----
  const q = query.trim().toLowerCase();
  const sorted = conversations
    .filter((c) => c.title.toLowerCase().includes(q))
    .sort((a, b) => new Date(b.updatedAt) - new Date(a.updatedAt));

  const sections = [];
  const starred = sorted.filter((c) => c.starred);
  if (starred.length > 0) sections.push({ label: 'Starred', items: starred });
  GROUP_ORDER.forEach((label) => {
    const items = sorted.filter((c) => !c.starred && getDateGroup(c.updatedAt) === label);
    if (items.length > 0) sections.push({ label, items });
  });

  return (
    <aside className="sidebar">
      <div className="sidebar-top">
        <button className="new-chat" onClick={startNewChat}>
          <PlusIcon />
          New chat
        </button>
        <label className="search">
          <SearchIcon className="search-icon" />
          <input
            className="search-input"
            placeholder="Search chats"
            value={query}
            onChange={(e) => setQuery(e.target.value)}
          />
        </label>
      </div>

      <nav className="sidebar-list" aria-label="Conversations">
        {loading &&
          Array.from({ length: 6 }, (_, i) => <div key={i} className="skeleton-row" />)}

        {!loading && sections.length === 0 && (
          <p className="sidebar-empty">No chats match “{query}”.</p>
        )}

        {sections.map((section) => (
          <section key={section.label} className="group">
            <h3 className="group-label">{section.label}</h3>
            <ul className="group-items">
              {section.items.map((c) => (
                <ConversationItem
                  key={c.id}
                  conversation={c}
                  active={c.id === activeId}
                  onSelect={onSelect}
                  onToggleStar={toggleStar}
                />
              ))}
            </ul>
          </section>
        ))}
      </nav>
    </aside>
  );
}
@@@ solution src/components/ConversationItem.jsx
import React from 'react';
import { StarIcon } from './Icon';

// Two sibling buttons (never nest a button inside a button).
export default function ConversationItem({ conversation, active, onSelect, onToggleStar }) {
  const { id, title, starred } = conversation;
  let className = 'item';
  if (active) className += ' is-active';
  if (starred) className += ' is-starred';

  return (
    <li className={className}>
      <button className="item-main" onClick={() => onSelect(id)} title={title}>
        {title}
      </button>
      <button
        className="item-star"
        onClick={() => onToggleStar(id)}
        aria-label={starred ? 'Unstar' : 'Star'}
        aria-pressed={starred}
      >
        <StarIcon filled={starred} />
      </button>
    </li>
  );
}
