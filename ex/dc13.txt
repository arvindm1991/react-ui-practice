// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /me/courses?status=... -> Course[]
// When shouldFail is true the request rejects (the "Simulate failure" checkbox).
export async function getCourses(status: CourseStatus, shouldFail: boolean): Promise<Course[]> {
  await randomDelay(400, 1200);
  if (shouldFail) {
    throw new Error('503 Service Unavailable');
  }
  return COURSES.filter((c) => c.status === status).map((c) => ({ ...c }));
}

function SkeletonCard() {
  return (
    <div className="course-card skeleton-card" aria-hidden="true">
      <div className="sk sk-cover" />
      <div className="course-body">
        <div className="sk sk-line" />
        <div className="sk sk-line short" />
        <div className="sk sk-bar" />
      </div>
    </div>
  );
}

type CourseListProps = {
  status: CourseStatus;
  shouldFail: boolean;
};

function CourseList({ status, shouldFail }: CourseListProps) {
  const [courses, setCourses] = useState<Course[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  // Bumping this number re-runs the effect: that is the whole Retry button.
  const [attempt, setAttempt] = useState(0);

  useEffect(() => {
    let cancelled = false;
    setLoading(true);
    setError(null);
    getCourses(status, shouldFail)
      .then((data) => {
        if (cancelled) {
          return;
        }
        setCourses(data);
        setLoading(false);
      })
      .catch(() => {
        if (cancelled) {
          return;
        }
        setError('Couldn’t load your courses.');
        setLoading(false);
      });
    return () => {
      // A newer request (tab switch, retry) replaces this one: ignore its answer.
      cancelled = true;
    };
  }, [status, shouldFail, attempt]);

  function handleRetry() {
    setAttempt((prev) => {
      return prev + 1;
    });
  }

  if (loading) {
    return (
      <div className="course-grid" aria-busy="true">
        <SkeletonCard />
        <SkeletonCard />
        <SkeletonCard />
      </div>
    );
  }

  if (error) {
    return (
      <div className="state-panel is-error" role="alert">
        <span className="state-icon">⚠️</span>
        <h3>{error}</h3>
        <p>Check your connection and try again.</p>
        <button type="button" className="primary-btn" onClick={() => { handleRetry(); }}>Retry</button>
      </div>
    );
  }

  if (courses.length === 0) {
    return (
      <div className="state-panel">
        <span className="state-icon">📚</span>
        <h3>No courses here yet</h3>
        <p>Save courses you like and they will show up in this list.</p>
        <button type="button" className="primary-btn">Browse catalog</button>
      </div>
    );
  }

  return (
    <div className="course-grid">
      {courses.map((c) => {
        return <CourseCard key={c.id} course={c} />;
      })}
    </div>
  );
}

export default function App() {
  const [tab, setTab] = useState<CourseStatus>('in-progress');
  const [fail, setFail] = useState(false);

  return (
    <div className="app">
      <Header />
      <main className="main">
        <div className="title-row">
          <div>
            <h1>My learning</h1>
            <p className="sub">Pick up where you left off, Jordan.</p>
          </div>
          <DevToolbar fail={fail} onFailChange={(v) => { setFail(v); }} />
        </div>
        <Tabs active={tab} onChange={(id) => { setTab(id); }} />
        <CourseList status={tab} shouldFail={fail} />
      </main>
    </div>
  );
}
