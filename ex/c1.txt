import React, { useState } from 'react';
import { initialComments, currentUser } from './data';

// Read the Task tab first. Several requirements use things this kit never
// taught you. That's the point: name each one, then look it up.
export default function App() {
  const [comments, setComments] = useState(initialComments);
  const [draft, setDraft] = useState('');

  function handleSubmit(e) {
    e.preventDefault();
    if (!draft.trim()) return;
    setComments((prev) => [
      ...prev,
      { id: 'c' + Date.now(), author: currentUser.name, body: draft.trim(), createdAt: new Date().toISOString() },
    ]);
    setDraft('');
  }

  return (
    <div className="thread">
      <h1>Discussion</h1>
      <ul className="comments">
        {comments.map((c) => (
          <li key={c.id} className="comment">
            <div className="comment-head">
              <strong>{c.author}</strong>
              <span className="time">{c.createdAt}</span>
            </div>
            <p className="comment-body">{c.body}</p>
          </li>
        ))}
      </ul>
      <form className="composer" onSubmit={handleSubmit}>
        <textarea
          className="composer-input"
          placeholder="Add a comment…"
          value={draft}
          onChange={(e) => setDraft(e.target.value)}
        />
        <div className="composer-foot">
          <button className="btn-primary" type="submit">Comment</button>
        </div>
      </form>
    </div>
  );
}

// Lookup: "js relative time format mdn" -> Intl.RelativeTimeFormat
const rtf = new Intl.RelativeTimeFormat('en', { numeric: 'auto' });

const UNITS = [
  ['year', 60 * 60 * 24 * 365],
  ['month', 60 * 60 * 24 * 30],
  ['week', 60 * 60 * 24 * 7],
  ['day', 60 * 60 * 24],
  ['hour', 60 * 60],
  ['minute', 60],
];

// "2 minutes ago", "3 hours ago", "yesterday", "just now"
export function timeAgo(iso, now = Date.now()) {
  const seconds = Math.round((new Date(iso).getTime() - now) / 1000); // negative = past
  if (Math.abs(seconds) < 45) return 'just now';
  for (const [unit, size] of UNITS) {
    if (Math.abs(seconds) >= size || unit === 'minute') {
      return rtf.format(Math.round(seconds / size), unit);
    }
  }
  return '';
}

// Full date for the tooltip: "Oct 5, 2026, 2:14 PM"
export function fullDate(iso) {
  return new Date(iso).toLocaleString('en-US', { dateStyle: 'medium', timeStyle: 'short' });
}

import { useEffect } from 'react';

// Lookup: "react click outside to close dropdown" -> ref + document mousedown + contains()
export function useClickOutside(ref, onOutside, enabled = true) {
  useEffect(() => {
    if (!enabled) return undefined;
    function handleDown(e) {
      if (ref.current && !ref.current.contains(e.target)) onOutside();
    }
    function handleKey(e) {
      if (e.key === 'Escape') onOutside();
    }
    document.addEventListener('mousedown', handleDown);
    document.addEventListener('keydown', handleKey);
    return () => {
      document.removeEventListener('mousedown', handleDown);
      document.removeEventListener('keydown', handleKey);
    };
  }, [ref, onOutside, enabled]);
}

import React, { useEffect, useRef, useState } from 'react';
import { timeAgo, fullDate } from '../utils/time';
import { useClickOutside } from '../hooks/useClickOutside';

export default function Comment({ comment, isNew, onDelete }) {
  const [copied, setCopied] = useState(false);
  const [menuOpen, setMenuOpen] = useState(false);
  const actionsRef = useRef(null);
  const itemRef = useRef(null);

  useClickOutside(actionsRef, () => setMenuOpen(false), menuOpen);

  // Lookup: "scrollIntoView mdn" (block: 'nearest' avoids scrolling the whole page)
  useEffect(() => {
    if (isNew && itemRef.current) itemRef.current.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
  }, [isNew]);

  // Reset "Copied" after 2s; clear the timer if the comment unmounts first
  useEffect(() => {
    if (!copied) return undefined;
    const id = setTimeout(() => setCopied(false), 2000);
    return () => clearTimeout(id);
  }, [copied]);

  // Lookup: "copy text to clipboard javascript mdn" -> navigator.clipboard.writeText
  async function copyLink() {
    const url = `https://example.com/thread#${comment.id}`;
    try {
      await navigator.clipboard.writeText(url);
      setCopied(true);
    } catch (err) {
      console.warn('Clipboard blocked:', err.message);
    }
  }

  return (
    <li ref={itemRef} className={'comment' + (isNew ? ' is-new' : '')}>
      <div className="comment-head">
        <strong>{comment.author}</strong>
        <time className="time" dateTime={comment.createdAt} title={fullDate(comment.createdAt)}>
          {timeAgo(comment.createdAt)}
        </time>
        <div className="comment-actions" ref={actionsRef}>
          <button className={'ghost-btn' + (copied ? ' is-done' : '')} onClick={copyLink}>
            {copied ? 'Copied!' : 'Copy link'}
          </button>
          <button
            className="ghost-btn"
            aria-label="More actions"
            aria-haspopup="menu"
            aria-expanded={menuOpen}
            onClick={() => setMenuOpen((o) => !o)}
          >
            ⋯
          </button>
          {menuOpen && (
            <div className="menu" role="menu">
              <button role="menuitem" className="danger" onClick={() => onDelete(comment.id)}>
                Delete comment
              </button>
            </div>
          )}
        </div>
      </div>
      <p className="comment-body">{comment.body}</p>
    </li>
  );
}

import React, { useEffect, useLayoutEffect, useRef, useState } from 'react';
import { initialComments, currentUser } from './data';
import Comment from './components/Comment';

const MAX = 500;
const DRAFT_KEY = 'comment-draft';

// Lookup: "localStorage mdn". Wrap in try/catch: it can throw in private mode.
function readDraft() {
  try {
    return localStorage.getItem(DRAFT_KEY) || '';
  } catch (e) {
    return '';
  }
}

export default function App() {
  const [comments, setComments] = useState(initialComments);
  const [draft, setDraft] = useState(readDraft); // lazy initial state: runs once
  const [newId, setNewId] = useState(null);
  const [, setTick] = useState(0);
  const inputRef = useRef(null);

  // Persist the draft as you type
  useEffect(() => {
    try {
      localStorage.setItem(DRAFT_KEY, draft);
    } catch (e) {
      /* storage unavailable */
    }
  }, [draft]);

  // Re-render every 30s so "just now" becomes "1 minute ago"
  useEffect(() => {
    const id = setInterval(() => setTick((t) => t + 1), 30000);
    return () => clearInterval(id);
  }, []);

  // Lookup: "textarea auto grow height react" -> reset height, then set to scrollHeight
  useLayoutEffect(() => {
    const el = inputRef.current;
    if (!el) return;
    el.style.height = 'auto';
    el.style.height = Math.min(el.scrollHeight, 200) + 'px';
  }, [draft]);

  const length = draft.trim().length;
  const tooLong = draft.length > MAX;
  const canSubmit = length > 0 && !tooLong;

  function submit() {
    if (!canSubmit) return;
    const comment = {
      id: 'c' + Date.now(),
      author: currentUser.name,
      body: draft.trim(),
      createdAt: new Date().toISOString(),
    };
    setComments((prev) => [...prev, comment]);
    setNewId(comment.id);
    setDraft('');
  }

  function handleSubmit(e) {
    e.preventDefault();
    submit();
  }

  // Lookup: "keyboard event metaKey ctrlKey mdn"
  function handleKeyDown(e) {
    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
      e.preventDefault();
      submit();
    }
  }

  function handleDelete(id) {
    setComments((prev) => prev.filter((c) => c.id !== id));
  }

  return (
    <div className="thread">
      <h1>Discussion</h1>
      <ul className="comments">
        {comments.map((c) => (
          <Comment key={c.id} comment={c} isNew={c.id === newId} onDelete={handleDelete} />
        ))}
      </ul>
      <form className="composer" onSubmit={handleSubmit}>
        <textarea
          ref={inputRef}
          className="composer-input"
          placeholder="Add a comment…"
          rows={2}
          value={draft}
          onChange={(e) => setDraft(e.target.value)}
          onKeyDown={handleKeyDown}
          aria-describedby="comment-counter"
        />
        <div className="composer-foot">
          <span id="comment-counter" className={'counter' + (tooLong ? ' is-over' : '')}>
            {draft.length} / {MAX}
          </span>
          <span className="hint">⌘/Ctrl + Enter to send</span>
          <button className="btn-primary" type="submit" disabled={!canSubmit}>Comment</button>
        </div>
      </form>
    </div>
  );
}
