let keys = [
  { id: 'k1', name: 'Production server', last4: 'b7d3', createdAt: '2025-11-03', lastUsedAt: '2026-10-05', status: 'active' },
  { id: 'k2', name: 'Staging', last4: '8e40', createdAt: '2026-01-17', lastUsedAt: '2026-09-28', status: 'active' },
  { id: 'k3', name: 'Data pipeline', last4: '9b73', createdAt: '2026-03-22', lastUsedAt: null, status: 'active' },
  { id: 'k4', name: 'Old laptop', last4: 'b4c7', createdAt: '2025-06-09', lastUsedAt: '2025-12-30', status: 'revoked' },
];

const wait = (ms) => new Promise((r) => setTimeout(r, ms));

// GET /keys -> ApiKey[]
// ApiKey: { id, name, last4, createdAt: 'YYYY-MM-DD',
//           lastUsedAt: 'YYYY-MM-DD' | null, status: 'active' | 'revoked' }
export async function getKeys() {
  await wait(700);
  return keys.map((k) => ({ ...k }));
}

// POST /keys -> { key: ApiKey, secret: string }
// The full secret is only ever returned here, once.
export async function createKey(name) {
  await wait(600);
  const secret = 'sk-live-' + Math.random().toString(16).slice(2, 14) + Math.random().toString(16).slice(2, 10);
  const key = {
    id: 'k' + Date.now(),
    name,
    last4: secret.slice(-4),
    createdAt: new Date().toLocaleDateString('en-CA'), // today, YYYY-MM-DD
    lastUsedAt: null,
    status: 'active',
  };
  keys = [key, ...keys];
  return { key: { ...key }, secret };
}

// POST /keys/:id/revoke
export async function revokeKey(id) {
  await wait(300);
  keys = keys.map((k) => (k.id === id ? { ...k, status: 'revoked' } : k));
}

// "Nov 3, 2025"
export function formatDate(isoDate) {
  if (!isoDate) return '';
  return new Date(isoDate + 'T00:00:00').toLocaleDateString('en-US', {
    month: 'short',
    day: 'numeric',
    year: 'numeric',
  });
}

// maskKey('b7d3') -> "sk-live-••••b7d3"
export function maskKey(last4) {
  return 'sk-live-••••' + last4;
}

// pluralize(1, 'active key') -> "1 active key"
// pluralize(3, 'active key') -> "3 active keys"
export function pluralize(count, word) {
  return count + ' ' + word + (count === 1 ? '' : 's');
}

import React, { useEffect, useState } from 'react';
import { getKeys, createKey, revokeKey } from '../api';
import Button from '../components/Button';
import Badge from '../components/Badge';
import { formatDate, maskKey, pluralize } from '../utils/format';

// Written in "my JSX standard": decisions above the return, no && or ternaries in JSX.
export default function ApiKeysPage() {
  // ---- state ----
  const [keys, setKeys] = useState([]);
  const [loading, setLoading] = useState(true);
  const [hideRevoked, setHideRevoked] = useState(false);
  const [newSecret, setNewSecret] = useState(null); // mid-session change

  useEffect(() => {
    let cancelled = false;
    getKeys().then((data) => {
      if (cancelled) {
        return;
      }
      setKeys(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // ---- handlers ----
  function handleRevoke(id) {
    // Optimistic: update the screen now, tell the API in the background.
    setKeys((prev) => {
      return prev.map((k) => {
        if (k.id !== id) {
          return k;
        }
        return { ...k, status: 'revoked' };
      });
    });
    revokeKey(id);
  }

  function handleCreated(key, secret) {
    setKeys((prev) => {
      return [key, ...prev];
    });
    setNewSecret(secret);
  }

  // ---- derived ----
  const activeCount = keys.filter((k) => {
    return k.status === 'active';
  }).length;

  let visible = keys;
  if (hideRevoked) {
    visible = keys.filter((k) => {
      return k.status === 'active';
    });
  }

  // ---- decisions ----
  let banner = null;
  if (newSecret) {
    banner = (
      <div className="secret-banner">
        <div className="secret-text">
          <p className="secret-title">Save this key now. You won't be able to see it again.</p>
          <code className="secret-value">{newSecret}</code>
        </div>
        <Button onClick={() => { setNewSecret(null); }}>Done</Button>
      </div>
    );
  }

  let body;
  if (loading) {
    body = (
      <tr>
        <td colSpan={5} className="table-message">Loading keys…</td>
      </tr>
    );
  } else if (visible.length === 0) {
    body = (
      <tr>
        <td colSpan={5} className="table-message">No API keys yet.</td>
      </tr>
    );
  } else {
    body = visible.map((k) => {
      return <KeyRow key={k.id} apiKey={k} onRevoke={handleRevoke} />;
    });
  }

  return (
    <div className="keys-page">
      <header className="keys-header">
        <div>
          <h1>API keys</h1>
          <p className="keys-sub">{pluralize(activeCount, 'active key')}</p>
        </div>
        <label className="toggle">
          <input
            type="checkbox"
            checked={hideRevoked}
            onChange={(e) => { setHideRevoked(e.target.checked); }}
          />
          Hide revoked
        </label>
      </header>

      <CreateKeyForm onCreated={handleCreated} />
      {banner}

      <div className="table-card">
        <table className="table">
          <thead>
            <tr>
              <th>Name</th>
              <th>Key</th>
              <th>Created</th>
              <th>Last used</th>
              <th aria-label="Actions" />
            </tr>
          </thead>
          <tbody>{body}</tbody>
        </table>
      </div>
    </div>
  );
}

function KeyRow({ apiKey, onRevoke }) {
  const isRevoked = apiKey.status === 'revoked';

  let lastUsed = 'Never';
  if (apiKey.lastUsedAt) {
    lastUsed = formatDate(apiKey.lastUsedAt);
  }

  let action = <Badge tone="muted">Revoked</Badge>;
  if (!isRevoked) {
    action = (
      <Button variant="danger" onClick={() => { onRevoke(apiKey.id); }}>
        Revoke
      </Button>
    );
  }

  let rowClass = '';
  if (isRevoked) {
    rowClass = 'is-revoked';
  }

  return (
    <tr className={rowClass}>
      <td className="key-name">{apiKey.name}</td>
      <td><code className="key-mask">{maskKey(apiKey.last4)}</code></td>
      <td className="muted">{formatDate(apiKey.createdAt)}</td>
      <td className="muted">{lastUsed}</td>
      <td className="actions">{action}</td>
    </tr>
  );
}

function CreateKeyForm({ onCreated }) {
  const [name, setName] = useState('');
  const [creating, setCreating] = useState(false);

  async function handleSubmit(e) {
    e.preventDefault();
    const value = name.trim();
    if (!value) {
      return;
    }
    setCreating(true);
    try {
      const result = await createKey(value);
      onCreated(result.key, result.secret);
      setName('');
    } finally {
      setCreating(false);
    }
  }

  let label = 'Create key';
  if (creating) {
    label = 'Creating…';
  }
  const disabled = creating || name.trim() === '';

  return (
    <form className="create-row" onSubmit={handleSubmit}>
      <input
        className="input"
        placeholder="Key name, e.g. Production server"
        value={name}
        onChange={(e) => { setName(e.target.value); }}
      />
      <Button variant="primary" type="submit" disabled={disabled}>{label}</Button>
    </form>
  );
}
