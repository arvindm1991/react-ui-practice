import React, { useState } from 'react';
import { Logo } from './components/Logo';

const LEVELS = ['', 'Weak', 'Fair', 'Strong'];

function BrandPanel() {
  return (
    <aside className="brand-panel">
      <Logo />
      <h2>Business banking that balances itself.</h2>
      <ul className="perks">
        <li>Free ACH and wires</li>
        <li>Cards for every teammate</li>
        <li>Books that close in minutes</li>
      </ul>
    </aside>
  );
}

type Rule = {
  id: string;
  label: string;
  met: boolean;
};

// Pure function: easy to reason about, easy to test.
function getRules(password: string): Rule[] {
  return [
    { id: 'length', label: 'At least 8 characters', met: password.length >= 8 },
    { id: 'number', label: 'Contains a number', met: /\d/.test(password) },
    { id: 'symbol', label: 'Contains a symbol', met: /[^A-Za-z0-9]/.test(password) },
  ];
}

type StrengthMeterProps = {
  password: string;
  rules: Rule[];
};

function StrengthMeter({ password, rules }: StrengthMeterProps) {
  const score = rules.filter((r) => {
    return r.met;
  }).length;

  // Anything typed is at least "Weak". Empty shows nothing.
  let level = 0;
  if (password !== '') {
    level = Math.max(score, 1);
  }

  let meterCls = 'strength';
  let label = '';
  if (level > 0) {
    label = LEVELS[level];
    meterCls = 'strength is-' + label.toLowerCase();
  }

  const segments = [1, 2, 3].map((n) => {
    let cls = 'segment';
    if (n <= level) {
      cls = 'segment is-on';
    }
    return <span key={n} className={cls} />;
  });

  return (
    <div className={meterCls}>
      <div className="segments">{segments}</div>
      <span className="strength-label">{label}</span>
    </div>
  );
}

// Change: checklist of rules.
function RuleList({ rules }: { rules: Rule[] }) {
  return (
    <ul className="rules">
      {rules.map((r) => {
        let cls = 'rule';
        let mark = '○';
        if (r.met) {
          cls = 'rule is-met';
          mark = '✓';
        }
        return (
          <li key={r.id} className={cls}>
            <span className="rule-mark">{mark}</span>
            {r.label}
          </li>
        );
      })}
    </ul>
  );
}

function SignUpForm() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [showPassword, setShowPassword] = useState(false);

  function handleToggleShow() {
    setShowPassword((prev) => {
      return !prev;
    });
  }

  // derived: never stored in state
  const rules = getRules(password);
  const allMet = rules.every((r) => {
    return r.met;
  });
  const canSubmit = email.includes('@') && allMet;

  let inputType = 'password';
  let toggleText = 'Show';
  if (showPassword) {
    inputType = 'text';
    toggleText = 'Hide';
  }
  const toggleLabel = toggleText + ' password';

  return (
    <form className="form" onSubmit={(e) => { e.preventDefault(); }}>
      <h1>Create your account</h1>
      <p className="sub">Takes about two minutes. No credit check.</p>

      <label className="label" htmlFor="email">Work email</label>
      <input
        id="email"
        className="input"
        type="email"
        placeholder="you@company.com"
        value={email}
        onChange={(e) => { setEmail(e.target.value); }}
      />

      <label className="label" htmlFor="password">Password</label>
      <div className="password-wrap">
        <input
          id="password"
          className="input password-input"
          type={inputType}
          value={password}
          onChange={(e) => { setPassword(e.target.value); }}
        />
        <button type="button" className="show-btn" aria-label={toggleLabel} onClick={() => { handleToggleShow(); }}>
          {toggleText}
        </button>
      </div>
      <StrengthMeter password={password} rules={rules} />
      <RuleList rules={rules} />

      <button type="submit" className="btn-primary" disabled={!canSubmit}>Create account</button>
      <p className="fine">Already have an account? <a href="#">Sign in</a></p>
    </form>
  );
}

export default function App() {
  return (
    <div className="page">
      <BrandPanel />
      <main className="form-side">
        <SignUpForm />
      </main>
    </div>
  );
}
