  id: string;
  title: string;
  subject: string;
  done: boolean;
};

const TASKS: Task[] = [
  { id: 't1', title: 'Read chapter 4: Cell respiration', subject: 'Biology', done: true },
  { id: 't2', title: 'Problem set 3, questions 1–12', subject: 'Calculus', done: false },
  { id: 't3', title: 'Outline the WW1 causes essay', subject: 'History', done: false },
  { id: 't4', title: 'Flashcards: irregular verbs', subject: 'Spanish', done: false },
];

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /planner/tasks -> Task[]
export async function getTasks(): Promise<Task[]> {
  await randomDelay(500, 1200);
  return TASKS.map((t) => {
    return { ...t };
  });
}

// makeId('t') -> "t_k3j9x2"  (unique enough for client-side ids)
export function makeId(prefix: string): string {
  return prefix + '_' + Math.random().toString(36).slice(2, 8);
}

import React, { useEffect, useState } from 'react';
import { getTasks } from './api';
import type { Task } from './api';
import { makeId } from './utils/id';

// NEXT: the "3 left" badge. remaining is null while tasks are still loading.
function PlannerHeader({ remaining }: { remaining: number | null }) {
  let badge = null;
  if (remaining === 0) {
    badge = <span className="left-badge is-done">All done</span>;
  } else if (remaining !== null) {
    badge = <span className="left-badge">{remaining} left</span>;
  }
  return (
    <header className="planner-head">
      <div>
        <p className="week">Week 6 · Oct 7 – 13</p>
        <h1>This week</h1>
      </div>
      {badge}
    </header>
  );
}

type TaskRowProps = {
  task: Task;
  onToggle: (id: string) => void;
  onDelete: (id: string) => void;
};

// Already wired: it reports clicks up with the task's id.
function TaskRow({ task, onToggle, onDelete }: TaskRowProps) {
  let cls = 'task';
  if (task.done) {
    cls = 'task is-done';
  }
  return (
    <li className={cls}>
      <input type="checkbox" className="task-check" checked={task.done} onChange={() => { onToggle(task.id); }} />
      <span className="task-title">{task.title}</span>
      <span className="subject">{task.subject}</span>
      <button type="button" className="delete-btn" aria-label="Delete task" onClick={() => { onDelete(task.id); }}>×</button>
    </li>
  );
}

// CORE: owns only the draft text. It reports a finished title up.
function AddTask({ onAdd }: { onAdd: (title: string) => void }) {
  const [draft, setDraft] = useState('');

  // A <form> gives us Enter-to-submit for free.
  function handleSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const title = draft.trim();
    if (title === '') {
      return;
    }
    onAdd(title);
    setDraft('');
  }

  // NEXT: Add disabled while blank
  const isEmpty = draft.trim() === '';

  return (
    <form className="add-row" onSubmit={handleSubmit}>
      <input className="add-input" placeholder="Add a task and press Enter" value={draft} onChange={(e) => { setDraft(e.target.value); }} />
      <button type="submit" className="add-btn" disabled={isEmpty}>Add</button>
    </form>
  );
}

function Planner() {
  const [tasks, setTasks] = useState<Task[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    getTasks().then((data) => {
      if (cancelled) {
        return;
      }
      setTasks(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // CORE: add (CHANGE: newest first, so it goes to the front: was [...prev, task])
  function handleAdd(title: string) {
    const task: Task = { id: makeId('t'), title, subject: 'Inbox', done: false };
    setTasks((prev) => {
      return [task, ...prev];
    });
  }

  // CORE: toggle with map + object spread (never mutate t.done)
  function handleToggle(id: string) {
    setTasks((prev) => {
      return prev.map((t) => {
        if (t.id !== id) {
          return t;
        }
        return { ...t, done: !t.done };
      });
    });
  }

  // NEXT: delete with filter
  function handleDelete(id: string) {
    setTasks((prev) => {
      return prev.filter((t) => {
        return t.id !== id;
      });
    });
  }

  // STRETCH: clear completed
  function handleClearCompleted() {
    setTasks((prev) => {
      return prev.filter((t) => {
        return !t.done;
      });
    });
  }

  // NEXT: derived, never stored in state
  const remaining = tasks.filter((t) => {
    return !t.done;
  }).length;
  const doneCount = tasks.length - remaining;

  let list = <p className="empty">Loading your tasks…</p>;
  // NEXT: a friendly empty state once everything is deleted
  if (!loading && tasks.length === 0) {
    list = <p className="empty">Nothing planned. Add your first task above.</p>;
  } else if (!loading) {
    list = (
      <ul className="task-list">
        {tasks.map((task) => {
          return <TaskRow key={task.id} task={task} onToggle={handleToggle} onDelete={handleDelete} />;
        })}
      </ul>
    );
  }

  // STRETCH
  let footer = null;
  if (doneCount > 0) {
    footer = (
      <div className="planner-foot">
        <button type="button" className="clear-btn" onClick={() => { handleClearCompleted(); }}>Clear completed ({doneCount})</button>
      </div>
    );
  }

  let headerCount: number | null = remaining;
  if (loading) {
    headerCount = null;
  }

  return (
    <section className="planner">
      <PlannerHeader remaining={headerCount} />
      <AddTask onAdd={handleAdd} />
      {list}
      {footer}
    </section>
  );
}

export default function App() {
  return (
    <div className="app">
      <div className="brand">studyhall</div>
      <Planner />
    </div>
  );
}
