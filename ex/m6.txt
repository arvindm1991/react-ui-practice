import React, { useState } from 'react';
import { INITIAL_TASKS } from '../data/tasks';
import TaskRow from '../components/TaskRow';
import Button from '../components/Button';
import NewTaskModal from '../components/NewTaskModal';

export default function TasksPage() {
  const [tasks, setTasks] = useState(INITIAL_TASKS);
  const [modalOpen, setModalOpen] = useState(false);

  function handleToggle(id) {
    setTasks((prev) => {
      return prev.map((t) => {
        if (t.id !== id) {
          return t;
        }
        return { ...t, done: !t.done };
      });
    });
  }

  function handleCreate(task) {
    setTasks((prev) => {
      return [task, ...prev];
    });
    setModalOpen(false);
  }

  const openCount = tasks.filter((t) => {
    return !t.done;
  }).length;

  // Rendering the modal only while open means its form state
  // starts fresh every time it opens.
  let modal = null;
  if (modalOpen) {
    modal = <NewTaskModal onCreate={handleCreate} onClose={() => { setModalOpen(false); }} />;
  }

  return (
    <div className="page">
      <header className="page-header">
        <div>
          <h1>Tasks</h1>
          <p className="page-sub">{openCount} open</p>
        </div>
        <Button variant="primary" onClick={() => { setModalOpen(true); }}>+ New task</Button>
      </header>

      <ul className="task-list">
        {tasks.map((t) => {
          return <TaskRow key={t.id} task={t} onToggle={handleToggle} />;
        })}
      </ul>

      {modal}
    </div>
  );
}
@@@ solution src/components/NewTaskModal.jsx
import React, { useEffect, useState } from 'react';
import Button from './Button';
import { PEOPLE } from '../data/people';
import { makeId } from '../utils/ids';

const PRIORITIES = [
  ['low', 'Low'],
  ['medium', 'Medium'],
  ['high', 'High'],
];

export default function NewTaskModal({ onCreate, onClose }) {
  const [title, setTitle] = useState('');
  const [assigneeId, setAssigneeId] = useState(PEOPLE[0].id);
  const [priority, setPriority] = useState('medium');

  // Mid-session change: Escape closes.
  useEffect(() => {
    function handleKey(e) {
      if (e.key === 'Escape') {
        onClose();
      }
    }
    document.addEventListener('keydown', handleKey);
    return () => {
      document.removeEventListener('keydown', handleKey);
    };
  }, [onClose]);

  function handleSubmit(e) {
    e.preventDefault();
    const value = title.trim();
    if (!value) {
      return;
    }
    onCreate({ id: makeId('t'), title: value, assigneeId: assigneeId, priority: priority, done: false });
  }

  // Mid-session change: clicking the dark backdrop closes,
  // clicking inside the panel does not.
  function handleBackdropClick(e) {
    if (e.target === e.currentTarget) {
      onClose();
    }
  }

  const canCreate = title.trim() !== '';

  return (
    <div className="backdrop" onClick={handleBackdropClick}>
      <form className="modal" onSubmit={handleSubmit} role="dialog" aria-label="New task">
        <h2>New task</h2>

        <label className="field">
          <span className="field-label">Name</span>
          <input
            className="input"
            autoFocus
            placeholder="What needs to be done?"
            value={title}
            onChange={(e) => { setTitle(e.target.value); }}
          />
        </label>

        <label className="field">
          <span className="field-label">Assignee</span>
          <select className="input" value={assigneeId} onChange={(e) => { setAssigneeId(e.target.value); }}>
            {PEOPLE.map((p) => {
              return <option key={p.id} value={p.id}>{p.name}</option>;
            })}
          </select>
        </label>

        <div className="field">
          <span className="field-label">Priority</span>
          <div className="segmented" role="group" aria-label="Priority">
            {PRIORITIES.map((opt) => {
              let segClass = 'seg';
              if (priority === opt[0]) {
                segClass = 'seg is-active';
              }
              return (
                <button
                  key={opt[0]}
                  type="button"
                  className={segClass}
                  onClick={() => { setPriority(opt[0]); }}
                >
                  {opt[1]}
                </button>
              );
            })}
          </div>
        </div>

        <div className="modal-actions">
          <Button type="button" onClick={() => { onClose(); }}>Cancel</Button>
          <Button variant="primary" type="submit" disabled={!canCreate}>Create</Button>
        </div>
      </form>
    </div>
  );
}
