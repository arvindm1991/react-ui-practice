// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /boards/q4/cards -> Card[]
export async function getCards(): Promise<Card[]> {
  await randomDelay(500, 1200);
  return CARDS.map((c) => ({ ...c }));
}

import React, { useEffect, useState } from 'react';
import { getCards } from './api';
import type { Card, ColumnId } from './api';
import { COLUMNS, LABEL_NAMES } from './board';

const WIP_LIMIT = 3;

type CardViewProps = {
  card: Card;
  canLeft: boolean;
  canRight: boolean;
  onMove: (id: string, step: number) => void;
};

function CardView({ card, canLeft, canRight, onMove }: CardViewProps) {
  return (
    <li className="card">
      <span className={'label label-' + card.label}>{LABEL_NAMES[card.label]}</span>
      <p className="card-title">{card.title}</p>
      <div className="card-actions">
        <button type="button" className="move-btn" aria-label="Move left" disabled={!canLeft} onClick={() => { onMove(card.id, -1); }}>←</button>
        <button type="button" className="move-btn" aria-label="Move right" disabled={!canRight} onClick={() => { onMove(card.id, 1); }}>→</button>
      </div>
    </li>
  );
}

function AddCardForm({ onAdd }: { onAdd: (title: string) => void }) {
  const [title, setTitle] = useState('');

  function handleSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    if (title.trim() === '') {
      return;
    }
    onAdd(title.trim());
    setTitle('');
  }

  return (
    <form className="add-form" onSubmit={handleSubmit}>
      <input
        className="add-input"
        placeholder="New card title"
        value={title}
        onChange={(e) => { setTitle(e.target.value); }}
      />
      <button type="submit" className="add-btn">Add</button>
    </form>
  );
}

type ColumnProps = {
  title: string;
  index: number;
  cards: Card[];
  limit: number | null;
  onMove: (id: string, step: number) => void;
  children?: React.ReactNode;
};

function Column({ title, index, cards, limit, onMove, children }: ColumnProps) {
  let empty = null;
  if (cards.length === 0) {
    empty = <p className="column-empty">No cards</p>;
  }

  const canLeft = index > 0;
  const canRight = index < COLUMNS.length - 1;

  let countText = String(cards.length);
  let countCls = 'count';
  if (limit !== null) {
    countText = cards.length + ' / ' + limit;
    if (cards.length >= limit) {
      countCls = 'count is-full';
    }
  }

  return (
    <section className="column">
      <header className="column-head">
        <h2>{title}</h2>
        <span className={countCls}>{countText}</span>
      </header>
      <ul className="card-list">
        {cards.map((card) => {
          return <CardView key={card.id} card={card} canLeft={canLeft} canRight={canRight} onMove={onMove} />;
        })}
      </ul>
      {empty}
      {children}
    </section>
  );
}

function Board() {
  const [cards, setCards] = useState<Card[]>([]);
  const [loading, setLoading] = useState(true);
  const [notice, setNotice] = useState('');

  useEffect(() => {
    let cancelled = false;
    getCards().then((data) => {
      if (cancelled) {
        return;
      }
      setCards(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // derived: which cards are in a column is never stored separately
  function cardsIn(column: ColumnId) {
    return cards.filter((c) => {
      return c.column === column;
    });
  }

  function moveCard(id: string, step: number) {
    const card = cards.find((c) => {
      return c.id === id;
    });
    if (!card) {
      return;
    }
    const from = COLUMNS.findIndex((col) => {
      return col.id === card.column;
    });
    const target = COLUMNS[from + step];
    if (!target) {
      return;
    }
    // WIP limit: check against the current cards before moving.
    if (target.id === 'doing' && cardsIn('doing').length >= WIP_LIMIT) {
      setNotice('Doing is at its WIP limit of ' + WIP_LIMIT + '. Finish something first.');
      return;
    }
    setNotice('');
    setCards((prev) => {
      return prev.map((c) => {
        if (c.id !== id) {
          return c;
        }
        return { ...c, column: target.id };
      });
    });
  }

  function addCard(title: string) {
    const card: Card = { id: 'new-' + Date.now(), title, column: 'todo', label: 'design' };
    setCards((prev) => {
      return [...prev, card];
    });
  }

  if (loading) {
    return <p className="board-loading">Loading board…</p>;
  }

  let noticeEl = null;
  if (notice !== '') {
    noticeEl = <p className="wip-notice" role="alert">{notice}</p>;
  }

  return (
    <div>
      {noticeEl}
      <div className="board">
        {COLUMNS.map((col, index) => {
          let limit: number | null = null;
          if (col.id === 'doing') {
            limit = WIP_LIMIT;
          }
          let form = null;
          if (col.id === 'todo') {
            form = <AddCardForm onAdd={addCard} />;
          }
          return (
            <Column key={col.id} title={col.title} index={index} cards={cardsIn(col.id)} limit={limit} onMove={moveCard}>
              {form}
            </Column>
          );
        })}
      </div>
    </div>
  );
}

export default function App() {
  return (
    <div className="app">
      <header className="topbar">
        <span className="logo">▦ Trellis</span>
        <span className="crumb">Product / <b>Q4 roadmap</b></span>
        <div className="people"><span>RK</span><span>MA</span><span>+3</span></div>
      </header>
      <Board />
    </div>
  );
}
