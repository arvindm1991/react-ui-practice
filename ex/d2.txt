@@@ solution src/App.jsx
import React from 'react';
import { projects } from './data';

const STATUS_LABEL = { active: 'Active', paused: 'Paused', done: 'Done' };

function ProjectCard({ project }) {
  return (
    <article className="card">
      <span className={`badge badge-${project.status}`}>
        {STATUS_LABEL[project.status]}
      </span>
      {/* title attribute shows the full name on hover when truncated */}
      <h3 className="card-title" title={project.name}>{project.name}</h3>
      <p className="card-desc">{project.description}</p>
      <footer className="card-meta">
        <span>
          {project.members} {project.members === 1 ? 'member' : 'members'}
        </span>
        <span>Updated {project.updated}</span>
      </footer>
    </article>
  );
}

export default function App() {
  return (
    <main className="page">
      <h1 className="title">Projects</h1>
      <div className="grid">
        {projects.map((p) => (
          <ProjectCard key={p.id} project={p} />
        ))}
      </div>
    </main>
  );
}
