// Fake API. Treat it like a real backend you can't change.
export type Photo = {
  id: string;
  title: string;
  author: string;
  likes: number;
  location: string;
  colors: [string, string]; // gradient stand-in for the image
};

export type PhotoPage = {
  photos: Photo[];
  nextCursor: number | null; // null = no more pages
};

const TITLES = ['Morning fog', 'Neon alley', 'Salt flats', 'Quiet harbor', 'Dune lines', 'Glass tower', 'Late tram', 'First snow', 'Copper hour', 'Tide pools'];
const AUTHORS = ['mira.k', 'jonas', 'aiko_lens', 'dev.shoots', 'lena.v', 'oskar'];
const PLACES = ['Lisbon', 'Kyoto', 'Reykjavik', 'Oaxaca', 'Cape Town', 'Oslo', 'Hanoi', 'Marfa'];
const PALETTE: [string, string][] = [
  ['#FF9A8B', '#6A5ACD'], ['#1D976C', '#93F9B9'], ['#F7971E', '#FFD200'], ['#2C3E50', '#4CA1AF'],
  ['#ee0979', '#ff6a00'], ['#4776E6', '#8E54E9'], ['#11998e', '#38ef7d'], ['#C33764', '#1D2671'],
];

const TOTAL = 40;
const ALL: Photo[] = [];
for (let i = 0; i < TOTAL; i++) {
  ALL.push({
    id: 'p' + (i + 1),
    title: TITLES[i % TITLES.length] + ' #' + (i + 1),
    author: AUTHORS[i % AUTHORS.length],
    likes: 120 + ((i * 937) % 4800),
    location: PLACES[i % PLACES.length],
    colors: PALETTE[i % PALETTE.length],
  });
}

// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /photos?cursor=0&limit=8 -> PhotoPage
// Pass the nextCursor from the previous page to get the next one.
export async function getPhotos(cursor: number, limit: number = 8): Promise<PhotoPage> {
  console.log('GET /photos?cursor=' + cursor);
  await randomDelay(500, 1100);
  const slice = ALL.slice(cursor, cursor + limit).map((p) => ({ ...p }));
  const next = cursor + limit;
  let nextCursor: number | null = next;
  if (next >= TOTAL) {
    nextCursor = null;
  }
  return { photos: slice, nextCursor: nextCursor };
}

// formatCount(1840) -> "1.8K", formatCount(950) -> "950"
export function formatCount(n: number): string {
  if (n >= 1000) {
    return (n / 1000).toFixed(1) + 'K';
  }
  return String(n);
}

function TopBar() {
  return (
    <header className="topbar">

function Feed() {
  const [photos, setPhotos] = useState<Photo[]>([]);
  const [loading, setLoading] = useState(true);

  // Only loads the first page. TODO: infinite scroll.

    getPhotos(0).then((page) => {

  return (

      <div className="feed">
        {photos.map((p) => {
          return <PhotoCard key={p.id} photo={p} />;
        })}
        {status}
      </div>
    </div>
  );

    <div className="app">

@@@ solution src/App.tsx
import React, { useEffect, useRef, useState } from 'react';
import { getPhotos } from './api';
import type { Photo } from './api';
import PhotoCard from './components/PhotoCard';
import Spinner from './components/Spinner';

// CHANGE: back-to-top threshold
const SHOW_TOP_AFTER = 800; // px scrolled before "Back to top" appears

function TopBar() {
  return (
    <header className="topbar">
      <span className="logo">◉ aperture</span>
      <nav className="tabs">
        <button type="button" className="tab is-active">For you</button>
        <button type="button" className="tab">Following</button>
      </nav>
    </header>
  );
}

function Feed() {
  // CORE: feed state (photos, cursor, loading) + refs for the scroll root, sentinel and in-flight guard
  const [photos, setPhotos] = useState<Photo[]>([]);
  const [nextCursor, setNextCursor] = useState<number | null>(0); // null = reached the end
  const [loading, setLoading] = useState(false);
  // CHANGE: back-to-top visibility
  const [showTop, setShowTop] = useState(false);
  // CORE: refs for the scroll root, the sentinel, and the in-flight guard
  const scrollRef = useRef<HTMLDivElement>(null);
  const sentinelRef = useRef<HTMLDivElement>(null);
  // A ref, not state: the observer callback must see the latest value immediately.
  const inFlight = useRef(false);

  // CORE: IntersectionObserver on the sentinel, root = scroll container; inFlight ref prevents duplicates
  // A new observer per cursor. When a page arrives, nextCursor changes, the observer is
  // recreated, and if the sentinel is still visible it fires again (fills tall screens).
  useEffect(() => {
    // NEXT: stop observing at the end of the feed
    if (nextCursor === null) {
      return; // end of feed: stop observing
    }
    // CORE
    const root = scrollRef.current;
    const sentinel = sentinelRef.current;
    if (!root || !sentinel) {
      return;
    }
    let cancelled = false;
    const cursor = nextCursor;
    const observer = new IntersectionObserver((entries) => {
      if (!entries[0].isIntersecting || inFlight.current) {
        return;
      }
      inFlight.current = true;
      setLoading(true);
      getPhotos(cursor).then((page) => {
        inFlight.current = false;
        if (cancelled) {
          return;
        }
        setPhotos((prev) => {
          return [...prev, ...page.photos];
        });
        setNextCursor(page.nextCursor);
        setLoading(false);
      });
    }, { root: root, rootMargin: '0px 0px 300px 0px' });
    observer.observe(sentinel);
    return () => {
      cancelled = true;
      observer.disconnect();
    };
  }, [nextCursor]);

  // CHANGE: track scroll position + smooth scroll to top
  function handleScroll(e: React.UIEvent<HTMLDivElement>) {
    const scrolled = e.currentTarget.scrollTop > SHOW_TOP_AFTER;
    setShowTop(scrolled); // React skips the re-render when the value didn't change
  }

  function handleBackToTop() {
    if (scrollRef.current) {
      scrollRef.current.scrollTo({ top: 0, behavior: 'smooth' });
    }
  }

  // CORE: loading row (reuse Spinner); NEXT: the "all caught up" end row
  // decisions
  let status = null;
  if (loading) {
    status = <Spinner label="Loading more photos…" />;
  } else if (nextCursor === null) {
    status = <p className="feed-end">You're all caught up · {photos.length} photos</p>;
  }

  // CHANGE: floating back-to-top button
  let topButton = null;
  if (showTop) {
    topButton = (
      <button type="button" className="back-to-top" onClick={() => { handleBackToTop(); }}>
        ↑ Back to top
      </button>
    );
  }

  // CORE: sentinel after the last card + status row (CHANGE: the .feed-wrap wrapper and {topButton})
  return (
    <div className="feed-wrap">
      <div className="feed-scroll" ref={scrollRef} onScroll={handleScroll}>
        <div className="feed">
          {photos.map((p) => {
            return <PhotoCard key={p.id} photo={p} />;
          })}
          <div ref={sentinelRef} className="sentinel" aria-hidden="true" />
          {status}
        </div>
      </div>
      {topButton}
    </div>
  );
}

export default function App() {
  return (
    <div className="app">
      <TopBar />
      <Feed />
    </div>
  );
}
