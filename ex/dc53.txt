// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// POST /cards/:id/move { toColumn, toIndex }
// toIndex is the position in the destination column AFTER the card is taken out of its old spot.
// Rejects (deterministically) when moving an unassigned card into "done".
export async function moveCard(cardId: string, toColumn: ColumnId, toIndex: number): Promise<void> {
  await randomDelay(400, 900);
  const pos = findCard(DB, cardId);
  if (!pos) {
    throw new Error('Card ' + cardId + ' not found.');
  }
  const card = DB[pos.columnId][pos.index];
  if (toColumn === 'done' && !card.assignee) {
    throw new Error('Assign someone before moving ' + card.id + ' to Done.');
  }
  DB = moveCardInBoard(DB, cardId, toColumn, toIndex);
}

function initials(name: string): string {
  return name
    .split(' ')
    .map((part) => part[0])
    .join('')
    .slice(0, 2)
    .toUpperCase();
}

export default function Avatar({ name }: { name: string | null }) {
  if (!name) {
    return <span className="avatar is-empty" title="Unassigned" aria-label="Unassigned" />;
  }
  let sum = 0;
  for (let i = 0; i < name.length; i++) {
    sum += name.charCodeAt(i);
  }
  const color = COLORS[sum % COLORS.length];
  return (
    <span className="avatar" style={{ background: color }} title={name} aria-label={name}>
      {initials(name)}
    </span>
  );
}

import type { Board, ColumnId } from '../api';

// Where is a card? findCard(board, 'ORB-41') -> { columnId: 'backlog', index: 0 }
export function findCard(board: Board, cardId: string): { columnId: ColumnId; index: number } | null {
  const ids = Object.keys(board) as ColumnId[];
  for (const columnId of ids) {
    const index = board[columnId].findIndex((c) => c.id === cardId);
    if (index !== -1) {
      return { columnId, index };
    }
  }
  return null;
}

// Pure: returns a NEW board with the card moved.
// toIndex is the index in the destination column after the card has been removed from its old spot.
export function moveCardInBoard(board: Board, cardId: string, toColumn: ColumnId, toIndex: number): Board {
  const from = findCard(board, cardId);
  if (!from) {
    return board;
  }
  const card = board[from.columnId][from.index];
  const next: Board = { ...board };
  next[from.columnId] = board[from.columnId].filter((c) => c.id !== cardId);
  const dest = [...next[toColumn]];
  const at = Math.max(0, Math.min(toIndex, dest.length));
  dest.splice(at, 0, card);
  next[toColumn] = dest;
  return next;
}

// Drag helper: where would a drop at this pointer Y land inside `container`?
// Returns the index of the first card (element with data-card-id) whose vertical middle
// is below clientY, or the number of cards if there is none.
// Note: if the dragged card is in this container, it is counted too.
export function getInsertIndex(container: HTMLElement, clientY: number): number {
  const cardEls = Array.from(container.querySelectorAll<HTMLElement>('[data-card-id]'));
  for (let i = 0; i < cardEls.length; i++) {
    const r = cardEls[i].getBoundingClientRect();
    if (clientY < r.top + r.height / 2) {
      return i;
    }
  }
  return cardEls.length;
}

type DropTarget = { columnId: ColumnId; index: number };

// CORE: each column is a drop zone (NEXT: indicatorAt, dragId, onDragLeave)
type ColumnProps = {
  id: ColumnId;
  title: string;
  cards: Card[];
  indicatorAt: number | null;
  dragId: string | null;
  onDragOver: (e: React.DragEvent<HTMLElement>, columnId: ColumnId) => void;
  onDragLeave: (e: React.DragEvent<HTMLElement>) => void;
  onDrop: (e: React.DragEvent<HTMLElement>, columnId: ColumnId) => void;
  cardHandlers: Pick<CardViewProps, 'onDragStart' | 'onDragEnd' | 'onKeyDown'>;
};

function Column({ id, title, cards, indicatorAt, dragId, onDragOver, onDragLeave, onDrop, cardHandlers }: ColumnProps) {
  // NEXT: build the list with the insert indicator spliced in at its index
  // (Core alone can just map the cards).
  const items: React.ReactNode[] = [];
  const indicator = <div key="drop-indicator" className="drop-indicator" />;
  cards.forEach((card, i) => {
    if (indicatorAt === i) {
      items.push(indicator);
    }
    items.push(
      <CardView key={card.id} card={card} dragging={card.id === dragId} {...cardHandlers} />
    );
  });
  if (indicatorAt === cards.length) {
    items.push(indicator);
  }

  // NEXT: tint the column under the pointer
  let cls = 'column column-' + id;
  if (indicatorAt !== null) {
    cls += ' is-over';
  }

  // CORE: drop zone handlers
  return (
    <section
      className={cls}
      onDragOver={(e) => { onDragOver(e, id); }}
      onDragLeave={(e) => { onDragLeave(e); }}
      onDrop={(e) => { onDrop(e, id); }}
    >
      <header className="column-head">
        <span className="column-dot" />
        <h2>{title}</h2>
        <span className="column-count">{cards.length}</span>
      </header>
      <div className="column-body">
        {items}
      </div>
    </section>
  );
}

export default function App() {
  const [board, setBoard] = useState<Board | null>(null);
  // CORE: drag state lives next to the board
  const [dragId, setDragId] = useState<string | null>(null);
  // NEXT: drives the indicator and the tint
  const [dropTarget, setDropTarget] = useState<DropTarget | null>(null);
  // NEXT: the rollback toast
  const [toast, setToast] = useState<string | null>(null);
  // STRETCH: which card to keep focused after a keyboard move
  const [focusId, setFocusId] = useState<string | null>(null);
  const boardRef = useRef<HTMLDivElement>(null);
  // STRETCH: screen reader announcement
  const [announcement, setAnnouncement] = useState('');

  useEffect(() => {
    let cancelled = false;
    getBoard().then((data) => {
      if (cancelled) {
        return;
      }
      setBoard(data);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // NEXT: the toast hides itself after 4 s. A new message restarts the timer.
  useEffect(() => {
    if (!toast) {
      return;
    }
    const t = setTimeout(() => {
      setToast(null);
    }, 4000);
    return () => {
      clearTimeout(t);
    };
  }, [toast]);

  // STRETCH: a card moved to another column is a NEW DOM node, so focus is lost. Put it back.
  useEffect(() => {
    if (!focusId || !boardRef.current) {
      return;
    }
    const el = boardRef.current.querySelector<HTMLElement>('[data-card-id="' + focusId + '"]');
    if (el) {
      el.focus();
    }
  }, [board, focusId]);

  // CORE: optimistic: move now, save in the background, put it back if the server says no.
  function commitMove(cardId: string, toColumn: ColumnId, toIndex: number) {
    if (!board) {
      return;
    }
    const from = findCard(board, cardId);
    if (!from || (from.columnId === toColumn && from.index === toIndex)) {
      return;
    }
    setBoard((prev) => {
      if (!prev) {
        return prev;
      }
      return moveCardInBoard(prev, cardId, toColumn, toIndex);
    });
    // STRETCH: announce the move
    const col = COLUMNS.find((c) => { return c.id === toColumn; });
    let colTitle: string = toColumn;
    if (col) {
      colTitle = col.title;
    }
    setAnnouncement('Moved ' + cardId + ' to ' + colTitle + '.');
    // CORE: save, roll back on failure
    moveCard(cardId, toColumn, toIndex).catch((err: Error) => {
      // Roll back only THIS card, so other moves made meanwhile survive.
      setBoard((prev) => {
        if (!prev) {
          return prev;
        }
        return moveCardInBoard(prev, cardId, from.columnId, from.index);
      });
      setToast(err.message); // NEXT: say why
    });
  }

  // CORE: mouse drag and drop
  function handleDragStart(e: React.DragEvent<HTMLElement>, cardId: string) {
    e.dataTransfer.effectAllowed = 'move';
    e.dataTransfer.setData('text/plain', cardId);
    setDragId(cardId);
    setFocusId(null); // STRETCH
  }

  function handleDragOver(e: React.DragEvent<HTMLElement>, columnId: ColumnId) {
    if (!dragId) {
      return;
    }
    e.preventDefault(); // without this, drop never fires
    // NEXT: show where it will land. dragover fires constantly: only set state when the target changes
    const index = getInsertIndex(e.currentTarget, e.clientY);
    if (!dropTarget || dropTarget.columnId !== columnId || dropTarget.index !== index) {
      setDropTarget({ columnId, index });
    }
  }

  // NEXT: ignore leaving into a child of the same column
  function handleDragLeave(e: React.DragEvent<HTMLElement>) {
    if (!e.currentTarget.contains(e.relatedTarget as Node | null)) {
      setDropTarget(null);
    }
  }

  // CORE
  function endDrag() {
    setDragId(null);
    setDropTarget(null);
  }

  function handleDrop(e: React.DragEvent<HTMLElement>, columnId: ColumnId) {
    e.preventDefault();
    if (!board || !dragId) {
      endDrag();
      return;
    }
    let index = getInsertIndex(e.currentTarget, e.clientY);
    // NEXT: the insert index counts the dragged card itself; the API index doesn't.
    const from = findCard(board, dragId);
    if (from && from.columnId === columnId && from.index < index) {
      index -= 1;
    }
    // CORE
    commitMove(dragId, columnId, index);
    endDrag();
  }

  // CHANGE: ← / → on a focused card moves it one column, same position (clamped)
  function handleCardKeyDown(e: React.KeyboardEvent<HTMLElement>, cardId: string) {
    if (!board || (e.key !== 'ArrowLeft' && e.key !== 'ArrowRight')) {
      return;
    }
    const from = findCard(board, cardId);
    if (!from) {
      return;
    }
    e.preventDefault();
    const colIndex = COLUMNS.findIndex((c) => { return c.id === from.columnId; });
    let step = 1;
    if (e.key === 'ArrowLeft') {
      step = -1;
    }
    const target = COLUMNS[colIndex + step];
    if (target) {
      setFocusId(cardId); // STRETCH: keep focus on it
      commitMove(cardId, target.id, Math.min(from.index, board[target.id].length));
    }
  }

  // CORE: one object so Column can spread it onto every card
  const cardHandlers = {
    onDragStart: handleDragStart,
    onDragEnd: endDrag,
    onKeyDown: handleCardKeyDown,
  };

  let content;
  if (!board) {
    content = <p className="loading">Loading board…</p>;
  } else {
    content = (
      <div className="board" ref={boardRef}>
        {COLUMNS.map((col) => {
          // NEXT: indicator only in the column under the pointer
          let indicatorAt: number | null = null;
          if (dropTarget && dropTarget.columnId === col.id) {
            indicatorAt = dropTarget.index;
          }
          // CORE
          return (
            <Column
              key={col.id}
              id={col.id}
              title={col.title}
              cards={board[col.id]}
              indicatorAt={indicatorAt}
              dragId={dragId}
              onDragOver={handleDragOver}
              onDragLeave={handleDragLeave}
              onDrop={handleDrop}
              cardHandlers={cardHandlers}
            />
          );
        })}
      </div>
    );
  }

  // NEXT: rollback toast
  let toastEl = null;
  if (toast) {
    toastEl = <Toast message={toast} onDismiss={() => { setToast(null); }} />;
  }

  return (
    <div className="app">
      <header className="topbar">
        <span className="logo">◆ Orbit</span>
        <h1>Sprint 24</h1>
        <span className="sprint-dates">Oct 6 – Oct 17</span>
      </header>
      {content}
      {/* STRETCH: visually hidden live region */}
      <p className="sr-only" aria-live="polite">{announcement}</p>
      {/* NEXT */}
      {toastEl}
    </div>
  );
}
