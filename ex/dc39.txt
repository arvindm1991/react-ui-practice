// Resolves after a random delay between min and max milliseconds.
function randomDelay(min: number, max: number): Promise<void> {
  const ms = min + Math.random() * (max - min);
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /employees -> Employee[] (all of them, unsorted; filtering/sorting/paging is done in the UI)
export async function getEmployees(): Promise<Employee[]> {
  await randomDelay(500, 1100);
  return EMPLOYEES.map((e) => ({ ...e }));
}

type PaginationProps = {
  page: number;       // 1-based
  pageCount: number;
  onChange: (page: number) => void;
};

// "‹ Prev  Page 2 of 4  Next ›". Buttons disable at the ends.
export function Pagination({ page, pageCount, onChange }: PaginationProps) {
  return (
    <div className="pagination">
      <button type="button" className="page-btn" disabled={page <= 1} onClick={() => { onChange(page - 1); }}>‹ Prev</button>
      <span className="page-info">Page {page} of {pageCount}</span>
      <button type="button" className="page-btn" disabled={page >= pageCount} onClick={() => { onChange(page + 1); }}>Next ›</button>
    </div>
  );
}

const MONTHS = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

// formatDate("2019-03-04") -> "Mar 4, 2019"
export function formatDate(iso: string): string {
  const parts = iso.split('-').map(Number);
  return MONTHS[parts[1] - 1] + ' ' + parts[2] + ', ' + parts[0];
}

import React, { useEffect, useState } from 'react';
import { DEPARTMENTS, getEmployees } from './api';
import type { Employee } from './api';
import { DeptBadge } from './components/DeptBadge';
import { Pagination } from './components/Pagination';
import { formatDate } from './utils/format';

function Sidebar() {
  return (
    <aside className="sidebar">
      <div className="logo"><span className="logo-mark">P</span>Peoplebase</div>
      <nav className="side-nav">
        <span className="side-link">Home</span>
        <span className="side-link is-active">Directory</span>
        <span className="side-link">Time off</span>
        <span className="side-link">Payroll</span>
      </nav>
    </aside>
  );
}

type ColumnKey = 'name' | 'title' | 'department' | 'location' | 'startDate';
type SortDir = 'asc' | 'desc';

// One config drives the header, the cells and the column toggles.
const COLUMNS: { key: ColumnKey; label: string }[] = [
  { key: 'name', label: 'Name' },
  { key: 'title', label: 'Title' },
  { key: 'department', label: 'Department' },
  { key: 'location', label: 'Location' },
  { key: 'startDate', label: 'Start date' },
];
const PAGE_SIZE = 8;

function renderCell(employee: Employee, key: ColumnKey) {
  if (key === 'department') {
    return <DeptBadge department={employee.department} />;
  }
  if (key === 'startDate') {
    return formatDate(employee.startDate);
  }
  return employee[key];
}

type ColumnMenuProps = {
  visible: Record<ColumnKey, boolean>;
  onToggle: (key: ColumnKey) => void;
};

// Change: column visibility. Name is always shown.
function ColumnMenu({ visible, onToggle }: ColumnMenuProps) {
  return (
    <details className="col-menu">
      <summary className="btn">Columns</summary>
      <div className="col-pop">
        {COLUMNS.map((col) => {
          if (col.key === 'name') {
            return null;
          }
          return (
            <label key={col.key} className="col-option">
              <input type="checkbox" checked={visible[col.key]} onChange={() => { onToggle(col.key); }} />
              {col.label}
            </label>
          );
        })}
      </div>
    </details>
  );
}

function EmployeeTable() {
  const [employees, setEmployees] = useState<Employee[]>([]);
  const [loading, setLoading] = useState(true);
  const [query, setQuery] = useState('');
  const [dept, setDept] = useState('');
  const [sortKey, setSortKey] = useState<ColumnKey>('name');
  const [sortDir, setSortDir] = useState<SortDir>('asc');
  const [page, setPage] = useState(1);
  const [visible, setVisible] = useState<Record<ColumnKey, boolean>>({
    name: true, title: true, department: true, location: true, startDate: true,
  });

  useEffect(() => {
    let cancelled = false;
    getEmployees().then((data) => {
      if (cancelled) {
        return;
      }
      setEmployees(data);
      setLoading(false);
    });
    return () => {
      cancelled = true;
    };
  }, []);

  // Every filter change sends you back to page 1 (page 3 of the old list may not exist).
  function handleQuery(value: string) {
    setQuery(value);
    setPage(1);
  }

  function handleDept(value: string) {
    setDept(value);
    setPage(1);
  }

  function handleClear() {
    setQuery('');
    setDept('');
    setPage(1);
  }

  function handleSort(key: ColumnKey) {
    if (key === sortKey) {
      setSortDir((prev) => {
        if (prev === 'asc') {
          return 'desc';
        }
        return 'asc';
      });
    } else {
      setSortKey(key);
      setSortDir('asc');
    }
    setPage(1);
  }

  function handleToggleColumn(key: ColumnKey) {
    setVisible((prev) => {
      return { ...prev, [key]: !prev[key] };
    });
  }

  if (loading) {
    return <div className="card"><p className="table-msg">Loading employees…</p></div>;
  }

  // derived pipeline: search -> department -> sort -> page. Nothing here is stored in state.
  const q = query.trim().toLowerCase();
  const filtered = employees.filter((e) => {
    const matchesQuery = q === '' || e.name.toLowerCase().includes(q) || e.title.toLowerCase().includes(q);
    const matchesDept = dept === '' || e.department === dept;
    return matchesQuery && matchesDept;
  });
  const sorted = [...filtered].sort((a, b) => {
    const result = a[sortKey].localeCompare(b[sortKey]);
    if (sortDir === 'desc') {
      return -result;
    }
    return result;
  });
  const pageCount = Math.max(1, Math.ceil(sorted.length / PAGE_SIZE));
  const start = (page - 1) * PAGE_SIZE;
  const pageRows = sorted.slice(start, start + PAGE_SIZE);
  const visibleColumns = COLUMNS.filter((col) => {
    return visible[col.key];
  });

  let clearBtn = null;
  if (query !== '' || dept !== '') {
    clearBtn = <button type="button" className="btn btn-link" onClick={() => { handleClear(); }}>Clear filters</button>;
  }

  let rows;
  if (pageRows.length === 0) {
    rows = (
      <tr>
        <td className="table-empty" colSpan={visibleColumns.length}>No employees match your filters.</td>
      </tr>
    );
  } else {
    rows = pageRows.map((e) => {
      return (
        <tr key={e.id}>
          {visibleColumns.map((col) => {
            let cls = '';
            if (col.key === 'name') {
              cls = 'cell-name';
            }
            return <td key={col.key} className={cls}>{renderCell(e, col.key)}</td>;
          })}
        </tr>
      );
    });
  }

  let rangeText = '0 of 0';
  if (sorted.length > 0) {
    rangeText = (start + 1) + '–' + (start + pageRows.length) + ' of ' + sorted.length;
  }

  return (
    <div className="card">
      <div className="toolbar">
        <input className="search" placeholder="Search name or title" value={query} onChange={(e) => { handleQuery(e.target.value); }} />
        <select className="select" value={dept} onChange={(e) => { handleDept(e.target.value); }}>
          <option value="">All departments</option>
          {DEPARTMENTS.map((d) => {
            return <option key={d} value={d}>{d}</option>;
          })}
        </select>
        {clearBtn}
        <span className="toolbar-spacer" />
        <ColumnMenu visible={visible} onToggle={handleToggleColumn} />
      </div>
      <table className="table">
        <thead>
          <tr>
            {visibleColumns.map((col) => {
              let arrow = '↕';
              let cls = 'sort-btn';
              let ariaSort: 'ascending' | 'descending' | 'none' = 'none';
              if (col.key === sortKey) {
                cls = 'sort-btn is-sorted';
                arrow = '▲';
                ariaSort = 'ascending';
                if (sortDir === 'desc') {
                  arrow = '▼';
                  ariaSort = 'descending';
                }
              }
              return (
                <th key={col.key} aria-sort={ariaSort}>
                  <button type="button" className={cls} onClick={() => { handleSort(col.key); }}>
                    {col.label}
                    <span className="sort-arrow">{arrow}</span>
                  </button>
                </th>
              );
            })}
          </tr>
        </thead>
        <tbody>{rows}</tbody>
      </table>
      <div className="table-foot">
        <span className="range">Showing {rangeText}</span>
        <Pagination page={page} pageCount={pageCount} onChange={setPage} />
      </div>
    </div>
  );
}

export default function App() {
  return (
    <div className="app">
      <Sidebar />
      <main className="main">
        <div className="page-head">
          <h1>Directory</h1>
          <p className="page-sub">Everyone at Northwind, in one place.</p>
        </div>
        <EmployeeTable />
      </main>
    </div>
  );
}
