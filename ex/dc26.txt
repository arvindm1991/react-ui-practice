function wait(ms: number): Promise<void> {
  return new Promise((resolve) => {
    setTimeout(resolve, ms);
  });
}

// GET /people?q=jan -> Person[] whose name contains q (case-insensitive)
// Shorter queries match more rows and are SLOWER: "j" takes ~1.3s, "jan" ~0.7s.
// So an older, shorter request can finish AFTER a newer, longer one.
export async function searchPeople(query: string): Promise<Person[]> {
  const q = query.trim().toLowerCase();
  await wait(Math.max(250, 1600 - q.length * 300));
  return PEOPLE.filter((p) => p.name.toLowerCase().includes(q)).map((p) => ({ ...p }));
}

import React from 'react';

const COLORS = ['#0D9488', '#7C3AED', '#EA580C', '#2563EB', '#DB2777', '#65A30D'];

// Initials in a colored circle. Color is stable per name.
export default function Avatar({ name }: { name: string }) {
  const parts = name.split(' ');
  let initials = parts[0][0];
  if (parts.length > 1) {
    initials = parts[0][0] + parts[parts.length - 1][0];
  }
  let sum = 0;
  for (let i = 0; i < name.length; i++) {
    sum += name.charCodeAt(i);
  }
  return (
    <span className="avatar" style={{ background: COLORS[sum % COLORS.length] }}>{initials}</span>
  );
}

import React, { useEffect, useState } from 'react';
import Avatar from './components/Avatar';
import type { Person } from './api';
import { searchPeople, TOTAL_PEOPLE } from './api';

const DEBOUNCE_MS = 300;

// CHANGE: highlight the first match
// Wraps the first case-insensitive match of `query` in <mark>.
function Highlight({ text, query }: { text: string; query: string }) {
  const index = text.toLowerCase().indexOf(query.toLowerCase());
  if (query === '' || index === -1) {
    return <span>{text}</span>;
  }
  const before = text.slice(0, index);
  const match = text.slice(index, index + query.length);
  const after = text.slice(index + query.length);
  return (
    <span>
      {before}
      <mark className="match">{match}</mark>
      {after}
    </span>
  );
}

// CHANGE: PersonRow gets the query to highlight
function PersonRow({ person, query }: { person: Person; query: string }) {
  return (
    <li className="person">
      <Avatar name={person.name} />
      <div className="person-main">
        <span className="person-name"><Highlight text={person.name} query={query} /></span>
        <span className="person-title">{person.title} · {person.team}</span>
      </div>
      <span className="person-loc">{person.location}</span>
    </li>
  );
}

function SearchBox({ value, onChange }: { value: string; onChange: (value: string) => void }) {
  return (
    <div className="search">
      <span className="search-icon">⌕</span>
      <input
        className="search-input"
        placeholder="Search people by name"
        value={value}
        onChange={(e) => { onChange(e.target.value); }}
      />
    </div>
  );
}

// CORE: results + loading, searched by a debounced effect
function Directory() {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState<Person[]>([]);
  const [loading, setLoading] = useState(false);
  // CHANGE: the query the current results belong to (used for highlighting).
  const [resultsFor, setResultsFor] = useState('');

  // CORE: debounce + ignore stale responses
  useEffect(() => {
    const q = query.trim();
    if (q === '') {
      setResults([]);
      setResultsFor(''); // CHANGE
      setLoading(false);
      return;
    }
    // One flag per effect run. When the query changes, the cleanup marks
    // this run as stale, so its late response is ignored.
    let cancelled = false;
    const timer = setTimeout(() => {
      setLoading(true);
      searchPeople(q).then((data) => {
        if (cancelled) {
          return;
        }
        setResults(data);
        setResultsFor(q); // CHANGE
        setLoading(false);
      });
    }, DEBOUNCE_MS);
    return () => {
      cancelled = true;
      clearTimeout(timer); // debounce: a new keystroke cancels the pending call
    };
  }, [query]);

  // CORE: Searching… line
  let status = null;
  if (loading) {
    status = <p className="dir-status" aria-live="polite">Searching…</p>;
  }

  // CORE: empty prompt, then the results (NEXT: no-match message)
  let body;
  if (query.trim() === '') {
    body = <p className="dir-message">Start typing to search {TOTAL_PEOPLE} people.</p>;
  } else if (resultsFor === '') {
    body = null; // first search still in flight: the status line says so
  } else if (results.length === 0) {
    body = <p className="dir-message">No people match "{resultsFor}".</p>;
  } else {
    body = (
      <ul className="people">
        {results.map((p) => {
          return <PersonRow key={p.id} person={p} query={resultsFor} />;
        })}
      </ul>
    );
  }

  // NEXT: dim old results while a newer search loads
  let cls = 'directory';
  if (loading) {
    cls = 'directory is-loading';
  }

  // CORE: render
  return (
    <section className={cls}>
      <SearchBox value={query} onChange={setQuery} />
      {status}
      {body}
    </section>
  );
}

export default function App() {
  return (
    <div className="app">
      <header className="topbar">
        <span className="logo">Roster</span>
        <nav className="tabs">
          <a className="tab is-active" href="#">People</a>
          <a className="tab" href="#">Teams</a>
          <a className="tab" href="#">Org chart</a>
        </nav>
      </header>
      <main className="page">
        <h1>People</h1>
        <Directory />
      </main>
    </div>
  );
}
