const FALLBACK = [
  'Good question. The short answer is to cut anything the reader already believes and spend those words on the one thing they do not. In your draft, the first two sentences restate the premise. Start at "Most teams never measure this", which is where the piece actually begins, and move the statistic up so it lands in the first screen.',
  'Another way to approach it: read the draft out loud and mark every place you take a breath mid-sentence. Those are your split points. Your third paragraph has two of them, and breaking it there turns one dense block into a clear claim followed by its evidence. The ending can stay as it is.',
  'If you only change one thing, change the verbs. "Is able to help" becomes "helps", "make a decision" becomes "decide". That alone trims about ten percent and makes the tone more confident without touching the structure you already like.',

const attempts: Record<string, number> = {};

function pickReply(prompt: string): string {
  const key = prompt.trim().toLowerCase();
  const n = attempts[key] || 0;
  attempts[key] = n + 1;
  const match = REPLIES.find((r) => key.includes(r.keyword));
  const variants = match ? match.variants : FALLBACK;
  // Asking the same prompt again (regenerate) gives the next variant.
  return variants[n % variants.length];
}

// POST /chat/stream
// Calls onToken(token) for each chunk of the reply, like a server-sent event stream.
// Resolves when the reply is complete.
// If `signal` is aborted, stops emitting and rejects with a DOMException named "AbortError".
export function streamReply(prompt: string, onToken: (token: string) => void, signal: AbortSignal): Promise<void> {
  const tokens = pickReply(prompt).match(/\S+\s*/g) || [];
  return new Promise((resolve, reject) => {
    let i = 0;
    let timer: ReturnType<typeof setTimeout>;

    function abort() {
      clearTimeout(timer);
      reject(new DOMException('The stream was aborted.', 'AbortError'));
    }
    if (signal.aborted) {
      abort();
      return;
    }
    signal.addEventListener('abort', abort, { once: true });

    function next() {
      if (i >= tokens.length) {
        signal.removeEventListener('abort', abort);
        resolve();
        return;
      }
      onToken(tokens[i]);
      i += 1;
      timer = setTimeout(next, 15 + Math.random() * 45);
    }
    // time to first token
    timer = setTimeout(next, 300 + Math.random() * 400);
  });
}

          })}

    </div>
  );
}

import React, { useEffect, useRef, useState } from 'react';
import { streamReply } from './api';
import type { ChatMessage } from './api';
import { makeId } from './utils/id';
import { SEED } from './seed';

// CORE: per-message status (client-only)
type Status = 'streaming' | 'done' | 'stopped' | 'error';
type Message = ChatMessage & { status?: Status };

function Header() {
  return (
    <header className="header">
      <span className="logo">✒</span>
      <div>
        <h1>Quill</h1>
        <p>Draft: “After hours”</p>
      </div>
    </header>
  );
}

// CHANGE: canRegenerate / onRegenerate props
type MessageBubbleProps = {
  message: Message;
  canRegenerate: boolean;
  onRegenerate: () => void;
};

function MessageBubble({ message, canRegenerate, onRegenerate }: MessageBubbleProps) {
  if (message.role === 'user') {
    return (
      <div className="msg msg-user">
        <p className="bubble">{message.text}</p>
      </div>
    );
  }

  // CORE: blinking cursor while streaming
  const streaming = message.status === 'streaming';

  let cursor = null;
  if (streaming) {
    cursor = <span className="cursor" aria-hidden="true" />;
  }

  // NEXT: thinking dots before the first token arrives
  // Before the first token arrives, show "thinking" dots instead of an empty line.
  let text = <p className="assistant-text">{message.text}{cursor}</p>;
  if (streaming && message.text === '') {
    text = <p className="thinking" aria-label="Quill is thinking"><span /><span /><span /></p>;
  }

  // CORE: Stopped / error pill
  let note = null;
  if (message.status === 'stopped') {
    note = <span className="msg-note">Stopped</span>;
  } else if (message.status === 'error') {
    note = <span className="msg-note is-error">Something went wrong</span>;
  }

  // CHANGE: Regenerate under the last answer
  let regen = null;
  if (canRegenerate) {
    regen = <button type="button" className="regen-btn" onClick={() => { onRegenerate(); }}>↻ Regenerate</button>;
  }

  // CORE: footer row
  let footer = null;
  if (note || regen) {
    footer = <div className="msg-footer">{note}{regen}</div>;
  }

  return (
    <div className="msg msg-assistant">
      <span className="avatar" aria-hidden="true">✦</span>
      <div className="assistant-body" aria-busy={streaming}>
        {text}
        {footer}
      </div>
    </div>
  );
}

// CORE: Composer knows about streaming and Stop
type ComposerProps = {
  value: string;
  streaming: boolean;
  onChange: (value: string) => void;
  onSend: () => void;
  onStop: () => void;
};

function Composer({ value, streaming, onChange, onSend, onStop }: ComposerProps) {
  // Enter sends, Shift+Enter makes a new line. Never while a reply is streaming.
  function handleKeyDown(e: React.KeyboardEvent<HTMLTextAreaElement>) {
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault();
      if (!streaming) {
        onSend();
      }
    }
  }

  let button = <button type="button" className="send-btn" disabled={value.trim() === ''} onClick={() => { onSend(); }}>Send</button>;
  if (streaming) {
    button = <button type="button" className="stop-btn" onClick={() => { onStop(); }}><span className="stop-icon" aria-hidden="true" />Stop</button>;
  }

  return (
    <div className="composer">
      <textarea
        className="composer-input"
        rows={2}
        placeholder="Ask Quill to rewrite, shorten, or suggest…"
        value={value}
        onChange={(e) => { onChange(e.target.value); }}
        onKeyDown={handleKeyDown}
      />
      {button}
    </div>
  );
}

export default function App() {
  const [messages, setMessages] = useState<Message[]>(SEED);
  const [input, setInput] = useState('');
  // CORE: which message is streaming (drives Stop) + the controller (never rendered, so a ref)
  const [streamingId, setStreamingId] = useState<string | null>(null);
  // NEXT: scroll-follow state
  // Is the user scrolled to the bottom? Only then do new tokens auto-scroll.
  const [atBottom, setAtBottom] = useState(true);
  // CORE
  const controllerRef = useRef<AbortController | null>(null);
  // NEXT
  const threadRef = useRef<HTMLDivElement>(null);

  // NEXT: follow the stream, but never yank someone who scrolled up to read.
  useEffect(() => {
    const el = threadRef.current;
    if (el && atBottom) {
      el.scrollTop = el.scrollHeight;
    }
  }, [messages, atBottom]);

  // NEXT: leaving the page mid-stream must not keep the request running.
  useEffect(() => {
    return () => {
      if (controllerRef.current) {
        controllerRef.current.abort();
      }
    };
  }, []);

  // CORE: one place that updates a message and one that ends a stream
  function patchMessage(id: string, patch: Partial<Message>) {
    setMessages((prev) => {
      return prev.map((m) => {
        if (m.id !== id) {
          return m;
        }
        return { ...m, ...patch };
      });
    });
  }

  function finish(id: string, status: Status) {
    patchMessage(id, { status });
    setStreamingId(null);
    controllerRef.current = null;
  }

  // CORE: start a stream into an existing assistant message
  function runStream(assistantId: string, prompt: string) {
    const controller = new AbortController();
    controllerRef.current = controller;
    setStreamingId(assistantId);
    setAtBottom(true); // NEXT: sending always scrolls down

    function handleToken(token: string) {
      // updater form: tokens arrive faster than renders
      setMessages((prev) => {
        return prev.map((m) => {
          if (m.id !== assistantId) {
            return m;
          }
          return { ...m, text: m.text + token };
        });
      });
    }

    streamReply(prompt, handleToken, controller.signal)
      .then(() => {
        finish(assistantId, 'done');
      })
      .catch((err) => {
        if (err && err.name === 'AbortError') {
          finish(assistantId, 'stopped');
          return;
        }
        finish(assistantId, 'error');
      });
  }

  // CORE: append both messages, then stream
  function handleSend() {
    const prompt = input.trim();
    if (prompt === '' || streamingId) {
      return;
    }
    const userMsg: Message = { id: makeId('msg'), role: 'user', text: prompt };
    const assistantMsg: Message = { id: makeId('msg'), role: 'assistant', text: '', status: 'streaming' };
    setMessages((prev) => {
      return [...prev, userMsg, assistantMsg];
    });
    setInput('');
    runStream(assistantMsg.id, prompt);
  }

  // CORE: Stop aborts; the catch above marks it 'stopped'
  function handleStop() {
    if (controllerRef.current) {
      controllerRef.current.abort();
    }
  }

  // CHANGE: clear the last answer in place and stream it again for the same prompt.
  function handleRegenerate() {
    const last = messages[messages.length - 1];
    const prev = messages[messages.length - 2];
    if (!last || !prev || last.role !== 'assistant' || prev.role !== 'user' || streamingId) {
      return;
    }
    patchMessage(last.id, { text: '', status: 'streaming' });
    runStream(last.id, prev.text);
  }

  // NEXT: track whether the user is at the bottom
  function handleScroll() {
    const el = threadRef.current;
    if (!el) {
      return;
    }
    // a few px of slack: sub-pixel rounding means "bottom" is rarely exactly 0
    setAtBottom(el.scrollHeight - el.scrollTop - el.clientHeight < 40);
  }

  function jumpToLatest() {
    setAtBottom(true);
  }

  // CHANGE: only the last message can regenerate
  const lastId = messages[messages.length - 1].id;

  // NEXT: jump pill
  let jump = null;
  if (!atBottom && streamingId) {
    jump = <button type="button" className="jump-btn" onClick={() => { jumpToLatest(); }}>↓ Jump to latest</button>;
  }

  // CORE: render (NEXT adds thread-wrap, ref, onScroll and the pill)
  return (
    <div className="app">
      <Header />
      <div className="thread-wrap">
        <div className="thread" ref={threadRef} onScroll={handleScroll}>
          <div className="thread-inner">
            {messages.map((m) => {
              const canRegenerate = m.id === lastId && m.role === 'assistant' && !streamingId;
              return <MessageBubble key={m.id} message={m} canRegenerate={canRegenerate} onRegenerate={handleRegenerate} />;
            })}
          </div>
        </div>
        {jump}
      </div>
      <Composer value={input} streaming={streamingId !== null} onChange={setInput} onSend={handleSend} onStop={handleStop} />
    </div>
  );
}
